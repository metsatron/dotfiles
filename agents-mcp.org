** Claude - Telegram Bot API
Minimal Bot API MCP server. Reads =TELEGRAM_BOT_TOKEN= and =TELEGRAM_CHAT_ID=
from the environment. Set those per workspace in local MCP client config.

#+BEGIN_SRC python :tangle all/.local/bin/claude-telegram-mcp :shebang "#!/usr/bin/env python3" :comments no
"""Telegram Bot API MCP server — send/receive messages via configured bot."""
import os
import requests
from mcp.server.fastmcp import FastMCP

mcp = FastMCP("telegram")
_TOKEN = os.environ["TELEGRAM_BOT_TOKEN"]
_CHAT = os.environ.get("TELEGRAM_CHAT_ID", "")
_API = f"https://api.telegram.org/bot{_TOKEN}"


@mcp.tool()
def send_message(text: str, chat_id: str = "") -> str:
    """Send a text message to the configured Telegram chat. Supports Markdown."""
    target = chat_id or _CHAT
    r = requests.post(f"{_API}/sendMessage", json={
        "chat_id": target, "text": text, "parse_mode": "Markdown"
    }, timeout=10)
    return "ok" if r.ok else f"error: {r.text}"


@mcp.tool()
def get_recent_messages(limit: int = 10) -> str:
    """Get recent messages sent to this bot (getUpdates)."""
    r = requests.get(f"{_API}/getUpdates", params={"limit": limit}, timeout=10)
    if not r.ok:
        return f"error: {r.text}"
    updates = r.json().get("result", [])
    if not updates:
        return "No recent messages."
    lines = []
    for u in updates:
        msg = u.get("message", {})
        user = msg.get("from", {}).get("first_name", "?")
        text = msg.get("text", "[no text]")
        lines.append(f"{user}: {text}")
    return "\n".join(lines)


if __name__ == "__main__":
    mcp.run(transport="stdio")
#+END_SRC

** Agentify Desktop - browser-session MCP (ChatGPT / Perplexity / more)
Local control center that drives the user's OWN logged-in browser over Chrome
DevTools Protocol and exposes it to MCP-capable agents (Claude Code, Codex,
OpenCode, Hermes) as =agentify_*= tools: query, navigate, read page, tabs,
artifacts. Six services: ChatGPT, Perplexity, Claude, Gemini, AI Studio, Grok.
This is the fleet organ for reaching web subscriptions (ChatGPT Projects,
Perplexity Spaces) from any agent without an API key — the subscription login
lives in the browser profile, never in config. State: =~/.agentify-desktop/=.
Loopback-only HTTP API; CAPTCHAs pause for the human by design.

Pinned 0.2.4 (encoded 2026-08-15; first consumer: the Haiku governor lane in
HelmCortex FORGE). The upstream ChatGPT controller types text one character at
a time through CDP =Input.insertText=. Literal newline insertion is not a lawful
multiline gesture for ChatGPT's ProseMirror composer and can submit fragments
before Agentify's explicit final send. Both launchers therefore use a locally
prepared, exact-version runtime. The preparer verifies the upstream controller
SHA-256 and patches the final typing boundary to normalize line separators and
emit Shift+Enter for each line break. Ordinary text is inserted in bounded
runs, avoiding the upstream per-character delay. The composer is verified empty
before any prompt bytes are inserted, with one exact selection retry before the
send fails closed. Text structure is preserved, no raw newline reaches
=Input.insertText=, and the existing explicit send remains the only submission.
Any unknown source or failed verification aborts launch; Agentify is never
allowed to run unguarded. One entrypoint, subcommand dispatch
(verified via =--help=): default/=gui= starts the control center, =mcp= runs the
stdio MCP server.

    title: 'ChatGPT',
    readyState: 'complete',

test('structured prompts use Shift+Enter and submit exactly once', async () => {
  const events = [];
  let waitForSendChecks = 0;
  let clearChecks = 0;
  const page = {
    async navigate() {},
    async evaluate(js) {
      if (js.includes('const hasTurnstile')) return readyState();
      if (js.includes('selectNodeContents')) {
        clearChecks += 1;
        return { ok: true, promptLen: 9 };
      }
      if (js.includes('const text = el.matches')) {
        clearChecks += 1;
        return { ok: true, promptLen: 0 };
      }
      if (js.includes('missing_prompt_textarea')) return { ok: true, rect: { x: 10, y: 10, w: 200, h: 40 } };
      if (js.includes('form.requestSubmit')) {
        events.push({ type: 'submit' });
        return true;
      }
      if (js.includes('already_generating')) return { ok: true, requestSubmit: true, host: 'chatgpt.com' };
      if (js.includes('promptLen')) {
        waitForSendChecks += 1;
        return waitForSendChecks >= 2
          ? { stopVisible: false, sendDisabled: true, promptLen: 0 }
          : { stopVisible: false, sendDisabled: false, promptLen: 32 };
      }
      throw new Error(`unexpected_eval:${js.slice(0, 80)}`);
    },
    async getUrl() { return 'https://chatgpt.com/'; },
    async sendKey(key, options = {}) {
      events.push({ type: 'key', key, modifiers: options.modifiers || [] });
    },
    async insertText(text) { events.push({ type: 'text', text }); },
    async moveMouse() {},
    async mouseDown() {},
    async mouseUp() {},
    async setFileInputFiles() {}
  };

  const controller = new ChatGPTController({
    page,
    selectors: {
      promptTextarea: '#prompt-textarea',
      sendButton: 'button[data-testid="send-button"]',
      stopButton: 'button[data-testid="stop-button"]',
      assistantMessage: '[data-message-author-role="assistant"]'
    }
  });

  const longLine = 'x'.repeat(5000);
  const prompt = `# Heading\r\n\r\n- first\u2028- second\u2029\`\`\`js\ncode();\n${longLine}\n\`\`\``;
  await controller.send({ text: prompt, timeoutMs: 5_000 });

  const clearAt = events.findLastIndex((event) => event.type === 'key' && event.key === 'Backspace');
  const submitAt = events.findIndex((event) => event.type === 'submit');
  const typing = events.slice(clearAt + 1, submitAt);
  assert.ok(clearAt >= 0);
  assert.ok(submitAt >= 0);
  assert.equal(clearChecks, 2);
  assert.equal(events.filter((event) => event.type === 'key' && event.key === 'Backspace').length, 2);
  assert.equal(events.filter((event) => event.type === 'submit').length, 1);
  assert.equal(typing.some((event) => event.type === 'text' && /[\r\n\u2028\u2029]/u.test(event.text)), false);
  assert.equal(typing.some((event) => event.type === 'key' && event.key === 'Enter' && event.modifiers.length === 0), false);
  assert.ok(typing.filter((event) => event.type === 'text').every((event) => event.text.length <= 2048));
  assert.ok(typing.filter((event) => event.type === 'text').length < 16);

  const reconstructed = typing.map((event) => {
    if (event.type === 'text') return event.text;
    if (event.type === 'key' && event.key === 'Enter' && event.modifiers.includes('shift')) return '\n';
    return '';
  }).join('');
  assert.equal(reconstructed, `# Heading\n\n- first\n- second\n\`\`\`js\ncode();\n${longLine}\n\`\`\``);
});

*** Agent users reach Agentify through a read-only share

Ruling 2026-09-26: all agents may drive Agentify, the control center stays Mètsàtron's (his browser, his logins), and the API token reaches agent users through a group, never by opening his home. =agentify-agent-share sync= copies only =state.json= (port, pid) and =token.txt= from =~/.agentify-desktop= into =/var/lib/agentify-share= (operator-owned, group =cortex=, directory 2750, files 0640), atomically and only when they change; =agentify-desktop-mcp= falls back to that directory when the caller cannot read the operator state. Loopback is reachable by every local uid, so the token is the only wall, and the group is exactly the set of agent users. =install= (run once as the operator, uses =sudo -n=) creates the directory and =/etc/cron.d/dotcortex-agentify-agent-share=, which syncs every minute, so a token rotation or control-center restart reaches agents within a minute.

#+BEGIN_SRC bash :tangle all/.local/bin/agentify-agent-share :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/agentify-agent-share
# Mirror the Agentify control center's port and token for agent users.
set -euo pipefail

SRC="${AGENTIFY_DESKTOP_STATE_DIR:-$HOME/.agentify-desktop}"
DST="${AGENTIFY_SHARE_DIR:-/var/lib/agentify-share}"
SELF="$(readlink -f "$0")"

usage() {
    cat <<'EOF'
usage: agentify-agent-share sync      copy state.json + token.txt to the share (operator)
       agentify-agent-share install   create the share and its cron owner (operator, once)
       agentify-agent-share status
EOF
}

cmd_sync() {
    [ -d "$DST" ] && [ -w "$DST" ] || { echo "agentify-agent-share: share not writable: $DST (run install)" >&2; exit 1; }
    local f tmp
    for f in state.json token.txt; do
        [ -r "$SRC/$f" ] || { echo "agentify-agent-share: missing $SRC/$f" >&2; exit 1; }
        cmp -s "$SRC/$f" "$DST/$f" 2>/dev/null && continue
        tmp="$(mktemp "$DST/.$f.XXXXXX")"
        cat "$SRC/$f" > "$tmp"
        chmod 0640 "$tmp"
        mv -f "$tmp" "$DST/$f"
    done
}

cmd_install() {
    local me cron tmp
    me="$(id -un)"
    cron=/etc/cron.d/dotcortex-agentify-agent-share
    sudo -n install -d -o "$me" -g cortex -m 2750 "$DST"
    tmp="$(mktemp)"
    cat > "$tmp" <<EOF
# DotCortex Agentify agent share; managed by agentify-agent-share install.
SHELL=/bin/sh
,* * * * * $me HOME=$HOME PATH=/usr/bin:/bin $SELF sync
EOF
    sudo -n install -m 644 "$tmp" "$cron"
    rm -f "$tmp"
    cmd_sync
    cmd_status
}

cmd_status() {
    echo "share: $(stat -c '%A %U:%G' "$DST" 2>/dev/null || echo missing) $DST"
    for f in state.json token.txt; do
        if cmp -s "$SRC/$f" "$DST/$f" 2>/dev/null; then echo "  $f: in sync"; else echo "  $f: STALE or missing"; fi
    done
}

case "${1:-}" in
    sync) cmd_sync ;;
    install) cmd_install ;;
    status) cmd_status ;;
    -h|--help|"") usage ;;
    *) usage >&2; exit 1 ;;
esac
