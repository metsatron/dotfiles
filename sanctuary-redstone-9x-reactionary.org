#+TITLE: Redstone 9X — Reactionary Collection (ReactOS look-and-feel by phob1an)
#+PROPERTY: header-args :mkdirp yes

The *Reactionary* collection projects phob1an's ReactOS-flavoured KDE look (https://www.opencode.net/phob1an/reactionary, GPLv3) across the Redstone 9X cross-widget stack. ReactOS recreates the classic Windows NT/2000 shell, so the Windows 2000 base (=Windows-Classic= chrome, =#d4d0c8=-family face) is the correct bed; each variant simply recolours it to one of phob1an's palettes and wears his icon set with Windows 2000 sounds.

Nothing here is new engine work — it reuses the projector wholesale. Palettes are read straight from the four vendored GTK themes (=linux/.themes/Reactionary*=), which carry phob1an's own =@define-color= values, so the GTK layer and every other toolkit share one source of truth. =redstone-9x-theme-convert= builds the IceWM chrome + qt6ct palette + Wine =.reg= + the =.scheme= file; the vendored GTK theme is left untouched because each carries an =.upstream= marker (the builder skips GTK emission for those). =redstone-9x-scheme-apply= then coats GTK/Qt/Wine/FOX/IceWM from the emitted scheme.

*Icon axis.* Four variants wear phob1an's own icon theme (=Reactionary=, the 3329 isometric SVGs vendored at =linux/.local/share/icons/Reactionary=, inheriting =SE98,Chicago95,hicolor= for anything it lacks). A fifth, *Reactionary 2000*, wears the genuine Windows 2000 =SE98= icon set on the same canonical chrome — the classic-icons counterpart.

*Known limits / future flourishes.*
- The isometric icons are SVG, so the IceWM desktop/menu icon graft (which needs rasterised PNGs) falls back to =SE98=; GTK/Qt apps render the real isometric set. Rasterising a curated subset to PNG for the graft is a future pass.
- phob1an's icons read a touch flat (late-2000s style). A future enhancement (Mètsàtron's idea, 2026-07-31): add an outline + perspective drop-shadow to the isometric SVGs to sell the 3D depth. Authentic-flat for now.
- A dark-face variant (Creatures =#707070=, Mystery =#687969=) keeps phob1an's black caption glyphs; if they read low-contrast against the dark bevel, that is a threshold refinement in the shared engine, not here.

* Builder — redstone-9x-reactionary-build

Host build tool (needs =~/Downloads/IceWM/Windows-95.tar.gz= + Pillow, like every classic builder). Rerun any time to regenerate the whole collection from the current vendored palettes. The generated =.scheme= files, IceWM theme dirs, qt6ct palettes and Wine =.reg= files are vendored payload (git-tracked directly, not tangled) — this driver is their canonical source.

#+BEGIN_SRC bash :tangle linux/.local/bin/redstone-9x-reactionary-build :shebang "#!/bin/bash" :tangle-mode (identity #o755)
# Build the Reactionary (ReactOS by phob1an) Desktop Scheme collection.
# Palettes are read from the vendored GTK themes (phob1an's own @define-color) —
# single source of truth. Chrome/qt6ct/wine/scheme come from theme-convert; the
# vendored GTK theme is protected by its .upstream marker (theme-convert skips
# GTK emission for upstream-marked themes).
set -euo pipefail
ROOT="${HOME}/DotCortex"
SCH="${ROOT}/linux/.config/redstone-9x/schemes"
TMP="$(mktemp -d)"; trap 'rm -rf "${TMP}"' EXIT

# <GtkTheme(=TNAME, camelCase to match the vendored dir)>|<scheme-id>|<Pretty Title>|<icons>
VARIANTS=(
  "Reactionary|reactionary|Reactionary|Reactionary"
  "ReactionaryCreatures|reactionary-creatures|Reactionary Creatures|Reactionary"
  "ReactionaryDaVinci|reactionary-davinci|Reactionary DaVinci|Reactionary"
  "ReactionaryMystery|reactionary-mystery|Reactionary Mystery|Reactionary"
)

getcol(){ grep -hE "@define-color $2 " "${ROOT}/linux/.themes/$1/gtk-3.0/gtk.css" 2>/dev/null | head -1 | grep -oE '#[0-9a-fA-F]{3,6}' | head -1; }
hex6(){ local h="${1#\#}"; if [ "${#h}" -eq 3 ]; then echo "#${h:0:1}${h:0:1}${h:1:1}${h:1:1}${h:2:1}${h:2:1}"; else echo "#${h}"; fi; }
dec(){ local h="${1#\#}"; printf '%d %d %d' "$((16#${h:0:2}))" "$((16#${h:2:2}))" "$((16#${h:4:2}))"; }
# shade toward black by <pct>% (e.g. 72 = keep 72% of each channel)
shade(){ local h="${1#\#}" p="$2"; printf '%d %d %d' "$((16#${h:0:2}*p/100))" "$((16#${h:2:2}*p/100))" "$((16#${h:4:2}*p/100))"; }

for v in "${VARIANTS[@]}"; do
  IFS='|' read -r gtk sid title icons <<< "${v}"
  # phob1an's .upstream marker protects the vendored GTK theme from recolour.
  touch "${ROOT}/linux/.themes/${gtk}/.upstream"
  FACE="$(hex6 "$(getcol "${gtk}" bg_color)")"
  ACCENT="$(hex6 "$(getcol "${gtk}" selected_bg_color)")"
  TITLE="$(hex6 "$(getcol "${gtk}" titlebar_bg_color)")"
  BASE="$(hex6 "$(getcol "${gtk}" base_color)")"
  [ -n "${FACE}" ] && [ -n "${TITLE}" ] || { echo "SKIP ${gtk}: palette not readable" >&2; continue; }
  # Windows .theme: active caption gradient = 72%-shade of phob1an's titlebar
  # (dark endpoint) -> the titlebar colour itself (light endpoint); desktop takes
  # the same dark endpoint so it sits under the caption harmoniously.
  cat > "${TMP}/${sid}.theme" <<EOF
[Control Panel\\Colors]
ActiveTitle=$(shade "${TITLE}" 72)
GradientActiveTitle=$(dec "${TITLE}")
TitleText=255 255 255
InactiveTitle=128 128 128
GradientInactiveTitle=192 192 192
InactiveTitleText=224 224 224
ButtonFace=$(dec "${FACE}")
ButtonText=0 0 0
Window=$(dec "${BASE}")
WindowText=0 0 0
Menu=$(dec "${FACE}")
MenuText=0 0 0
Hilight=$(dec "${ACCENT}")
HilightText=255 255 255
Background=$(shade "${TITLE}" 72)
EOF
  ICON_OVERRIDE="${icons}" redstone-9x-theme-convert "${TMP}/${sid}.theme" "${gtk}" "${sid}"
  # Pretty display title + Windows 2000 sounds (theme-convert names the scheme
  # after the camelCase GTK theme and defaults sound to the scheme id).
  sed -i -e "s/^SCHEME_TITLE=.*/SCHEME_TITLE=\"${title}\"/" \
         -e 's/^SOUND_SCHEME=.*/SOUND_SCHEME="windows-2000"/' "${SCH}/${sid}.scheme"
  echo "built ${sid} (gtk=${gtk} face=${FACE} title=${TITLE} accent=${ACCENT} icons=${icons})"
done

# Reactionary 2000 — the canonical palette on the genuine Windows 2000 SE98 icons.
# Same chrome/GTK/qt6ct/wine as reactionary; only the icon set differs.
cp "${SCH}/reactionary.scheme" "${SCH}/reactionary-2000.scheme"
sed -i -e 's/^SCHEME_TITLE=.*/SCHEME_TITLE="Reactionary 2000"/' \
       -e 's/^ICON_THEME=.*/ICON_THEME="SE98"/' "${SCH}/reactionary-2000.scheme"
echo "built reactionary-2000 (canonical chrome + SE98 win2k icons)"
echo "Reactionary collection complete."
#+END_SRC
