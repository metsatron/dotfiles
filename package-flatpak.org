Reproducible Flatpak state from your dotfiles, Guix style. Repo-backed Flatpaks and GitHub-release Flatpak bundles are both owned here. =loom flatpak:apply= is the single authority for Flatpak install, update, and uninstall decisions.

* App-level Flatpak overrides

Per-app persistent overrides stowed to =~/.local/share/flatpak/overrides/<app-id>=. Flatpak merges these with the app's manifest permissions for every launch path (panel launcher, file manager, CLI, =flatpak run=) — no wrapper script required.

** KeePassXC

Enable Kvantum Qt theming for every launch path. The =org.kde.KStyle.Kvantum//5.15-25.08= extension (declared in the Flatpak manifest above) provides =libkvantum.so= at =/usr/share/runtime/lib/plugins/styles/= inside the sandbox; =~/.config/Kvantum/kvantum.kvconfig= (stowed from =linux/.config/Kvantum/kvantum.kvconfig=, tangled from =sanctuary-kvantum.org=) selects the =QTStep-DarkAdobe= palette. =filesystems=host= in the KeePassXC manifest makes the stow symlink chain fully reachable without additional filesystem overrides.

* Local Flatpak derivatives

Local derivatives keep a second application identity while tracking the payload
of a declared upstream Flatpak.  This is used for Redstone Obsidian: the distinct
application ID gives Electron a distinct user-data directory and singleton, so
Redstone cannot capture host =obsidian://= requests.

* Sanctuary Flatpak runner

Run a host-installed Flatpak payload from inside a sanctuary.  The app/runtime
objects are read from the host Flatpak installations, but process environment,
theme lookup, and default config/cache paths come from the sanctuary guest home.

#+BEGIN_SRC bash :tangle linux/.local/bin/sanctuary-flatpak-run :shebang "#!/bin/bash" :tangle-mode (identity #o755)
set -euo pipefail

usage() {
  echo "Usage: sanctuary-flatpak-run <app-id> [flatpak-run-args...]" >&2
}

if [[ "$#" -lt 1 ]]; then
  usage
  exit 2
fi

case "${HOME}" in
  */.local/share/dotcortex/guests/*/home) ;;
  *)
    echo "sanctuary-flatpak-run: refusing outside a DotCortex guest home: HOME=${HOME}" >&2
    exit 2
    ;;
esac

app_id="$1"
shift

redstone_home="${HOME%/}/"
redstone_home="${redstone_home%/}"
redstone_official_id="md.obsidian.Obsidian"
redstone_derived_id="com.tiagopires.ObsidianRedstone"
if [[ "${redstone_home}" = */.local/share/dotcortex/guests/sanctuary-redstone-9x/home && \
      "${app_id}" = "${redstone_official_id}" ]]; then
  old_app_home="${HOME}/.var/app/${redstone_official_id}"
  new_app_home="${HOME}/.var/app/${redstone_derived_id}"

  # Preserve the Redstone profile when crossing the new Flatpak identity.  The
  # copy is one-way and atomic; the old tree remains as a rollback copy.  Never
  # copy a live Electron singleton because that can produce a split profile.
  if [[ ! -e "${new_app_home}" && -d "${old_app_home}" ]]; then
    old_profile="${old_app_home}/config/obsidian"
    for singleton in SingletonLock SingletonSocket SingletonCookie; do
      if [[ -e "${old_profile}/${singleton}" || -L "${old_profile}/${singleton}" ]]; then
        echo "sanctuary-flatpak-run: close the old Redstone Obsidian before profile migration (${singleton} is live or stale)" >&2
        exit 1
      fi
    done

    migration="${new_app_home}.migrating.$$"
    if [[ -e "${migration}" ]]; then
      echo "sanctuary-flatpak-run: refusing unexpected migration path: ${migration}" >&2
      exit 1
    fi
    cp -a "${old_app_home}" "${migration}"
    mv "${migration}" "${new_app_home}"
    echo "sanctuary-flatpak-run: copied Redstone Obsidian profile to isolated application ID" >&2
  fi

  app_id="${redstone_derived_id}"
fi

host_home="${SANCTUARY_HOST_HOME:-/home/metsatron}"
if [[ -n "${SANCTUARY_FLATPAK_BIN:-}" ]]; then
  flatpak_bin="${SANCTUARY_FLATPAK_BIN}"
elif command -v flatpak >/dev/null 2>&1; then
  flatpak_bin="$(command -v flatpak)"
else
  flatpak_bin="${host_home}/.guix-extra-profiles/core/core/bin/flatpak"
fi
export FLATPAK_USER_DIR="${SANCTUARY_FLATPAK_USER_DIR:-${host_home}/.local/share/flatpak}"
export FLATPAK_SYSTEM_DIR="${SANCTUARY_FLATPAK_SYSTEM_DIR:-/run/host/var/lib/flatpak}"

if [[ ! -x "${flatpak_bin}" ]]; then
  echo "sanctuary-flatpak-run: missing Flatpak binary: ${flatpak_bin}" >&2
  exit 1
fi

if [[ ! -d "${FLATPAK_USER_DIR}/app" ]]; then
  echo "sanctuary-flatpak-run: missing host user Flatpak app store: ${FLATPAK_USER_DIR}" >&2
  exit 1
fi

if [[ ! -d "${FLATPAK_SYSTEM_DIR}/runtime" && ! -d "${FLATPAK_USER_DIR}/runtime" ]]; then
  echo "sanctuary-flatpak-run: missing Flatpak runtime store (checked user=${FLATPAK_USER_DIR} system=${FLATPAK_SYSTEM_DIR})" >&2
  exit 1
fi

if [[ ! -e /var/lib/dbus/machine-id && -r /run/host/var/lib/dbus/machine-id ]]; then
  mkdir -p /var/lib/dbus
  ln -sfnT /run/host/var/lib/dbus/machine-id /var/lib/dbus/machine-id
fi

export XDG_CONFIG_HOME="${XDG_CONFIG_HOME:-${HOME}/.config}"
export XDG_DATA_HOME="${XDG_DATA_HOME:-${HOME}/.local/share}"
export XDG_CACHE_HOME="${XDG_CACHE_HOME:-${HOME}/.cache}"

prepend_xdg_data() {
  local dir="$1"
  [[ -d "$dir" ]] || return 0
  case ":${XDG_DATA_DIRS:-}:" in
    *":${dir}:"*) ;;
    *) XDG_DATA_DIRS="${dir}${XDG_DATA_DIRS:+:${XDG_DATA_DIRS}}" ;;
  esac
}

prepend_xdg_data "${FLATPAK_USER_DIR}/exports/share"
prepend_xdg_data "${FLATPAK_SYSTEM_DIR}/exports/share"
export XDG_DATA_DIRS

safe_link() {
  local target="$1"
  local link="$2"
  mkdir -p "$(dirname "$link")"
  if [[ -e "$link" && ! -L "$link" ]]; then
    echo "sanctuary-flatpak-run: preserving existing real path $link" >&2
    return 0
  fi
  ln -sfnT "$target" "$link"
}

shared_link() {
  local target="$1"
  local link="$2"
  local saved="${link}.local-before-shared"
  mkdir -p "$(dirname "$link")"
  if [[ -L "$link" ]]; then
    ln -sfnT "$target" "$link"
    return 0
  fi
  if [[ -e "$link" ]]; then
    if [[ "$(readlink -f "$link" 2>/dev/null || true)" = "$(readlink -f "$target" 2>/dev/null || true)" ]]; then
      return 0
    fi
    if [[ -e "$saved" ]]; then
      echo "sanctuary-flatpak-run: preserving existing real path $link because $saved already exists" >&2
      return 0
    fi
    mv "$link" "$saved"
  fi
  ln -sfnT "$target" "$link"
}

resolve_shared_root() {
  local name="$1"
  local candidate resolved
  # ${HOME}/${name} (the guest-home symlink) is tried FIRST: each sanctuary
  # points its own guest-home RetroPie/Games at the mount that is actually
  # reachable inside that guest (Redstone: /run/dotcortex/retropie), whereas the
  # host-home symlink (…/RetroPie -> …/mnt/x230/RetroPie) is an autofs path the
  # guest cannot traverse. The `-d "${resolved}"` guard skips such dead symlinks
  # so a broken host-home candidate falls through to a working one.
  for candidate in \
    "${HOME}/${name}" \
    "${host_home}/${name}" \
    "${host_home}/mnt/x230/${name}" \
    "/run/host/home/metsatron/${name}"
  do
    [[ -e "$candidate" || -L "$candidate" ]] || continue
    resolved="$(readlink -f "$candidate" 2>/dev/null || true)"
    [[ -n "${resolved}" ]] || continue
    [[ -d "${resolved}" ]] || continue
    case "${resolved}" in
      /run/host/*) continue ;;
    esac
    printf '%s\n' "${resolved}"
    return 0
  done
  return 1
}

add_shared_root_args() {
  local name="$1"
  local candidate resolved seen=" "
  for candidate in \
    "${HOME}/${name}" \
    "${host_home}/${name}" \
    "${host_home}/mnt/x230/${name}" \
    "/run/host/home/metsatron/${name}"
  do
    [[ -e "$candidate" || -L "$candidate" ]] || continue
    case "${candidate}" in
      /run/host/*) continue ;;
    esac
    case "${seen}" in
      *" ${candidate} "*) ;;
      *)
        run_args+=("--filesystem=${candidate}")
        seen="${seen}${candidate} "
        ;;
    esac
    resolved="$(readlink -f "$candidate" 2>/dev/null || true)"
    [[ -n "${resolved}" ]] || continue
    case "${resolved}" in
      /run/host/*) continue ;;
    esac
    case "${seen}" in
      *" ${resolved} "*) ;;
      *)
        run_args+=("--filesystem=${resolved}")
        seen="${seen}${resolved} "
        ;;
    esac
  done
}

run_args=()
# Forward the sanctuary's X display into the flatpak sandbox so the app
# renders inside the sanctuary (e.g. Xephyr :93) rather than on the host
# display (:0). --socket=x11 shares the X socket for $DISPLAY, and the
# explicitly-set DISPLAY env points the app at it. Without this flatpak
# inherits the host DISPLAY and the window never reaches the sanctuary.
if [[ -n "${DISPLAY:-}" ]]; then
  run_args+=("--socket=x11" "--env=DISPLAY=${DISPLAY}")
fi
# An Xpra display needs both its X11 socket and the matching MIT-MAGIC-COOKIE.
# The host home is visible in the Distrobox, but the nested Flatpak sandbox is
# not allowed to read the authority file unless this exact path is granted.
# Keep the grant read-only and narrow: never expose the host Xauthority tree.
if [[ -n "${XAUTHORITY:-}" && -r "${XAUTHORITY}" ]]; then
  run_args+=("--env=XAUTHORITY=${XAUTHORITY}" "--filesystem=${XAUTHORITY}:ro")
fi
for root_name in Games RetroPie Applications; do
  add_shared_root_args "$root_name"
done

# OpenRA's manifest already permits /run/media and /media, but Redstone's
# optical broker is deliberately visible inside the sanctuary at this exact
# path.  Grant only that read-only root to OpenRA in this sanctuary; the D:
# drive symlink can then be followed by OpenRA's C&C content detector without
# widening the filesystem contract for unrelated Flatpaks.
if [[ "${app_id}" = "net.openra.OpenRA" &&
      "${redstone_home}" = */.local/share/dotcortex/guests/sanctuary-redstone-9x/home &&
      -d /run/dotcortex/redstone-media ]]; then
  run_args+=("--filesystem=/run/dotcortex/redstone-media:ro")
fi

if [[ "${app_id}" = "net.lutris.Lutris" ]]; then
  lutris_app_home="${HOME}/.var/app/net.lutris.Lutris"
  # Second-generation Lutris topology (lutris.org, 2026-07-23): live state is
  # MACHINE-LOCAL, never the shared NFS tree. The guest-home links point at the
  # host machine's real ~/.config/lutris and ~/.local/share/lutris (the host
  # home is a RW bind inside every sanctuary), so every sanctuary on a machine
  # shares that machine's state. Cross-machine sync and the playtime lease are
  # lutris-play/lutris-sync's job — never a shared live mount.
  lutris_cfg="${HOME}/.config/lutris"
  lutris_data="${HOME}/.local/share/lutris"
  [[ -e "${lutris_cfg}" || -L "${lutris_cfg}" ]] || lutris_cfg="${host_home}/.config/lutris"
  [[ -e "${lutris_data}" || -L "${lutris_data}" ]] || lutris_data="${host_home}/.local/share/lutris"
  lutris_cfg="$(readlink -f "${lutris_cfg}" 2>/dev/null || printf '%s' "${lutris_cfg}")"
  lutris_data="$(readlink -f "${lutris_data}" 2>/dev/null || printf '%s' "${lutris_data}")"
  # Glycin (the GNOME 49 runtime's image loader) hard-requires the
  # org.freedesktop.portal.Flatpak name for its flatpak-spawn --sandbox
  # workers. Private in-sanctuary session buses (dbus-run-session) do not
  # carry it, which killed every SVG load and left a 10x10 unrendered window
  # (diagnosed live 2026-07-22). The session launcher projects the HOST bus
  # address into the guest; run the outer flatpak against that bus so portal
  # activation reaches the host's flatpak-portal. X11 forwarding is explicit
  # (--socket=x11 + DISPLAY above), so the window still lands in the Xephyr.
  sanctuary_bus_file="${HOME}/.config/sanctuary/host-session-bus"
  if [[ -r "${sanctuary_bus_file}" ]]; then
    sanctuary_host_bus="$(head -n1 "${sanctuary_bus_file}")"
    if [[ -n "${sanctuary_host_bus}" ]]; then
      export DBUS_SESSION_BUS_ADDRESS="${sanctuary_host_bus}"
    fi
  fi
  mkdir -p \
    "${lutris_cfg}" \
    "${lutris_data}" \
    "${lutris_app_home}/config" \
    "${lutris_app_home}/data" \
    "${lutris_app_home}/cache"
  shared_link "${lutris_cfg}" "${lutris_app_home}/config/lutris"
  shared_link "${lutris_data}" "${lutris_app_home}/data/lutris"
  if [[ -f "${HOME}/.config/gtk-3.0/settings.ini" ]]; then
    mkdir -p "${lutris_app_home}/config/gtk-3.0"
    [[ -L "${lutris_app_home}/config/gtk-3.0/settings.ini" ]] && unlink "${lutris_app_home}/config/gtk-3.0/settings.ini"
    cp -f "${HOME}/.config/gtk-3.0/settings.ini" "${lutris_app_home}/config/gtk-3.0/settings.ini"
  fi
  if [[ -f "${HOME}/.config/gtk-4.0/settings.ini" ]]; then
    mkdir -p "${lutris_app_home}/config/gtk-4.0"
    [[ -L "${lutris_app_home}/config/gtk-4.0/settings.ini" ]] && unlink "${lutris_app_home}/config/gtk-4.0/settings.ini"
    cp -f "${HOME}/.config/gtk-4.0/settings.ini" "${lutris_app_home}/config/gtk-4.0/settings.ini"
  fi
  run_args+=(
    "--filesystem=${lutris_cfg}"
    "--filesystem=${lutris_data}"
    # RO view of the host bin dir so catalogue entries can exec
    # lutris-ports-bridge (which immediately flatpak-spawn --host's out of
    # the sandbox to relay ports scripts into the sanctuary lane).
    "--filesystem=${host_home}/.local/bin:ro"
    "--filesystem=${HOME}/.config/gtk-3.0:ro"
    "--filesystem=${HOME}/.config/gtk-4.0:ro"
    "--env=GTK_THEME=Default"
    "--env=GTK_CSD=0"
    "--env=GTK_OVERLAY_SCROLLING=0"
    "--unset-env=FONTCONFIG_FILE"
    "--unset-env=FONTCONFIG_PATH"
    "--env=WINEESYNC=0"
    "--env=WINEFSYNC=0"
    "--env=XDG_CACHE_HOME=${lutris_app_home}/cache"
  )
fi

flatpak_args=()
app_args=()
before_separator=1
for arg in "$@"; do
  # Flatpak desktop files commonly spell the default app command as
  # --command=. Passing that empty value through to `flatpak run` makes
  # bubblewrap try to exec an empty filename. The default command is already
  # selected by the application ID, so omit only the empty form and preserve
  # real --command=/path values.
  if [[ "$arg" = "--command=" ]]; then
    continue
  fi
  if [[ "$arg" = "--" && "$before_separator" -eq 1 ]]; then
    before_separator=0
    continue
  fi
  if [[ "$before_separator" -eq 1 ]]; then
    flatpak_args+=("$arg")
  else
    app_args+=("$arg")
  fi
done

# Obsidian's Terminal plugin runs inside the Flatpak but its interactive shell
# belongs to the sanctuary guest home.  Expose only the fleet wrapper file and
# grant only flatpak-spawn's host-launch bus name.  The lane registry supplies
# the matching vault, so Cortex, Redstone 9X, and Redstone NT all converge the
# same plugin profile without embedding machine-specific paths here.
if [[ "${app_id}" = "md.obsidian.Obsidian" || \
      "${app_id}" = "com.tiagopires.ObsidianRedstone" ]]; then
  obsidian_terminal_wrapper="${HOME}/.local/bin/obsidian-host-zsh"
  if [[ -x "${obsidian_terminal_wrapper}" ]]; then
    run_args+=(
      "--filesystem=${obsidian_terminal_wrapper}:ro"
      "--talk-name=org.freedesktop.Flatpak"
    )
  fi

  obsidian_lane_env="${XDG_CONFIG_HOME}/dotcortex/capture-lane.env"
  obsidian_terminal_bridge="${HOME}/.local/bin/flatpak-obsidian-terminal-bridge"
  if [[ -r "${obsidian_lane_env}" && -x "${obsidian_terminal_bridge}" ]]; then
    # shellcheck disable=SC1090
    . "${obsidian_lane_env}"
    if [[ "${CAPTURE_LANE_OBSIDIAN_APP_ID:-}" = "${app_id}" &&
          -d "${CAPTURE_LANE_VAULT_PATH:-}" ]]; then
      if ! OBSIDIAN_APP_ID="${app_id}" \
          OBSIDIAN_VAULT="${CAPTURE_LANE_VAULT_PATH}" \
          OBSIDIAN_HOST_ZSH="${obsidian_terminal_wrapper}" \
          GUIX_ZSH="${HOME}/.guix-extra-profiles/core/core/bin/zsh" \
          FLATPAK_BIN="${flatpak_bin}" \
          "${obsidian_terminal_bridge}"; then
        echo "sanctuary-flatpak-run: Obsidian terminal bridge did not converge" >&2
      fi
    fi
  fi
fi

if [[ "${app_id}" = "org.garudalinux.firedragon" ]]; then
  firedragon_profile_link="${SANCTUARY_CORTEX_PROFILE_LINK:-${HOME}/.config/sanctuary/cortex-firefox-profile}"
  if [[ ! -d "${firedragon_profile_link}" ]]; then
    echo "sanctuary-flatpak-run: missing FireDragon profile: ${firedragon_profile_link}" >&2
    exit 1
  fi
  # Flatpak keeps the parent .firedragon directory persistent. Passing the
  # child symlink as a filesystem mount then collides with that persistent
  # mount in bubblewrap, so use the resolved private HelmCortex path for both
  # the mount and the browser profile argument.
  firedragon_profile="$(readlink -f "${firedragon_profile_link}" 2>/dev/null || true)"
  if [[ -z "${firedragon_profile}" || ! -d "${firedragon_profile}" ]]; then
    echo "sanctuary-flatpak-run: resolved FireDragon profile is unavailable: ${firedragon_profile_link}" >&2
    exit 1
  fi
  run_args+=("--filesystem=${firedragon_profile}")
  has_profile_arg=0
  for arg in "${app_args[@]}"; do
    [[ "${arg}" = "--profile" || "${arg}" = --profile=* ]] && has_profile_arg=1
  done
  if [[ "${has_profile_arg}" -eq 0 ]]; then
    app_args=("--profile" "${firedragon_profile}" "${app_args[@]}")
  fi
fi

# The Cortex Obsidian entry opens the private HelmCortex vault from a path
# projected into the guest home.  The container can traverse that symlink, but
# Flatpak's `home` permission does not follow it into the host-backed target.
# Expose only this exact vault to official Obsidian in sanctuary-cortex; keep
# other sanctuary Flatpaks on their existing narrower filesystem contract.
if [[ "${app_id}" = "md.obsidian.Obsidian" &&
      "${redstone_home}" = */.local/share/dotcortex/guests/sanctuary-cortex/home ]]; then
  obsidian_vault="${SANCTUARY_CORTEX_OBSIDIAN_VAULT:-${HOME}/HelmCortex}"
  if [[ ! -d "${obsidian_vault}" ]]; then
    echo "sanctuary-flatpak-run: Cortex Obsidian vault is unavailable: ${obsidian_vault}" >&2
    exit 1
  fi
  run_args+=("--filesystem=${obsidian_vault}")

  # The nested Flatpak cannot mount the host document portal's FUSE tree, so
  # its folder picker falls back to Quick Start.  A bare menu launch therefore
  # opens the registered private vault directly.  An obsidian:// URI is already
  # a complete transport envelope whose vault= query selects the vault; never
  # combine it with the CLI --vault selector, because Obsidian interprets the
  # mixed forms as a second vault/session launch.
  has_vault_arg=0
  for arg in "${app_args[@]}"; do
    case "${arg}" in
      --vault|--vault=*|vault=*|obsidian://*) has_vault_arg=1 ;;
    esac
  done
  if [[ "${has_vault_arg}" -eq 0 ]]; then
    app_args+=(--vault HelmCortex)
  fi

  # The full Cortex desktop has its own private D-Bus session. Its
  # xdg-desktop-portal is present there, but the document portal's FUSE mount
  # belongs to the live Xpra app session on :100. Obsidian's folder picker
  # therefore returns to Quick Start unless this app talks to the bus that
  # owns org.freedesktop.portal.Documents. Keep DISPLAY=:101 unchanged so
  # the window remains in the full Cortex desktop.
  cortex_portal_bus_file="${SANCTUARY_CORTEX_PORTAL_BUS_FILE:-/run/user/1000/xpra/100/dbus.env}"
  if [[ -r "${cortex_portal_bus_file}" ]]; then
    # Xpra records the app-session bus in a stable, guest-visible file.  Do not
    # inspect /proc here: the sanctuary's proc view hides host process
    # environments even though the shared Xpra runtime directory is readable.
    candidate_bus="$(sed -n 's/^DBUS_SESSION_BUS_ADDRESS=//p' \
      "${cortex_portal_bus_file}" | head -n 1)"
    if [[ -n "${candidate_bus}" ]]; then
      portal_names_ok=0
      if command -v gdbus >/dev/null 2>&1 &&
         DBUS_SESSION_BUS_ADDRESS="${candidate_bus}" \
           gdbus call --session --dest org.freedesktop.DBus \
             --object-path /org/freedesktop/DBus \
             --method org.freedesktop.DBus.ListNames 2>/dev/null |
           grep -Fq 'org.freedesktop.portal.Documents'; then
        portal_names_ok=1
      fi
      if [[ "${portal_names_ok}" -eq 1 ]]; then
        export DBUS_SESSION_BUS_ADDRESS="${candidate_bus}"
        run_args+=("--env=DBUS_SESSION_BUS_ADDRESS=${candidate_bus}")
      fi
    fi
  fi
fi

run_system_flatpak() {

add_remote() {
  local scope="$1" name="$2" url="$3"
  unset_ssl
  if [ "$scope" = "system" ]; then
    if flatpak remote-list --columns=name | grep -qx "$name"; then return 0; fi
    if ! run_system_flatpak remote-add --if-not-exists "$name" "$url"; then

      run_system_flatpak remote-add --if-not-exists --from "$name" "$f"
    fi
  else
    if flatpak --user remote-list --columns=name | grep -qx "$name"; then return 0; fi
    if ! "${CLEAN_ENV[@]}" flatpak --user remote-add --if-not-exists "$name" "$url"; then
      mkdir -p "$HOME/.cache/flatpak-remotes"
      local f="$HOME/.cache/flatpak-remotes/${name}.flatpakrepo"
      "${CLEAN_ENV[@]}" curl -fsSL "$url" -o "$f"
      "${CLEAN_ENV[@]}" flatpak --user remote-add --if-not-exists --from "$name" "$f"
    fi
  fi
}

* Capture live state into the manifest
Merges your current user and system apps into =apps.ssv=. Keeps comments. Does not remove existing lines.
#+BEGIN_SRC bash :tangle linux/.local/bin/flatpak-capture :shebang "#!/usr/bin/env bash"
#!/usr/bin/env bash
set -euo pipefail
OUT="${HOME}/DotCortex/linux/.flatpak/manifest/apps.ssv"

# Run flatpak in a clean env to avoid SSL/Guix contamination
CLEAN_ENV=(env -i HOME="$HOME" PATH="/usr/sbin:/usr/bin:/bin"
  XDG_RUNTIME_DIR="${XDG_RUNTIME_DIR:-/run/user/$(id -u)}"
  DBUS_SESSION_BUS_ADDRESS="${DBUS_SESSION_BUS_ADDRESS:-}")

dump_scope() { # user|system
  local scope="$1"
  if [ "$scope" = user ]; then
    "${CLEAN_ENV[@]}" flatpak --user list --app --columns=origin,application,branch
  else
    "${CLEAN_ENV[@]}" flatpak --system list --app --columns=origin,application,branch
  fi | awk -v s="$scope" '
    NF>=3 { printf "%s %s %s %s\n", $1, $2, $3, s }
  '
}

tmp="$(mktemp)"; trap 'rm -f "$tmp"' EXIT

{
  # keep comments / blanks
  [ -f "$OUT" ] && grep -E '^\s*#|^\s*$' "$OUT" || true
  # merge existing data + fresh dumps, uniq
  {
    [ -f "$OUT" ] && grep -Ev '^\s*#|^\s*$' "$OUT" || true
    dump_scope user
    dump_scope system
  } | awk 'NF>=4 {print $1, $2, $3, $4}' | sort -u
} > "$tmp"

mv "$tmp" "$OUT"
echo "Wrote $OUT"

* GitHub release-backed Flatpak bundles
These helpers keep Flatpak bundles cached under =${XDG_CACHE_HOME:-$HOME/.cache}/flatpak/releases= and install them through Flatpak, while still letting =loom flatpak:apply= own the full desired state.

#+BEGIN_SRC bash :tangle linux/.local/bin/flatpak-release-sync :shebang "#!/usr/bin/env bash"
#!/usr/bin/env bash
set -euo pipefail

REL_MAN="${HOME}/DotCortex/linux/.flatpak/manifest/releases.ssv"
CACHE_BASE="${XDG_CACHE_HOME:-$HOME/.cache}/flatpak/releases"
FORCE_SUDO_SYSTEM="${FORCE_SUDO_SYSTEM:-0}"

CLEAN_ENV=(env -i HOME="$HOME" PATH="/usr/sbin:/usr/bin:/bin:${HOME}/.local/bin"
  LANG="${LANG:-C.UTF-8}" LC_ALL="${LC_ALL:-C.UTF-8}"
  TERM="${TERM:-xterm-256color}"
  XDG_RUNTIME_DIR="${XDG_RUNTIME_DIR:-/run/user/$(id -u)}"
  DBUS_SESSION_BUS_ADDRESS="${DBUS_SESSION_BUS_ADDRESS:-}")

[ -f "$REL_MAN" ] || exit 0

gh_latest_json() {
  local repo="$1" url="https://api.github.com/repos/${repo}/releases/latest"
  if [ -n "${GH_TOKEN:-}" ]; then
    curl -fsSL -H "Authorization: Bearer $GH_TOKEN" "$url"
  else
    curl -fsSL "$url"
  fi
}

cache_dir() {
  local app_id="$1"
  printf '%s/%s\n' "$CACHE_BASE" "$app_id"
}

installed_version() {
  local app_id="$1" scope="$2"
  if [ "$scope" = user ]; then
    { "${CLEAN_ENV[@]}" flatpak --user info "$app_id" 2>/dev/null | awk -F': *' '/^[[:space:]]*Version:/ {print $2}'; } || true
  else
    { "${CLEAN_ENV[@]}" flatpak --system info "$app_id" 2>/dev/null | awk -F': *' '/^[[:space:]]*Version:/ {print $2}'; } || true
  fi
}

run_install() {
  local scope="$1" bundle="$2"
  if [ "$scope" = system ] && [ "$FORCE_SUDO_SYSTEM" = 1 ]; then
    sudo "${CLEAN_ENV[@]}" flatpak install -y --or-update --system --bundle "$bundle"
    return 0
  fi
  if [ "$scope" = user ]; then
    "${CLEAN_ENV[@]}" flatpak install -y --or-update --user --bundle "$bundle"
  else
    if ! "${CLEAN_ENV[@]}" flatpak install -y --or-update --system --bundle "$bundle"; then
      echo "Retrying with sudo: flatpak install --or-update --system --bundle $bundle" >&2
      sudo "${CLEAN_ENV[@]}" flatpak install -y --or-update --system --bundle "$bundle"
    fi
  fi
}

sync_one() {
  local repo="$1" app_id="$2" asset_name="$3" scope="$4"
  local json latest_tag latest_ver asset_url asset_digest dir bundle old_name installed tmp resolved_asset_name asset_entry_name asset_entry_url asset_entry_digest

  json="$(gh_latest_json "$repo")"
  latest_tag="$(jq -r '.tag_name' <<<"$json")"
  latest_ver="$(sed -E 's/^v//' <<<"$latest_tag")"
  while IFS=$'\t' read -r asset_entry_name asset_entry_url asset_entry_digest; do
    case "$asset_entry_name" in
      $asset_name)
        resolved_asset_name="$asset_entry_name"
        asset_url="$asset_entry_url"
        asset_digest="$asset_entry_digest"
        break
        ;;
    esac
  done < <(jq -r '.assets[] | [.name, .browser_download_url, (.digest // "")] | @tsv' <<<"$json")

  [ -n "${asset_url:-}" ] || { echo "No asset matching $asset_name for $repo" >&2; return 1; }

  dir="$(cache_dir "$app_id")"
  mkdir -p "$dir"
  bundle="$dir/$resolved_asset_name"
  old_name="$( [ -f "$dir/asset.name" ] && cat "$dir/asset.name" || true )"
  if [ -n "$old_name" ] && [ "$old_name" != "$resolved_asset_name" ] && [ -f "$dir/$old_name" ]; then
    rm -f "$dir/$old_name"
  fi

  if [ ! -s "$bundle" ] || [ "$( [ -f "$dir/asset.digest" ] && cat "$dir/asset.digest" || true )" != "$asset_digest" ]; then
    tmp="$bundle.part"
    curl -fL "$asset_url" -o "$tmp"
    mv -f "$tmp" "$bundle"
  fi

  printf '%s\n' "$json" > "$dir/latest.json"
  printf '%s\n' "$latest_tag" > "$dir/tag"
  printf '%s\n' "$resolved_asset_name" > "$dir/asset.name"
  printf '%s\n' "$asset_url" > "$dir/asset.url"
  printf '%s\n' "$asset_digest" > "$dir/asset.digest"

  installed="$(installed_version "$app_id" "$scope")"
  echo "  release: repo=$repo id=$app_id scope=$scope latest=$latest_ver installed=${installed:-none}"
  if [ -n "$installed" ] && [ "$installed" = "$latest_ver" ]; then
    echo "  -> already current, skipping bundle reinstall for $app_id"
    return 0
  fi
  run_install "$scope" "$bundle"
}

echo "== Syncing release-backed Flatpaks =="
awk '$0!~/^#/ && NF>=4 {print $1 "\t" $2 "\t" $3 "\t" $4}' "$REL_MAN" \
  | while IFS=$'\t' read -r repo app_id asset_name scope; do
      sync_one "$repo" "$app_id" "$asset_name" "$scope"
    done

  | while IFS=$'\t' read -r repo app_id asset_name scope; do
      have="$(installed_version "$app_id" "$scope")"
      if [ -z "$have" ]; then
        echo "  + $app_id $scope (missing, release bundle from $repo)"
        continue
      fi
      json="$(gh_latest_json "$repo")"
      latest_tag="$(jq -r '.tag_name' <<<"$json")"
      latest_ver="$(sed -E 's/^v//' <<<"$latest_tag")"
      resolved_asset_name=""
      while IFS=$'\t' read -r asset_entry_name _asset_entry_url _asset_entry_digest; do
        case "$asset_entry_name" in
          $asset_name)
            resolved_asset_name="$asset_entry_name"
            break
            ;;
        esac
      done < <(jq -r '.assets[] | [.name, .browser_download_url, (.digest // "")] | @tsv' <<<"$json")
      if [ -z "$resolved_asset_name" ]; then
        echo "  ? $app_id $scope (no release asset matching $asset_name in $repo)"
        continue
      fi
      newest="$(printf '%s\n%s\n' "$have" "$latest_ver" | sort -V | tail -n1)"
      if [ "$newest" = "$have" ]; then
        echo "  = $app_id $scope ($have, asset $resolved_asset_name)"
      else
        echo "  ~ $app_id $scope ($have -> $latest_ver, asset $resolved_asset_name)"
      fi
    done

* Local Flatpak derivative builder

Build the Redstone identity from the currently declared and installed official
Obsidian deployment.  Flatpak's OSTree repository deduplicates the unchanged
payload; only exported identity files and metadata differ.  The source commit is
recorded in the derivative metadata so =flatpak:diff= and =flatpak:apply= can
detect and refresh it after an upstream update.

#+BEGIN_SRC bash :tangle linux/.local/bin/flatpak-local-derive :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

MAN="${HOME}/DotCortex/linux/.flatpak/manifest/local-derivatives.ssv"
REMOTE="dotcortex-local"
REPO="${HOME}/.local/share/dotcortex/flatpak/repo"
CACHE_ROOT="${XDG_CACHE_HOME:-${HOME}/.cache}/flatpak/local-derive"
MODE="${1:-apply}"
ACTIVE_WORK=""

cleanup_active_work() {
  [[ -n "${ACTIVE_WORK}" ]] || return 0
  case "${ACTIVE_WORK}" in
    "${CACHE_ROOT}/"*)
      [[ -d "${ACTIVE_WORK}" ]] && find "${ACTIVE_WORK}" -depth -delete
      ;;
    *)
      echo "flatpak-local-derive: refusing unsafe cleanup path: ${ACTIVE_WORK}" >&2
      return 1
      ;;
  esac
  ACTIVE_WORK=""
}

trap cleanup_active_work EXIT

if [[ "${MODE}" != "apply" && "${MODE}" != "--check" ]]; then
  echo "Usage: flatpak-local-derive [--check]" >&2
  exit 2
fi

[[ -f "${MAN}" ]] || exit 0

installed_source_commit() {
  flatpak --user info --show-commit "$1" 2>/dev/null || true
}

derived_source_commit() {
  { flatpak --user info --show-metadata "$1" 2>/dev/null || true; } \
    | awk -F= '$0=="[X-DotCortex]"{section=1; next} /^\[/{section=0} section && $1=="source-commit"{print $2; exit}'
}

check_one() {
  local source_id="$1" derived_id="$2" branch="$3" scope="$4"
  local source_commit derived_commit
  if [[ "${scope}" != "user" ]]; then
    echo "  ! ${derived_id}: unsupported local derivative scope ${scope}" >&2
    return 1
  fi
  source_commit="$(installed_source_commit "${source_id}")"
  derived_commit="$(derived_source_commit "${derived_id}")"
  if [[ -z "${source_commit}" ]]; then
    echo "  ! ${derived_id}: source ${source_id} is not installed"
  elif [[ "${source_commit}" = "${derived_commit}" ]]; then
    echo "  = ${derived_id} ${scope} (tracks ${source_id} ${source_commit:0:12})"
  elif [[ -z "${derived_commit}" ]]; then
    echo "  + ${derived_id} ${scope} (derive from ${source_id} ${source_commit:0:12})"
  else
    echo "  ~ ${derived_id} ${scope} (${derived_commit:0:12} -> ${source_commit:0:12})"
  fi
}

derive_one() {
  local source_id="$1" derived_id="$2" branch="$3" scope="$4"
  local source_commit derived_commit source_dir work build metadata_tmp
  local source_desktop derived_desktop source_metainfo derived_metainfo icon old_name new_name

  [[ "${scope}" = "user" ]] || {
    echo "flatpak-local-derive: unsupported scope for ${derived_id}: ${scope}" >&2
    return 1
  }

  source_commit="$(installed_source_commit "${source_id}")"
  [[ -n "${source_commit}" ]] || {
    echo "flatpak-local-derive: source app is not installed: ${source_id}" >&2
    return 1
  }
  derived_commit="$(derived_source_commit "${derived_id}")"

  if [[ "${source_commit}" != "${derived_commit}" ]]; then
    source_dir="$(flatpak --user info --show-location "${source_id}")"
    [[ -d "${source_dir}/files" && -f "${source_dir}/metadata" ]] || {
      echo "flatpak-local-derive: incomplete source deployment: ${source_dir}" >&2
      return 1
    }

    mkdir -p "${CACHE_ROOT}" "${REPO}"
    work="$(mktemp -d "${CACHE_ROOT}/${derived_id}.XXXXXX")"
    case "${work}" in
      "${CACHE_ROOT}/"*) ;;
      *) echo "flatpak-local-derive: unsafe work directory: ${work}" >&2; return 1 ;;
    esac
    ACTIVE_WORK="${work}"
    build="${work}/build"
    mkdir -p "${build}"

    # Hard-link the immutable deployment for a cheap build tree.  Every file we
    # modify is written to a new inode before replacement, so the official
    # deployment is never edited.
    cp -al "${source_dir}/files" "${build}/files"

    metadata_tmp="${work}/metadata"
    awk -v old="${source_id}" -v new="${derived_id}" -v commit="${source_commit}" '
      { gsub(old,new); print }
      END {
        print ""
        print "[X-DotCortex]"
        print "source-id=" old
        print "source-commit=" commit
      }
    ' "${source_dir}/metadata" >"${metadata_tmp}"
    mv "${metadata_tmp}" "${build}/metadata"

    source_desktop="${build}/files/share/applications/${source_id}.desktop"
    derived_desktop="${build}/files/share/applications/${derived_id}.desktop"
    [[ -f "${source_desktop}" ]] || {
      echo "flatpak-local-derive: source desktop file missing: ${source_desktop}" >&2
      return 1
    }
    awk -v old="${source_id}" -v new="${derived_id}" '
      /^Name=/ { print "Name=Obsidian (Redstone)"; next }
      { gsub(old,new); print }
    ' "${source_desktop}" >"${work}/desktop"
    mv "${work}/desktop" "${derived_desktop}"
    unlink "${source_desktop}"

    source_metainfo="${build}/files/share/metainfo/${source_id}.metainfo.xml"
    derived_metainfo="${build}/files/share/metainfo/${derived_id}.metainfo.xml"
    [[ -f "${source_metainfo}" ]] || {
      echo "flatpak-local-derive: source metainfo missing: ${source_metainfo}" >&2
      return 1
    }
    awk -v old="${source_id}" -v new="${derived_id}" '
      { gsub(old,new); sub("<name>Obsidian</name>","<name>Obsidian (Redstone)</name>"); print }
    ' "${source_metainfo}" >"${work}/metainfo"
    mv "${work}/metainfo" "${derived_metainfo}"
    unlink "${source_metainfo}"

    if [[ -f "${build}/files/bin/obsidian.sh" ]]; then
      awk -v old="${source_id}" -v new="${derived_id}" '{ gsub(old,new); print }' \
        "${build}/files/bin/obsidian.sh" >"${work}/obsidian.sh"
      chmod --reference="${build}/files/bin/obsidian.sh" "${work}/obsidian.sh"
      mv "${work}/obsidian.sh" "${build}/files/bin/obsidian.sh"
    fi

    while IFS= read -r -d '' icon; do
      old_name="$(basename "${icon}")"
      new_name="${old_name/#${source_id}/${derived_id}}"
      mv "${icon}" "$(dirname "${icon}")/${new_name}"
    done < <(find "${build}/files/share/icons" -type f -name "${source_id}.*" -print0 2>/dev/null)

    # Appstream data is regenerated from the renamed metainfo during export.
    if [[ -d "${build}/files/share/app-info" ]]; then
      find "${build}/files/share/app-info" -depth -delete
    fi

    flatpak build-finish "${build}"
    flatpak build-export --no-update-summary "${REPO}" "${build}" "${branch}"
    flatpak build-update-repo "${REPO}"
    cleanup_active_work
  fi

  if flatpak --user remote-list --columns=name | grep -Fxq "${REMOTE}"; then
    flatpak --user remote-modify --url="file://${REPO}" "${REMOTE}"
  else
    flatpak --user remote-add --no-gpg-verify "${REMOTE}" "file://${REPO}"
  fi
  flatpak --user install -y --or-update "${REMOTE}" "${derived_id}" "${branch}"
  flatpak override --user --env=ZYPAK_ZYGOTE_STRATEGY_SPAWN=0 "${derived_id}"
  echo "flatpak-local-derive: ${derived_id} tracks ${source_id} ${source_commit:0:12}"
}

if [[ "${MODE}" = "--check" ]]; then
  echo "[local derivatives]"
  while read -r source_id derived_id branch scope; do
    [[ -n "${source_id:-}" ]] || continue
    check_one "${source_id}" "${derived_id}" "${branch}" "${scope}"
  done < <(awk '$0!~/^#/ && NF>=4{print $1,$2,$3,$4}' "${MAN}")
  exit 0
fi

while read -r source_id derived_id branch scope; do
  [[ -n "${source_id:-}" ]] || continue
  derive_one "${source_id}" "${derived_id}" "${branch}" "${scope}"
done < <(awk '$0!~/^#/ && NF>=4{print $1,$2,$3,$4}' "${MAN}")

#!/usr/bin/env bash
set -euo pipefail

MAN="${HOME}/DotCortex/linux/.flatpak/manifest/apps.ssv"
REL_MAN="${HOME}/DotCortex/linux/.flatpak/manifest/releases.ssv"
LOCAL_MAN="${HOME}/DotCortex/linux/.flatpak/manifest/local-derivatives.ssv"

[ -x "$REL_SYNC_HELPER" ] || REL_SYNC_HELPER="${HOME}/DotCortex/linux/.local/bin/flatpak-release-sync"
LOCAL_SYNC_HELPER="${HOME}/.local/bin/flatpak-local-derive"

[ -x "$OBSIDIAN_HOST_URI_HELPER" ] || OBSIDIAN_HOST_URI_HELPER="${HOME}/DotCortex/linux/.local/bin/obsidian-host-uri-apply"

UNINSTALL="${UNINSTALL:-1}"
UPDATE="${UPDATE:-1}"
FORCE_SUDO_SYSTEM="${FORCE_SUDO_SYSTEM:-0}"
DEBUG="${DEBUG:-0}"

[ "$DEBUG" = 1 ] && set -x

CLEAN_TERM="${TERM:-xterm-256color}"
CLEAN_PATH="/usr/sbin:/usr/bin:/bin:${HOME}/.local/bin"
CLEAN_XDG_DATA_DIRS="${XDG_DATA_DIRS:-/usr/local/share:/usr/share}"

  case ":$CLEAN_XDG_DATA_DIRS:" in
    *":$d:"*) ;;

  LANG="$CLEAN_LANG" LC_ALL="$CLEAN_LANG"
  TERM="$CLEAN_TERM"
  SSL_CERT_FILE="$CA_BUNDLE" SSL_CERT_DIR="$CA_DIR"
  CURL_CA_BUNDLE="$CA_BUNDLE" GIT_SSL_CAINFO="$CA_BUNDLE"

ensure_remotes(){

  [ -x "$helper" ] && "${CLEAN_ENV[@]}" "$helper" || true
}

run_fp(){ local scope="$1"; shift; unset_ssl
  if [ "$scope" = system ] && [ "$FORCE_SUDO_SYSTEM" = 1 ]; then
    sudo "${CLEAN_ENV[@]}" flatpak "$@" </dev/null; return
  fi
  if "${CLEAN_ENV[@]}" flatpak "$@" </dev/null; then return; fi

    echo "Retrying with sudo: flatpak $*" >&2
    sudo "${CLEAN_ENV[@]}" flatpak "$@" </dev/null
  fi
}

# ----- parsing & plan (normalize to TAB-delimited) -----
desired_repo_refs(){
  awk -v FS="$FS" -v OFS="$tab" '$0!~/^#/ && NF>=4{print $2,$3,$4}' "$MAN" | sort -u
}
desired_app_ids(){
  { awk -v FS="$FS" '$0!~/^#/ && NF>=4{print $2}' "$MAN"
    [ -f "$REL_MAN" ] && awk -v FS="$FS" '$0!~/^#/ && NF>=4{print $2}' "$REL_MAN" || true
    [ -f "$LOCAL_MAN" ] && awk -v FS="$FS" '$0!~/^#/ && NF>=4{print $2}' "$LOCAL_MAN" || true; } \
  | sort -u
}
installed_refs(){
  { "${CLEAN_ENV[@]}" flatpak --user   list --columns=application,branch | awk -v OFS="$tab" 'NF>=2{print $1,$2,"user"}'
    "${CLEAN_ENV[@]}" flatpak --system list --columns=application,branch | awk -v OFS="$tab" 'NF>=2{print $1,$2,"system"}'; } \
  | sort -u
}
installed_apps(){
  { "${CLEAN_ENV[@]}" flatpak --user   list --app --columns=application | awk -v OFS="$tab" 'NF{print $1,"user"}'
    "${CLEAN_ENV[@]}" flatpak --system list --app --columns=application | awk -v OFS="$tab" 'NF{print $1,"system"}'; } \
  | sort -u
}

print_plan(){
  local d r; d="$(mktemp)"; r="$(mktemp)"; trap 'rm -f "$d" "$r"' RETURN
  desired_repo_refs >"$d"; installed_refs >"$r"
  echo "== Plan =="
  echo "[to install]"
  comm -23 "$d" "$r" | sed 's/^/  + /' || true
  echo "[to uninstall]"
  compute_extras | sed 's/^/  - /' || true
}

already_installed(){ # id branch scope
  if [ "$3" = user ]; then
    "${CLEAN_ENV[@]}" flatpak --user list --columns=application,branch | awk -v id="$1" -v branch="$2" '$1==id && $2==branch{found=1} END{exit !found}'
  else
    "${CLEAN_ENV[@]}" flatpak --system list --columns=application,branch | awk -v id="$1" -v branch="$2" '$1==id && $2==branch{found=1} END{exit !found}'
  fi
}

apply_installs(){
  echo "== Installing =="
  awk -v FS="$FS" '$0!~/^#/ && NF>=4{printf "  manifest: origin=%s id=%s branch=%s scope=%s\n",$1,$2,$3,$4}' "$MAN" || true

  awk -v FS="$FS" -v OFS="$tab" '$0!~/^#/ && NF>=4{print $1,$2,$3,$4}' "$MAN" |
  while read -r origin id branch scope; do
    [ -n "$origin$id$branch$scope" ] || continue
    already_installed "$id" "$branch" "$scope" && continue
    echo "  -> installing $id ($scope) from $origin:$branch"
    args=(install --noninteractive); [ "$scope" = user ] && args+=(--user) || args+=(--system)
    run_fp "$scope" "${args[@]}" "$origin" "${id}//${branch}"
  done
}

update_manifested(){
  [ "$UPDATE" = 1 ] || return 0
  echo "== Updating =="
  mapfile -t IDS_USER < <(awk -v FS="$FS" '$0!~/^#/ && NF>=4 && $4=="user"{print $2}' "$MAN" | sort -u)
  if [ "${#IDS_USER[@]}" -gt 0 ]; then
    "${CLEAN_ENV[@]}" flatpak --user update -y "${IDS_USER[@]}" || true
  fi

  mapfile -t IDS_SYS  < <(awk -v FS="$FS" '$0!~/^#/ && NF>=4 && $4=="system"{print $2}' "$MAN" | sort -u)
  if [ "${#IDS_SYS[@]}" -gt 0 ]; then
    run_fp system update -y "${IDS_SYS[@]}" || true
  fi

  if [ -x "$REL_SYNC_HELPER" ]; then
    FORCE_SUDO_SYSTEM="$FORCE_SUDO_SYSTEM" "$REL_SYNC_HELPER"
  fi
}

compute_extras(){ # installed applications whose ID is absent from all manifests
  local ids i; ids="$(mktemp)"; i="$(mktemp)"; trap 'rm -f "$ids" "$i"' RETURN
  desired_app_ids >"$ids"; installed_apps >"$i"
  while read -r id scope; do
    grep -Fxq "$id" "$ids" || printf '%s\t%s\n' "$id" "$scope"
  done <"$i"
}

apply_uninstalls(){
  [ "$ENFORCE" = 1 ] && [ "$UNINSTALL" = 1 ] || return 0
  echo "== Uninstalling extras =="
  compute_extras | while read -r id scope; do
    [ -n "${id:-}" ] || continue
    echo "  -> uninstalling $id ($scope)"
    if [ "$scope" = user ]; then
      "${CLEAN_ENV[@]}" flatpak --user uninstall -y "$id" || true
    else
      "${CLEAN_ENV[@]}" flatpak --system uninstall -y "$id" || true
    fi
  done
}

ensure_keepassxc_system_theme(){
  # KeePassXC's `classic` preference calls QApplication::setStyle("Fusion")
  # during startup and defeats QT_STYLE_OVERRIDE=kvantum.  Keep the mutable
  # per-app tree outside DotCortex, but converge this one allowlisted scalar
  # during the declarative Flatpak apply.  Guest homes are included so the
  # sanctuary runner and direct host launches share the same preference.
  local _home="$1"
  local ini="${_home}/.var/app/org.keepassxc.KeePassXC/config/keepassxc/keepassxc.ini"
  [ -f "$ini" ] || return 0

  local classic_count
  classic_count="$(grep -c '^ApplicationTheme=classic$' "$ini" || true)"
  case "$classic_count" in
    0) return 0 ;;
    1) ;;
    *)
      echo "  [keepassxc] refusing ambiguous ApplicationTheme in $ini" >&2
      return 1
      ;;
  esac

  local backup_dir="${_home}/.local/state/dotcortex/backups/keepassxc"
  local backup="${backup_dir}/keepassxc.ini.$(date +%Y%m%d-%H%M%S)"
  mkdir -p "$backup_dir"
  cp -p "$ini" "$backup"
  sed -i 's/^ApplicationTheme=classic$/ApplicationTheme=system/' "$ini"
  grep -qx 'ApplicationTheme=system' "$ini"
  echo "  [keepassxc] ApplicationTheme=system ($ini; backup=$backup)"
}

link_kvantum_theme_dirs(){
  # Flatpak overrides XDG_CONFIG_HOME to $HOME/.var/app/<id>/config/ inside
  # the sandbox.  Kvantum reads its active theme from $XDG_CONFIG_HOME/Kvantum/,
  # so without a bridge it ignores the stowed kvantum.kvconfig and falls back to
  # its built-in default.  Symlink each Qt Flatpak's per-app Kvantum dir to the
  # shared stowed config so the theme selection and theme files are always in sync.
  # Runs in the operator-home context; also covers sanctuary guest homes whose
  # Flatpak data lives under $HOME/.local/share/dotcortex/guests/*/home/.var/app/.
  local id
  for id in org.keepassxc.KeePassXC; do
    local _home
    for _home in "$HOME" "$HOME"/.local/share/dotcortex/guests/*/home; do
      [ -d "${_home}/.var/app/${id}" ] || continue
      ensure_keepassxc_system_theme "${_home}"
      local kvantum_src="${_home}/.config/Kvantum"
      [ -e "$kvantum_src" ] || continue
      local target="${_home}/.var/app/${id}/config/Kvantum"
      if [ -L "$target" ] && [ "$(readlink "$target")" = "$kvantum_src" ]; then
        continue
      fi
      if [ -e "$target" ] && [ ! -L "$target" ]; then
        echo "  [kvantum-link] $target is a real directory; skipping" >&2
        continue
      fi
      [ -L "$target" ] && unlink "$target"
      ln -s "$kvantum_src" "$target"
      echo "  [kvantum-link] $target -> $kvantum_src"
    done
  done
}

ensure_remotes
print_plan
apply_installs
update_manifested
if [ -x "$LOCAL_SYNC_HELPER" ]; then
  "${CLEAN_ENV[@]}" "$LOCAL_SYNC_HELPER"
fi
apply_uninstalls
link_kvantum_theme_dirs

if [ -x "$OBSIDIAN_HOST_URI_HELPER" ]; then
  "${CLEAN_ENV[@]}" "$OBSIDIAN_HOST_URI_HELPER"
fi
echo "Done."

[Context]

** Apply script (maps the common fields → flatpak override)
#+BEGIN_SRC bash :tangle linux/.local/bin/flatpak-perms-apply :shebang "#!/usr/bin/env bash" :mkdirp yes
set -euo pipefail
BASE="${HOME}/DotCortex/linux/.flatpak/overrides-perms"

CLEAN_ENV=(env -i HOME="$HOME" PATH="/usr/sbin:/usr/bin:/bin"
  LANG="${LANG:-C.UTF-8}" LC_ALL="${LC_ALL:-C.UTF-8}" TERM="${TERM:-xterm-256color}"
  XDG_RUNTIME_DIR="${XDG_RUNTIME_DIR:-/run/user/$(id -u)}"
  DBUS_SESSION_BUS_ADDRESS="${DBUS_SESSION_BUS_ADDRESS:-}")

apply_app(){
  local app="$1" f="$BASE/$1.perms"
  [ -f "$f" ] || { echo "skip $app (no $f)"; return 0; }

  # Build override flags from sections
  awk -v app="$app" '
    function splitcsv(s, arr){ gsub(/ /,"",s); n=split(s,arr,/[,;]/); return n }
    BEGIN{ sec=""; }
    /^\[/ { gsub(/[\[\]]/,""); sec=$0; next }
    /^[[:space:]]*$/ { next }
    { line=$0 }
    sec=="Context" {
      if (line ~ /^sockets=/) {
        sub(/^sockets=/,"",line); n=splitcsv(line,a); for(i=1;i<=n;i++) if(a[i]!="") printf "--socket=%s\n", a[i];
      } else if (line ~ /^devices=/) {
        sub(/^devices=/,"",line); n=splitcsv(line,a); for(i=1;i<=n;i++) if(a[i]!="") printf "--device=%s\n", a[i];
      } else if (line ~ /^filesystems=/) {
        sub(/^filesystems=/,"",line); n=splitcsv(line,a); for(i=1;i<=n;i++) if(a[i]!="") { if(a[i] ~ /^!/) printf "--nofilesystem=%s\n", substr(a[i],2); else printf "--filesystem=%s\n", a[i]; }
      } else if (line ~ /^persistent=/) {
        sub(/^persistent=/,"",line); n=splitcsv(line,a); for(i=1;i<=n;i++) if(a[i]!="") printf "--persist=%s\n", a[i];
      } else if (line ~ /^unset-environment=/) {
        sub(/^unset-environment=/,"",line); n=splitcsv(line,a); for(i=1;i<=n;i++) if(a[i]!="") printf "--unset-env=%s\n", a[i];
      } else if (line ~ /^features=/) {
        sub(/^features=/,"",line); n=splitcsv(line,a); for(i=1;i<=n;i++) if(a[i]!="") printf "--allow=%s\n", a[i];
      } else if (line ~ /^shared=/) {
        sub(/^shared=/,"",line); n=splitcsv(line,a); for(i=1;i<=n;i++) if(a[i]!="") printf "--share=%s\n", a[i];
      } else if (line ~ /^system-bus=/) {
        sub(/^system-bus=/,"",line); n=splitcsv(line,a); for(i=1;i<=n;i++) if(a[i]!="") printf "--system-talk-name=%s\n", a[i];
      } else if (line ~ /^session-bus=/) {
        sub(/^session-bus=/,"",line); n=splitcsv(line,a); for(i=1;i<=n;i++) if(a[i]!="") printf "--talk-name=%s\n", a[i];
      } else if (line ~ /^env=/) {
        sub(/^env=/,"",line); n=split(line,a,/[;,]/);
        for(i=1;i<=n;i++){ split(a[i],kv,/=/); if(kv[1]!="") printf "--env=%s=%s\n", kv[1], kv[2]; }
      }
    }
    sec=="Session Bus Policy" {
      split(line, kv, /=/)
      if (kv[1] != "" && kv[2] == "talk") printf "--talk-name=%s\n", kv[1];
    }
    sec=="System Bus Policy" {
      split(line, kv, /=/)
      if (kv[1] != "" && kv[2] == "talk") printf "--system-talk-name=%s\n", kv[1];
    }
    sec=="Environment" {
      split(line, kv, /=/)
      if (kv[1] != "") printf "--env=%s=%s\n", kv[1], substr(line, length(kv[1]) + 2);
    }
  ' "$f" | sed '/^$/d' | while IFS= read -r flag; do
    case "$flag" in
      --filesystem=/usr/share/fonts|--filesystem=/usr/share/fonts:*|\
      --filesystem=/usr/local/share/fonts|--filesystem=/usr/local/share/fonts:*|\
      --filesystem=/etc/fonts|--filesystem=/etc/fonts:*)
        printf 'skip reserved filesystem override %s for %s\n' "${flag#--filesystem=}" "$app" >&2
        continue
        ;;
    esac
    printf '%s\n' "$flag"
  done > "/tmp/.fpflags.$app"

  # FireDragon is crossing from the old X230 launcher to the T1700 profile.
  # Its historical override accumulated host-etc, reserved font paths, and
  # the old X230 profile mount.  Reset only this app's override so the
  # declared perms block below becomes authoritative again; other Flatpak
  # applications and all profile data remain untouched.
  if [[ "$app" = "org.garudalinux.firedragon" ]]; then
    "${CLEAN_ENV[@]}" flatpak override --user --reset "$app"
  fi

  # Apply at user scope
  if [ -s "/tmp/.fpflags.$app" ]; then
    mapfile -t FLAGS < "/tmp/.fpflags.$app"
    "${CLEAN_ENV[@]}" flatpak override --user "${FLAGS[@]}" "$app"
    echo "applied overrides → $app"
  else
    echo "no overrides to apply for $app"
  fi
  rm -f "/tmp/.fpflags.$app"
}

export -f apply_app
for p in "$BASE"/*.perms; do
  [ -e "$p" ] || { echo "no *.perms found in $BASE"; exit 0; }
  app="$(basename "$p" .perms)"
  apply_app "$app"
done

** Obsidian Windows 98 theme

The Redstone Obsidian theme is deployed from a pinned HelmCortex NEXUS mirror,
not copied into DotCortex's public overlays and not left as mutable GUI-only
state.  The Redstone launcher supplies its exact guest vault, verifies the
pinned CSS payload, installs the modern Obsidian theme wrapper, and selects it
only there.  General Flatpak lifecycle operations never apply themes, and the
helper refuses to infer HelmCortex or any other vault.

To review an upstream change without applying it:

#+BEGIN_SRC sh
obsidian-win98-theme-apply --check-upstream
#+END_SRC

If the candidate is wanted, update =PINNED_COMMIT= and =EXPECTED_SHA256= below,
tangle this module, then launch Redstone or invoke the helper with an explicit
=OBSIDIAN_VAULT=.  That makes theme upgrades a reviewed declarative pin change
rather than silent network drift.

* Moonlight DS4 right-stick deadzone
Apply a kernel-level deadzone to the hall-effect modded DS4 right stick
(axes 3=ABS_RX, 4=ABS_RY) so Moonlight does not see drift at rest.

Moonlight uses SDL2 for gamepad input. The kernel flatness alone is not
enough — SDL must also be told to honour it via
=SDL_LINUX_JOYSTICK_DEADZONES=1=. That override is applied once by
=flatpak override= (see below) and persists across launches.

#+BEGIN_SRC bash :tangle linux/.local/bin/ds4-right-deadzone :shebang "#!/usr/bin/env bash" :mkdirp yes
# ~/.local/bin/ds4-right-deadzone
#!/usr/bin/env bash
set -euo pipefail

# Hall-effect modded DS4, conservative right-stick deadzone.
# 35 covers observed drift (~10 counts RX, ~16 counts RY) with headroom.
DEADZONE="${DS4_DEADZONE:-35}"
WAIT_SECS=20

matches_target_name() {
  local name="$1"
  [[ "$name" == "Sony Interactive Entertainment Wireless Controller" ]] \
    || [[ "$name" == "Wireless Controller" ]]
}

find_controller_event() {
  local evsys name_file name dev
  for evsys in /sys/class/input/event*; do
    [[ -e "$evsys" ]] || continue
    dev="/dev/input/${evsys##*/}"
    name_file="$evsys/device/name"
    [[ -r "$name_file" ]] || continue
    name="$(<"$name_file")"
    matches_target_name "$name" || continue
    if evdev-joystick --showcal "$dev" >/dev/null 2>&1; then
      printf '%s\n' "$dev"
      return 0
    fi
  done
  return 1
}

main() {
  command -v evdev-joystick >/dev/null 2>&1 || {
    echo "Missing required command: evdev-joystick (apt install joystick)" >&2
    exit 1
  }

  local dev=""
  for ((i=0; i<WAIT_SECS; i++)); do
    if dev="$(find_controller_event)"; then
      break
    fi
    sleep 1
  done

  if [[ -z "$dev" ]]; then
    echo "DS4 controller not found after ${WAIT_SECS}s" >&2
    exit 1
  fi

  # ABS_RX = axis 3, ABS_RY = axis 4
  evdev-joystick --evdev "$dev" --axis 3 --deadzone "$DEADZONE"
  evdev-joystick --evdev "$dev" --axis 4 --deadzone "$DEADZONE"

  echo "Applied right-stick deadzone ${DEADZONE} to ${dev}"
  evdev-joystick --showcal "$dev"
}

main "$@"

* DS4 audio-output hijack suppression (fleet-wide)
The DualShock 4 exposes a USB audio interface that audio servers keep
auto-selecting as the default output sink, forcing a manual "disable the
controller" step on every reconnect. This udev rule makes the controller's
sound card invisible to both PulseAudio (=PULSE_IGNORE=) and PipeWire
(=ACP_IGNORE=) across the whole Linux fleet. Gamepad input (HID / =js0= /
motion / touchpad) is a separate kernel subsystem and is unaffected.

Covers both DS4 hardware revisions: =05c4= (CUH-ZCT1) and =09cc= (CUH-ZCT2).

Root rule, so it ships as an installer (same pattern as =nfs-*-setup=), not a
stowed file. Deploy per Linux machine:
#+BEGIN_SRC sh
tangle-one package-flatpak.org      # then stow the linux overlay on the machine
ds4-no-audio-install        # uses sudo internally; idempotent
#+END_SRC

** Managed udev rule
#+BEGIN_SRC conf :tangle linux/.local/share/udev/rules.d/62-ds4-no-audio.rules :mkdirp yes :comments no
# Managed by DotCortex via package-flatpak.org.
# DualShock 4 exposes a USB audio interface that audio servers keep selecting as
# the default output sink. Make both PulseAudio (PULSE_IGNORE) and PipeWire
# (ACP_IGNORE) ignore the controller's sound card. Gamepad input (HID/js) is a
# separate subsystem and is unaffected. Covers DS4 v1 (05c4) and v2 (09cc).
SUBSYSTEM=="sound", ATTRS{idVendor}=="054c", ATTRS{idProduct}=="05c4", ENV{PULSE_IGNORE}="1", ENV{ACP_IGNORE}="1"
SUBSYSTEM=="sound", ATTRS{idVendor}=="054c", ATTRS{idProduct}=="09cc", ENV{PULSE_IGNORE}="1", ENV{ACP_IGNORE}="1"
#+END_SRC

** Installer
Copies the managed rule into =/etc/udev/rules.d/=, reloads udev, and drops any
already-loaded DS4 card from the live audio server so the fix applies without a
replug. Idempotent and safe to re-run on every machine; no-ops cleanly on
systems without udev (e.g. Termux).
#+BEGIN_SRC bash :tangle linux/.local/bin/ds4-no-audio-install :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755) :mkdirp yes :comments no
set -euo pipefail

RULE_NAME="62-ds4-no-audio.rules"
OWNER_USER="${SUDO_USER:-${USER:-$(id -un)}}"
OWNER_HOME="$(getent passwd "$OWNER_USER" | cut -d: -f6)"
RULE_SRC="$OWNER_HOME/.local/share/udev/rules.d/$RULE_NAME"
RULE_DEST="/etc/udev/rules.d/$RULE_NAME"
LEGACY_DEST="/etc/udev/rules.d/61-dualshock4-no-audio.rules"   # ad-hoc T480 file, superseded

run_root() { if [ "$(id -u)" -eq 0 ]; then "$@"; else sudo "$@"; fi; }

command -v udevadm >/dev/null 2>&1 || { echo "no udevadm — not a udev system; nothing to do"; exit 0; }
[ -r "$RULE_SRC" ] || { echo "Missing $RULE_SRC — tangle + stow DotCortex first" >&2; exit 1; }

run_root install -d -m 0755 /etc/udev/rules.d
run_root install -m 0644 "$RULE_SRC" "$RULE_DEST"
echo "Installed $RULE_DEST"

if [ -e "$LEGACY_DEST" ]; then
  run_root rm -f "$LEGACY_DEST"
  echo "Removed superseded $LEGACY_DEST"
fi

run_root udevadm control --reload-rules

# Drop an already-connected DS4 from the live audio server so no replug is
# needed (PulseAudio: unload the card's owning module). We deliberately do NOT
# blanket re-trigger the sound subsystem (udevadm trigger --subsystem-match=sound):
# that fires synthetic "change" events at every card and silences running audio
# apps such as Moonlight/SDL, which hold a stream open but stop feeding it. The
# rule applies to the DS4 on its next plug event regardless.
if command -v pactl >/dev/null 2>&1; then
  owner="$(pactl list cards 2>/dev/null | awk '/Name: alsa_card.usb-Sony/{f=1} f&&/Owner Module:/{print $3; exit}')" || owner=""
  if [ -n "${owner:-}" ]; then
    if pactl unload-module "$owner" 2>/dev/null; then
      echo "Dropped live DS4 card (module $owner)"
    fi
  fi
fi

echo "DS4 audio suppression active. If the controller is connected, replug it to be sure."
#+END_SRC
