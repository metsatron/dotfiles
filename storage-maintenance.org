* Storage maintenance

This module owns a deliberately small cleanup allowlist.  It collects
unreferenced Guix store items; clears Nala's downloaded-package archive;
removes unused, unpinned Flatpak refs; prunes stale Homebrew downloads and
dangling uv cache entries; caps the known Redstone 9X =clipmenud.log= runaway;
and expires old entries from Ductor's regenerable candidate cache.  It does
not run dependency autoremoval, delete package-manager rollback generations,
or sweep generic caches, temporary directories, Flatpak objects, Borg state,
model files, agent sessions, or HelmCortex data.

The cleanup command is host-gated and singleton.  Dry-run mode reports the
same targets without changing them.  The cron installer is separate so
tangling never changes the live scheduler.

** Cleanup command

#+BEGIN_SRC bash :tangle all/.local/bin/dotcortex-storage-maintenance :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

TAG="dotcortex-storage-maintenance"
DRY_RUN=0
EXPECTED_HOST=""
CANDIDATE_MAX_AGE_DAYS="${STORAGE_MAINTENANCE_CANDIDATE_DAYS:-14}"
CLIPMENU_MAX_BYTES="${STORAGE_MAINTENANCE_CLIPMENU_MAX_BYTES:-16777216}"
TELEGRAM_STATE_MAX_BYTES="${STORAGE_MAINTENANCE_TELEGRAM_STATE_MAX_BYTES:-268435456}"
CLAUDE_STATE_MAX_BYTES="${STORAGE_MAINTENANCE_CLAUDE_STATE_MAX_BYTES:-268435456}"
CADDY_LOG_MAX_BYTES="${STORAGE_MAINTENANCE_CADDY_LOG_MAX_BYTES:-134217728}"

usage() {
  cat <<'EOF'
Usage: dotcortex-storage-maintenance --expected-host EXACT_HOST [--dry-run]

Run the narrow DotCortex cleanup allowlist:
  - collect unreferenced Guix store items
  - clear Nala's downloaded-package archive
  - remove unused, unpinned user Flatpak refs
  - prune Homebrew downloads older than 30 days
  - prune dangling uv cache entries
  - truncate Redstone 9X clipmenud.log files larger than 16 MiB
  - remove Ductor candidate-cache entries older than 14 days

No dependency autoremove, package-generation deletion, generic cache,
temporary-directory, Flatpak object surgery, Borg, model, agent-session, or
HelmCortex cleanup is performed.
EOF
}

while [ "$#" -gt 0 ]; do
  case "$1" in
    --dry-run) DRY_RUN=1 ;;
    --expected-host)
      [ "$#" -ge 2 ] || { printf '%s: --expected-host requires a value\n' "$TAG" >&2; exit 2; }
      EXPECTED_HOST="$2"
      shift ;;
    -h|--help|help) usage; exit 0 ;;
    *) printf '%s: unknown argument: %s\n' "$TAG" "$1" >&2; usage >&2; exit 2 ;;
  esac
  shift
done

[ -n "$EXPECTED_HOST" ] || { printf '%s: --expected-host is required\n' "$TAG" >&2; exit 2; }
CURRENT_HOST="${STORAGE_MAINTENANCE_HOSTNAME:-$(hostname)}"
[ "$CURRENT_HOST" = "$EXPECTED_HOST" ] || {
  printf '%s: refusing on %s; expected %s\n' "$TAG" "$CURRENT_HOST" "$EXPECTED_HOST" >&2
  exit 1
}

MAINTENANCE_HOME="${STORAGE_MAINTENANCE_HOME:-$HOME}"
STATE_DIR="${STORAGE_MAINTENANCE_STATE_DIR:-$MAINTENANCE_HOME/.local/state/dotcortex-storage-maintenance}"
LOCK_FILE="$STATE_DIR/lock"
CANDIDATE_DIR="$MAINTENANCE_HOME/.cache/ductor-candidates"
GUIX_BIN="${STORAGE_MAINTENANCE_GUIX:-$MAINTENANCE_HOME/.guix-extra-profiles/core/core/bin/guix}"
NALA_BIN="${STORAGE_MAINTENANCE_NALA:-$(command -v nala || true)}"
FLATPAK_BIN="${STORAGE_MAINTENANCE_FLATPAK:-$(command -v flatpak || true)}"
UV_BIN="${STORAGE_MAINTENANCE_UV:-$(command -v uv || true)}"
BREW_BIN="${STORAGE_MAINTENANCE_BREW:-$(command -v brew || true)}"
SUDO_BIN="${STORAGE_MAINTENANCE_SUDO:-$(command -v sudo || true)}"
APT_ARCHIVE_DIR="${STORAGE_MAINTENANCE_APT_ARCHIVE_DIR:-/var/cache/apt/archives}"

if [ -z "$BREW_BIN" ] && [ -x /home/linuxbrew/.linuxbrew/bin/brew ]; then
  BREW_BIN=/home/linuxbrew/.linuxbrew/bin/brew
fi

case "$CANDIDATE_MAX_AGE_DAYS:$CLIPMENU_MAX_BYTES:$TELEGRAM_STATE_MAX_BYTES:$CLAUDE_STATE_MAX_BYTES:$CADDY_LOG_MAX_BYTES" in
  *[!0-9:]*|:*|*:) printf '%s: cleanup limits must be non-negative integers\n' "$TAG" >&2; exit 2 ;;
esac

if [ ! -x "$GUIX_BIN" ]; then
  GUIX_BIN="$(command -v guix || true)"
fi
[ -n "$GUIX_BIN" ] && [ -x "$GUIX_BIN" ] || {
  printf '%s: Guix executable not found\n' "$TAG" >&2
  exit 1
}
command -v flock >/dev/null 2>&1 || { printf '%s: flock is required\n' "$TAG" >&2; exit 1; }

mkdir -p "$STATE_DIR"
exec 9>"$LOCK_FILE"
flock -n 9 || { printf '%s: another cleanup is already running\n' "$TAG" >&2; exit 1; }

disk_line() {
  df -PB1 "$MAINTENANCE_HOME" | awk 'NR == 2 { printf "used=%s available=%s capacity=%s", $3, $4, $5 }'
}

run_root() {
  if [ "$(id -u)" = 0 ]; then
    "$@"
  else
    [ -n "$SUDO_BIN" ] && [ -x "$SUDO_BIN" ] || {
      printf '%s: sudo is required for %s\n' "$TAG" "$1" >&2
      return 1
    }
    "$SUDO_BIN" -n "$@"
  fi
}

optional_executable() {
  [ -n "$1" ] && [ -x "$1" ]
}

estimate_dead_bytes() {
  local path total=0
  while IFS= read -r path; do
    [ -e "$path" ] || continue
    total=$((total + $(du -sB1 "$path" 2>/dev/null | awk '{print $1}')))
  done < <("$GUIX_BIN" gc --list-dead)
  printf '%s' "$total"
}

report_tree_high_water() {
  local label="$1" path="$2" maximum="$3" bytes=0
  [ -e "$path" ] || return 0
  bytes="$(du -sB1 "$path" 2>/dev/null | awk '{print $1}')" || true
  bytes="${bytes:-0}"
  if [ "$bytes" -gt "$maximum" ]; then
    printf 'WARNING runtime-log high-water exceeded: label=%s bytes=%s maximum=%s path=%s\n' \
      "$label" "$bytes" "$maximum" "$path"
  else
    printf 'runtime-log high-water okay: label=%s bytes=%s maximum=%s\n' "$label" "$bytes" "$maximum"
  fi
}

report_file_family_high_water() {
  local label="$1" directory="$2" basename="$3" maximum="$4" candidate bytes=0 size
  [ -d "$directory" ] || return 0
  while IFS= read -r -d '' candidate; do
    size="$(stat -c '%s' "$candidate")"
    bytes=$((bytes + size))
  done < <(find "$directory" -maxdepth 1 -type f -name "$basename*" -print0)
  if [ "$bytes" -gt "$maximum" ]; then
    printf 'WARNING runtime-log high-water exceeded: label=%s bytes=%s maximum=%s path=%s/%s*\n' \
      "$label" "$bytes" "$maximum" "$directory" "$basename"
  else
    printf 'runtime-log high-water okay: label=%s bytes=%s maximum=%s\n' "$label" "$bytes" "$maximum"
  fi
}

printf '%s: start host=%s dry_run=%s %s\n' "$TAG" "$CURRENT_HOST" "$DRY_RUN" "$(disk_line)"

if [ "$DRY_RUN" = 1 ]; then
  DEAD_COUNT="$("$GUIX_BIN" gc --list-dead | awk 'NF { count++ } END { print count + 0 }')"
  DEAD_BYTES="$(estimate_dead_bytes)"
  printf 'would collect Guix garbage: paths=%s estimated_bytes=%s\n' "$DEAD_COUNT" "$DEAD_BYTES"
else
  printf 'collecting Guix garbage\n'
  GC_OUTPUT="$(mktemp "$STATE_DIR/guix-gc.XXXXXX")"
  if "$GUIX_BIN" gc >"$GC_OUTPUT" 2>&1; then
    tail -n 3 "$GC_OUTPUT"
    rm -f "$GC_OUTPUT"
  else
    cat "$GC_OUTPUT" >&2
    rm -f "$GC_OUTPUT"
    printf '%s: Guix garbage collection failed\n' "$TAG" >&2
    exit 1
  fi
fi

shopt -s nullglob
for logfile in "$MAINTENANCE_HOME"/.local/share/dotcortex/guests/*/home/.cache/redstone-9x/clipmenud.log; do
  [ -f "$logfile" ] || continue
  size="$(stat -c '%s' "$logfile")"
  [ "$size" -gt "$CLIPMENU_MAX_BYTES" ] || continue
  if [ "$DRY_RUN" = 1 ]; then
    printf 'would truncate oversized Redstone log: bytes=%s path=%s\n' "$size" "$logfile"
  else
    : >"$logfile"
    printf 'truncated oversized Redstone log: prior_bytes=%s path=%s\n' "$size" "$logfile"
  fi
done

if [ -d "$CANDIDATE_DIR" ]; then
  while IFS= read -r -d '' candidate; do
    case "$candidate" in
      "$CANDIDATE_DIR"/*) ;;
      *) printf '%s: refusing unexpected candidate path: %s\n' "$TAG" "$candidate" >&2; exit 1 ;;
    esac
    if [ "$DRY_RUN" = 1 ]; then
      printf 'would remove expired Ductor candidate cache entry: %s\n' "$candidate"
    else
      rm -rf -- "$candidate"
      printf 'removed expired Ductor candidate cache entry: %s\n' "$candidate"
    fi
  done < <(find "$CANDIDATE_DIR" -mindepth 1 -maxdepth 1 -mtime "+$CANDIDATE_MAX_AGE_DAYS" -print0)
fi

PACKAGE_FAILURES=0

if optional_executable "$NALA_BIN"; then
  if [ "$DRY_RUN" = 1 ]; then
    archive_bytes=0
    if [ -d "$APT_ARCHIVE_DIR" ]; then
      archive_bytes="$(du -sB1 "$APT_ARCHIVE_DIR" 2>/dev/null | awk '{print $1}')" || true
      archive_bytes="${archive_bytes:-0}"
    fi
    printf 'would clear Nala/APT downloaded-package archive: bytes=%s\n' "$archive_bytes"
  elif run_root "$NALA_BIN" clean; then
    printf 'cleaned Nala/APT downloaded-package archive\n'
  else
    printf '%s: Nala archive cleanup failed\n' "$TAG" >&2
    PACKAGE_FAILURES=$((PACKAGE_FAILURES + 1))
  fi
else
  printf 'skipped Nala archive cleanup: executable unavailable\n'
fi

if optional_executable "$FLATPAK_BIN"; then
  if [ "$DRY_RUN" = 1 ]; then
    printf 'would remove unused, unpinned user Flatpak refs\n'
  elif "$FLATPAK_BIN" uninstall --user --unused --assumeyes --noninteractive; then
    printf 'removed unused, unpinned user Flatpak refs\n'
  else
    printf '%s: Flatpak unused-ref cleanup failed\n' "$TAG" >&2
    PACKAGE_FAILURES=$((PACKAGE_FAILURES + 1))
  fi
else
  printf 'skipped Flatpak unused-ref cleanup: executable unavailable\n'
fi

if optional_executable "$BREW_BIN"; then
  if [ "$DRY_RUN" = 1 ]; then
    "$BREW_BIN" cleanup --dry-run --prune=30
  elif "$BREW_BIN" cleanup --prune=30; then
    printf 'pruned Homebrew downloads older than 30 days\n'
  else
    printf '%s: Homebrew cleanup failed\n' "$TAG" >&2
    PACKAGE_FAILURES=$((PACKAGE_FAILURES + 1))
  fi
else
  printf 'skipped Homebrew cleanup: executable unavailable\n'
fi

if optional_executable "$UV_BIN"; then
  if [ "$DRY_RUN" = 1 ]; then
    printf 'would prune dangling uv cache entries\n'
  elif "$UV_BIN" cache prune; then
    printf 'pruned dangling uv cache entries\n'
  else
    printf '%s: uv cache cleanup failed\n' "$TAG" >&2
    PACKAGE_FAILURES=$((PACKAGE_FAILURES + 1))
  fi
else
  printf 'skipped uv cache cleanup: executable unavailable\n'
fi

# A root-owned host-private configuration is the explicit opt-in. No home discovery.
pip_cache_config="${STORAGE_MAINTENANCE_PIP_CACHE_CONFIG:-/etc/dotcortex/pip-cache-maintenance.json}"
if [ -e "$pip_cache_config" ] || [ -L "$pip_cache_config" ]; then
  script_dir="$(dirname "$(readlink -f "${BASH_SOURCE[0]}")")"
  pip_cache_args=(--config "$pip_cache_config")
  [ "$DRY_RUN" != 1 ] || pip_cache_args+=(--dry-run)
  if ! run_root /usr/bin/python3 "$script_dir/dotcortex-pip-cache-maintenance" "${pip_cache_args[@]}"; then
    printf '%s: opted-in pip cache maintenance refused or failed\n' "$TAG" >&2
    PACKAGE_FAILURES=$((PACKAGE_FAILURES + 1))
  fi
fi

report_tree_high_water \
  telegram-agents "$MAINTENANCE_HOME/.local/state/telegram-agents" "$TELEGRAM_STATE_MAX_BYTES"
report_tree_high_water \
  claude-idle-compaction "$MAINTENANCE_HOME/.local/state/claude-idle-compaction" "$CLAUDE_STATE_MAX_BYTES"
report_file_family_high_water \
  caddy-refresh "$MAINTENANCE_HOME/.local/state" caddy-refresh.log "$CADDY_LOG_MAX_BYTES"

printf '%s: finish %s\n' "$TAG" "$(disk_line)"
[ "$PACKAGE_FAILURES" -eq 0 ] || exit 1
#+END_SRC

** Daily Kikin schedule

The Kikin contract runs at 06:30, after the overnight Borg window and before
the morning fleet report.  The installer derives the account identity at
apply time and requires an exact hostname, keeping the public source free of
machine-local identity values.

#+BEGIN_SRC bash :tangle linux/.local/bin/dotcortex-storage-maintenance-cron-apply :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

TAG="dotcortex-storage-maintenance-cron-apply"
DEFAULT_TARGET="/etc/cron.d/dotcortex-storage-maintenance"
TARGET="${STORAGE_MAINTENANCE_CRON_TARGET:-$DEFAULT_TARGET}"
DRY_RUN=0
CHECK=0
REMOVE=0
EXPECTED_HOST=""

usage() {
  cat <<'EOF'
Usage: dotcortex-storage-maintenance-cron-apply --host EXACT_HOST [--dry-run|--check|--remove]

Install, verify, preview, or remove the host-gated 06:30 daily cleanup job.
EOF
}

while [ "$#" -gt 0 ]; do
  case "$1" in
    --host)
      [ "$#" -ge 2 ] || { printf '%s: --host requires a value\n' "$TAG" >&2; exit 2; }
      EXPECTED_HOST="$2"
      shift ;;
    --dry-run) DRY_RUN=1 ;;
    --check) CHECK=1 ;;
    --remove) REMOVE=1 ;;
    -h|--help|help) usage; exit 0 ;;
    *) printf '%s: unknown argument: %s\n' "$TAG" "$1" >&2; usage >&2; exit 2 ;;
  esac
  shift
done

[ "$((DRY_RUN + CHECK + REMOVE))" -le 1 ] || {
  printf '%s: choose only one of --dry-run, --check, or --remove\n' "$TAG" >&2
  exit 2
}

run_priv() { if [ "$(id -u)" = 0 ]; then "$@"; else sudo -n "$@"; fi; }

if [ "$REMOVE" = 1 ]; then
  if [ "$TARGET" = "$DEFAULT_TARGET" ]; then
    run_priv rm -f "$TARGET"
  else
    rm -f "$TARGET"
  fi
  printf '%s: removed %s\n' "$TAG" "$TARGET"
  exit 0
fi

[ -n "$EXPECTED_HOST" ] || { printf '%s: --host EXACT_HOST is required\n' "$TAG" >&2; exit 2; }
CURRENT_HOST="${STORAGE_MAINTENANCE_HOSTNAME:-$(hostname)}"
[ "$CURRENT_HOST" = "$EXPECTED_HOST" ] || {
  printf '%s: refusing installation on %s; expected %s\n' "$TAG" "$CURRENT_HOST" "$EXPECTED_HOST" >&2
  exit 1
}

TARGET_HOME="${STORAGE_MAINTENANCE_HOME:-$HOME}"
USER_NAME="${STORAGE_MAINTENANCE_USER:-$(id -un)}"
USER_ID="${STORAGE_MAINTENANCE_UID:-$(id -u)}"
CLEANUP="$TARGET_HOME/.local/bin/dotcortex-storage-maintenance"
STATE_DIR="$TARGET_HOME/.local/state/dotcortex-storage-maintenance"
CORE_BIN="$TARGET_HOME/.guix-extra-profiles/core/core/bin"
CRON_PATH="$CORE_BIN:$TARGET_HOME/.local/bin:/usr/local/bin:/usr/bin:/bin"

case "$TARGET_HOME:$USER_NAME:$USER_ID:$EXPECTED_HOST" in
  *[[:space:]]*) printf '%s: whitespace is not valid in cron fields\n' "$TAG" >&2; exit 2 ;;
esac
[ "$DRY_RUN" = 1 ] || [ -x "$CLEANUP" ] || {
  printf '%s: missing %s; tangle and stow first\n' "$TAG" "$CLEANUP" >&2
  exit 1
}

TMP="$(mktemp)"
trap 'rm -f "$TMP"' EXIT
cat >"$TMP" <<EOF
# DotCortex deterministic Kikin storage maintenance; managed by $TAG.
SHELL=/bin/sh
30 6 * * * $USER_NAME HOME=$TARGET_HOME PATH=$CRON_PATH XDG_RUNTIME_DIR=/run/user/$USER_ID $CLEANUP --expected-host $EXPECTED_HOST >>$STATE_DIR/cron.log 2>&1
EOF

if [ "$DRY_RUN" = 1 ]; then
  cat "$TMP"
  exit 0
fi
if [ "$CHECK" = 1 ]; then
  cmp -s "$TMP" "$TARGET" || {
    printf '%s: drift or missing target: %s\n' "$TAG" "$TARGET" >&2
    exit 1
  }
  printf '%s: installed contract matches %s\n' "$TAG" "$TARGET"
  exit 0
fi

mkdir -p "$STATE_DIR"
if [ "$TARGET" = "$DEFAULT_TARGET" ]; then
  run_priv install -o root -g root -m 0644 "$TMP" "$TARGET"
else
  install -m 0644 "$TMP" "$TARGET"
fi
printf '%s: installed %s\n' "$TAG" "$TARGET"
#+END_SRC

** Contract tests

#+BEGIN_SRC python :tangle linux/.local/share/dotcortex/tests/test-storage-maintenance.py :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
from __future__ import annotations

import os
import subprocess
import tempfile
import time
import unittest
from pathlib import Path


ROOT = Path(__file__).resolve().parents[5]
CLEANUP = ROOT / "all/.local/bin/dotcortex-storage-maintenance"
INSTALLER = ROOT / "linux/.local/bin/dotcortex-storage-maintenance-cron-apply"


class StorageMaintenanceTests(unittest.TestCase):
    def fixture(self, root: Path) -> tuple[dict[str, str], Path, Path, Path, Path]:
        fake_guix = root / "guix"
        dead = root / "dead-store-item"
        dead.mkdir()
        (dead / "payload").write_bytes(b"x" * 4096)
        marker = root / "guix-gc-ran"
        fake_guix.write_text(
            "#!/bin/sh\n"
            f"if [ \"$*\" = 'gc --list-dead' ]; then printf '%s\\n' '{dead}'; exit 0; fi\n"
            f"if [ \"$*\" = 'gc' ]; then : >'{marker}'; "
            "printf '%s\\n' detail-1 detail-2 detail-3 summary-1 summary-2; exit 0; fi\n"
            "exit 2\n",
            encoding="utf-8",
        )
        fake_guix.chmod(0o755)

        package_calls = root / "package-calls"
        package_bins: dict[str, Path] = {}
        for name in ("nala", "flatpak", "brew", "uv"):
            executable = root / name
            executable.write_text(
                "#!/bin/sh\n"
                f"printf '%s %s\\n' '{name}' \"$*\" >>'{package_calls}'\n",
                encoding="utf-8",
            )
            executable.chmod(0o755)
            package_bins[name] = executable
        fake_sudo = root / "sudo"
        fake_sudo.write_text(
            "#!/bin/sh\n"
            "[ \"${1:-}\" = -n ] && shift\n"
            "exec \"$@\"\n",
            encoding="utf-8",
        )
        fake_sudo.chmod(0o755)

        runaway = root / ".local/share/dotcortex/guests/test/home/.cache/redstone-9x/clipmenud.log"
        runaway.parent.mkdir(parents=True)
        with runaway.open("wb") as handle:
            handle.truncate(17 * 1024 * 1024)

        candidates = root / ".cache/ductor-candidates"
        old = candidates / "old"
        fresh = candidates / "fresh"
        old.mkdir(parents=True)
        fresh.mkdir()
        old_time = time.time() - (20 * 86400)
        os.utime(old, (old_time, old_time))

        unrelated = root / ".cache/must-survive"
        unrelated.write_text("keeper", encoding="utf-8")
        telegram_state = root / ".local/state/telegram-agents"
        claude_state = root / ".local/state/claude-idle-compaction"
        telegram_state.mkdir(parents=True)
        claude_state.mkdir()
        (telegram_state / "agent.log").write_text("large", encoding="utf-8")
        (claude_state / "diagnostic.log").write_text("large", encoding="utf-8")
        (root / ".local/state/caddy-refresh.log").write_text("large", encoding="utf-8")
        env = os.environ.copy()
        env.update(
            {
                "STORAGE_MAINTENANCE_PIP_CACHE_CONFIG": str(root / "absent-pip-config.json"),
                "STORAGE_MAINTENANCE_HOME": str(root),
                "STORAGE_MAINTENANCE_HOSTNAME": "kikin-kushi",
                "STORAGE_MAINTENANCE_GUIX": str(fake_guix),
                "STORAGE_MAINTENANCE_NALA": str(package_bins["nala"]),
                "STORAGE_MAINTENANCE_FLATPAK": str(package_bins["flatpak"]),
                "STORAGE_MAINTENANCE_BREW": str(package_bins["brew"]),
                "STORAGE_MAINTENANCE_UV": str(package_bins["uv"]),
                "STORAGE_MAINTENANCE_SUDO": str(fake_sudo),
                "STORAGE_MAINTENANCE_APT_ARCHIVE_DIR": str(root / "apt-archives"),
                "STORAGE_MAINTENANCE_TELEGRAM_STATE_MAX_BYTES": "1",
                "STORAGE_MAINTENANCE_CLAUDE_STATE_MAX_BYTES": "1",
                "STORAGE_MAINTENANCE_CADDY_LOG_MAX_BYTES": "1",
            }
        )
        return env, marker, runaway, old, package_calls

    def test_help(self) -> None:
        result = subprocess.run([CLEANUP, "--help"], text=True, capture_output=True)
        self.assertEqual(result.returncode, 0, result.stderr)
        self.assertIn("narrow DotCortex cleanup allowlist", result.stdout)

    def test_host_guard(self) -> None:
        env = os.environ.copy()
        env["STORAGE_MAINTENANCE_HOSTNAME"] = "not-kikin"
        result = subprocess.run(
            [CLEANUP, "--expected-host", "kikin-kushi", "--dry-run"],
            env=env,
            text=True,
            capture_output=True,
        )
        self.assertNotEqual(result.returncode, 0)
        self.assertIn("refusing", result.stderr)

    def test_dry_run_is_non_mutating(self) -> None:
        with tempfile.TemporaryDirectory() as tmp:
            root = Path(tmp)
            env, marker, runaway, old, package_calls = self.fixture(root)
            result = subprocess.run(
                [CLEANUP, "--expected-host", "kikin-kushi", "--dry-run"],
                env=env,
                text=True,
                capture_output=True,
            )
            self.assertEqual(result.returncode, 0, result.stderr)
            self.assertFalse(marker.exists())
            self.assertEqual(runaway.stat().st_size, 17 * 1024 * 1024)
            self.assertTrue(old.exists())
            self.assertIn("would truncate", result.stdout)
            self.assertIn("would remove", result.stdout)
            self.assertEqual(result.stdout.count("WARNING runtime-log high-water exceeded"), 3)
            self.assertEqual(
                package_calls.read_text(encoding="utf-8").splitlines(),
                ["brew cleanup --dry-run --prune=30"],
            )

    def test_real_run_changes_only_allowlisted_targets(self) -> None:
        with tempfile.TemporaryDirectory() as tmp:
            root = Path(tmp)
            env, marker, runaway, old, package_calls = self.fixture(root)
            result = subprocess.run(
                [CLEANUP, "--expected-host", "kikin-kushi"],
                env=env,
                text=True,
                capture_output=True,
            )
            self.assertEqual(result.returncode, 0, result.stderr)
            self.assertTrue(marker.exists())
            self.assertEqual(runaway.stat().st_size, 0)
            self.assertFalse(old.exists())
            self.assertTrue((root / ".cache/ductor-candidates/fresh").exists())
            self.assertEqual((root / ".cache/must-survive").read_text(), "keeper")
            self.assertNotIn("detail-1", result.stdout)
            self.assertIn("summary-2", result.stdout)
            self.assertEqual(
                package_calls.read_text(encoding="utf-8").splitlines(),
                [
                    "nala clean",
                    "flatpak uninstall --user --unused --assumeyes --noninteractive",
                    "brew cleanup --prune=30",
                    "uv cache prune",
                ],
            )

    def test_cron_contract(self) -> None:
        with tempfile.TemporaryDirectory() as tmp:
            root = Path(tmp)
            env = os.environ.copy()
            env.update(

** Opt-in account pip download caches

The existing daily cleanup also invokes the sibling pip-cache helper when
=/etc/dotcortex/pip-cache-maintenance.json= exists. That file and its ancestry
must be root-owned, local, and not group/world writable. Without the file,
existing cleanup scope is unchanged. Root may run only the helper to preview or
apply this lane without invoking the other maintenance operations. The runner
accepts =STORAGE_MAINTENANCE_PIP_CACHE_CONFIG= for an alternate root-owned
configuration path; the helper applies identical custody checks to that path.

Create a private JSON file with actual machine/account values replacing the
placeholders, then install it as root with mode 0600 under a root-owned 0755
=/etc/dotcortex/= directory. Do not commit the populated file to DotCortex:

#+BEGIN_EXAMPLE
{"host":"EXACT_HOST","accounts":["ACCOUNT"],"max_bytes":1073741824,"quiet_hours":24}
sudo /usr/bin/python3 /PATH/TO/DotCortex/all/.local/bin/dotcortex-pip-cache-maintenance --dry-run
sudo /usr/bin/python3 /PATH/TO/DotCortex/all/.local/bin/dotcortex-pip-cache-maintenance
#+END_EXAMPLE

Only explicitly named non-root passwd accounts are selected. The helper drops
supplementary groups, GID and UID before account Python execution or cache work.
It refuses symlinked/nonlocal/wrong-owner roots, foreign cache entries, nested
mounts, and a pip-configured destination outside the exact account =~/.cache/pip=.
Unavailable pip is skipped; it is never installed. The 1 GiB default is a trigger,
not a hard quota: an oversized cache is purged only after every entry has been
quiet for 24 hours. Sizes and account names are logged, never process arguments.

The helper takes the canonical managed =~/.local/state/dotcortex/dependencies/user-site.lock=
and skips active raw pip install/download/wheel processes for that UID before and
after measurement. Accounts using a custom XDG state root must reconcile their
installer locking before opting in. Raw installers do not honor this lock; the
process checks are a conservative activity check, not a system-wide exclusion
mechanism. Native =pip cache purge= runs as the selected account. Installed code,
model caches and all other cache roots are outside this lane. Dry-run creates no
account directories or lock files and never calls purge. A busy lock or unsafe
state returns nonzero; the existing maintenance runner records the failure.

#!/usr/bin/env python3
"""Opt-in account pip download-cache maintenance; never removes installed packages."""
import argparse
from contextlib import contextmanager
import fcntl
import json
import os
from pathlib import Path
import pwd
import re
import shutil
import socket
import stat
import subprocess
import sys
import time

DEFAULT_CONFIG = Path('/etc/dotcortex/pip-cache-maintenance.json')


class Refused(RuntimeError):
    pass


def local(path):
    if not path.is_absolute() or len(path.parts) < 3 or '..' in path.parts:
        raise Refused('unsafe or redirected root')
    mounts = []
    for line in Path('/proc/self/mountinfo').read_text().splitlines():
        left, right = line.split(' - ', 1)
        mount = Path(left.split()[4].replace('\\040', ' '))
        if path == mount or mount in path.parents:
            mounts.append((len(mount.parts), right.split()[0]))
    if mounts and max(mounts)[1] in {'autofs', 'nfs', 'nfs4', 'cifs', 'sshfs', 'fuse.sshfs'}:
        raise Refused('nonlocal root')
    # Reject links in order before touching descendants or resolving a remote target.
    for ancestor in [*reversed(path.parents), path]:
        if ancestor.is_symlink():
            raise Refused('unsafe or redirected root')


def config_read(path):
    if os.geteuid() != 0:
        raise Refused('root is required for the explicit account allowlist')
    local(path)
    for parent in path.parents:
        info = parent.lstat()
        if not stat.S_ISDIR(info.st_mode) or info.st_uid != 0 or info.st_mode & 0o022:
            raise Refused('configuration ancestry must have root custody')
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW)
    stream = os.fdopen(fd)
    try:
        info = os.fstat(fd)
        if not stat.S_ISREG(info.st_mode) or info.st_uid != 0 or info.st_mode & 0o022:
            raise Refused('configuration must be root-owned and not group/world writable')
        fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
        config = json.load(stream)
        if (not isinstance(config, dict) or set(config) - {'host', 'accounts', 'max_bytes', 'quiet_hours'}
                or config.get('host') != socket.gethostname()):
            raise Refused('invalid configuration or host mismatch')
        accounts = config.get('accounts')
        if (not isinstance(accounts, list) or not accounts or len(set(accounts)) != len(accounts)
                or any(not isinstance(a, str) or not re.fullmatch(r'[a-z_][a-z0-9_-]*', a) for a in accounts)):
            raise Refused('declare unique explicit account names')
        for key, default in [('max_bytes', 1073741824), ('quiet_hours', 24)]:
            config.setdefault(key, default)
            if type(config[key]) is not int or config[key] <= 0:
                raise Refused('limits must be positive integers')
        return stream, config
    except BaseException:
        stream.close()
        raise


def owned_directory(path, uid):
    info = path.lstat()
    if not stat.S_ISDIR(info.st_mode) or info.st_uid != uid or info.st_mode & 0o022:
        raise Refused('unsafe directory ownership or permissions')


def cache_root(account):
    home = Path(account.pw_dir)
    if account.pw_uid == 0 or home in {Path('/home'), Path('/root'), Path('/var'), Path('/tmp')}:
        raise Refused('root or broad account home')
    local(home)
    owned_directory(home, account.pw_uid)
    cache = home / '.cache/pip'
    try:
        owned_directory(home / '.cache', account.pw_uid)
    except FileNotFoundError:
        return home, None
    local(cache)
    if not cache.exists():
        return home, None
    owned_directory(cache, account.pw_uid)
    return home, cache


def active_installer(uid, proc=Path('/proc')):
    for process in proc.iterdir():
        if not process.name.isdecimal():
            continue
        try:
            if process.stat().st_uid != uid:
                continue
            with (process / 'cmdline').open('rb') as stream:
                argv = stream.read(65536).decode(errors='replace').split('\0')
        except FileNotFoundError:
            continue  # process exited
        # Permission failures for a selected UID fail closed, rather than hiding activity.
        pip = any(re.fullmatch(r'pip(?:3(?:\.\d+)?)?', Path(a).name) for a in argv[:4])
        if pip and any(a in {'install', 'download', 'wheel'} for a in argv):
            return True
    return False


def measure(cache, uid):
    device = cache.stat().st_dev
    total, newest = 0, cache.stat().st_mtime
    def unreadable(error):
        raise error
    for parent, dirs, files in os.walk(cache, followlinks=False, onerror=unreadable):
        for name in dirs + files:
            info = (Path(parent) / name).lstat()
            if (info.st_dev != device or info.st_uid != uid
                    or not (stat.S_ISDIR(info.st_mode) or stat.S_ISREG(info.st_mode))):
                raise Refused('foreign owner, symlink, special file or nested mount in pip cache')
            newest = max(newest, info.st_mtime, info.st_ctime)
            if stat.S_ISREG(info.st_mode):
                total += info.st_size
    return total, newest


@contextmanager
def installer_lock(home, uid, dry_run):
    # Matches dotcortex-deps' default XDG state lock. No redirects/custom state roots.
    parts = [home / '.local', home / '.local/state', home / '.local/state/dotcortex',
             home / '.local/state/dotcortex/dependencies']
    for directory in parts:
        if not directory.exists() and not directory.is_symlink():
            if dry_run:
                yield
                return
            directory.mkdir(mode=0o700)
        owned_directory(directory, uid)
    lock = parts[-1] / 'user-site.lock'
    if dry_run and not lock.exists():
        yield
        return
    fd = os.open(lock, os.O_RDWR | os.O_NOFOLLOW | (0 if dry_run else os.O_CREAT), 0o600)
    try:
        info = os.fstat(fd)
        if not stat.S_ISREG(info.st_mode) or info.st_uid != uid or info.st_mode & 0o022:
            raise Refused('unsafe managed installer lock')
        fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
        yield
    finally:
        os.close(fd)


def pip_run(python, home, args):
    env = dict(HOME=str(home), USER=pwd.getpwuid(os.geteuid()).pw_name,
               LOGNAME=pwd.getpwuid(os.geteuid()).pw_name, PATH='/usr/local/bin:/usr/bin:/bin')
    return subprocess.run([python, '-m', 'pip', '--disable-pip-version-check', *args],
                          env=env, text=True, capture_output=True, timeout=120)


def maintain(account, config, dry_run):
    home, cache = cache_root(account)
    if cache is None:
        return 'skipped: no pip cache'
    python = shutil.which('python3', path=str(home / '.guix-extra-profiles/core/core/bin') + ':/usr/local/bin:/usr/bin:/bin')
    if not python:
        return 'skipped: Python/pip unavailable'
    with installer_lock(home, account.pw_uid, dry_run):
        if active_installer(account.pw_uid):
            return 'skipped: active pip installer'
        result = pip_run(python, home, ['cache', 'dir'])
        if result.returncode:
            return 'skipped: pip unavailable'
        if result.stdout.strip() != str(cache):
            raise Refused('customized pip cache destination refused')
        size, newest = measure(cache, account.pw_uid)
        if size <= config['max_bytes']:
            return f'skipped: below size cap bytes={size}'
        if time.time() - newest < config['quiet_hours'] * 3600:
            return f'skipped: pip cache is not quiet bytes={size}'
        if active_installer(account.pw_uid):
            return 'skipped: active pip installer'
        if dry_run:
            return f'would purge pip download cache bytes={size}'
        # Recheck the exact root immediately before the native, account-owned mutation.
        if cache_root(account) != (home, cache):
            raise Refused('cache root changed')
        result = pip_run(python, home, ['--cache-dir', str(cache), 'cache', 'purge'])
        if result.returncode:
            raise Refused('native pip cache purge failed')
        return f'purged pip download cache prior_bytes={size}'


def run_account(account, config, dry_run):
    # Never execute account Python code with root credentials.
    sys.stdout.flush()
    child = os.fork()
    if child == 0:
        try:
            os.initgroups(account.pw_name, account.pw_gid)
            os.setgid(account.pw_gid)
            os.setuid(account.pw_uid)
            os.environ.clear()
            os.environ.update(HOME=account.pw_dir, XDG_STATE_HOME=account.pw_dir + '/.local/state')
            print('pip-cache account=' + account.pw_name + ' ' + maintain(account, config, dry_run), flush=True)
            os._exit(0)
        except (OSError, Refused, subprocess.TimeoutExpired):
            print('pip-cache account=' + account.pw_name + ' refused: unsafe state, busy lock or failed cache operation', flush=True)
            os._exit(2)
    return os.waitstatus_to_exitcode(os.waitpid(child, 0)[1])


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--config', type=Path, default=DEFAULT_CONFIG)
    parser.add_argument('--dry-run', action='store_true')
    args = parser.parse_args()
    try:
        stream, config = config_read(args.config)
        with stream:
            # Resolve every account before changing any cache; root is never a target.
            accounts = [pwd.getpwnam(name) for name in config['accounts']]
            if any(a.pw_uid == 0 for a in accounts):
                raise Refused('root account refused')
            failed = False
            for account in accounts:
                failed = bool(run_account(account, config, args.dry_run)) or failed
            return 2 if failed else 0
    except (OSError, ValueError, KeyError, TypeError, Refused) as exc:
        print('pip-cache maintenance refused: ' + str(exc), file=sys.stderr)
        return 2


if __name__ == '__main__':
    raise SystemExit(main())
