* Boundary

The package manifest is sourced in =package-nala.org=. This module only plans
and applies that small manifest, then owns Syncthing's boot lifecycle on the
three initial MailCortex hosts. Private device IDs, credentials, peer addresses,
and folder membership stay in each host's own Syncthing state. Activation requires
an explicit local readiness marker at =~/.config/mailcortex/syncthing-ready= and
an existing =~/Mail/= directory; these are provisioned only after private
configuration is reviewed. No public source writes either one.

** Narrow additive package apply

The default is a read-only apt simulation. =--apply= installs only the single
package in the micro-manifest; no repository setup, upgrade, removal, release
sync, or unrelated Nala manifest action is reachable from this entrypoint.

#+BEGIN_SRC bash :tangle debian/.local/bin/mailcortex-syncthing-package :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

usage() {
  cat <<'EOF'
Usage: mailcortex-syncthing-package [--dry-run|--apply|-h|--help]
Plan the isolated MailCortex Syncthing package transaction (default).
--apply explicitly authorizes the simulated, additive transaction.
EOF
}

case "${1:---dry-run}" in
  -h|--help) usage; exit 0 ;;
  --dry-run) mode=dry-run ;;
  --apply) mode=apply ;;
  *) usage >&2; exit 2 ;;
esac
[ "$#" -le 1 ] || { usage >&2; exit 2; }

host="${MAILCORTEX_TEST_HOSTNAME:-$(hostname -s)}"
case "$host" in
  kikin-kushi|ThinkPad-T1700|ThinkPad-X230|x230|ThinkPad-T480s|t480s) ;;
  *) echo "mailcortex-syncthing-package: unsupported host: $host" >&2; exit 2 ;;
esac

manifest="${MAILCORTEX_TEST_MANIFEST:-$HOME/.nala/manifest/mailcortex.ssv}"
[ -r "$manifest" ] || { echo "mailcortex-syncthing-package: missing manifest: $manifest" >&2; exit 1; }
count=0
while read -r pkg version category scope extra trailing; do
  case "${pkg:-}" in ''|\#*) continue ;; esac
  if [ "$pkg" != syncthing ] || [ "$version" != '""' ] \
     || [ "$category" != network ] || [ "$scope" != mailcortex ] \
     || [ "$extra" != '""' ] || [ -n "${trailing:-}" ]; then
    echo "mailcortex-syncthing-package: unexpected manifest row" >&2
    exit 2
  fi
  count=$((count + 1))
done < "$manifest"
[ "$count" -eq 1 ] || { echo "mailcortex-syncthing-package: expected exactly one Syncthing row" >&2; exit 2; }

flags=(--no-remove --no-upgrade --no-install-recommends -o APT::Get::AutomaticRemove=false)
if ! simulation="$(apt-get -s "${flags[@]}" install syncthing 2>&1)"; then
  printf '%s\n' "$simulation" >&2
  echo 'mailcortex-syncthing-package: apt simulation failed' >&2
  exit 3
fi
printf '%s\n' "$simulation"
if printf '%s\n' "$simulation" | grep -Eq '^(Remv|Purg) '; then
  echo 'mailcortex-syncthing-package: refused removal or purge' >&2
  exit 3
fi
while read -r installed_name; do
  [ -n "$installed_name" ] || continue
  if [ "$(dpkg-query -W -f='${db:Status-Abbrev}' "$installed_name" 2>/dev/null || true)" = 'ii ' ]; then
    echo "mailcortex-syncthing-package: refused upgrade of $installed_name" >&2
    exit 3
  fi
done < <(printf '%s\n' "$simulation" | awk '$1 == "Inst" {print $2}')

if [ "$mode" = dry-run ]; then
  echo 'mailcortex-syncthing-package: dry-run only; use --apply for this exact manifest'
  exit 0
fi
sudo -n env DEBIAN_FRONTEND=noninteractive apt-get install -y "${flags[@]}" syncthing
[ "$(dpkg-query -W -f='${db:Status-Abbrev}' syncthing 2>/dev/null || true)" = 'ii ' ] || {
  echo 'mailcortex-syncthing-package: install command returned but Syncthing is not installed' >&2
  exit 1
}
#+END_SRC

** Boot owner

X230 uses the distro's systemd user unit and lingering for login-independent
startup. Kikin and T480s use one managed cron entry at reboot and every three
minutes. The launcher holds a =flock= for its entire Syncthing lifetime, so
the self-heal cannot create a second owned process. An already-running,
non-owned Syncthing process causes a refusal rather than a duplicate.

#+BEGIN_SRC bash :tangle linux/.local/bin/mailcortex-syncthing-service :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

usage() {
  cat <<'EOF'
Usage: mailcortex-syncthing-service [--dry-run|--check|--apply|--run|-h|--help]
--dry-run  print the host's boot contract without changing it (default)
--check    verify the installed boot contract and daemon
--apply    install/enable the boot contract after private setup is ready
--run      SysV cron's singleton foreground daemon entrypoint
EOF
}

case "${1:---dry-run}" in
  -h|--help) usage; exit 0 ;;
  --dry-run|--check|--apply|--run) mode="${1:---dry-run}" ;;
  *) usage >&2; exit 2 ;;
esac
[ "$#" -le 1 ] || { usage >&2; exit 2; }

host="${MAILCORTEX_TEST_HOSTNAME:-$(hostname -s)}"
case "$host" in
  kikin-kushi|ThinkPad-T1700|ThinkPad-T480s|t480s) expected=sysv ;;
  ThinkPad-X230|x230) expected=systemd ;;
  *) echo "mailcortex-syncthing-service: unsupported host: $host" >&2; exit 2 ;;
esac
observed="${MAILCORTEX_TEST_INIT:-$(ps -p 1 -o comm= | tr -d '[:space:]')}"
case "$observed" in init) observed=sysv ;; esac
[ "$observed" = "$expected" ] || {
  echo "mailcortex-syncthing-service: expected $expected init on $host, observed $observed" >&2
  exit 2
}

owner="$HOME/.local/bin/mailcortex-syncthing-service"
bin="${MAILCORTEX_TEST_SYNCTHING_BIN:-/usr/bin/syncthing}"
ready="$HOME/.config/mailcortex/syncthing-ready"
cron_target="${MAILCORTEX_TEST_CRON_TARGET:-/etc/cron.d/mailcortex-syncthing}"
state="$HOME/.local/state/mailcortex"

require_ready() {
  [ -f "$ready" ] && [ -d "$HOME/Mail" ] && [ -x "$bin" ] || {
    echo 'mailcortex-syncthing-service: private readiness marker, Mail directory, or Syncthing binary missing' >&2
    exit 1
  }
}

render_cron() {
  local cron_path="$HOME/.local/bin:/usr/local/bin:/usr/bin:/bin" user_name
  user_name="$(id -un)"
  case "$HOME:$user_name" in *[[:space:]]*) echo 'mailcortex-syncthing-service: invalid cron identity' >&2; exit 2 ;; esac
  printf '# MailCortex Syncthing boot owner; generated by mailcortex-syncthing-service.\n'
  printf 'SHELL=/bin/sh\n'
  printf '@reboot %s HOME=%s PATH=%s %s --run 2>&1 | /usr/bin/logger -t mailcortex-syncthing\n' "$user_name" "$HOME" "$cron_path" "$owner"
  printf '*/3 * * * * %s HOME=%s PATH=%s %s --run 2>&1 | /usr/bin/logger -t mailcortex-syncthing\n' "$user_name" "$HOME" "$cron_path" "$owner"
}

if [ "$expected" = systemd ]; then
  [ "$mode" != --run ] || { echo 'mailcortex-syncthing-service: --run is SysV only' >&2; exit 2; }
  if [ "$mode" = --dry-run ]; then
    printf '%s\n' 'systemctl --user cat syncthing.service' \
      'systemctl --user show -P FragmentPath syncthing.service' \
      'sudo -n loginctl enable-linger CURRENT_USER' \
      'systemctl --user enable --now syncthing.service'
    exit 0
  fi
  command -v systemctl >/dev/null || { echo 'mailcortex-syncthing-service: systemctl missing' >&2; exit 1; }
  fragment="$(systemctl --user show -P FragmentPath syncthing.service)"
  case "$fragment" in
    /usr/lib/systemd/user/syncthing.service|/lib/systemd/user/syncthing.service) ;;
    *) echo "mailcortex-syncthing-service: unexpected user unit: $fragment" >&2; exit 1 ;;
  esac
  systemctl --user cat syncthing.service >/dev/null
  if [ "$mode" = --check ]; then
    [ "$(loginctl show-user "$(id -un)" -p Linger --value)" = yes ]
    systemctl --user is-enabled --quiet syncthing.service
    systemctl --user is-active --quiet syncthing.service
    echo 'mailcortex-syncthing-service: X230 user unit active, enabled, and lingering'
    exit 0
  fi
  require_ready
  sudo -n loginctl enable-linger "$(id -un)"
  systemctl --user enable --now syncthing.service
  exit 0
fi

case "$mode" in
  --dry-run) render_cron ;;
  --check)
    render_cron | cmp -s - "$cron_target" || {
      echo "mailcortex-syncthing-service: cron contract missing or drifted: $cron_target" >&2; exit 1;
    }
    pgrep -u "$(id -u)" -x syncthing >/dev/null || {
      echo 'mailcortex-syncthing-service: Syncthing is not running' >&2; exit 1;
    }
    echo 'mailcortex-syncthing-service: SysV cron contract and daemon present'
    ;;
  --apply)
    require_ready
    [ -x "$owner" ] || { echo "mailcortex-syncthing-service: missing stowed owner: $owner" >&2; exit 1; }
    command -v flock >/dev/null || { echo 'mailcortex-syncthing-service: flock missing' >&2; exit 1; }
    command -v logger >/dev/null || { echo 'mailcortex-syncthing-service: logger missing' >&2; exit 1; }
    tmp="$(mktemp)"
    trap 'rm -f "$tmp"' EXIT
    render_cron > "$tmp"
    if [ "$cron_target" = /etc/cron.d/mailcortex-syncthing ]; then
      sudo -n install -m 0644 "$tmp" "$cron_target"
    else
      install -m 0644 "$tmp" "$cron_target"
    fi
    echo "mailcortex-syncthing-service: installed $cron_target"
    ;;
  --run)
    require_ready
    mkdir -p "$state"
    exec 9>"$state/syncthing.lock"
    flock -n 9 || exit 0
    if pgrep -u "$(id -u)" -x syncthing >/dev/null; then
      echo 'mailcortex-syncthing-service: another Syncthing owner is running' >&2
      exit 3
    fi
    exec "$bin" --no-browser --no-upgrade --no-restart --no-default-folder
    ;;
esac
#+END_SRC

** Focused transport prerequisite fixtures

#+BEGIN_SRC python :tangle linux/.local/share/dotcortex/tests/test-mailcortex-syncthing.py :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
from __future__ import annotations

import os
import pathlib
import subprocess
import tempfile
import time
import unittest


ROOT = pathlib.Path(__file__).resolve().parents[5]
PACKAGE = ROOT / "debian/.local/bin/mailcortex-syncthing-package"
SERVICE = ROOT / "linux/.local/bin/mailcortex-syncthing-service"


class MailCortexSyncthingTests(unittest.TestCase):
    def setUp(self) -> None:
        self.temp = tempfile.TemporaryDirectory()
        self.addCleanup(self.temp.cleanup)
        self.base = pathlib.Path(self.temp.name)
        self.fakebin = self.base / "bin"
        self.fakebin.mkdir()
        self.manifest = self.base / "mailcortex.ssv"
        self.manifest.write_text('syncthing "" network mailcortex ""\n')
        self.calls = self.base / "calls"
        self.env = os.environ.copy()
        self.env.update(
            HOME=str(self.base),
            PATH=f"{self.fakebin}:/usr/bin:/bin",
            MAILCORTEX_TEST_HOSTNAME="kikin-kushi",
            MAILCORTEX_TEST_INIT="sysv",
            MAILCORTEX_TEST_MANIFEST=str(self.manifest),
            MAILCORTEX_TEST_CRON_TARGET=str(self.base / "cron"),
            CALLS=str(self.calls),
        )
        for name, body in {
            "apt-get": """case " $* " in *' -s '*) printf '%s\\n' "${SIMULATION:-Inst syncthing (1.0)}" ;; *) echo "apt-get $*" >> "$CALLS" ;; esac""",
            "dpkg-query": """name="${@: -1}"; if [ "${INSTALLED_NAME:-}" = "$name" ] || { [ "${POST_INSTALLED:-}" = "$name" ] && [ -e "$CALLS" ]; }; then printf 'ii '; else exit 1; fi""",
            "sudo": 'echo "sudo $*" >> "$CALLS"',
            "pgrep": "exit 1",
        }.items():
            path = self.fakebin / name
            path.write_text("#!/usr/bin/env bash\n" + body + "\n")
            path.chmod(0o755)

    def run_cmd(self, path: pathlib.Path, *args: str, env: dict[str, str] | None = None) -> subprocess.CompletedProcess[str]:
        return subprocess.run([str(path), *args], env=env or self.env, text=True, capture_output=True)

    def test_package_dry_run_and_explicit_apply(self) -> None:
        result = self.run_cmd(PACKAGE)
        self.assertEqual(result.returncode, 0, result.stderr)
        self.assertIn("dry-run only", result.stdout)
        self.assertFalse(self.calls.exists())
        env = self.env | {"POST_INSTALLED": "syncthing"}
        result = self.run_cmd(PACKAGE, "--apply", env=env)
        self.assertEqual(result.returncode, 0, result.stderr)
        calls = self.calls.read_text()
        self.assertIn("apt-get install -y --no-remove --no-upgrade --no-install-recommends", calls)
        self.assertNotIn("upgrade", calls.replace("--no-upgrade", ""))
        self.assertNotIn("remove -y", calls)

    def test_package_refuses_removal_upgrade_and_extra_manifest_row(self) -> None:
        for simulation in ("Remv unrelated", "Purg unrelated"):
            result = self.run_cmd(PACKAGE, "--apply", env=self.env | {"SIMULATION": simulation})
            self.assertEqual(result.returncode, 3)
            self.assertFalse(self.calls.exists())
        result = self.run_cmd(PACKAGE, "--apply", env=self.env | {"INSTALLED_NAME": "syncthing"})
        self.assertEqual(result.returncode, 3)
        self.manifest.write_text(self.manifest.read_text() + 'unrelated "" network mailcortex ""\n')
        result = self.run_cmd(PACKAGE, "--apply")
        self.assertEqual(result.returncode, 2)

    def test_package_refuses_unknown_host_and_help_has_no_effect(self) -> None:
        result = self.run_cmd(PACKAGE, "--apply", env=self.env | {"MAILCORTEX_TEST_HOSTNAME": "other"})
        self.assertEqual(result.returncode, 2)
        self.assertFalse(self.calls.exists())
        result = self.run_cmd(PACKAGE, "-h")
        self.assertEqual(result.returncode, 0)
        self.assertFalse(self.calls.exists())

    def test_service_render_and_host_refusal(self) -> None:
        result = self.run_cmd(SERVICE)
        self.assertEqual(result.returncode, 0, result.stderr)
        self.assertIn("@reboot", result.stdout)
        self.assertIn("*/3 * * * *", result.stdout)
        self.assertFalse((self.base / "cron").exists())
        x230 = self.env | {"MAILCORTEX_TEST_HOSTNAME": "ThinkPad-X230", "MAILCORTEX_TEST_INIT": "systemd"}
        result = self.run_cmd(SERVICE, "--dry-run", env=x230)
        self.assertEqual(result.returncode, 0, result.stderr)
        self.assertIn("systemctl --user enable --now syncthing.service", result.stdout)
        refused = self.run_cmd(SERVICE, "--dry-run", env=self.env | {"MAILCORTEX_TEST_HOSTNAME": "unknown"})
        self.assertEqual(refused.returncode, 2)
        mismatch = self.run_cmd(SERVICE, "--dry-run", env=self.env | {"MAILCORTEX_TEST_INIT": "systemd"})
        self.assertEqual(mismatch.returncode, 2)

    def test_service_apply_paths_are_fixture_bound(self) -> None:
        (self.base / "Mail").mkdir()
        ready = self.base / ".config/mailcortex/syncthing-ready"
        ready.parent.mkdir(parents=True)
        ready.touch()
        binary = self.base / "syncthing"
        binary.write_text("#!/bin/sh\nexit 0\n")
        binary.chmod(0o755)
        owner = self.base / ".local/bin/mailcortex-syncthing-service"
        owner.parent.mkdir(parents=True)
        owner.symlink_to(SERVICE)
        env = self.env | {"MAILCORTEX_TEST_SYNCTHING_BIN": str(binary)}
        result = self.run_cmd(SERVICE, "--apply", env=env)
        self.assertEqual(result.returncode, 0, result.stderr)
        cron = (self.base / "cron").read_text()
        self.assertIn("@reboot", cron)
        self.assertIn(str(owner) + " --run", cron)
        systemctl = self.fakebin / "systemctl"
        systemctl.write_text(
            '#!/bin/sh\n'
            'echo "systemctl $*" >> "$CALLS"\n'
            'case " $* " in *" show "*) echo /usr/lib/systemd/user/syncthing.service ;; esac\n'
        )
        systemctl.chmod(0o755)
        x230 = env | {"MAILCORTEX_TEST_HOSTNAME": "ThinkPad-X230", "MAILCORTEX_TEST_INIT": "systemd"}
        result = self.run_cmd(SERVICE, "--apply", env=x230)
        self.assertEqual(result.returncode, 0, result.stderr)
        calls = self.calls.read_text()
        self.assertIn("sudo -n loginctl enable-linger", calls)
        self.assertIn("systemctl --user enable --now syncthing.service", calls)

    def test_sysv_singleton_owner(self) -> None:
        (self.base / "Mail").mkdir()
        ready = self.base / ".config/mailcortex/syncthing-ready"
        ready.parent.mkdir(parents=True)
        ready.touch()
        binary = self.base / "syncthing"
        binary.write_text('#!/bin/sh\nprintf "started\\n" >> "$CALLS"\nsleep 20\n')
        binary.chmod(0o755)
        env = self.env | {"MAILCORTEX_TEST_SYNCTHING_BIN": str(binary)}
        first = subprocess.Popen([str(SERVICE), "--run"], env=env, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        try:
            for _ in range(40):
                if self.calls.exists():
                    break
                time.sleep(0.05)
            self.assertTrue(self.calls.exists(), "first owner did not start")
            second = self.run_cmd(SERVICE, "--run", env=env)
            self.assertEqual(second.returncode, 0, second.stderr)
            self.assertEqual(self.calls.read_text().splitlines(), ["started"])
        finally:
            first.terminate()
            first.wait(timeout=5)


if __name__ == "__main__":
    unittest.main()
