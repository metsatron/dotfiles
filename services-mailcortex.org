* Boundary

The package manifest is sourced in =package-nala.org=. This module only plans
and applies that small manifest, then owns Syncthing's boot lifecycle on each
configured node. Private device IDs, credentials, peer addresses,
and folder membership stay in a dedicated per-host Syncthing home at
=~/.config/mailcortex/syncthing/=. Configuration isolation prevents MailCortex
settings from rewriting an older general-purpose Syncthing profile. Both
profiles may claim the same network ports, so runtime startup refuses a second
Syncthing owner until the private port plan is reviewed.
Activation requires an explicit local readiness marker at
=~/.config/mailcortex/syncthing-ready=, a complete private identity/config, and
an existing =~/Mail/= directory; these are provisioned only after private
configuration is reviewed. No public source writes them.

** Narrow additive package apply

The default is a read-only apt simulation. =--apply= installs only the single
package in the micro-manifest; no repository setup, upgrade, removal, release
sync, or unrelated Nala manifest action is reachable from this entrypoint.

* Maildir command surface

The public command carries no fleet topology.  It maps a restricted RFC-style
address onto an already-provisioned Maildir below =~/Mail=: a bare local part is
a consort, while =seat+=, =machine+=, and =sanctuary+= select the other three
namespaces.  The domain must end in =.helm=, but remains provenance rather than
a network destination because Syncthing carries the entire tree.

=provision= creates validated Maildirs with private directory permissions;
=send= refuses unknown Maildirs so a typo cannot silently create a new address.
It writes a complete RFC 822 message under =tmp/=, fsyncs it, and atomically
renames it into =new/=.  =inbox= is read-only; =read= moves a message from
=new/= to =cur/= with the standard seen flag before rendering it.  Bodies may
refer to HelmCortex paths. Repeatable =send --attach PATH= adds MIME attachments
with the plain-text body first; missing, symlink and non-regular inputs are
refused before delivery. Total raw attachments are capped at 25 MiB. Types are
guessed from filenames, falling back to =application/octet-stream=. Without
=--attach= the original single-part wire format is unchanged.

#+BEGIN_SRC python :tangle all/.local/bin/mailcortex :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
"""MailCortex filesystem-native mail transport."""

from __future__ import annotations

import argparse
import hashlib
import json
import mimetypes
import os
import re
import socket
import stat
import sys
import time
import uuid
from email import policy
from email.message import EmailMessage
from email.parser import BytesParser
from email.utils import format_datetime, make_msgid
from datetime import datetime
from pathlib import Path


ADDRESS_RE = re.compile(
    r"^(?:(consort|seat|machine|sanctuary)\+)?"
    r"([a-z0-9][a-z0-9._-]*)@"
    r"([a-z0-9][a-z0-9.-]*\.helm)$"
)
NAMESPACE = {
    None: "consorts",
    "consort": "consorts",
    "seat": "seats",
    "machine": "machines",
    "sanctuary": "sanctuaries",
}


class MailCortexError(Exception):
    """Expected command failure suitable for a concise CLI diagnostic."""


WAKE_STATES = {
    "manual", "observed", "queued", "injecting", "injected", "delivery_unknown",
}
HANDLING_STATES = {
    "handling_completed", "handling_deferred", "handling_rejected", "handling_failed",
}


def mail_root() -> Path:
    return Path(os.environ.get("MAILCORTEX_ROOT", "~/Mail")).expanduser()


def runtime_root() -> Path:
    return Path(os.environ.get("MAILCORTEX_RUNTIME", "~/.local/state/mailcortex")).expanduser()


def atomic_json(path: Path, value: object) -> None:
    path.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
    temporary = path.with_name(f".{path.name}.{os.getpid()}.{uuid.uuid4().hex}.tmp")
    try:
        descriptor = os.open(temporary, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
        with os.fdopen(descriptor, "w", encoding="utf-8") as stream:
            json.dump(value, stream, sort_keys=True, separators=(",", ":"))
            stream.write("\n")
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(temporary, path)
        fsync_directory(path.parent)
    except Exception:
        try:
            temporary.unlink()
        except FileNotFoundError:
            pass
        raise


def receipt_path(seat: str, message_id: str) -> Path:
    if not message_id.startswith("<") or not message_id.endswith(">") or "@" not in message_id:
        raise MailCortexError("invalid Message-ID")
    key = hashlib.sha256(message_id.encode("utf-8")).hexdigest()
    return runtime_root() / "receipts" / seat / f"{key}.json"


def load_receipt(seat: str, message_id: str) -> dict[str, object] | None:
    path = receipt_path(seat, message_id)
    if not path.exists():
        return None
    if path.is_symlink() or not path.is_file():
        raise MailCortexError("receipt is not a regular file")
    try:
        current = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError) as error:
        raise MailCortexError("receipt is unreadable") from error
    if (
        not isinstance(current, dict)
        or current.get("schema") != "mailcortex.receipt.v2"
        or current.get("seat") != seat
        or current.get("message_id") != message_id
        or current.get("wake_state") not in WAKE_STATES
        or current.get("handling_state") not in HANDLING_STATES | {None}
    ):
        raise MailCortexError("receipt is invalid")
    return current


def write_wake_receipt(seat: str, message_id: str, state: str, **extra: object) -> None:
    if state not in WAKE_STATES - {"manual"}:
        raise MailCortexError(f"unknown wake state: {state}")
    path = receipt_path(seat, message_id)
    current = load_receipt(seat, message_id) or {
        "schema": "mailcortex.receipt.v2", "seat": seat,
        "message_id": message_id, "handling_state": None,
    }
    current.update({"wake_state": state, "wake_updated_at": time.time(), **extra})
    atomic_json(path, current)


def write_handling_receipt(seat: str, message_id: str, state: str) -> None:
    if state not in HANDLING_STATES:
        raise MailCortexError(f"unknown handling state: {state}")
    path = receipt_path(seat, message_id)
    current = load_receipt(seat, message_id) or {
        "schema": "mailcortex.receipt.v2", "seat": seat,
        "message_id": message_id, "wake_state": "manual",
    }
    current.update({"handling_state": state, "handling_updated_at": time.time()})
    atomic_json(path, current)


def receipt(args: argparse.Namespace) -> int:
    write_handling_receipt(args.seat, args.message_id, args.state)
    return 0


def address_maildir(address: str) -> Path:
    match = ADDRESS_RE.fullmatch(address.lower())
    if not match:
        raise MailCortexError(
            "invalid address; use name@machine.helm or "
            "seat+name@machine.helm (also machine+ and sanctuary+)"
        )
    kind, name, _domain = match.groups()
    return mail_root() / NAMESPACE[kind] / name


def require_maildir(address: str) -> Path:
    path = address_maildir(address)
    missing = [name for name in ("tmp", "new", "cur") if not (path / name).is_dir()]
    if missing:
        raise MailCortexError(f"maildir is not provisioned for {address}: {path}")
    return path


def provision(args: argparse.Namespace) -> int:
    # Validate every address before creating any directory.
    paths = list(dict.fromkeys(address_maildir(address) for address in args.addresses))
    root = mail_root()
    for path in paths:
        for component in (root, path.parent, path):
            if component.is_symlink():
                raise MailCortexError(f"refusing symlink in maildir path: {component}")
    old_umask = os.umask(0o077)
    try:
        for path in paths:
            for directory in (root, path.parent, path, *(path / leaf for leaf in ("tmp", "new", "cur"))):
                directory.mkdir(mode=0o700, parents=True, exist_ok=True)
                if directory.is_symlink() or not directory.is_dir():
                    raise MailCortexError(f"unsafe maildir path: {directory}")
                directory.chmod(0o700)
            print(path)
    finally:
        os.umask(old_umask)
    return 0


def safe_name() -> str:
    host = re.sub(r"[^A-Za-z0-9_-]", "_", socket.gethostname())
    return f"{time.time_ns()}.{os.getpid()}_{uuid.uuid4().hex}.{host}"


def fsync_directory(path: Path) -> None:
    descriptor = os.open(path, os.O_RDONLY | os.O_DIRECTORY)
    try:
        os.fsync(descriptor)
    finally:
        os.close(descriptor)


ATTACHMENT_LIMIT = 25 * 1024 * 1024


def attachments(paths: list[str]):
    remaining = ATTACHMENT_LIMIT
    for value in paths:
        path = Path(value).expanduser()
        # O_NONBLOCK avoids hanging on a FIFO; fstat binds validation to the
        # opened inode, and O_NOFOLLOW refuses a raced-in leaf symlink.
        descriptor = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
        with os.fdopen(descriptor, "rb") as stream:
            info = os.fstat(stream.fileno())
            if not stat.S_ISREG(info.st_mode):
                raise MailCortexError("attachment must be a regular file")
            if info.st_size > remaining:
                raise MailCortexError("attachments exceed 25 MiB")
            data = stream.read(remaining + 1)
        if len(data) > remaining:
            raise MailCortexError("attachments exceed 25 MiB")
        remaining -= len(data)
        mime, encoding = mimetypes.guess_type(path.name)
        yield path.name, mime if mime and not encoding else "application/octet-stream", data


def deliver(args: argparse.Namespace) -> int:
    target = require_maildir(args.to)
    if args.body is not None:
        body = args.body
    else:
        body = Path(args.body_file).read_text(encoding="utf-8")

    message = EmailMessage(policy=policy.SMTP)
    message["Date"] = format_datetime(datetime.now().astimezone())
    message["From"] = args.sender
    message["To"] = args.to
    message["Subject"] = args.subject
    message["Message-ID"] = make_msgid(domain=args.sender.rsplit("@", 1)[-1])
    message.set_content(body)
    for filename, mime, data in attachments(getattr(args, "attach", [])):
        major, minor = mime.split("/", 1)
        message.add_attachment(data, maintype=major, subtype=minor, filename=filename)

    name = safe_name()
    temporary = target / "tmp" / name
    delivered = target / "new" / name
    try:
        descriptor = os.open(temporary, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
        with os.fdopen(descriptor, "wb") as stream:
            stream.write(message.as_bytes())
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(temporary, delivered)
        fsync_directory(target / "new")
    except Exception:
        try:
            temporary.unlink()
        except FileNotFoundError:
            pass
        raise
    print(name)
    return 0


def parse_message(path: Path) -> EmailMessage:
    with path.open("rb") as stream:
        return BytesParser(policy=policy.default).parse(stream)


def messages(maildir: Path):
    for state in ("new", "cur"):
        for path in (maildir / state).iterdir():
            # Dot-prefixed names are in-flight writes (Syncthing's
            # .syncthing.*.tmp), never deliverable messages.
            if path.is_file() and not path.name.startswith("."):
                yield state, path


def show_inbox(args: argparse.Namespace) -> int:
    maildir = require_maildir(args.address)
    rows = sorted(messages(maildir), key=lambda item: (item[1].stat().st_mtime_ns, item[1].name))
    for state, path in rows:
        message = parse_message(path)
        fields = (
            state,
            path.name,
            str(message.get("Date", "")),
            str(message.get("From", "")),
            str(message.get("Subject", "")),
        )
        print("\t".join(value.replace("\t", " ").replace("\n", " ") for value in fields))
    return 0


def locate_message(maildir: Path, name: str) -> tuple[str, Path]:
    if not name or name in {".", ".."} or Path(name).name != name:
        raise MailCortexError("message must be an exact inbox filename")
    for state in ("new", "cur"):
        candidate = maildir / state / name
        if candidate.is_file():
            return state, candidate
    raise MailCortexError(f"message not found: {name}")


def read_message(args: argparse.Namespace) -> int:
    maildir = require_maildir(args.address)
    state, path = locate_message(maildir, args.message)
    message = parse_message(path)
    if state == "new":
        seen_name = path.name if ":2," in path.name else path.name + ":2,S"
        destination = maildir / "cur" / seen_name
        if destination.exists():
            raise MailCortexError(f"seen-message destination already exists: {seen_name}")
        os.replace(path, destination)
        fsync_directory(maildir / "new")
        fsync_directory(maildir / "cur")

    for header in ("Date", "From", "To", "Subject", "Message-ID"):
        print(f"{header}: {message.get(header, '')}")
    print()
    body = message.get_body(preferencelist=("plain",))
    print(body.get_content() if body is not None else "", end="")
    return 0


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="mailcortex",
        description="Deliver and read fleet-local RFC 822 messages through Maildir.",
    )
    subparsers = parser.add_subparsers(dest="command", required=True)

    create = subparsers.add_parser("provision", help="create private Maildirs for validated addresses")
    create.add_argument("addresses", nargs="+", type=validate_address)
    create.set_defaults(function=provision)

    send = subparsers.add_parser("send", help="atomically deliver one message")
    send.add_argument("--from", dest="sender", required=True, type=validate_address)
    send.add_argument("--to", required=True, type=validate_address)
    send.add_argument("--subject", required=True)
    body = send.add_mutually_exclusive_group(required=True)
    body.add_argument("--body")
    body.add_argument("--body-file")
    send.add_argument("--attach", action="append", default=[], metavar="PATH",
                      help="attach a regular file (repeatable; total <= 25 MiB)")
    send.set_defaults(function=deliver)

    inbox = subparsers.add_parser("inbox", help="list new and seen messages")
    inbox.add_argument("address", type=validate_address)
    inbox.set_defaults(function=show_inbox)

    read = subparsers.add_parser("read", help="render one message and mark it seen")
    read.add_argument("address", type=validate_address)
    read.add_argument("message")
    read.set_defaults(function=read_message)
    acknowledge = subparsers.add_parser("receipt", help="record a durable handling receipt")
    acknowledge.add_argument("seat", type=validate_component)
    acknowledge.add_argument("message_id")
    acknowledge.add_argument("state", choices=sorted(HANDLING_STATES))
    acknowledge.set_defaults(function=receipt)
    return parser


def validate_component(value: str) -> str:
    if not re.fullmatch(r"[a-z0-9][a-z0-9._-]{0,63}", value):
        raise argparse.ArgumentTypeError("invalid seat")
    return value


def validate_address(value: str) -> str:
    address_maildir(value)
    return value.lower()


def main() -> int:
    parser = build_parser()
    try:
        args = parser.parse_args()
        return args.function(args)
    except (MailCortexError, OSError, ValueError) as error:
        print(f"mailcortex: {error}", file=sys.stderr)
        return 2


if __name__ == "__main__":
    raise SystemExit(main())

* Module ownership

The seat watcher and its boot owner live in =services-mailcortex-watcher.org=.
Unchanged command, watcher, and lifecycle fixtures live in
=services-mailcortex-tests.org=. Tangle all three modules for the complete
transport and watcher surface; the generated paths remain unchanged.

* Syncthing boot owner

A systemd node uses a dedicated user unit and lingering for login-independent
startup. A SysV node uses one managed cron entry at reboot and every three
minutes. Both invoke the same launcher, which checks private readiness at each
start, holds a =flock= for Syncthing's lifetime, and records the exact PID and
process start time. =--check= verifies that record and the private command line.
An already-running Syncthing process causes a refusal while port isolation is
unsettled. The generic =syncthing.service= and legacy profile are untouched.

#+BEGIN_SRC bash :tangle linux/.local/bin/mailcortex-syncthing-service :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

usage() {
  cat <<'EOF'
Usage: mailcortex-syncthing-service [--dry-run|--check|--apply|--run|-h|--help]
--dry-run  print the host's boot contract without changing it (default)
--check    verify the installed boot contract and daemon
--apply    install/enable the boot contract after private setup is ready
--run      singleton foreground daemon entrypoint
EOF
}

case "${1:---dry-run}" in
  -h|--help) usage; exit 0 ;;
  --dry-run|--check|--apply|--run) mode="${1:---dry-run}" ;;
  *) usage >&2; exit 2 ;;
esac
[ "$#" -le 1 ] || { usage >&2; exit 2; }

observed="${MAILCORTEX_TEST_INIT:-$(ps -p 1 -o comm= | tr -d '[:space:]')}"
case "$observed" in
  systemd) expected=systemd ;;
  init|sysv) expected=sysv ;;
  *) echo "mailcortex-syncthing-service: unsupported init: $observed" >&2; exit 2 ;;
esac

owner="$HOME/.local/bin/mailcortex-syncthing-service"
bin="${MAILCORTEX_TEST_SYNCTHING_BIN:-/usr/bin/syncthing}"
ready="$HOME/.config/mailcortex/syncthing-ready"
cron_target="${MAILCORTEX_TEST_CRON_TARGET:-/etc/cron.d/mailcortex-syncthing}"
state="$HOME/.local/state/mailcortex"
config="$HOME/.config/mailcortex/syncthing"
unit=mailcortex-syncthing.service
pid_record="$state/syncthing.pid"

require_ready() {
  [ -f "$ready" ] && [ -d "$HOME/Mail" ] && [ -x "$bin" ] \
    && [ -r "$config/config.xml" ] && [ -r "$config/cert.pem" ] && [ -r "$config/key.pem" ] || {
    echo 'mailcortex-syncthing-service: private readiness marker, Mail directory, identity/config, or Syncthing binary missing' >&2
    exit 1
  }
}

private_owner_running() {
  local pid started current argv
  local -a args
  [ -r "$pid_record" ] || return 1
  read -r pid started < "$pid_record" || return 1
  [[ "$pid" =~ ^[1-9][0-9]*$ && "$started" =~ ^[0-9]+$ ]] || return 1
  [ -r "/proc/$pid/stat" ] && [ -r "/proc/$pid/cmdline" ] || return 1
  [ "$(stat -c %u "/proc/$pid")" = "$(id -u)" ] || return 1
  current="$(awk '{print $22}' "/proc/$pid/stat")"
  [ "$current" = "$started" ] || return 1
  mapfile -d '' -t args < "/proc/$pid/cmdline"
  # A shebang fixture has an interpreter before the binary path.
  if [ "${args[0]:-}" = "$bin" ]; then
    args=("${args[@]:1}")
  elif [ "${args[1]:-}" = "$bin" ]; then
    args=("${args[@]:2}")
  else
    return 1
  fi
  [ "${args[0]:-}" = serve ] && [ "${args[1]:-}" = "--home=$config" ] \
    && [ "${args[2]:-}" = --no-browser ] \
    && [ "${args[3]:-}" = --no-upgrade ] \
    && [ "${args[4]:-}" = --no-restart ] \
    && [ "${args[5]:-}" = --no-default-folder ] \
    && [ "${#args[@]}" -eq 6 ]
}

render_cron() {
  local cron_path="$HOME/.local/bin:/usr/local/bin:/usr/bin:/bin" user_name
  user_name="$(id -un)"
  case "$HOME:$user_name" in *[[:space:]]*) echo 'mailcortex-syncthing-service: invalid cron identity' >&2; exit 2 ;; esac
  printf '# MailCortex Syncthing boot owner; generated by mailcortex-syncthing-service.\n'
  printf 'SHELL=/bin/sh\n'
  printf '@reboot %s HOME=%s PATH=%s %s --run 2>&1 | /usr/bin/logger -t mailcortex-syncthing\n' "$user_name" "$HOME" "$cron_path" "$owner"
  printf '*/3 * * * * %s HOME=%s PATH=%s %s --run 2>&1 | /usr/bin/logger -t mailcortex-syncthing\n' "$user_name" "$HOME" "$cron_path" "$owner"
}

systemd_owner_enabled() {
  local state wants
  state="$(systemctl --user is-enabled "$unit" 2>/dev/null || true)"
  case "$state" in
    enabled) return 0 ;;
    linked)
      wants="$HOME/.config/systemd/user/default.target.wants/$unit"
      [ -L "$wants" ] \
        && [ "$(readlink -f "$wants")" = "$(readlink -f "$fragment")" ]
      ;;
    *) return 1 ;;
  esac
}

if [ "$expected" = systemd ]; then
  if [ "$mode" = --dry-run ]; then
    printf '%s\n' "systemctl --user cat $unit" \
      "systemctl --user show -P FragmentPath $unit" \
      'sudo -n loginctl enable-linger CURRENT_USER' \
      'systemctl --user daemon-reload' \
      "systemctl --user enable --now $unit"
    exit 0
  fi
  if [ "$mode" != --run ]; then
    command -v systemctl >/dev/null || { echo 'mailcortex-syncthing-service: systemctl missing' >&2; exit 1; }
    fragment="$(systemctl --user show -P FragmentPath "$unit")"
    [ "$fragment" = "$HOME/.config/systemd/user/$unit" ] || {
      echo "mailcortex-syncthing-service: unexpected user unit: $fragment" >&2; exit 1;
    }
    systemctl --user cat "$unit" >/dev/null
    if [ "$mode" = --check ]; then
      require_ready
      [ "$(loginctl show-user "$(id -un)" -p Linger --value)" = yes ]
      systemd_owner_enabled || {
        echo 'mailcortex-syncthing-service: user unit is not enabled for default.target' >&2; exit 1;
      }
      systemctl --user is-active --quiet "$unit"
      private_owner_running || { echo 'mailcortex-syncthing-service: private Syncthing owner absent' >&2; exit 1; }
      read -r owned_pid _ < "$pid_record"
      [ "$(systemctl --user show -P MainPID "$unit")" = "$owned_pid" ] || {
        echo 'mailcortex-syncthing-service: user unit owns a different PID' >&2; exit 1;
      }
      echo 'mailcortex-syncthing-service: MailCortex user unit and private daemon verified'
      exit 0
    fi
    require_ready
    sudo -n loginctl enable-linger "$(id -un)"
    systemctl --user daemon-reload
    systemctl --user enable --now "$unit"
    exit 0
  fi
fi

case "$mode" in
  --dry-run) render_cron ;;
  --check)
    require_ready
    render_cron | cmp -s - "$cron_target" || {
      echo "mailcortex-syncthing-service: cron contract missing or drifted: $cron_target" >&2; exit 1;
    }
    private_owner_running || {
      echo 'mailcortex-syncthing-service: private Syncthing owner absent' >&2; exit 1;
    }
    echo 'mailcortex-syncthing-service: SysV cron contract and private daemon verified'
    ;;
  --apply)
    require_ready
    [ -x "$owner" ] || { echo "mailcortex-syncthing-service: missing stowed owner: $owner" >&2; exit 1; }
    command -v flock >/dev/null || { echo 'mailcortex-syncthing-service: flock missing' >&2; exit 1; }
    command -v logger >/dev/null || { echo 'mailcortex-syncthing-service: logger missing' >&2; exit 1; }
    tmp="$(mktemp)"
    trap 'rm -f "$tmp"' EXIT
    render_cron > "$tmp"
    if [ "$cron_target" = /etc/cron.d/mailcortex-syncthing ]; then
      sudo -n install -m 0644 "$tmp" "$cron_target"
    else
      install -m 0644 "$tmp" "$cron_target"
    fi
    echo "mailcortex-syncthing-service: installed $cron_target"
    ;;
  --run)
    require_ready
    umask 077
    mkdir -p -m 0700 "$state"
    exec 9>"$state/syncthing.lock"
    flock -n 9 || exit 0
    if pgrep -u "$(id -u)" -x syncthing >/dev/null; then
      echo 'mailcortex-syncthing-service: another Syncthing owner is running' >&2
      exit 3
    fi
    printf '%s %s\n' "$$" "$(awk '{print $22}' "/proc/$$/stat")" > "$pid_record"
    exec "$bin" serve --home="$config" --no-browser --no-upgrade --no-restart --no-default-folder
    ;;
esac
