* Boundary

The package manifest is sourced in =package-nala.org=. This module only plans
and applies that small manifest, then owns Syncthing's boot lifecycle on each
configured node. Private device IDs, credentials, peer addresses,
and folder membership stay in a dedicated per-host Syncthing home at
=~/.config/mailcortex/syncthing/=. Configuration isolation prevents MailCortex
settings from rewriting an older general-purpose Syncthing profile. Both
profiles may claim the same network ports, so runtime startup refuses a second
Syncthing owner until the private port plan is reviewed.
Activation requires an explicit local readiness marker at
=~/.config/mailcortex/syncthing-ready=, a complete private identity/config, and
an existing =~/Mail/= directory; these are provisioned only after private
configuration is reviewed. No public source writes them.

** Narrow additive package apply

The default is a read-only apt simulation. =--apply= installs only the single
package in the micro-manifest; no repository setup, upgrade, removal, release
sync, or unrelated Nala manifest action is reachable from this entrypoint.

* Maildir command surface

The public command carries no fleet topology.  It maps a restricted RFC-style
address onto an already-provisioned Maildir below =~/Mail=: a bare local part is
a consort, while =seat+=, =machine+=, and =sanctuary+= select the other three
namespaces.  The domain must end in =.helm=, but remains provenance rather than
a network destination because Syncthing carries the entire tree.

=provision= creates validated Maildirs with private directory permissions;
=send= refuses unknown Maildirs so a typo cannot silently create a new address.
It writes a complete RFC 822 message under =tmp/=, fsyncs it, and atomically
renames it into =new/=.  =inbox= is read-only; =read= moves a message from
=new/= to =cur/= with the standard seen flag before rendering it.  Bodies may
refer to HelmCortex paths, but the command has no attachment feature.

"""MailCortex filesystem-native mail transport."""

from __future__ import annotations

import argparse
import hashlib
import json
import os
import re
import socket
import sys
import time
import uuid
from email import policy
from email.message import EmailMessage
from email.parser import BytesParser
from email.utils import format_datetime, make_msgid
from datetime import datetime
from pathlib import Path


ADDRESS_RE = re.compile(
    r"^(?:(consort|seat|machine|sanctuary)\+)?"
    r"([a-z0-9][a-z0-9._-]*)@"
    r"([a-z0-9][a-z0-9.-]*\.helm)$"
)
NAMESPACE = {
    None: "consorts",
    "consort": "consorts",
    "seat": "seats",
    "machine": "machines",
    "sanctuary": "sanctuaries",
}


class MailCortexError(Exception):
    """Expected command failure suitable for a concise CLI diagnostic."""


WAKE_STATES = {
    "manual", "observed", "queued", "injecting", "injected", "delivery_unknown",
}
HANDLING_STATES = {
    "handling_completed", "handling_deferred", "handling_rejected", "handling_failed",
}


def mail_root() -> Path:
    return Path(os.environ.get("MAILCORTEX_ROOT", "~/Mail")).expanduser()


def runtime_root() -> Path:
    return Path(os.environ.get("MAILCORTEX_RUNTIME", "~/.local/state/mailcortex")).expanduser()


def atomic_json(path: Path, value: object) -> None:
    path.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
    temporary = path.with_name(f".{path.name}.{os.getpid()}.{uuid.uuid4().hex}.tmp")
    try:
        descriptor = os.open(temporary, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
        with os.fdopen(descriptor, "w", encoding="utf-8") as stream:
            json.dump(value, stream, sort_keys=True, separators=(",", ":"))
            stream.write("\n")
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(temporary, path)
        fsync_directory(path.parent)
    except Exception:
        try:
            temporary.unlink()
        except FileNotFoundError:
            pass
        raise


def receipt_path(seat: str, message_id: str) -> Path:
    if not message_id.startswith("<") or not message_id.endswith(">") or "@" not in message_id:
        raise MailCortexError("invalid Message-ID")
    key = hashlib.sha256(message_id.encode("utf-8")).hexdigest()
    return runtime_root() / "receipts" / seat / f"{key}.json"


def load_receipt(seat: str, message_id: str) -> dict[str, object] | None:
    path = receipt_path(seat, message_id)
    if not path.exists():
        return None
    if path.is_symlink() or not path.is_file():
        raise MailCortexError("receipt is not a regular file")
    try:
        current = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError) as error:
        raise MailCortexError("receipt is unreadable") from error
    if (
        not isinstance(current, dict)
        or current.get("schema") != "mailcortex.receipt.v2"
        or current.get("seat") != seat
        or current.get("message_id") != message_id
        or current.get("wake_state") not in WAKE_STATES
        or current.get("handling_state") not in HANDLING_STATES | {None}
    ):
        raise MailCortexError("receipt is invalid")
    return current


def write_wake_receipt(seat: str, message_id: str, state: str, **extra: object) -> None:
    if state not in WAKE_STATES - {"manual"}:
        raise MailCortexError(f"unknown wake state: {state}")
    path = receipt_path(seat, message_id)
    current = load_receipt(seat, message_id) or {
        "schema": "mailcortex.receipt.v2", "seat": seat,
        "message_id": message_id, "handling_state": None,
    }
    current.update({"wake_state": state, "wake_updated_at": time.time(), **extra})
    atomic_json(path, current)


def write_handling_receipt(seat: str, message_id: str, state: str) -> None:
    if state not in HANDLING_STATES:
        raise MailCortexError(f"unknown handling state: {state}")
    path = receipt_path(seat, message_id)
    current = load_receipt(seat, message_id) or {
        "schema": "mailcortex.receipt.v2", "seat": seat,
        "message_id": message_id, "wake_state": "manual",
    }
    current.update({"handling_state": state, "handling_updated_at": time.time()})
    atomic_json(path, current)


def receipt(args: argparse.Namespace) -> int:
    write_handling_receipt(args.seat, args.message_id, args.state)
    return 0


def address_maildir(address: str) -> Path:
    match = ADDRESS_RE.fullmatch(address.lower())
    if not match:
        raise MailCortexError(
            "invalid address; use name@machine.helm or "
            "seat+name@machine.helm (also machine+ and sanctuary+)"
        )
    kind, name, _domain = match.groups()
    return mail_root() / NAMESPACE[kind] / name


def require_maildir(address: str) -> Path:
    path = address_maildir(address)
    missing = [name for name in ("tmp", "new", "cur") if not (path / name).is_dir()]
    if missing:
        raise MailCortexError(f"maildir is not provisioned for {address}: {path}")
    return path


def provision(args: argparse.Namespace) -> int:
    # Validate every address before creating any directory.
    paths = list(dict.fromkeys(address_maildir(address) for address in args.addresses))
    root = mail_root()
    for path in paths:
        for component in (root, path.parent, path):
            if component.is_symlink():
                raise MailCortexError(f"refusing symlink in maildir path: {component}")
    old_umask = os.umask(0o077)
    try:
        for path in paths:
            for directory in (root, path.parent, path, *(path / leaf for leaf in ("tmp", "new", "cur"))):
                directory.mkdir(mode=0o700, parents=True, exist_ok=True)
                if directory.is_symlink() or not directory.is_dir():
                    raise MailCortexError(f"unsafe maildir path: {directory}")
                directory.chmod(0o700)
            print(path)
    finally:
        os.umask(old_umask)
    return 0


def safe_name() -> str:
    host = re.sub(r"[^A-Za-z0-9_-]", "_", socket.gethostname())
    return f"{time.time_ns()}.{os.getpid()}_{uuid.uuid4().hex}.{host}"


def fsync_directory(path: Path) -> None:
    descriptor = os.open(path, os.O_RDONLY | os.O_DIRECTORY)
    try:
        os.fsync(descriptor)
    finally:
        os.close(descriptor)


def deliver(args: argparse.Namespace) -> int:
    target = require_maildir(args.to)
    if args.body is not None:
        body = args.body
    else:
        body = Path(args.body_file).read_text(encoding="utf-8")

    message = EmailMessage(policy=policy.SMTP)
    message["Date"] = format_datetime(datetime.now().astimezone())
    message["From"] = args.sender
    message["To"] = args.to
    message["Subject"] = args.subject
    message["Message-ID"] = make_msgid(domain=args.sender.rsplit("@", 1)[-1])
    message.set_content(body)

    name = safe_name()
    temporary = target / "tmp" / name
    delivered = target / "new" / name
    try:
        descriptor = os.open(temporary, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
        with os.fdopen(descriptor, "wb") as stream:
            stream.write(message.as_bytes())
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(temporary, delivered)
        fsync_directory(target / "new")
    except Exception:
        try:
            temporary.unlink()
        except FileNotFoundError:
            pass
        raise
    print(name)
    return 0


def parse_message(path: Path) -> EmailMessage:
    with path.open("rb") as stream:
        return BytesParser(policy=policy.default).parse(stream)


def messages(maildir: Path):
    for state in ("new", "cur"):
        for path in (maildir / state).iterdir():
            # Dot-prefixed names are in-flight writes (Syncthing's
            # .syncthing.*.tmp), never deliverable messages.
            if path.is_file() and not path.name.startswith("."):
                yield state, path


def show_inbox(args: argparse.Namespace) -> int:
    maildir = require_maildir(args.address)
    rows = sorted(messages(maildir), key=lambda item: (item[1].stat().st_mtime_ns, item[1].name))
    for state, path in rows:
        message = parse_message(path)
        fields = (
            state,
            path.name,
            str(message.get("Date", "")),
            str(message.get("From", "")),
            str(message.get("Subject", "")),
        )
        print("\t".join(value.replace("\t", " ").replace("\n", " ") for value in fields))
    return 0


def locate_message(maildir: Path, name: str) -> tuple[str, Path]:
    if not name or name in {".", ".."} or Path(name).name != name:
        raise MailCortexError("message must be an exact inbox filename")
    for state in ("new", "cur"):
        candidate = maildir / state / name
        if candidate.is_file():
            return state, candidate
    raise MailCortexError(f"message not found: {name}")


def read_message(args: argparse.Namespace) -> int:
    maildir = require_maildir(args.address)
    state, path = locate_message(maildir, args.message)
    message = parse_message(path)
    if state == "new":
        seen_name = path.name if ":2," in path.name else path.name + ":2,S"
        destination = maildir / "cur" / seen_name
        if destination.exists():
            raise MailCortexError(f"seen-message destination already exists: {seen_name}")
        os.replace(path, destination)
        fsync_directory(maildir / "new")
        fsync_directory(maildir / "cur")

    for header in ("Date", "From", "To", "Subject", "Message-ID"):
        print(f"{header}: {message.get(header, '')}")
    print()
    body = message.get_body(preferencelist=("plain",))
    print(body.get_content() if body is not None else "", end="")
    return 0


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="mailcortex",
        description="Deliver and read fleet-local RFC 822 messages through Maildir.",
    )
    subparsers = parser.add_subparsers(dest="command", required=True)

    create = subparsers.add_parser("provision", help="create private Maildirs for validated addresses")
    create.add_argument("addresses", nargs="+", type=validate_address)
    create.set_defaults(function=provision)

    send = subparsers.add_parser("send", help="atomically deliver one message")
    send.add_argument("--from", dest="sender", required=True, type=validate_address)
    send.add_argument("--to", required=True, type=validate_address)
    send.add_argument("--subject", required=True)
    body = send.add_mutually_exclusive_group(required=True)
    body.add_argument("--body")
    body.add_argument("--body-file")
    send.set_defaults(function=deliver)

    inbox = subparsers.add_parser("inbox", help="list new and seen messages")
    inbox.add_argument("address", type=validate_address)
    inbox.set_defaults(function=show_inbox)

    read = subparsers.add_parser("read", help="render one message and mark it seen")
    read.add_argument("address", type=validate_address)
    read.add_argument("message")
    read.set_defaults(function=read_message)
    acknowledge = subparsers.add_parser("receipt", help="record a durable handling receipt")
    acknowledge.add_argument("seat", type=validate_component)
    acknowledge.add_argument("message_id")
    acknowledge.add_argument("state", choices=sorted(HANDLING_STATES))
    acknowledge.set_defaults(function=receipt)
    return parser


def validate_component(value: str) -> str:
    if not re.fullmatch(r"[a-z0-9][a-z0-9._-]{0,63}", value):
        raise argparse.ArgumentTypeError("invalid seat")
    return value


def validate_address(value: str) -> str:
    address_maildir(value)
    return value.lower()


def main() -> int:
    parser = build_parser()
    try:
        args = parser.parse_args()
        return args.function(args)
    except (MailCortexError, OSError, ValueError) as error:
        print(f"mailcortex: {error}", file=sys.stderr)
        return 2


if __name__ == "__main__":
    raise SystemExit(main())

* Seat wake watcher

The watcher is a deliberately small, private-runtime consumer.  Its registry
is an operator-provisioned JSON contract, never inferred from mailbox paths or
from a neighbouring host.  A record is eligible only when its exact host,
canonical Maildir, harness, transport, and transport-specific target validate.
Herdr retains its exact target lookup and rest-state rules.  The DeepSeek
adapter uses the harness's authenticated local RPC seam and refuses a running
session before using =session/prompt= in queue mode.  Ductor's local API can
accept a prompt, but exposes no remote busy-status query; its adapter therefore
fails loud until the daemon provides one rather than risking delivery into an
active turn.  Receipts are keyed by a SHA-256 digest of the complete RFC
Message-ID and retain separate wake and handling state.  Busy and rate-limited
deliveries remain queued and are replayed on the next scan; an injection
already started is never retried automatically because its delivery outcome
may be ambiguous.

"""Idle-only, exactly-once MailCortex seat wake watcher."""
from __future__ import annotations

SAFE = re.compile(r"[a-z0-9][a-z0-9._-]{0,63}\Z")
PANE = re.compile(r"w[0-9A-Za-z]+:p[0-9A-Za-z]+\Z")
TRANSPORTS = {"herdr", "ductor", "deepseek"}


class WatcherError(Exception):
    pass


def load_registry(path: Path, host: str) -> dict:
    if path.is_symlink() or not path.is_file() or path.stat().st_mode & 0o077:
        raise WatcherError("registry is not a private regular file")
    try:
        data = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError) as error:
        raise WatcherError("registry is unreadable") from error
    if data.get("schema") != "mailcortex.seats.v1" or data.get("host") != host:
        raise WatcherError("registry is not owned by this host")
    seats = data.get("seats")
    if not isinstance(seats, dict) or not seats:
        raise WatcherError("registry seats must be a non-empty object")
    for seat, record in seats.items():
        if not isinstance(seat, str) or not SAFE.fullmatch(seat) or not isinstance(record, dict):
            raise WatcherError("invalid seat registry entry")
        transport = record.get("transport")
        if transport not in TRANSPORTS:
            raise WatcherError("unsupported seat registry transport")
        required = {
            "host", "mailbox", "harness", "transport", "rate_limit_seconds",
        } | {
            "herdr": {"herdr_session", "herdr_target"},
            "ductor": {"ductor_config"},
            "deepseek": {"deepseek_url", "deepseek_cookie_file", "deepseek_session"},
        }[transport]
        if set(record) != required:
            raise WatcherError("seat registry fields are not exact")
        if (
            record["host"] != host
            or record["mailbox"] not in {f"seats/{seat}", f"consorts/{seat}"}
            or not isinstance(record["harness"], str)
            or record["harness"] not in {"claude", "codex", "ductor", "deepseek"}
        ):
            raise WatcherError("seat registry ownership or transport mismatch")
        if transport == "herdr":
            if record["harness"] not in {"claude", "codex"}:
                raise WatcherError("seat registry ownership or transport mismatch")
            if not all(isinstance(record[key], str) and record[key] and len(record[key]) <= 128 for key in ("herdr_session", "herdr_target")):
                raise WatcherError("invalid Herdr target")
        elif transport == "ductor":
            if record["harness"] != "ductor" or not isinstance(record["ductor_config"], str) or not record["ductor_config"] or len(record["ductor_config"]) > 512:
                raise WatcherError("invalid Ductor configuration path")
        elif record["harness"] != "deepseek":
            raise WatcherError("seat registry ownership or transport mismatch")
        else:
            if not all(isinstance(record[key], str) and record[key] and len(record[key]) <= 512 for key in ("deepseek_url", "deepseek_cookie_file", "deepseek_session")):
                raise WatcherError("invalid DeepSeek target")
        if not isinstance(record["rate_limit_seconds"], (int, float)) or record["rate_limit_seconds"] < 0 or record["rate_limit_seconds"] > 86400:
            raise WatcherError("invalid rate limit")
    return data


def message_id(path: Path) -> str:
    try:
        message = BytesParser().parsebytes(path.read_bytes())
    except (OSError, ValueError):
        raise WatcherError("malformed mail")
    value = str(message.get("Message-ID", "")).strip()
    if not re.fullmatch(r"<[A-Za-z0-9.!#$%&'*+/=?^_`{|}~-]+@[A-Za-z0-9.-]+>", value):
        raise WatcherError("mail has no valid Message-ID")
    return value


def safe_receipt(seat: str, mid: str) -> dict | None:
    try:
        return MAILCORTEX.load_receipt(seat, mid)
    except MAILCORTEX.MailCortexError as error:
        raise WatcherError(str(error)) from error


def latest_injection(seat: str) -> float:
    latest = 0.0
    directory = MAILCORTEX.runtime_root() / "receipts" / seat
    if not directory.is_dir():
        return latest
    for path in directory.glob("*.json"):
        try:
            value = json.loads(path.read_text(encoding="utf-8"))
        except (OSError, json.JSONDecodeError) as error:
            raise WatcherError("receipt is unreadable") from error
        if value.get("schema") != "mailcortex.receipt.v2":
            raise WatcherError("receipt is invalid")
        if value.get("wake_state") == "injected":
            latest = max(latest, float(value.get("wake_updated_at", 0)))
    return latest


class HerdrAdapter:
    def __init__(self, runner=subprocess.run):
        self.runner = runner

    def resolve_idle_pane(self, record: dict) -> str | None:
        command = ["herdr", "--session", record["herdr_session"],
                   "agent", "get", record["herdr_target"]]
        result = self.runner(command, text=True, capture_output=True,
                             check=False, timeout=15)
        if result.returncode != 0:
            raise WatcherError("Herdr target lookup failed")
        try:
            agent = json.loads(result.stdout)["result"]["agent"]
        except json.JSONDecodeError as error:
            raise WatcherError("Herdr target response is not JSON") from error
        except (KeyError, TypeError) as error:
            raise WatcherError("Herdr target response is invalid") from error
        if agent.get("name") != record["herdr_target"]:
            raise WatcherError("Herdr resolved a different target")
        if agent.get("agent_status") not in {"idle", "done"}:
            return None
        pane_id = agent.get("pane_id")
        if not isinstance(pane_id, str) or not PANE.fullmatch(pane_id):
            raise WatcherError("Herdr target has no valid pane")
        return pane_id

    def deliver(self, record: dict, pane_id: str, prompt: str) -> None:
        command = ["herdr", "--session", record["herdr_session"],
                   "pane", "run", pane_id, prompt]
        result = self.runner(command, text=True, capture_output=True,
                             check=False, timeout=60)
        if result.returncode != 0:
            raise WatcherError("Herdr delivery outcome is unknown")


def private_file(path: Path, label: str) -> Path:
    if path.is_symlink() or not path.is_file() or path.stat().st_mode & 0o077:
        raise WatcherError(f"{label} is not a private regular file")
    return path


def loopback_url(value: str, label: str) -> str:
    parsed = urllib.parse.urlsplit(value)
    if parsed.scheme not in {"http", "https"} or parsed.username or parsed.password or not parsed.hostname:
        raise WatcherError(f"{label} is not a loopback URL")
    try:
        loopback = ipaddress.ip_address(parsed.hostname).is_loopback
    except ValueError:
        loopback = parsed.hostname.lower() == "localhost"
    if not loopback:
        raise WatcherError(f"{label} is not a loopback URL")
    return value.rstrip("/")

class DeepSeekAdapter:
    """Authenticated local DeepSeek RPC adapter with authoritative busy checks."""

    def __init__(self, opener=urllib.request.urlopen):
        self.opener = opener

    def _rpc(self, record: dict, method: str, args: dict) -> dict:
        base = loopback_url(record["deepseek_url"], "DeepSeek URL")
        cookie_path = private_file(Path(record["deepseek_cookie_file"]).expanduser(), "DeepSeek cookie")
        cookie = cookie_path.read_text(encoding="utf-8").strip()
        if not cookie:
            raise WatcherError("DeepSeek cookie is empty")
        # Wire shape verified live against dsh's typert gateway: the envelope
        # needs type "client-request", and args are keyed by each method's
        # declared parameter wire name (_request for session/list, request
        # for session/prompt).
        payload = {
            "type": "client-request",
            "rpcId": uuid.uuid4().hex,
            "method": method,
            "payload": {"args": args},
        }
        request = urllib.request.Request(
            f"{base}/api/{method}",
            data=json.dumps(payload, separators=(",", ":")).encode("utf-8"),
            headers={"Content-Type": "application/json", "Cookie": cookie},
            method="POST",
        )
        try:
            with self.opener(request, timeout=15) as response:
                if response.status != 200:
                    raise WatcherError("DeepSeek RPC request failed")
                result = json.loads(response.read().decode("utf-8"))
        except WatcherError:
            raise
        except (OSError, urllib.error.URLError, json.JSONDecodeError) as error:
            raise WatcherError("DeepSeek RPC request failed") from error
        try:
            rpc_result = result["result"]
            if not rpc_result["ok"]:
                raise WatcherError("DeepSeek RPC returned an error")
            value = rpc_result["value"]
        except (KeyError, TypeError) as error:
            raise WatcherError("DeepSeek RPC response is invalid") from error
        if not isinstance(value, dict):
            raise WatcherError("DeepSeek RPC response is invalid")
        return value

    def resolve_idle_pane(self, record: dict) -> str | None:
        value = self._rpc(record, "session/list", {"_request": {}})
        for item in value.get("items", []):
            if not isinstance(item, dict) or item.get("sessionId") != record["deepseek_session"]:
                continue
            if item.get("running") is True:
                return None
            if item.get("running") is not False:
                raise WatcherError("DeepSeek session has no authoritative busy state")
            return record["deepseek_session"]
        raise WatcherError("DeepSeek session was not found")

    def deliver(self, record: dict, target: str, prompt: str) -> None:
        value = self._rpc(
            record,
            "session/prompt",
            {
                "request": {
                    "requestId": uuid.uuid4().hex,
                    "sessionId": target,
                    "mode": "queue",
                    "content": [{"type": "text", "text": prompt}],
                },
            },
        )
        if value.get("accepted") is not True:
            raise WatcherError("DeepSeek prompt was not accepted")


class TransportAdapters:
    def __init__(self):
        self.adapters = {
            "herdr": HerdrAdapter(),
            "ductor": DuctorAdapter(),
            "deepseek": DeepSeekAdapter(),
        }

    def for_record(self, record: dict) -> object:
        try:
            return self.adapters[record["transport"]]
        except KeyError as error:
            raise WatcherError("unsupported transport") from error


def record_adapter(adapter: object, record: dict) -> object:
    factory = getattr(adapter, "for_record", None)
    return factory(record) if factory is not None else adapter


def scan_once(registry: dict, adapter: object, now: float | None = None) -> int:
    now = time.time() if now is None else now
    count = 0
    seen: set[tuple[str, str]] = set()
    for seat, record in registry["seats"].items():
        mailbox_name = record.get("mailbox")
        if mailbox_name not in {f"seats/{seat}", f"consorts/{seat}"}:
            raise WatcherError("seat mailbox is not canonical")
        mailbox = MAILCORTEX.mail_root() / record["mailbox"]
        namespace = mailbox_name.split("/", 1)[0]
        components = (MAILCORTEX.mail_root(), MAILCORTEX.mail_root() / namespace, mailbox)
        if any(item.is_symlink() for item in components) or not all(item.is_dir() for item in components) or not (mailbox / "new").is_dir():
            raise WatcherError("canonical mailbox is absent or unsafe")
        last_injected = latest_injection(seat)
        for path in sorted(mailbox.joinpath("new").iterdir(), key=lambda p: p.name):
            if not path.is_file() or path.is_symlink() or path.name.startswith("."):
                continue
            mid = message_id(path)
            if (seat, mid) in seen:
                continue
            seen.add((seat, mid))
            old = safe_receipt(seat, mid)
            if old and old["wake_state"] in {"injecting", "injected", "delivery_unknown"}:
                continue
            if last_injected and last_injected + record["rate_limit_seconds"] > now:
                MAILCORTEX.write_wake_receipt(seat, mid, "queued", reason="rate_limited",
                                              wake_updated_at=now)
                continue
            MAILCORTEX.write_wake_receipt(seat, mid, "observed", filename=path.name,
                                          wake_updated_at=now)
            transport_adapter = record_adapter(adapter, record)
            pane_id = transport_adapter.resolve_idle_pane(record)
            if pane_id is None:
                MAILCORTEX.write_wake_receipt(seat, mid, "queued", reason="busy",
                                              wake_updated_at=now)
                continue
            MAILCORTEX.write_wake_receipt(seat, mid, "injecting", filename=path.name,
                                          pane_id=pane_id, wake_updated_at=now)
            try:
                transport_adapter.deliver(record, pane_id, "/inbox")
            except Exception as error:
                MAILCORTEX.write_wake_receipt(
                    seat, mid, "delivery_unknown", filename=path.name,
                    pane_id=pane_id, reason="delivery_error", wake_updated_at=now,
                )
                if isinstance(error, WatcherError):
                    raise
                raise WatcherError("transport delivery outcome is unknown") from error
            MAILCORTEX.write_wake_receipt(seat, mid, "injected", filename=path.name,
                                          pane_id=pane_id, wake_updated_at=now)
            count += 1
    return count


def locked_scan(registry: dict, adapter: object, now: float | None = None) -> int:
    lock_path = MAILCORTEX.runtime_root() / "watcher.lock"
    lock_path.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
    with lock_path.open("a+", encoding="utf-8") as lock:
        try:
            fcntl.flock(lock.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
        except BlockingIOError as error:
            raise WatcherError("another watcher owns the scan lock") from error
        try:
            return scan_once(registry, adapter, now=now)
        finally:
            fcntl.flock(lock.fileno(), fcntl.LOCK_UN)


def main() -> int:
    parser = argparse.ArgumentParser(description="Wake owned idle MailCortex seats exactly once.")
    parser.add_argument("--registry", default=os.environ.get("MAILCORTEX_SEAT_REGISTRY", "~/.config/mailcortex/seats.json"))
    parser.add_argument("--once", action="store_true", help="scan once; otherwise poll")
    parser.add_argument("--interval", type=float, default=2.0)
    args = parser.parse_args()
    try:
        registry = load_registry(Path(args.registry).expanduser(), os.environ.get("MAILCORTEX_HOSTNAME", socket.gethostname()))
        while True:
            locked_scan(registry, TransportAdapters())
            if args.once:
                return 0
            time.sleep(max(args.interval, 0.1))
    except (WatcherError, OSError, ValueError) as error:
        print(f"mailcortex-seat-watcher: {error}", file=sys.stderr)
        return 2


if __name__ == "__main__":
    raise SystemExit(main())

** Seat watcher boot owner

The seat watcher has a separate, opt-in boot contract. A systemd host uses a
user unit; a SysV host uses a managed cron entry. Both start the existing
watcher with its default private registry. The launcher validates that registry
before activation and at every start. A missing or invalid registry exits
without a retry loop; the user unit also limits other restart failures. Private
seat names and Herdr targets remain solely in =~/.config/mailcortex/seats.json=.
Systemd activation is declared per approved host by a tracked stow-layer
=default.target.wants= symlink; capability-only hosts omit that link and remain
disabled even though the shared unit and launcher are installed.

private_owner_running() {
  local pid started current
  local -a args
  [ -r "$pid_record" ] || return 1
  read -r pid started < "$pid_record" || return 1
  [[ "$pid" =~ ^[1-9][0-9]*$ && "$started" =~ ^[0-9]+$ ]] || return 1
  [ -r "/proc/$pid/stat" ] && [ -r "/proc/$pid/cmdline" ] || return 1
  [ "$(stat -c %u "/proc/$pid")" = "$(id -u)" ] || return 1
  current="$(awk '{print $22}' "/proc/$pid/stat")"
  [ "$current" = "$started" ] || return 1
  mapfile -d '' -t args < "/proc/$pid/cmdline"
  if [ "${args[0]:-}" = "$watcher" ]; then
    args=("${args[@]:1}")
  elif [ "${args[1]:-}" = "$watcher" ]; then
    args=("${args[@]:2}")
  else
    return 1
  fi
  [ "${args[0]:-}" = --registry ] && [ "${args[1]:-}" = "$registry" ] \
    && [ "${#args[@]}" -eq 2 ]
}

require_ready() {
  [ -f "$ready" ] && [ -d "$HOME/Mail" ] && [ -x "$bin" ] \
    && [ -r "$config/config.xml" ] && [ -r "$config/cert.pem" ] && [ -r "$config/key.pem" ] || {
    echo 'mailcortex-syncthing-service: private readiness marker, Mail directory, identity/config, or Syncthing binary missing' >&2
    exit 1
  }
}

private_owner_running() {
  local pid started current argv
  local -a args
  [ -r "$pid_record" ] || return 1
  read -r pid started < "$pid_record" || return 1
  [[ "$pid" =~ ^[1-9][0-9]*$ && "$started" =~ ^[0-9]+$ ]] || return 1
  [ -r "/proc/$pid/stat" ] && [ -r "/proc/$pid/cmdline" ] || return 1
  [ "$(stat -c %u "/proc/$pid")" = "$(id -u)" ] || return 1
  current="$(awk '{print $22}' "/proc/$pid/stat")"
  [ "$current" = "$started" ] || return 1
  mapfile -d '' -t args < "/proc/$pid/cmdline"
  # A shebang fixture has an interpreter before the binary path.
  if [ "${args[0]:-}" = "$bin" ]; then
    args=("${args[@]:1}")
  elif [ "${args[1]:-}" = "$bin" ]; then
    args=("${args[@]:2}")
  else
    return 1
  fi
  [ "${args[0]:-}" = serve ] && [ "${args[1]:-}" = "--home=$config" ] \
    && [ "${args[2]:-}" = --no-browser ] \
    && [ "${args[3]:-}" = --no-upgrade ] \
    && [ "${args[4]:-}" = --no-restart ] \
    && [ "${args[5]:-}" = --no-default-folder ] \
    && [ "${#args[@]}" -eq 6 ]
}

render_cron() {
  local cron_path="$HOME/.local/bin:/usr/local/bin:/usr/bin:/bin" user_name
  user_name="$(id -un)"
  case "$HOME:$user_name" in *[[:space:]]*) echo 'mailcortex-syncthing-service: invalid cron identity' >&2; exit 2 ;; esac
  printf '# MailCortex Syncthing boot owner; generated by mailcortex-syncthing-service.\n'
  printf 'SHELL=/bin/sh\n'
  printf '@reboot %s HOME=%s PATH=%s %s --run 2>&1 | /usr/bin/logger -t mailcortex-syncthing\n' "$user_name" "$HOME" "$cron_path" "$owner"
  printf '*/3 * * * * %s HOME=%s PATH=%s %s --run 2>&1 | /usr/bin/logger -t mailcortex-syncthing\n' "$user_name" "$HOME" "$cron_path" "$owner"
}

systemd_owner_enabled() {
  local state wants
  state="$(systemctl --user is-enabled "$unit" 2>/dev/null || true)"
  case "$state" in
    enabled) return 0 ;;
    linked)
      wants="$HOME/.config/systemd/user/default.target.wants/$unit"
      [ -L "$wants" ] \
        && [ "$(readlink -f "$wants")" = "$(readlink -f "$fragment")" ]
      ;;
    *) return 1 ;;
  esac
}

  --dry-run) render_cron ;;
  --check)
    require_ready
    render_cron | cmp -s - "$cron_target" || {
      echo "mailcortex-syncthing-service: cron contract missing or drifted: $cron_target" >&2; exit 1;
    }
    private_owner_running || {
      echo 'mailcortex-syncthing-service: private Syncthing owner absent' >&2; exit 1;
    }
    echo 'mailcortex-syncthing-service: SysV cron contract and private daemon verified'
    ;;
  --apply)
    require_ready
    [ -x "$owner" ] || { echo "mailcortex-syncthing-service: missing stowed owner: $owner" >&2; exit 1; }
    command -v flock >/dev/null || { echo 'mailcortex-syncthing-service: flock missing' >&2; exit 1; }
    command -v logger >/dev/null || { echo 'mailcortex-syncthing-service: logger missing' >&2; exit 1; }
    tmp="$(mktemp)"
    trap 'rm -f "$tmp"' EXIT
    render_cron > "$tmp"
    if [ "$cron_target" = /etc/cron.d/mailcortex-syncthing ]; then
      sudo -n install -m 0644 "$tmp" "$cron_target"
    else
      install -m 0644 "$tmp" "$cron_target"
    fi
    echo "mailcortex-syncthing-service: installed $cron_target"
    ;;
  --run)
    require_ready
    umask 077
    mkdir -p -m 0700 "$state"
    exec 9>"$state/syncthing.lock"
    flock -n 9 || exit 0
    if pgrep -u "$(id -u)" -x syncthing >/dev/null; then
      echo 'mailcortex-syncthing-service: another Syncthing owner is running' >&2
      exit 3
    fi
    printf '%s %s\n' "$$" "$(awk '{print $22}' "/proc/$$/stat")" > "$pid_record"
    exec "$bin" serve --home="$config" --no-browser --no-upgrade --no-restart --no-default-folder
    ;;
esac
