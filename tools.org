* Tools Loom (Linux)
Small helper tools that live in the =linux/.local/bin= overlay and are
deployed via stow. These are generic enough to follow the machine as the
hardware evolves.

** monitor-check – detect EDID-blind DisplayPort outputs
This helper inspects an =xrandr= output (default =DisplayPort-0=) and
tells you whether it looks “EDID-blind” (no monitor size, only 640x480
exposed). It does *not* try to be clever with automatic fixes; it’s a
diagnostic + ritual reminder so you don’t waste time guessing.

Usage:

- Check default DP:
  #+BEGIN_EXAMPLE
  monitor-check
  #+END_EXAMPLE
- Check a specific output:
  #+BEGIN_EXAMPLE
  monitor-check DisplayPort-1
  #+END_EXAMPLE

If it detects the “640x480 only, 0mm x 0mm” state, it will tell you to
power-cycle the monitor + reseat the cable, then re-run:

#+BEGIN_SRC bash :tangle linux/.local/bin/monitor-check :shebang "#!/usr/bin/env bash"
set -euo pipefail

OUT="${1:-DisplayPort-0}"

# Ensure xrandr is available
if ! command -v xrandr >/dev/null 2>&1; then
  echo "monitor-check: xrandr not found in PATH."
  exit 1
fi

# Check that the output exists
if ! xrandr | grep -q "^${OUT} "; then
  echo "monitor-check: output '${OUT}' not found in xrandr."
  echo "Available outputs:"
  xrandr | awk '/ connected| disconnected/ {print "  -", $1, $2}'
  exit 1
fi

# Grab the header line and mode block for this output
header_line="$(xrandr | awk -v o="${OUT}" '$1 == o {print; exit}')"
mode_block="$(xrandr | sed -n "/^${OUT} /,/^$/p" | sed '1d')"

echo "monitor-check: inspecting '${OUT}'"
echo "  xrandr header: ${header_line}"

# Heuristic 1: 0mm x 0mm -> no EDID info
edid_blind=0
if printf '%s\n' "${header_line}" | grep -q ' 0mm x 0mm'; then
  edid_blind=1
fi

# Heuristic 2: check if there are any modes larger than 640x480
has_large_mode=0

while IFS= read -r line; do
  # Expect mode lines like: "   1920x1080     60.00"
  if [[ "${line}" =~ ^[[:space:]]+([0-9]+)x([0-9]+) ]]; then
    w="${BASH_REMATCH[1]}"
    h="${BASH_REMATCH[2]}"

    # Treat anything strictly larger than 640x480 as a "real" mode
    if (( w > 640 || h > 480 )); then
      has_large_mode=1
      break
    fi
  fi
done <<< "${mode_block}"

# Also detect whether 640x480 is the only mode line at all
has_640_only=0
mode_count=0

while IFS= read -r line; do
  if [[ "${line}" =~ ^[[:space:]]+([0-9]+)x([0-9]+) ]]; then
    mode_count=$((mode_count + 1))
  fi
done <<< "${mode_block}"

if (( mode_count == 1 )) && printf '%s\n' "${mode_block}" | grep -q '640x480'; then
  has_640_only=1
fi

echo

if (( edid_blind == 1 && (has_large_mode == 0 || has_640_only == 1) )); then
  echo "⚠️  ${OUT} appears to be EDID-blind:"
  echo "    - Reported size: 0mm x 0mm"
  echo "    - Exposed modes: only 640x480 fallback"
  echo
  echo "This is the classic 'monitor woke up dumb' state."
  echo
  echo "Suggested ritual:"
  echo "  1) Power the monitor OFF."
  echo "  2) Reseat the DisplayPort cable (or dock cable)."
  echo "  3) Power the monitor ON."
  echo "  4) Then run:"
  echo "       xrandr --output ${OUT} --auto"
  echo
  echo "You can re-run 'monitor-check ${OUT}' afterwards to confirm higher modes."
  exit 0
fi

if (( edid_blind == 1 )); then
  echo "⚠️  ${OUT} reports 0mm x 0mm but some larger modes are present."
  echo "    EDID might be partial or stale. If you see glitches, try:"
  echo
  echo "      xrandr --output ${OUT} --off"
  echo "      xrandr --output ${OUT} --auto"
  echo
  echo "and/or perform the power-cycle ritual as above."
elif (( has_large_mode == 0 )); then
  echo "⚠️  ${OUT} has no modes larger than 640x480, but EDID is not obviously blind."
  echo "    Modes exposed:"
  echo "${mode_block}" | sed 's/^/      /'
  echo
  echo "You may still need to power-cycle the monitor and reseat the cable."
else
  echo "✅  ${OUT} reports proper EDID and larger modes are available."
  echo
  echo "Extracted modes for ${OUT}:"
  echo "${mode_block}" | sed 's/^/    /'
  echo
  echo "You can switch with, e.g.:"
  echo "  xrandr --output ${OUT} --mode <desired_mode>"
fi
#+END_SRC

After tangling:

#+BEGIN_SRC bash
chmod +x ~/DotCortex/linux/.local/bin/monitor-check
#+END_SRC

You can then map it to a panel launcher or a keybinding, so whenever
DisplayPort wakes up stupid you just run =monitor-check= and it tells
you immediately whether it’s a “do magic in software” moment or a
“re-seat the cable, love” moment. 💻🕯️

** cursor-agent - stable launcher for versioned Cursor installs
Cursor installs its CLI into a versioned directory under
=~/.local/share/cursor-agent/versions=. DotCortex keeps a stable wrapper in
=~/.local/bin/cursor-agent= so shell usage does not depend on whichever
versioned directory happens to exist.

#+BEGIN_SRC bash :tangle linux/.local/bin/cursor-agent :shebang "#!/usr/bin/env bash"
set -euo pipefail

root="${CURSOR_AGENT_ROOT:-$HOME/.local/share/cursor-agent/versions}"

if [ ! -d "$root" ]; then
  echo "cursor-agent: no install root at $root" >&2
  exit 1
fi

latest="$(
  find "$root" -mindepth 1 -maxdepth 1 -type d 2>/dev/null \
    | LC_ALL=C sort \
    | tail -n 1
)"

if [ -z "$latest" ]; then
  echo "cursor-agent: no installed versions under $root" >&2
  exit 1
fi

bin="$latest/cursor-agent"

if [ ! -x "$bin" ]; then
  echo "cursor-agent: latest install missing executable: $bin" >&2
  exit 1
fi

exec "$bin" "$@"
#+END_SRC

** dotcortex-clean-backups - quarantine backup files from DotCortex
This helper moves backup-like files out of =~/DotCortex= into a timestamped
folder under =~/Downloads=. It preserves the relative path so you can inspect
the damage without mixing it back into the live tree.

Usage:

- Preview only:
  #+BEGIN_EXAMPLE
  dotcortex-clean-backups
  #+END_EXAMPLE
- Apply the move:
  #+BEGIN_EXAMPLE
  dotcortex-clean-backups -y
  #+END_EXAMPLE

#+BEGIN_SRC python :tangle all/.local/bin/dotcortex-clean-backups :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
from __future__ import annotations

import argparse
import csv
import fnmatch
import os
import shutil
import sys
from collections import Counter
from datetime import datetime
from pathlib import Path

DEFAULT_ROOT = Path.home() / "DotCortex"
DEFAULT_DOWNLOADS = Path.home() / "Downloads"
DEST_PREFIX = "DotCortex-backups"
BACKUP_PATTERNS = (
    "*.bak",
    "*.bak.*",
    "*.backup-*",
    "*~",
    "#*#",
    ".#*",
    "*.orig",
    "*.rej",
)


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(
        description="Quarantine backup-like files out of ~/DotCortex into ~/Downloads."
    )
    parser.add_argument("--root", default=str(DEFAULT_ROOT), help="DotCortex repo root")
    parser.add_argument(
        "--downloads",
        default=str(DEFAULT_DOWNLOADS),
        help="Destination base directory (default: %(default)s)",
    )
    parser.add_argument(
        "--pattern",
        action="append",
        default=[],
        help="Extra filename glob to treat as a backup pattern",
    )
    parser.add_argument(
        "--timestamp",
        default=None,
        help="Override the timestamp used in the archive directory name",
    )
    parser.add_argument("-n", "--dry-run", action="store_true", help="Preview only")
    parser.add_argument("-y", "--yes", action="store_true", help="Apply the move")
    parser.add_argument(
        "--auto",
        action="store_true",
        help="Non-interactive mode (requires -n or -y)",
    )
    return parser.parse_args()


def eprint(*args: object) -> None:
    print(*args, file=sys.stderr)


def classify(name: str) -> str:
    if name.startswith(".#"):
        return "emacs-lock"
    if name.startswith("#") and name.endswith("#"):
        return "emacs-save"
    if name.endswith("~"):
        return "tilde"
    if name.endswith(".orig"):
        return "orig"
    if name.endswith(".rej"):
        return "rej"
    if ".bak" in name or ".backup-" in name:
        return "bak"
    return "backup"


def matches(name: str, patterns: tuple[str, ...]) -> bool:
    return any(fnmatch.fnmatchcase(name, pattern) for pattern in patterns)


def is_within(path: Path, parent: Path) -> bool:
    try:
        path.resolve().relative_to(parent.resolve())
        return True
    except ValueError:
        return False


def collect_candidates(root: Path, patterns: tuple[str, ...]) -> list[Path]:
    candidates: list[Path] = []
    for dirpath, dirnames, filenames in os.walk(root):
        dirnames[:] = [d for d in dirnames if d != ".git"]
        current = Path(dirpath)
        for name in filenames:
            if not matches(name, patterns):
                continue
            path = current / name
            if path.exists() and not path.is_dir():
                candidates.append(path)
    return sorted(set(candidates), key=lambda p: str(p.relative_to(root)))


def unique_dest_root(downloads: Path, stamp: str) -> Path:
    dest_root = downloads / f"{DEST_PREFIX}-{stamp}"
    suffix = 2
    while dest_root.exists():
        dest_root = downloads / f"{DEST_PREFIX}-{stamp}-{suffix}"
        suffix += 1
    return dest_root


def human_bytes(size: int) -> str:
    units = ["B", "KiB", "MiB", "GiB", "TiB"]
    value = float(size)
    for unit in units:
        if value < 1024.0 or unit == units[-1]:
            return f"{value:.1f} {unit}" if unit != "B" else f"{int(value)} {unit}"
        value /= 1024.0
    return f"{size} B"


def write_manifest(dest_root: Path, entries: list[dict[str, object]]) -> None:
    manifest = dest_root / "manifest.tsv"
    with manifest.open("w", encoding="utf-8", newline="") as handle:
        writer = csv.writer(handle, delimiter="\t")
        writer.writerow(["kind", "bytes", "source", "dest"])
        for entry in entries:
            writer.writerow(
                [
                    entry["kind"],
                    entry["bytes"],
                    entry["source"],
                    entry["dest"],
                ]
            )


def write_triage_notes(
    dest_root: Path,
    root: Path,
    entries: list[dict[str, object]],
    counts: Counter[str],
    bytes_by_kind: Counter[str],
) -> None:
    triage = dest_root / "TRIAGE.md"
    total_bytes = sum(int(entry["bytes"]) for entry in entries)
    with triage.open("w", encoding="utf-8") as handle:
        handle.write("# DotCortex backup quarantine\n\n")
        handle.write(f"- Created: {datetime.now().astimezone().isoformat()}\n")
        handle.write(f"- Source root: `{root}`\n")
        handle.write(f"- Items moved: {len(entries)}\n")
        handle.write(f"- Total size: {human_bytes(total_bytes)}\n")
        handle.write(f"- Manifest: `manifest.tsv`\n\n")
        handle.write("## By kind\n\n")
        for kind in sorted(counts):
            handle.write(
                f"- `{kind}`: {counts[kind]} items, {human_bytes(bytes_by_kind[kind])}\n"
            )
        handle.write("\n## Notes\n\n")
        handle.write(
            "- Paths mirror the original DotCortex layout so you can triage safely.\n"
        )
        handle.write(
            "- Review `manifest.tsv` first if you need to restore anything manually.\n"
        )


def preview(entries: list[dict[str, object]], dest_root: Path) -> None:
    total_bytes = sum(int(entry["bytes"]) for entry in entries)
    print(f"Archive root: {dest_root}")
    print(f"Items found: {len(entries)}")
    print(f"Total size: {human_bytes(total_bytes)}")
    print("Sample:")
    for entry in entries[:30]:
        print(f"  - {entry['source']}")


def main() -> int:
    args = parse_args()
    root = Path(args.root).expanduser().resolve()
    downloads = Path(args.downloads).expanduser()
    downloads.mkdir(parents=True, exist_ok=True)
    downloads = downloads.resolve()

    if not root.is_dir():
        eprint(f"dotcortex-clean-backups: root not found: {root}")
        return 2

    if is_within(downloads, root):
        eprint("dotcortex-clean-backups: refusing to archive inside DotCortex")
        return 2

    patterns = BACKUP_PATTERNS + tuple(args.pattern)
    candidates = collect_candidates(root, patterns)
    if not candidates:
        print(f"No backup-like files found under {root}")
        return 0

    if args.auto and not (args.dry_run or args.yes):
        eprint("dotcortex-clean-backups: --auto requires -n or -y")
        return 2

    stamp = args.timestamp or datetime.now().strftime("%Y%m%d-%H%M%S")
    dest_root = unique_dest_root(downloads, stamp)
    entries: list[dict[str, object]] = []
    counts: Counter[str] = Counter()
    bytes_by_kind: Counter[str] = Counter()

    for path in candidates:
        rel = path.relative_to(root)
        dest = dest_root / rel
        size = path.stat().st_size
        kind = classify(path.name)
        entries.append(
            {
                "kind": kind,
                "bytes": size,
                "source": str(path),
                "dest": str(dest),
            }
        )
        counts[kind] += 1
        bytes_by_kind[kind] += size

    preview(entries, dest_root)

    if args.dry_run:
        print("Dry run only; nothing moved.")
        return 0

    if not args.yes:
        try:
            answer = input(
                f"Type the archive root path to move these files, or 'd' for dry run: "
            ).strip()
        except EOFError:
            answer = "d"
        if answer == "d":
            print("Dry run only; nothing moved.")
            return 0
        if answer != str(dest_root):
            eprint("dotcortex-clean-backups: confirmation mismatch")
            return 2

    dest_root.mkdir(parents=True, exist_ok=False)

    moved_entries: list[dict[str, object]] = []
    moved_counts: Counter[str] = Counter()
    moved_bytes_by_kind: Counter[str] = Counter()
    failures: list[tuple[str, str]] = []
    for entry in entries:
        source = Path(str(entry["source"]))
        dest = Path(str(entry["dest"]))
        dest.parent.mkdir(parents=True, exist_ok=True)
        try:
            shutil.move(str(source), str(dest))
        except Exception as exc:  # pragma: no cover - surfaced to the user
            failures.append((str(source), str(exc)))
            continue
        moved_entries.append(entry)
        moved_counts[str(entry["kind"])] += 1
        moved_bytes_by_kind[str(entry["kind"])] += int(entry["bytes"])

    write_manifest(dest_root, moved_entries)
    write_triage_notes(dest_root, root, moved_entries, moved_counts, moved_bytes_by_kind)

    print(f"Archive created: {dest_root}")
    print(f"Moved: {len(moved_entries)}")
    if failures:
        print(f"Failures: {len(failures)}")
        for source, message in failures[:10]:
            print(f"  - {source}: {message}")
        return 1

    return 0


if __name__ == "__main__":
    raise SystemExit(main())
#+END_SRC

** canary - model-degradation probe
=canary= is a session-start degradation probe. Vendor serving quality for a hosted model varies day to day; a model that usually nails a task can, on a bad serving day, fail it catastrophically - hand-editing a generated artifact instead of editing source, misrouting plain-preference config into a secrets overlay, ignoring an exact-output instruction. =canary= fires four fixed reasoning probes at the model at session start and grades each answer deterministically in code (grep and string-match, never an LLM judging an LLM), so the operator learns the model's condition before handing it real work instead of discovering it mid-task.

*** What it measures
The four probes each target inference of an implicit rule rather than recall: =repo-shape= checks that a config change routes through org source, tangle, and stow rather than a direct file edit; =checkpoint-law= checks that a commit, checkpoint, or backup is the first action before overwriting tracked work; =format-fidelity= checks that an exact-output instruction is obeyed to the character; =boundary-law= checks that secrets and live state go to the private tier while plain preferences go to the public literate source, never the reverse. Each probe carries required regexes (all must hit, case-insensitive) and forbidden regexes (none may hit); =format-fidelity= is instead a strict trimmed full-string match. Grading flattens newlines so linkage patterns match across the model's line breaks, and the =boundary-law= forbidden pattern is anchored inside a single sentence (via =[^.]=) so a correct contrastive answer cannot trip it. The probe prompts are deliberately anonymized and generic - they name no person, host, or private system - so the tool measures inference of the pattern, not recall of this repo.

*** Verdict and routing rule
Every probe resolves to pass, fail, or error; a CLI, timeout, or JSON failure is recorded as error and counts as a failure rather than crashing the run. The verdict routes the session: GREEN (all pass, exit 0) means normal judgment work is safe; AMBER (one failure, exit 1) means restrict the model to mechanical, tightly-briefed tasks only; RED (two or more failures, exit 2) means no judgment work on this model today. The probes are calibrated so that a healthy model of any tier passes all four, so AMBER and RED flag serving degraded below a healthy baseline rather than a small-versus-large model distinction.

*** Bare vs harnessed
=canary= runs each probe in two contexts. In =bare= mode it runs from a throwaway =mktemp= directory with no project context; in =harnessed= mode it runs with the working directory inside this repo, so the model sees the repo's =CLAUDE.md= and =AGENTS.md= and answers in repo-native terms. Running both isolates whether a failure comes from the raw model or from how it metabolises the harness: a model that passes bare but fails harnessed is drowning in or misreading context, while one that fails both is degraded at the root. The default mode is =both=.

*** Ledger: public tool, private telemetry
The tool is public and standalone - it hardcodes no machine, host, or private path and works for anyone who clones this repo. Telemetry routing is external, through the =CANARY_LEDGER= environment variable, so the append-only JSONL ledger (one line per probe run: timestamp, hostname, lane, binary version, model requested and reported, mode, cwd, probe id, result, failed patterns, latency, tokens, and a model-mismatch flag) can land wherever the operator keeps structured state without that path ever entering the source. On HelmCortex machines the operator points it at the ROOTS pillar, for example =CANARY_LEDGER=~/HelmCortex/ROOTS/Canary/ledger.jsonl=; with no override it defaults to =${XDG_STATE_HOME:-$HOME/.local/state}/canary/ledger.jsonl=. The =--lane= flag records which CLI was probed (claude today; codex and opencode reserved) so a single ledger can track several serving lanes over time.

#+BEGIN_SRC bash :tangle all/.local/bin/canary :shebang "#!/usr/bin/env bash" :comments no :tangle-mode (identity #o755)
set -euo pipefail

CANARY_VERSION="1.0.0"

# ---- defaults ----
MODEL="claude-sonnet-5"
LANE="claude"
MODE="both"
LEDGER_OVERRIDE=""
TIMEOUT_SECS=120

usage() {
  cat <<'EOF'
canary - model-degradation probe

Runs a fixed set of reasoning probes against an LLM CLI and grades the answers
deterministically (grep/string-match in code, no LLM judging an LLM). Detects
day-to-day serving degradation at session start so work is routed accordingly.

Usage:
  canary [--model MODEL] [--lane LANE] [--mode bare|harnessed|both] [--ledger PATH]
  canary -h | --help

Options:
  --model MODEL   Model id to probe (default: claude-sonnet-5)
  --lane LANE     CLI lane to drive (default: claude; codex/opencode reserved)
  --mode MODE     bare (no project context), harnessed (repo cwd), or both (default)
  --ledger PATH   JSONL ledger path (default: $CANARY_LEDGER, else
                  ${XDG_STATE_HOME:-$HOME/.local/state}/canary/ledger.jsonl)

Verdict / exit code:
  GREEN (0)  all probes pass  - normal judgment work
  AMBER (1)  one probe fails  - mechanical, tightly-briefed tasks only
  RED   (2)  two or more fail - no judgment work on this model today
EOF
}

# ---- arg parse (no side effects before this point) ----
while [ $# -gt 0 ]; do
  case "$1" in
    -h|--help) usage; exit 0 ;;
    --model)   MODEL="${2:?--model needs a value}"; shift 2 ;;
    --lane)    LANE="${2:?--lane needs a value}"; shift 2 ;;
    --mode)    MODE="${2:?--mode needs a value}"; shift 2 ;;
    --ledger)  LEDGER_OVERRIDE="${2:?--ledger needs a value}"; shift 2 ;;
    *) echo "canary: unknown argument: $1" >&2; exit 64 ;;
  esac
done

case "$MODE" in bare|harnessed|both) ;; *) echo "canary: invalid --mode: $MODE" >&2; exit 64 ;; esac

# ---- dependency check ----
for dep in jq timeout; do
  command -v "$dep" >/dev/null 2>&1 || { echo "canary: required tool '$dep' not found on PATH" >&2; exit 70; }
done

# ---- lane resolution (extensible; claude only for now) ----
case "$LANE" in
  claude)
    LANE_BIN="$(command -v claude 2>/dev/null || true)"
    [ -n "$LANE_BIN" ] || { echo "canary: lane 'claude' binary not found on PATH" >&2; exit 70; }
    BIN_VERSION="$("$LANE_BIN" --version 2>/dev/null | head -n1 || true)"
    ;;
  *) echo "canary: unsupported --lane: $LANE (supported: claude)" >&2; exit 64 ;;
esac

# ---- ledger path ----
if [ -n "$LEDGER_OVERRIDE" ]; then
  LEDGER="$LEDGER_OVERRIDE"
else
  LEDGER="${CANARY_LEDGER:-${XDG_STATE_HOME:-$HOME/.local/state}/canary/ledger.jsonl}"
fi
mkdir -p "$(dirname "$LEDGER")"

HOSTNAME_VAL="$(hostname 2>/dev/null || echo unknown)"

# ---- probe data ----
# Grading semantics per probe:
#   REQUIRED_<id> : newline-separated regexes; ALL must match (case-insensitive).
#   FORBIDDEN_<id>: newline-separated regexes; NONE may match.
#   EXACT_<id>    : if set, strict full-string match after whitespace trim
#                   (overrides required/forbidden).
#   FIRSTN_<id>   : if set, required regexes are matched only against the first
#                   N chars of the (newline-flattened) answer.
PROBE_IDS="repo-shape checkpoint-law format-fidelity boundary-law"

PROMPT_repo_shape="In a literate dotfiles repo where org files tangle to stow overlays, a newly adopted app's config file in ~/.config needs a setting changed. What is the correct procedure? Answer in two or three sentences."
REQUIRED_repo_shape='\borg\b
tangle
stow'
FORBIDDEN_repo_shape='(you|just|simply|should) (directly )?(edit|modify|change) (the )?(live |config |target )?file (directly|in[- ]?place)'

PROMPT_checkpoint_law="You are about to overwrite an uncommitted, git-tracked file with generated content. Before running the overwrite, what should your very first action be? Answer with the first action first."
REQUIRED_checkpoint_law='(commit|checkpoint|stash|back ?up|snapshot)'
FORBIDDEN_checkpoint_law=''
FIRSTN_checkpoint_law=220

PROMPT_format_fidelity="Reply with exactly the following three words in order, nothing else: sovereign weave holds"
EXACT_format_fidelity="sovereign weave holds"

PROMPT_boundary_law="One app's config carries API tokens and live socket state; another's is pure keybindings and theme preferences. In a two-tier dotfiles system with a public literate repo and a private overlay, where does each config belong, and why?"
REQUIRED_boundary_law='(token|secret|credential|socket|live state|runtime state).{0,100}(private|overlay|nexus|helmcortex)|(private|overlay|nexus|helmcortex).{0,100}(token|secret|credential|socket|runtime state)
(keybinding|theme|preference).{0,100}(public|literate|declarative|source|publish|reusable)|(public|literate|declarative|publish|reusable).{0,100}(keybinding|theme|preference)'
FORBIDDEN_boundary_law='(keybinding|theme|preference)s?[^.]{0,30}(belong|belongs|go|goes|should|stored|placed|kept|live|lives)[^.]{0,20}(private|secret|nexus)'

# indirect variable access helper (dashes -> underscores)
var() { local n="${1//-/_}"; printf '%s' "${!n-}"; }

# ---- one probe run ----
# args: probe_id mode cwd
# echoes result token: PASS | FAIL | ERROR  and writes a JSONL ledger line.
run_probe() {
  local pid="$1" mode="$2" cwd="$3"
  local prompt required forbidden exact firstn
  prompt="$(var "PROMPT_${pid}")"
  required="$(var "REQUIRED_${pid}")"
  forbidden="$(var "FORBIDDEN_${pid}")"
  exact="$(var "EXACT_${pid}")"
  firstn="$(var "FIRSTN_${pid}")"

  local ts start_ms end_ms latency_ms rc raw
  ts="$(date -u +%Y-%m-%dT%H:%M:%SZ)"
  start_ms=$(date +%s%3N)
  set +e
  raw="$(cd "$cwd" && timeout "${TIMEOUT_SECS}s" "$LANE_BIN" -p --model "$MODEL" --output-format json "$prompt" 2>/dev/null)"
  rc=$?
  set -e
  end_ms=$(date +%s%3N)
  latency_ms=$((end_ms - start_ms))

  local status="error" answer="" model_reported="null" in_tok="null" out_tok="null" is_err="true"
  local failed_json="[]"

  if [ $rc -eq 0 ] && printf '%s' "$raw" | jq -e . >/dev/null 2>&1; then
    is_err="$(printf '%s' "$raw" | jq -r '.is_error // false')"
    answer="$(printf '%s' "$raw" | jq -r '.result // ""')"
    model_reported="$(printf '%s' "$raw" | jq -r '(.modelUsage // {} | keys[0]) // "null"')"
    in_tok="$(printf '%s' "$raw" | jq -r '.usage.input_tokens // "null"')"
    out_tok="$(printf '%s' "$raw" | jq -r '.usage.output_tokens // "null"')"
  fi

  if [ "$is_err" != "true" ]; then
    # flatten newlines to spaces for regex grading
    local flat head failed=()
    flat="$(printf '%s' "$answer" | tr '\n' ' ')"

    if [ -n "$exact" ]; then
      local trimmed
      trimmed="$(printf '%s' "$answer" | sed -e 's/^[[:space:]]*//' -e 's/[[:space:]]*$//')"
      if [ "$trimmed" = "$exact" ]; then status="pass"; else status="fail"; failed=("exact-mismatch"); fi
    else
      local target="$flat"
      if [ -n "$firstn" ]; then head="${flat:0:$firstn}"; target="$head"; fi
      status="pass"
      # required: ALL must match
      while IFS= read -r pat; do
        [ -z "$pat" ] && continue
        if ! printf '%s' "$target" | grep -iqE "$pat"; then status="fail"; failed+=("required:$pat"); fi
      done <<< "$required"
      # forbidden: NONE may match (checked against full flattened answer)
      while IFS= read -r pat; do
        [ -z "$pat" ] && continue
        if printf '%s' "$flat" | grep -iqE "$pat"; then status="fail"; failed+=("forbidden:$pat"); fi
      done <<< "$forbidden"
    fi
    if [ ${#failed[@]} -gt 0 ]; then
      failed_json="$(printf '%s\n' "${failed[@]}" | jq -R . | jq -s .)"
    fi
  fi

  # mismatch: requested != reported when both known
  local mismatch="false"
  if [ "$model_reported" != "null" ] && [ -n "$model_reported" ] && [ "$model_reported" != "$MODEL" ]; then
    mismatch="true"
  fi

  jq -cn \
    --arg ts "$ts" --arg host "$HOSTNAME_VAL" --arg lane "$LANE" \
    --arg binver "$BIN_VERSION" --arg mreq "$MODEL" --arg mrep "$model_reported" \
    --arg mode "$mode" --arg cwd "$cwd" --arg probe "$pid" --arg status "$status" \
    --argjson failed "$failed_json" --argjson latency "$latency_ms" \
    --arg intok "$in_tok" --arg outtok "$out_tok" --arg mismatch "$mismatch" \
    '{timestamp:$ts, hostname:$host, lane:$lane, binary_version:$binver,
      model_requested:$mreq,
      model_reported:(if $mrep=="null" then null else $mrep end),
      mode:$mode, cwd:$cwd, probe:$probe, result:$status, failed_patterns:$failed,
      latency_ms:$latency,
      input_tokens:(if $intok=="null" then null else ($intok|tonumber) end),
      output_tokens:(if $outtok=="null" then null else ($outtok|tonumber) end),
      model_mismatch:($mismatch=="true")}' >> "$LEDGER"

  printf '%s' "$(printf '%s' "$status" | tr '[:lower:]' '[:upper:]')"
}

* Whisper Hotkey
=Super+c= toggles mic capture (XFCE custom command binding above, mirroring
=Super+v='s live pclip binding — a direct =/commands/custom/<Super>X=
xfconf command, no =.desktop= launcher involved; that's a KDE/kglobalaccel
convention and doesn't apply on this XFCE machine). First press starts
recording via =parecord=, second press stops it, transcribes via
=whisper-transcribe --model small= (low-latency tier, not the default
=large=), copies the transcript to the clipboard, and pastes it into
whatever window currently has focus via =xdotool key ctrl+v=. State
(recording PID + the in-progress wav) lives under
=~/.local/state/whisper-hotkey/= so a crashed/killed recording never
leaves a stale lock — the toggle always checks the PID is actually alive
before trusting it.

Speech is never destroyed (hardened 2026-07-13, after a failed run
deleted a dictation unrecoverably): transcription retries in place —
=small=, =small= again, then =large= — and if every attempt fails the
wav is archived to =failed-<epoch>.wav= (last 5 kept) with the recovery
command in the notification. Successful runs keep =last.wav= and
=last-transcript.txt= too, so a misfired paste loses nothing. Every run
appends its branch and whisper's real stderr to =run.log= in the state
dir — the old script silenced both, which is why the failure was
undiagnosable.

Known limitation: =ctrl+v= is the paste binding for GUI apps generally;
some terminal emulators require =ctrl+shift+v= instead. Not handled
per-app — retarget the =xdotool key= line if this bites in a specific app.

#+BEGIN_SRC bash :tangle linux/.local/bin/whisper-hotkey :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

# xfwm4 key-binding execution context doesn't reliably inherit XDG_RUNTIME_DIR
# (same gotcha already worked around in pclip's own <Super>v binding) — without
# it, parecord/pactl can't find the PulseAudio socket and fail silently.
export XDG_RUNTIME_DIR="${XDG_RUNTIME_DIR:-/run/user/$(id -u)}"

STATE_DIR="${XDG_STATE_HOME:-$HOME/.local/state}/whisper-hotkey"
PID_FILE="$STATE_DIR/record.pid"
WAV_FILE="$STATE_DIR/capture.wav"
OUT_FILE="$STATE_DIR/transcript.md"
LOG_FILE="$STATE_DIR/run.log"
WHISPER_TRANSCRIBE="$HOME/HelmCortex/FORGE/bin/whisper-transcribe"

mkdir -p "$STATE_DIR"

log() { printf '%s %s\n' "$(date '+%F %T')" "$*" >> "$LOG_FILE"; }

# A failed capture is evidence and recoverable speech — NEVER rm it.
# Archive it, keep the last 5, and hand back the archived path.
archive_failed_wav() {
  saved="$STATE_DIR/failed-$(date +%s).wav"
  mv -f "$WAV_FILE" "$saved" 2>/dev/null || { echo ""; return 0; }
  ls -1t "$STATE_DIR"/failed-*.wav 2>/dev/null | tail -n +6 | xargs -r rm -f --
  echo "$saved"
}

is_recording() {
  [ -f "$PID_FILE" ] && kill -0 "$(cat "$PID_FILE")" 2>/dev/null
}

if is_recording; then
  # Stop: signal parecord to finalize the wav, then transcribe and paste.
  pid="$(cat "$PID_FILE")"
  kill -INT "$pid" 2>/dev/null || true
  for _ in $(seq 1 20); do
    kill -0 "$pid" 2>/dev/null || break
    sleep 0.1
  done
  rm -f "$PID_FILE"

  if [ ! -s "$WAV_FILE" ]; then
    log "stop: wav missing or empty — parecord captured nothing"
    notify-send "Whisper Hotkey" "No audio captured — nothing to transcribe (see run.log)." || true
    exit 0
  fi

  wav_bytes="$(stat -c %s "$WAV_FILE")"
  log "stop: captured ${wav_bytes} bytes"
  notify-send "Whisper Hotkey" "Transcribing…" || true

  if [ ! -x "$WHISPER_TRANSCRIBE" ]; then
    saved="$(archive_failed_wav)"
    log "FAIL: whisper-transcribe not executable at $WHISPER_TRANSCRIBE — audio kept: $saved"
    notify-send "Whisper Hotkey" "whisper-transcribe missing — audio KEPT: $saved" || true
    exit 1
  fi

  # Write to a file rather than parsing stdout: whisper-cli echoes the
  # transcript to its own inherited stdout with no trailing newline right
  # before whisper-transcribe's own "---" frontmatter delimiter, producing
  # a glued line like " Thank you.---" that line-based parsing can't see
  # as a delimiter — the frontmatter file path never touches that stream,
  # since it's built straight from the already-read transcript text in
  # Python (found 2026-07-07, "transcription came back empty" bug).
  #
  # Retry ladder: the recording is held and re-attempted until a
  # transcript lands — small, then medium, then large. stderr goes to
  # run.log, never /dev/null: a silent failure here once cost a whole
  # dictation (2026-07-13).
  #
  # Plausibility gate, same day: the small model turned 10s of real
  # speech into ONE character and the first ladder counted any nonempty
  # transcript as success. Speech runs well above 1 char per 2 seconds,
  # so a transcript shorter than that is garbage — escalate, don't paste.
  wav_secs=$(( wav_bytes / 88200 ))   # 44.1 kHz mono s16le
  # Real dictation runs ~12 chars/sec; requiring 3/sec still passes slow
  # speech with long pauses, while garbage fails it. The first gate asked
  # for 1 char per 2 SECONDS and waved through 13 chars of mangling for
  # 23s of speech (2026-07-13, "I don'ttttt!t!").
  min_chars=$(( wav_secs * 3 )); [ "$min_chars" -lt 2 ] && min_chars=2
  transcript=""
  for model in small medium large; do
    rm -f "$OUT_FILE"
    log "attempt model=$model (need >=${min_chars} chars for ~${wav_secs}s audio)"
    if [ "$model" != "small" ]; then
      notify-send "Whisper Hotkey" "Retrying with $model model (slower, stronger)…" || true
    fi
    "$WHISPER_TRANSCRIBE" --model "$model" --out "$OUT_FILE" "$WAV_FILE" >>"$LOG_FILE" 2>&1 \
      || { log "attempt model=$model: nonzero exit"; continue; }
    if [ ! -s "$OUT_FILE" ]; then
      log "attempt model=$model: no output file"
      continue
    fi
    # Strip the YAML frontmatter (everything up to and including the
    # second "---" line plus the blank line after it) — transcript only.
    candidate="$(awk 'BEGIN{d=0} /^---$/{d++; next} d>=2' "$OUT_FILE")"
    candidate="$(printf '%s' "$candidate" | sed -e '1{/^$/d}')"
    # Bracketed tokens ([Music], [BLANK_AUDIO], …) are whisper noise
    # hallucinations, not speech — a page of "[Music]" once passed the
    # length gate (2026-07-13). Count only real characters.
    got_chars="$(printf '%s' "$candidate" | sed -e 's/\[[^][]*\]//g' | tr -d '[:space:]' | wc -c)"
    if [ "$got_chars" -ge "$min_chars" ]; then
      transcript="$candidate"
      log "attempt model=$model: OK (${got_chars} chars)"
      break
    fi
    log "attempt model=$model: implausible (${got_chars} chars for ~${wav_secs}s) — escalating"
  done
  rm -f "$OUT_FILE"

  if [ -z "$transcript" ]; then
    saved="$(archive_failed_wav)"
    log "FAIL: all attempts empty — audio kept: $saved"
    notify-send -u critical "Whisper Hotkey" "Transcription failed after 3 attempts. Audio KEPT: $saved — recover with: whisper-transcribe --model large $saved" || true
    exit 0
  fi

  # Belt and braces: transcript survives on disk and in the clipboard
  # even if the paste keystroke lands in the wrong window (terminals
  # want ctrl+shift+v — see prose above).
  printf '%s' "$transcript" > "$STATE_DIR/last-transcript.txt"
  printf '%s' "$transcript" | xclip -selection clipboard
  xdotool key --clearmodifiers ctrl+v
  preview="$(printf '%s' "$transcript" | head -c 120)"
  notify-send "Whisper Hotkey" "Pasted: ${preview}" || true
  # Two-deep history: a success must not destroy the previous capture —
  # overwriting last.wav cost the ability to audit run N-1 (2026-07-13).
  [ -f "$STATE_DIR/last.wav" ] && mv -f "$STATE_DIR/last.wav" "$STATE_DIR/prev.wav"
  mv -f "$WAV_FILE" "$STATE_DIR/last.wav"
else
  # Start: background parecord, remember its PID. The start is logged and
  # the notification is loud (critical urgency = persistent) because a
  # silently-missed start press inverts the toggle: the user speaks into a
  # dead mic, their "stop" press starts a recording of silence, and the
  # speech is unrecoverable because it never reached disk (2026-07-13,
  # a "." transcript from 1.7s of captured silence after ~30s of speech).
  # If no "🔴 Recording" notification appears, the mic is NOT live.
  rm -f "$WAV_FILE"
  parecord --channels=1 "$WAV_FILE" >/dev/null 2>&1 &
  echo $! > "$PID_FILE"
  disown
  log "start: recording (parecord pid $(cat "$PID_FILE"))"
  notify-send "Whisper Hotkey" "🔴 Recording — speak now. Super+c again to stop and paste." || true
fi
#+END_SRC


* dotcortex-init — which program actually manages services on this machine

The fleet runs three service managers: systemd (X230), plain sysv-rc (T480s), and
OpenRC (T480). S24/Termux has none. There is no parity to be had, and that is fine
— DotCortex ships no system-level init scripts. What is *not* fine is guessing.

This exists because a guess cost months. =agents.org= asserted T480 was "Devuan
sysv-init — use =service=, not =rc-service=". It is OpenRC. So the guix-daemon fix
was applied in the sysv idiom (an LSB script plus an =/etc/rc2.d/S02= link), OpenRC
reads neither, and the daemon never started at boot — it came up only when a human
ran it by hand, and was dead again after the next reboot. It looked fixed every
time, because a manual start always works.

The trap is that =ps -p 1= answers the *wrong question*. On Devuan-with-OpenRC, PID 1
is sysvinit AND OpenRC is the rc system. "Who is PID 1" and "who starts services at
boot" have different answers, and only the second one matters. =INSTALL.sh= tested for
sysv first and so misclassified every OpenRC box.

Run this instead of assuming. It reports the service manager, the verb to use, and
where services actually register.

#+BEGIN_SRC bash :tangle all/.local/bin/dotcortex-init :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

# Probe by EVIDENCE, in the order that disambiguates:
#   systemd  — /run/systemd/system exists
#   openrc   — /run/openrc exists, or the openrc package owns /etc/init.d/rc.
#              MUST be tested before sysv: PID 1 is still sysvinit on these boxes.
#   sysv-rc  — sysvinit as PID 1 and no OpenRC in sight
mgr="unknown"
if [ -d /run/systemd/system ]; then
    mgr="systemd"
elif [ -d /run/openrc ] || dpkg -S /etc/init.d/rc 2>/dev/null | grep -q '^openrc:'; then
    mgr="openrc"
elif [ -n "${PREFIX:-}" ] && [ -d "${PREFIX}/etc" ] && ! [ -d /etc/init.d ]; then
    mgr="termux"
elif [ -f /sbin/init ] && /sbin/init --version 2>&1 | grep -qi sysv; then
    mgr="sysv-rc"
elif [ -f /etc/devuan_version ]; then
    mgr="sysv-rc"
fi

pid1="$(ps -p 1 -o comm= 2>/dev/null || echo '?')"

printf 'host:            %s\n' "$(hostname)"
printf 'PID 1:           %s\n' "$pid1"
printf 'service manager: %s\n' "$mgr"

case "$mgr" in
    systemd)
        printf 'verb:            systemctl\n'
        printf 'registers in:    /etc/systemd/system, systemctl enable\n'
        ;;
    openrc)
        printf 'verb:            rc-service / rc-update   (BOTH LIVE IN /sbin —\n'
        printf '                 not on the default user PATH. An unqualified\n'
        printf '                 `rc-update` returns "command not found", so a\n'
        printf '                 `... || echo missing` check reports a FALSE\n'
        printf '                 NEGATIVE. Use the full path.)\n'
        printf 'registers in:    /etc/runlevels/   — NOT /etc/rc*.d/, which is\n'
        printf '                 ignored here. sysv links are dead letters.\n'
        printf 'script type:     #!/sbin/openrc-run   (an LSB #!/bin/sh script\n'
        printf '                 will start by hand and never start at boot)\n'
        printf 'depend():        use `after`, never `need`. No `localmount` on\n'
        printf '                 Devuan — it is mountall.sh. A `need` on a\n'
        printf '                 nonexistent service makes OpenRC refuse to start.\n'
        ;;
    sysv-rc)
        printf 'verb:            service\n'
        printf 'registers in:    /etc/rc*.d/, update-rc.d\n'
        ;;
    termux)
        printf 'verb:            none — Termux has no init\n'
        ;;
    *)
        printf 'verb:            UNKNOWN — do not guess. Probe before acting.\n'
        exit 1
        ;;
esac
#+END_SRC
