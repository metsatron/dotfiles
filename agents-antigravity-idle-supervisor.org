* Design notes

This launcher is the owner for interactive Antigravity CLI processes. It uses Python's =pty.fork= so the Antigravity child receives a real controlling terminal while the parent owns the PTY master for the entire process lifetime. The parent proxies bytes in both directions, propagates terminal geometry and signals, and is the only component permitted to inject supervisor-controlled submissions. Hook helpers communicate with the per-session Unix socket; they never write to a terminal.

This follows the established =codex-warm= and =claude-warm= architecture in DotCortex:
- Channel prefix: =antigravity-= (e.g. =antigravity-<32-hex-uuid>=)
- Environment namespace: =ANTIGRAVITY_IDLE_*=
- Runtime directory: =$XDG_RUNTIME_DIR/antigravity-idle-compaction/<channel>=
- Herdr integration: agent label =Antigravity=, socket-based heartbeat and state reporting
- Hook integration: =antigravity-hook-idle-event= consumes Antigravity's =hooks.json= events and forwards validated lifecycle signals to the control socket.

def write_all(fd: int, data: bytes) -> None:
    view = memoryview(data)
    while view:
        written = os.write(fd, view)
        view = view[written:]


def child_alive(pid: int | None) -> bool:
    if not pid:
        return False
    try:
        os.kill(pid, 0)
        return True
    except OSError:
        return False


def set_terminal_raw(fd: int) -> list[object]:
    original = termios.tcgetattr(fd)
    tty.setraw(fd)
    return original


def restore_terminal(fd: int, attributes: list[object] | None) -> None:
    if attributes is not None:
        try:
            termios.tcsetattr(fd, termios.TCSADRAIN, attributes)
        except OSError:
            pass


def get_terminal_size(fd: int) -> tuple[int, int]:
    try:
        raw = fcntl.ioctl(fd, termios.TIOCGWINSZ, b"\0" * 8)
        rows, columns = struct.unpack("HH", raw[:4])
        return (rows or 24, columns or 80)
    except OSError:
        return (24, 80)


def set_terminal_size(fd: int, rows: int, columns: int) -> None:
    try:
        raw = struct.pack("HHHH", max(1, rows), max(1, columns), 0, 0)
        fcntl.ioctl(fd, termios.TIOCSWINSZ, raw)
    except OSError:
        pass


def shell_exit_code(return_value: object) -> int:
    if return_value is None:
        return 0
    if isinstance(return_value, int):
        return return_value
    return 1

class TerminalInputClassifier:
    """Classify terminal input bytes without consuming or rewriting them."""

    DEVICE_ATTRIBUTES_RE = re.compile(rb"^\x1b\[\?[0-9;]*c$")
    CURSOR_POSITION_RE = re.compile(rb"^\x1b\[[0-9]+;[0-9]+R$")
    FOCUS_EVENT_RE = re.compile(rb"^\x1b\[[IO]$")
    PASTE_BRACKET_RE = re.compile(rb"^\x1b\[20[01]~$")
    MOUSE_EVENT_RE = re.compile(rb"^\x1b\[<[0-9]+;[0-9]+;[0-9]+[Mm]$")

    def __init__(self, escape_timeout: float = 0.05) -> None:
        self.buffer = bytearray()
        self.escape_timeout = escape_timeout
        self.escape_started_at: float | None = None
        self.bracketed_paste_active = False

    def feed(self, data: bytes) -> list[tuple[str, int, bool]]:
        observations: list[tuple[str, int, bool]] = []
        if not data:
            return observations
        self.buffer.extend(data)
        while self.buffer:
            if self.buffer.startswith(b"\x1b[200~"):
                del self.buffer[:6]
                self.bracketed_paste_active = True
                observations.append(("bracketed-paste-start", 6, False))
                continue
            if self.buffer.startswith(b"\x1b[201~"):
                del self.buffer[:6]
                self.bracketed_paste_active = False
                observations.append(("bracketed-paste-end", 6, False))
                continue
            if self.bracketed_paste_active:
                chunk = bytes(self.buffer)
                self.buffer.clear()
                observations.append(("bracketed-paste-content", len(chunk), True))
                break
            if self.buffer[0] != 0x1B:
                index = 0
                while index < len(self.buffer) and self.buffer[index] != 0x1B:
                    index += 1
                chunk = bytes(self.buffer[:index])
                del self.buffer[:index]
                self.escape_started_at = None
                observations.append(("human-input", len(chunk), True))
                continue
            matched, consumed = self._match_escape(bytes(self.buffer))
            if matched is not None:
                del self.buffer[:consumed]
                self.escape_started_at = None
                observations.append((matched, consumed, False))
                continue
            if self.escape_started_at is None:
                self.escape_started_at = monotonic()
            break
        return observations

    def flush_expired(self) -> list[tuple[str, int, bool]]:
        if not self.buffer or self.escape_started_at is None:
            return []
        if monotonic() - self.escape_started_at < self.escape_timeout:
            return []
        chunk = bytes(self.buffer)
        self.buffer.clear()
        self.escape_started_at = None
        return [("human-input", len(chunk), True)]

    def _match_escape(self, chunk: bytes) -> tuple[str | None, int]:
        if len(chunk) < 2:
            return None, 0
        second = chunk[1]
        if second == ord("["):
            for length in range(3, min(len(chunk) + 1, 32)):
                slice_bytes = chunk[:length]
                if self.DEVICE_ATTRIBUTES_RE.fullmatch(slice_bytes):
                    return "device-attributes", length
                if self.CURSOR_POSITION_RE.fullmatch(slice_bytes):
                    return "cursor-position-report", length
                if self.FOCUS_EVENT_RE.fullmatch(slice_bytes):
                    return "focus-event", length
                if self.PASTE_BRACKET_RE.fullmatch(slice_bytes):
                    return "bracketed-paste-marker", length
                if self.MOUSE_EVENT_RE.fullmatch(slice_bytes):
                    return "mouse-report", length
            final_bytes = b"~ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz"
            for index in range(2, len(chunk)):
                if chunk[index] in final_bytes:
                    return "csi-sequence", index + 1
            return None, 0
        if second == ord("]"):
            for terminator in (b"\x07", b"\x1b\\"):
                index = chunk.find(terminator, 2)
                if index != -1:
                    return "osc-sequence", index + len(terminator)
            return None, 0
        if second in (ord("O"), ord("N")):
            if len(chunk) >= 3:
                return "ss3-sequence", 3
            return None, 0
        return "escape-sequence", 2


class AntigravityScreenClassifier:
    """Classify visible Antigravity PTY output without consuming or rewriting it."""

    SPINNER_RE = re.compile(r"(?:^| )[⠋⠙⠹⠸⠼⠴⠦⠧⠇⠏](?: |$)")
    OSC_TITLE_RE = re.compile(
        r"\x1b\](?:0|1|2);(.*?)(?:\x07|\x1b\\)", re.DOTALL
    )
    ANSI_RE = re.compile(
        r"\x1b(?:\[[0-?]*[ -/]*[@-~]|\][^\x07\x1b]*(?:\x07|\x1b\\)|[@-_])"
    )
    STRONG_BLOCKERS = (
        "press enter to confirm",
        "enter to submit",
        "allow command?",
        "do you want to execute",
        "are you sure",
    )

    def __init__(self) -> None:
        self.raw = bytearray()
        self.osc_title = ""
        self.state: str | None = None

    def reset(self) -> None:
        self.raw.clear()
        self.osc_title = ""
        self.state = None

    def _visible_text(self) -> str:
        text = self.ANSI_RE.sub("", bytes(self.raw).decode("utf-8", "replace"))
        return "".join(character for character in text if character in "\n\r\t" or ord(character) >= 0x20)

    def _classify(self, visible: str) -> str:
        title = self.osc_title
        if "Action Required" in title or "Approval" in title:
            return "blocked"
        if self.SPINNER_RE.search(title):
            return "working"

        recent = visible[-HERDR_SCREEN_BYTES:]
        lowered = recent.lower()
        if any(marker in lowered for marker in self.STRONG_BLOCKERS):
            return "blocked"
        if "[y/n]" in lowered or "yes (y)" in lowered or "(y/n)" in lowered:
            return "blocked"

        lines = [line.strip() for line in recent.splitlines() if line.strip()]
        bottom = "\n".join(lines[-3:]) if lines else ""
        if self.SPINNER_RE.search(bottom) or "thinking" in bottom.lower() or "working" in bottom.lower():
            return "working"
        if title.strip():
            return "idle"
        return "unknown"

    def feed(self, data: bytes) -> str | None:
        if not data:
            return None
        self.raw.extend(data)
        if len(self.raw) > HERDR_SCREEN_BYTES:
            del self.raw[:-HERDR_SCREEN_BYTES]
        titles = list(self.OSC_TITLE_RE.finditer(bytes(self.raw).decode("utf-8", "replace")))
        if titles:
            self.osc_title = titles[-1].group(1)
        state = self._classify(self._visible_text())
        if state == self.state:
            return None
        self.state = state
        return state


class HerdrReporter:
    """Register identity and optionally report PTY-derived state through Herdr."""

    def __init__(self) -> None:
        socket_path = os.environ.get("HERDR_SOCKET_PATH", "")
        pane_id = os.environ.get("HERDR_PANE_ID", "")
        agent = os.environ.get("ANTIGRAVITY_WARM_HERDR_AGENT", "") or "Antigravity"
        self.socket_path = pathlib.Path(socket_path) if socket_path else None
        self.pane_id = pane_id if HERDR_PANE_RE.fullmatch(pane_id) else None
        self.agent = agent if HERDR_AGENT_RE.fullmatch(agent) else None
        self.enabled = (
            os.environ.get("HERDR_ENV") == "1"
            and self.socket_path is not None
            and self.socket_path.is_absolute()
            and len(str(self.socket_path)) < 100
            and self.pane_id is not None
            and self.agent is not None
        )
        self.source = HERDR_SOURCE
        self.sequence = 0
        self.registered = False
        self.last_report_at: float | None = None
        self.last_report_state: str | None = None
        self.last_error: str | None = None
        self.next_heartbeat = 0.0
        self.child_pid: int | None = None
        self.agent_session_id: str | None = None

    def state(self) -> dict[str, object]:
        return {
            "herdr_enabled": self.enabled,
            "herdr_pane_id": self.pane_id,
            "herdr_agent": self.agent,
            "herdr_source": self.source,
            "herdr_registered": self.registered,
            "herdr_last_report_at": self.last_report_at,
            "herdr_last_report_state": self.last_report_state,
            "herdr_last_error": self.last_error,
            "herdr_child_pid": self.child_pid,
            "herdr_agent_session_id": self.agent_session_id,
        }

    @staticmethod
    def _valid_session_id(value: object) -> str | None:
        if isinstance(value, str) and SESSION_RE.fullmatch(value):
            return value
        return None

    def _send(self, method: str, params: dict[str, object]) -> bool:
        if not self.enabled or self.socket_path is None:
            return False
        self.sequence += 1
        request = {
            "id": f"{self.source}:{self.sequence}",
            "method": method,
            "params": params,
        }
        try:
            with socket.socket(socket.AF_UNIX, socket.SOCK_STREAM) as connection:
                connection.settimeout(0.5)
                connection.connect(str(self.socket_path))
                connection.sendall((json.dumps(request, separators=(",", ":")) + "\n").encode())
                connection.recv(4096)
            self.last_error = None
            return True
        except (OSError, ValueError) as error:
            self.last_error = str(error)
            return False

    def register(self, child_pid: int | None = None) -> bool:
        if not self.enabled or self.pane_id is None or self.agent is None:
            return False
        self.child_pid = child_pid
        params: dict[str, object] = {
            "pane_id": self.pane_id,
            "agent": self.agent,
            "source": self.source,
            "tokens": {"pty_owner": "antigravity-warm"},
        }
        if child_pid:
            params["child_pid"] = child_pid
        ok = self._send("agent_register", params)
        if ok:
            self.registered = True
        return ok

    def report_state(self, state: str, session_id: str | None = None) -> bool:
        if not self.enabled or self.pane_id is None:
            return False
        clean_session = self._valid_session_id(session_id)
        if clean_session is not None:
            self.agent_session_id = clean_session
        params: dict[str, object] = {
            "pane_id": self.pane_id,
            "state": state,
            "source": self.source,
        }
        if self.agent_session_id is not None:
            params["agent_session"] = {"value": self.agent_session_id}
        ok = self._send("agent_state", params)
        if ok:
            self.last_report_at = now()
            self.last_report_state = state
        return ok

    def heartbeat(self, state: str | None, session_id: str | None = None) -> None:
        if not self.enabled:
            return
        current = monotonic()
        if current < self.next_heartbeat:
            return
        self.next_heartbeat = current + 30.0
        if not self.registered:
            self.register(self.child_pid)
        if state is not None:
            self.report_state(state, session_id)

        )

        self.socket_path = self.session_dir / "socket"
        self.state_path = self.session_dir / "state.json"
        state_dir = pathlib.Path(
            os.environ.get("XDG_STATE_HOME") or (pathlib.Path.home() / ".local/state")
        ) / "antigravity-idle-compaction"
        self.last_session_dir = state_dir
        self.last_session_path = self.last_session_dir / "last-session.json"

        self.executable = (
            os.environ.get("ANTIGRAVITY_IDLE_REAL_EXECUTABLE")
            or shutil.which("agy")
            or "/home/metsatron/.local/bin/agy"
        )

    def _debug(self, message: str) -> None:
        if not DEBUG:
            return
        stamp = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime(now()))
        line = f"[{stamp}] {message}\n"
        try:
            log_path = self.session_dir / "supervisor.log"
            if log_path.exists() and log_path.stat().st_size > DEBUG_LOG_MAX_BYTES:
                log_path.unlink()
            with log_path.open("a", encoding="utf-8") as handle:
                handle.write(line)
        except OSError:
            pass

    def _write_state(self) -> None:
        payload = self.state()
        temporary = self.state_path.with_suffix(".tmp")
        try:
            self.session_dir.mkdir(parents=True, exist_ok=True, mode=0o700)
            temporary.write_text(json.dumps(payload, indent=2), encoding="utf-8")
            os.chmod(temporary, 0o600)
            temporary.replace(self.state_path)
        except OSError:
            pass

    def _write_last_session(self) -> None:
        if not self.session_id:
            return
        payload = {
            "session_id": self.session_id,
            "channel": self.channel,
            "recorded_at": now(),
            "supervisor_pid": os.getpid(),
        }
        label = os.environ.get("ANTIGRAVITY_WARM_HERDR_AGENT")
        filenames = ["last-session.json"]
        if label and HERDR_AGENT_RE.fullmatch(label):
            filenames.append(f"last-session.{label}.json")
        try:
            self.last_session_dir.mkdir(parents=True, exist_ok=True, mode=0o700)
            for name in filenames:
                target = self.last_session_dir / name
                temporary = target.with_suffix(".tmp")
                temporary.write_text(json.dumps(payload, indent=2), encoding="utf-8")
                os.chmod(temporary, 0o600)
                temporary.replace(target)
        except OSError:
            pass

        return {

            "supervisor_pid": os.getpid(),
            "child_pid": self.pid,
            "status": self.status,
            "enabled": self.enabled,
            "session_id": self.session_id,

            "last_state_transition": self.last_state_transition,
            "last_gate_rejection_reason": self.last_gate_rejection_reason,
            "last_cancellation_reason": self.last_cancellation_reason,
            "last_injection_result": self.last_injection_result,
            "last_injection_at": self.last_injection_at,
            "child_status": self.child_status,
            "child_exited_at": self.child_exited_at,
            "screen_state": self.screen_state,
            "herdr": self.herdr.state(),
        }

    def _setup_ipc(self) -> None:
        self.session_dir.mkdir(parents=True, exist_ok=True, mode=0o700)
        self.server_socket = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
        self.server_socket.bind(str(self.socket_path))
        os.chmod(self.socket_path, 0o600)
        self.server_socket.listen(16)
        self.server_socket.setblocking(False)
        self.selector.register(self.server_socket, selectors.EVENT_READ, self._accept_ipc)

    def _accept_ipc(self, key: selectors.SelectorKey) -> None:
        try:
            connection, _ = key.fileobj.accept()
            self._serve_one(connection)
        except OSError:
            pass

    def _serve_one(self, connection: socket.socket) -> None:
        try:
            connection.settimeout(1.0)
            data = connection.recv(65536)
            if not data:
                return
            message = json.loads(data.split(b"\n", 1)[0].decode("utf-8"))
            if not isinstance(message, dict):
                response = {"ok": False, "error": "invalid-json"}
            elif message.get("kind") == "event":
                response = self._handle_event(message)
            else:
                response = self._handle_control(message)
            connection.sendall((json.dumps(response, separators=(",", ":")) + "\n").encode())
        except (OSError, ValueError, UnicodeError) as error:
            self._debug(f"ipc-error channel={short_id(self.channel)} error={error}")
        finally:
            connection.close()

    def _handle_event(self, message: dict[str, object]) -> dict[str, object]:
        event = message.get("event")
        session_id = message.get("session_id")
        if not isinstance(event, str) or event not in EVENTS:
            return {"ok": False, "error": "invalid-event"}
        if isinstance(session_id, str) and SESSION_RE.fullmatch(session_id):
            if self.session_id != session_id:
                self.session_id = session_id
                self.session_bound_at = now()
                self._write_last_session()

        current = now()
        if event in {"session-start", "session-end"}:
            self.last_lifecycle_activity = current
            self.last_state_transition = f"event:{event}"
        elif event == "user-prompt-submit":
            self.status = "active"
            self.dirty = True
            self.last_lifecycle_activity = current
            self.last_state_transition = "event:user-prompt-submit"
        elif event == "session-activity":
            self.last_lifecycle_activity = current
            self.last_state_transition = "event:session-activity"
        elif event == "stop":
            self.status = "idle"
            self.last_normal_stop = current
            self.rearm_count += 1
            self.last_state_transition = "event:stop"
        elif event == "post-compact":
            self.compacting = False
            self.dirty = False
            self.last_lifecycle_activity = current
            self.last_state_transition = "event:post-compact"
        elif event == "stop-failure":
            self.last_lifecycle_activity = current
            self.last_state_transition = "event:stop-failure"

        self._write_state()
        return {"ok": True, "state": self.state()}

    def _handle_control(self, message: dict[str, object]) -> dict[str, object]:
        command = message.get("command")
        if command not in CONTROL_COMMANDS:
            return {"ok": False, "error": f"unrecognized-command: {command}"}
        if command in {"state", "show"}:
            return {"ok": True, "state": self.state()}
        if command == "enable":
            self.enabled = True
            self.last_state_transition = "control:enable"
            self._write_state()
            return {"ok": True, "state": self.state()}
        if command == "disable":
            self.enabled = False
            self.last_state_transition = "control:disable"
            self._write_state()
            return {"ok": True, "state": self.state()}
        if command == "cancel":
            reason = str(message.get("reason", "manual-cancel"))
            self.last_cancellation_reason = reason
            self.last_state_transition = f"control:cancel:{reason}"
            self._write_state()
            return {"ok": True, "state": self.state()}
        if command == "inject":
            text = str(message.get("text", ""))
            if not text:
                return {"ok": False, "error": "empty-injection"}
            if self.master_fd is None:
                return {"ok": False, "error": "master-not-available"}
            try:
                write_all(self.master_fd, text.encode("utf-8"))
                self.last_injection_at = now()
                self.last_injection_result = "success"
                self._write_state()
                return {"ok": True, "state": self.state()}
            except OSError as error:
                return {"ok": False, "error": str(error)}
        if command == "trigger":
            return {"ok": True, "message": "trigger-accepted", "state": self.state()}
        return {"ok": False, "error": "unhandled-command"}

    def launch(self) -> None:
        self._setup_ipc()
        self._write_state()
        environment = os.environ.copy()
        environment.update(
            {
                "ANTIGRAVITY_IDLE_CHANNEL": self.channel,
                "ANTIGRAVITY_IDLE_OWNER": "pty-supervisor",
                "ANTIGRAVITY_IDLE_CONTROL_SOCKET": str(self.socket_path),
                "ANTIGRAVITY_IDLE_RUNTIME_DIR": str(self.session_dir),
                "ANTIGRAVITY_IDLE_SUPERVISOR_PID": str(os.getpid()),
            }
        )
        pid, master_fd = pty.fork()
        if pid == 0:
            try:
                os.execvpe(self.executable, [self.executable, *self.argv], environment)
            except BaseException as error:
                print(f"antigravity-warm: failed to exec Antigravity: {error}", file=sys.stderr)
                os._exit(127)
        self.pid = pid
        self.master_fd = master_fd
        os.set_blocking(master_fd, False)
        self.selector.register(master_fd, selectors.EVENT_READ, self._read_master)
        if sys.stdin.isatty():
            self.selector.register(sys.stdin.fileno(), selectors.EVENT_READ, self._read_stdin)
        self.herdr.register(pid)
        self._write_state()

    def _read_stdin(self, key: selectors.SelectorKey) -> None:
        try:
            data = os.read(sys.stdin.fileno(), 65536)
        except (OSError, BlockingIOError):
            return
        if not data:
            self.stdin_eof = True
            if self.master_fd is not None:
                try:
                    write_all(self.master_fd, b"\x04")
                except OSError:
                    pass
            try:
                self.selector.unregister(sys.stdin.fileno())
            except Exception:
                pass
            return
        if self.master_fd is not None:
            try:
                write_all(self.master_fd, data)
            except OSError:
                self.shutdown_requested = True
        try:
            observations = self.input_classifier.feed(data)
        except Exception:
            observations = [("unknown-input", len(data), True)]
        if observations and any(is_human for _, _, is_human in observations):
            self.last_local_input = now()
            self._write_state()

    def _read_master(self, key: selectors.SelectorKey) -> None:
        if self.master_fd is None:
            return
        try:
            data = os.read(self.master_fd, 65536)
        except OSError as error:
            if error.errno in {errno.EIO, errno.EBADF}:
                self.shutdown_requested = True
            return
        if not data:
            self.shutdown_requested = True
            return
        state = self.screen_classifier.feed(data)
        try:
            write_all(sys.stdout.fileno(), data)
            sys.stdout.flush()
        except OSError:
            self.shutdown_requested = True
        if state is not None and state != self.screen_state:
            self.screen_state = state
            self.herdr.report_state(state, self.session_id)
            self._write_state()

    def _child_has_exited(self) -> bool:
        if self.child_status is not None or not self.pid:
            return self.child_status is not None
        try:
            waited_pid, status = os.waitpid(self.pid, os.WNOHANG)
        except ChildProcessError:
            self.child_status = 1 << 8
            return True
        if waited_pid == self.pid:
            self.child_status = status
            self.child_exited_at = now()
            self.status = "exited"
            return True
        return False

    def run(self) -> int:
        self.launch()
        original_termios = None
        if sys.stdin.isatty():
            try:
                original_termios = set_terminal_raw(sys.stdin.fileno())
                rows, cols = get_terminal_size(sys.stdin.fileno())
                if self.master_fd is not None:
                    set_terminal_size(self.master_fd, rows, cols)
            except OSError:
                pass

        def handle_sigwinch(_signum, _frame):
            if sys.stdin.isatty() and self.master_fd is not None:
                r, c = get_terminal_size(sys.stdin.fileno())
                set_terminal_size(self.master_fd, r, c)

        signal.signal(signal.SIGWINCH, handle_sigwinch)

        try:
            while not self.shutdown_requested:
                if self._child_has_exited():
                    break
                self.herdr.heartbeat(self.screen_state, self.session_id)
                events = self.selector.select(timeout=0.1)
                for key, _mask in events:
                    callback = key.data
                    callback(key)
        finally:
            if sys.stdin.isatty() and original_termios is not None:
                restore_terminal(sys.stdin.fileno(), original_termios)
            self.cleanup()

        if self.child_status is not None:
            if os.WIFEXITED(self.child_status):
                return os.WEXITSTATUS(self.child_status)
            if os.WIFSIGNALED(self.child_status):
                return 128 + os.WTERMSIG(self.child_status)
        return 0

    def cleanup(self) -> None:
        try:
            self.selector.close()
        except Exception:
            pass
        if self.server_socket:
            try:
                self.server_socket.close()
            except OSError:
                pass
        try:
            if self.socket_path.exists():
                self.socket_path.unlink()
        except OSError:
            pass


def control_cli(argv: list[str]) -> int:
    """Entry point for antigravity-warmctl."""
    runtime_root = pathlib.Path(
        os.environ.get("XDG_RUNTIME_DIR") or f"/run/user/{os.getuid()}"
    ) / "antigravity-idle-compaction"
    sockets = sorted(runtime_root.glob("*/socket"))
    if not sockets:
        print("antigravity-warmctl: no running antigravity-warm supervisor found", file=sys.stderr)
        return 1
    target_socket = sockets[-1]
    cmd = argv[0] if argv else "state"
    payload = {"command": cmd, "args": argv[1:]}
    try:
        with socket.socket(socket.AF_UNIX, socket.SOCK_STREAM) as client:
            client.settimeout(2.0)
            client.connect(str(target_socket))
            client.sendall((json.dumps(payload) + "\n").encode())
            reply = client.recv(65536)
            parsed = json.loads(reply.decode())
            print(json.dumps(parsed, indent=2))
            return 0 if parsed.get("ok") else 2
    except OSError as err:
        print(f"antigravity-warmctl: connection error: {err}", file=sys.stderr)
        return 1


def main(argv: list[str]) -> int:
    if argv and argv[0] == "--control":
        return control_cli(argv[1:])
    supervisor = Supervisor(argv)
    return supervisor.run()


if __name__ == "__main__":
    raise SystemExit(main(sys.argv[1:]))
