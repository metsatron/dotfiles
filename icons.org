* Home sync helper
Keep icon and theme payloads in DotCortex while making the live home roots real
directories. Icon themes become real directories with repo-backed contents so
only cache files live in =$HOME=. GTK/XFWM theme entries stay repo-backed as
top-level symlinks inside real home theme roots so Stow never tries to replace
the whole tree.

#+BEGIN_SRC python :tangle all/.local/bin/icons-home-sync :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
from __future__ import annotations

import argparse
import shutil
import subprocess
from pathlib import Path


ICON_SYNC_MAP = (
    ("all/.icons", ".icons"),
    ("linux/.icons", ".icons"),
    ("all/.local/share/icons", ".local/share/icons"),
    ("linux/.local/share/icons", ".local/share/icons"),
)

THEME_SYNC_MAP = (
    ("all/.themes", ".themes"),
    ("linux/.themes", ".themes"),
    ("all/.local/share/themes", ".local/share/themes"),
    ("linux/.local/share/themes", ".local/share/themes"),
)

CACHE_NAMES = {"icon-theme.cache", ".icon-theme.cache"}
ROOT_SKIP_NAMES = {".git"}
ICON_SKIP_NAMES = CACHE_NAMES | ROOT_SKIP_NAMES


def remove_path(path: Path) -> None:
    if path.is_dir() and not path.is_symlink():
        shutil.rmtree(path)
    elif path.exists() or path.is_symlink():
        path.unlink()


def ensure_real_dir(path: Path, dry_run: bool) -> None:
    if dry_run:
        if path.is_symlink():
            print(f"plan detach root {path}")
        elif not path.exists():
            print(f"plan mkdir {path}")
        return
    if path.is_symlink():
        path.unlink()
    elif path.exists() and not path.is_dir():
        path.unlink()
    path.mkdir(parents=True, exist_ok=True)


def sync_icon_dir(srcs: list[Path], dst: Path, dry_run: bool) -> None:
    if dry_run:
        if dst.is_symlink():
            print(f"plan detach theme {dst}")
        elif dst.exists() and not dst.is_dir():
            print(f"plan replace non-dir {dst}")
        elif not dst.exists():
            print(f"plan mkdir {dst}")
    else:
        if dst.is_symlink() or (dst.exists() and not dst.is_dir()):
            remove_path(dst)
        dst.mkdir(parents=True, exist_ok=True)

    desired: dict[str, Path] = {}
    for src in srcs:
        for child in sorted(src.iterdir()):
            if child.name in ICON_SKIP_NAMES:
                continue
            desired[child.name] = child

    if dst.exists() and dst.is_dir() and not dst.is_symlink():
        for existing in sorted(dst.iterdir()):
            if existing.name in CACHE_NAMES:
                continue
            if existing.name not in desired:
                if dry_run:
                    print(f"plan remove stale {existing}")
                else:
                    remove_path(existing)

    for name, child in sorted(desired.items()):
        target = dst / name
        child_resolved = child.resolve(strict=False)
        if target.is_symlink() and target.resolve(strict=False) == child_resolved:
            continue
        if dry_run:
            print(f"plan link {target} -> {child}")
            continue
        if target.exists() or target.is_symlink():
            remove_path(target)
        target.symlink_to(child)


def remove_repo_caches(repo_root: Path, dry_run: bool) -> int:
    removed = 0
    seen: set[Path] = set()
    for src_rel, _dst_rel in ICON_SYNC_MAP:
        src_root = repo_root / src_rel
        if not src_root.exists():
            continue
        for cache_name in CACHE_NAMES:
            for cache_path in src_root.rglob(cache_name):
                if cache_path in seen:
                    continue
                seen.add(cache_path)
                if dry_run:
                    print(f"plan remove repo cache {cache_path}")
                else:
                    remove_path(cache_path)
                removed += 1
    return removed


def sync_icon_entry(srcs: list[Path], dst_root: Path, dry_run: bool) -> bool:
    src = srcs[-1]
    dst = dst_root / src.name
    if src.is_dir():
        sync_icon_dir(srcs, dst, dry_run)
    else:
        src_resolved = src.resolve(strict=False)
        if dst.is_symlink() and dst.resolve(strict=False) == src_resolved:
            return False
        if dry_run:
            print(f"plan link {dst} -> {src}")
        else:
            dst.parent.mkdir(parents=True, exist_ok=True)
            if dst.exists() or dst.is_symlink():
                remove_path(dst)
            dst.symlink_to(src)
    return True


def is_under_any(path: Path, roots: list[Path]) -> bool:
    resolved = path.resolve(strict=False)
    for root in roots:
        root_resolved = root.resolve(strict=False)
        try:
            resolved.relative_to(root_resolved)
            return True
        except ValueError:
            continue
    return False


def sync_theme_entries(src_roots: list[Path], dst_root: Path, dry_run: bool) -> int:
    desired: dict[str, Path] = {}
    for src_root in src_roots:
        for src in sorted(src_root.iterdir()):
            if src.name in ROOT_SKIP_NAMES:
                continue
            desired[src.name] = src

    synced = 0
    if dst_root.exists() and dst_root.is_dir() and not dst_root.is_symlink():
        for existing in sorted(dst_root.iterdir()):
            if existing.name in desired:
                continue
            if existing.is_symlink() and is_under_any(existing, src_roots):
                if dry_run:
                    print(f"plan remove stale {existing}")
                else:
                    existing.unlink()

    for name, src in sorted(desired.items()):
        dst = dst_root / name
        src_resolved = src.resolve(strict=False)
        if dst.is_symlink() and dst.resolve(strict=False) == src_resolved:
            continue
        if dry_run:
            action = "relink" if dst.exists() or dst.is_symlink() else "link"
            print(f"plan {action} {dst} -> {src}")
        else:
            if dst.exists() or dst.is_symlink():
                remove_path(dst)
            dst.symlink_to(src)
        synced += 1
    return synced


def rebuild_caches(themes: set[Path], dry_run: bool) -> int:
    gtk_cache = shutil.which("gtk-update-icon-cache")
    if not gtk_cache:
        return 0
    count = 0
    for theme in sorted(themes):
        if not theme.exists() or not theme.is_dir() or theme.is_symlink():
            continue
        if not (theme / "index.theme").is_file():
            continue
        for stale in (theme / ".icon-theme.cache", theme / "icon-theme.cache"):
            if stale.exists() or stale.is_symlink():
                if dry_run:
                    print(f"plan remove stale cache {stale}")
                else:
                    stale.unlink()
        if dry_run:
            print(f"plan cache {theme}")
        else:
            subprocess.run(
                [gtk_cache, "-f", "-t", str(theme)],
                check=False,
                stdout=subprocess.DEVNULL,
                stderr=subprocess.DEVNULL,
            )
        count += 1
    return count


def main() -> int:
    parser = argparse.ArgumentParser(
        description="Keep repo-backed icons and themes while rebuilding only local icon caches in home roots."
    )
    parser.add_argument(
        "--repo-root",
        default=str(Path.home() / "DotCortex"),
        help="DotCortex repo root",
    )
    parser.add_argument("--dry-run", action="store_true", help="show planned actions only")
    args = parser.parse_args()

    repo_root = Path(args.repo_root).expanduser().resolve()
    home = Path.home()
    repo_caches_removed = remove_repo_caches(repo_root, args.dry_run)
    synced = 0
    icon_themes_to_rebuild: set[Path] = set()

    sync_roots: dict[str, list[Path]] = {}
    for src_rel, dst_rel in ICON_SYNC_MAP:
        src_root = repo_root / src_rel
        if not src_root.exists():
            continue
        sync_roots.setdefault(dst_rel, []).append(src_root)

    for dst_rel, src_roots in sync_roots.items():
        dst_root = home / dst_rel
        ensure_real_dir(dst_root, args.dry_run)
        entries: dict[str, list[Path]] = {}
        for src_root in src_roots:
            for src in sorted(src_root.iterdir()):
                if src.name in ICON_SKIP_NAMES:
                    continue
                entries.setdefault(src.name, []).append(src)
        for name in sorted(entries):
            if sync_icon_entry(entries[name], dst_root, args.dry_run):
                synced += 1
                dst = dst_root / name
                if dst.exists() and dst.is_dir() and not dst.is_symlink():
                    icon_themes_to_rebuild.add(dst)

    theme_roots: dict[str, list[Path]] = {}
    for src_rel, dst_rel in THEME_SYNC_MAP:
        src_root = repo_root / src_rel
        if not src_root.exists():
            continue
        theme_roots.setdefault(dst_rel, []).append(src_root)

    for dst_rel, src_roots in theme_roots.items():
        dst_root = home / dst_rel
        ensure_real_dir(dst_root, args.dry_run)
        synced += sync_theme_entries(src_roots, dst_root, args.dry_run)

    cached = rebuild_caches(icon_themes_to_rebuild, args.dry_run)

    mode = "planned" if args.dry_run else "linked"
    print(
        f"icons-home-sync: {mode}_entries={synced} "
        f"repo_caches_removed={repo_caches_removed} cache_roots={cached}"
    )
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

* App icon injection helper
Scale a source image into one or more repo-managed icon themes, using the
theme's directory grammar. This is for app icon aliases such as =kitty=,
=firedragon=, or =org.garudalinux.firedragon=.

Pixel art defaults to nearest-neighbor scaling so crisp source pixels do not get
blurred. Use =--resample lanczos= for painterly or antialiased source art.
Use =--magic-pink= when an icon generator gives you a PNG with a fake
=#ff00ff= transparency matte instead of an alpha channel.
Use =--trim-transparent= after color-keying to crop away transparent borders
while preserving the icon's aspect ratio on a square transparent canvas.
The first =--names= entry is written as the real PNG. Additional aliases are
relative symlinks by default to avoid duplicate binary payloads.

#+BEGIN_SRC python :tangle all/.local/bin/icon-theme-inject :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
from __future__ import annotations

import argparse
from pathlib import Path

from PIL import Image


THEME_SIZES = {
    "Irixium": (16, 22, 24, 32, 48, 64, 128, 256),
    "BeOS-r5-Icons": (16, 22, 24, 32, 48, 64, 128, 256),
    "indigo-reality": (16, 22, 32, 48, 64, 128, 256),
}


def theme_app_dir(theme: str, size: int) -> Path:
    if theme == "Irixium":
        return Path(f"linux/.local/share/icons/{theme}/{size}x{size}/apps")
    if theme == "indigo-reality" and size == 256:
        return Path(f"linux/.local/share/icons/{theme}/apps/256FIXED")
    return Path(f"linux/.local/share/icons/{theme}/apps/{size}")


def parse_csv(value: str) -> list[str]:
    return [item.strip() for item in value.split(",") if item.strip()]


def replace_path(path: Path) -> None:
    if path.is_symlink() or path.is_file():
        path.unlink()
    elif path.exists():
        raise SystemExit(f"refusing to replace non-file path: {path}")


def write_icon_alias(alias: Path, target_name: str, copy_aliases: bool) -> None:
    if alias.exists() or alias.is_symlink():
        replace_path(alias)
    if copy_aliases:
        source = alias.with_name(target_name)
        alias.write_bytes(source.read_bytes())
    else:
        alias.symlink_to(target_name)


def parse_hex_color(value: str) -> tuple[int, int, int]:
    color = value.strip()
    if color.startswith("#"):
        color = color[1:]
    if len(color) != 6:
        raise argparse.ArgumentTypeError("expected RGB hex color such as #ff00ff")
    try:
        return tuple(int(color[i : i + 2], 16) for i in (0, 2, 4))
    except ValueError as exc:
        raise argparse.ArgumentTypeError("expected RGB hex color such as #ff00ff") from exc


def remove_color_key(image: Image.Image, color: tuple[int, int, int], tolerance: int) -> Image.Image:
    keyed = image.convert("RGBA")
    pixels = keyed.load()
    width, height = keyed.size
    cr, cg, cb = color
    for y in range(height):
        for x in range(width):
            r, g, b, a = pixels[x, y]
            if (
                abs(r - cr) <= tolerance
                and abs(g - cg) <= tolerance
                and abs(b - cb) <= tolerance
            ):
                pixels[x, y] = (r, g, b, 0)
            elif a == 0:
                pixels[x, y] = (r, g, b, 0)
    return keyed


def trim_transparent(image: Image.Image, padding: int) -> Image.Image:
    alpha = image.getchannel("A")
    bbox = alpha.getbbox()
    if bbox is None:
        return image

    left, top, right, bottom = bbox
    if padding > 0:
        left = max(0, left - padding)
        top = max(0, top - padding)
        right = min(image.width, right + padding)
        bottom = min(image.height, bottom + padding)

    cropped = image.crop((left, top, right, bottom))
    side = max(cropped.width, cropped.height)
    square = Image.new("RGBA", (side, side), (0, 0, 0, 0))
    square.paste(cropped, ((side - cropped.width) // 2, (side - cropped.height) // 2))
    return square


def main() -> int:
    parser = argparse.ArgumentParser(
        description="Inject scaled app icons into repo-managed DotCortex icon themes."
    )
    parser.add_argument("source", help="source image, usually a large PNG")
    parser.add_argument("--themes", default="Irixium,BeOS-r5-Icons", help="comma-separated theme names")
    parser.add_argument("--names", required=True, help="comma-separated icon names without .png")
    parser.add_argument("--sizes", default="", help="comma-separated sizes; default follows each theme")
    parser.add_argument(
        "--copy-aliases",
        action="store_true",
        help="write every --names entry as a physical PNG instead of symlinking aliases",
    )
    parser.add_argument(
        "--resample",
        choices=("nearest", "lanczos"),
        default="nearest",
        help="nearest for pixel art, lanczos for smooth artwork",
    )
    parser.add_argument(
        "--transparent-color",
        type=parse_hex_color,
        default=None,
        help="make matching RGB pixels transparent before resizing, for color-keyed source art",
    )
    parser.add_argument(
        "--transparent-tolerance",
        type=int,
        default=0,
        help="per-channel tolerance for --transparent-color, default exact match",
    )
    parser.add_argument(
        "--magic-pink",
        action="store_true",
        help="shortcut for --transparent-color #ff00ff",
    )
    parser.add_argument(
        "--trim-transparent",
        action="store_true",
        help="crop transparent borders after color-keying and pad back to square before resizing",
    )
    parser.add_argument(
        "--trim-padding",
        type=int,
        default=0,
        help="source-pixel padding to preserve around --trim-transparent crops",
    )
    parser.add_argument("--repo-root", default=".", help="DotCortex repository root")
    args = parser.parse_args()

    repo_root = Path(args.repo_root).expanduser().resolve()
    source = Path(args.source).expanduser().resolve()
    if not source.is_file():
        raise SystemExit(f"missing source image: {source}")

    names = parse_csv(args.names)
    themes = parse_csv(args.themes)
    explicit_sizes = tuple(int(size) for size in parse_csv(args.sizes))
    if not names:
        raise SystemExit("--names must contain at least one icon name")

    resample = Image.Resampling.NEAREST if args.resample == "nearest" else Image.Resampling.LANCZOS
    image = Image.open(source).convert("RGBA")
    transparent_color = (255, 0, 255) if args.magic_pink else args.transparent_color
    if transparent_color is not None:
        if args.transparent_tolerance < 0:
            raise SystemExit("--transparent-tolerance must be zero or greater")
        image = remove_color_key(image, transparent_color, args.transparent_tolerance)
    if args.trim_padding < 0:
        raise SystemExit("--trim-padding must be zero or greater")
    if args.trim_transparent:
        image = trim_transparent(image, args.trim_padding)

    written = 0
    for theme in themes:
        sizes = explicit_sizes or THEME_SIZES.get(theme)
        if not sizes:
            raise SystemExit(f"unknown theme without explicit --sizes: {theme}")
        for size in sizes:
            app_dir = repo_root / theme_app_dir(theme, size)
            app_dir.mkdir(parents=True, exist_ok=True)
            scaled = image.resize((size, size), resample)
            primary = app_dir / f"{names[0]}.png"
            if primary.exists() or primary.is_symlink():
                replace_path(primary)
            scaled.save(primary)
            print(f"{primary.relative_to(repo_root)} {size}x{size}")
            written += 1
            for name in names[1:]:
                alias = app_dir / f"{name}.png"
                write_icon_alias(alias, primary.name, args.copy_aliases)
                mode = "copy" if args.copy_aliases else "symlink"
                print(f"{alias.relative_to(repo_root)} -> {primary.name} ({mode})")
                written += 1

    print(f"icon-theme-inject: wrote {written} icon files")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

* System-wide icon cache rebuild
=icons-home-sync='s own cache rebuild only covers themes it just synced into
=$HOME=. A missing =icon-theme.cache= on a theme outside that scope (a
system-installed pack under =/usr/share/icons=, root-owned, never touched by
DotCortex) silently forces GTK to fall back to a full directory stat/walk on
every icon lookup instead of one indexed read — found chasing a slow Rofi
=show-icons: true= launch back to =Catalina-Dark= having no cache at all
(2026-07-09). This sweeps every icon theme root, user and system alike, and
rebuilds whatever cache is missing (or all of them, with =--force=), batching
the root-owned ones behind a single =sudo= prompt instead of one per theme.

#+BEGIN_SRC python :tangle all/.local/bin/icon-cache-rebuild :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
from __future__ import annotations

import argparse
import os
import shutil
import subprocess
import sys
from pathlib import Path

CACHE_NAMES = ("icon-theme.cache", ".icon-theme.cache")

DEFAULT_ROOTS = (
    Path("/usr/share/icons"),
    Path("/usr/local/share/icons"),
    Path.home() / ".local/share/icons",
    Path.home() / ".icons",
)


def find_theme_dirs(roots: list[Path]) -> list[Path]:
    themes: list[Path] = []
    seen: set[Path] = set()
    for root in roots:
        if not root.is_dir():
            continue
        for entry in sorted(root.iterdir()):
            if not entry.is_dir() or entry.is_symlink():
                continue
            if not (entry / "index.theme").is_file():
                continue
            resolved = entry.resolve(strict=False)
            if resolved in seen:
                continue
            seen.add(resolved)
            themes.append(entry)
    return themes


def has_cache(theme: Path) -> bool:
    return any((theme / name).is_file() for name in CACHE_NAMES)


def main() -> int:
    parser = argparse.ArgumentParser(
        description="Rebuild GTK icon-theme.cache files wherever missing, across "
        "both user and root-owned icon theme roots (not just DotCortex-synced ones)."
    )
    parser.add_argument(
        "--roots",
        help="Comma-separated theme roots to scan (default: system + user icon dirs)",
    )
    parser.add_argument(
        "--force",
        action="store_true",
        help="Rebuild every theme's cache, not just themes missing one",
    )
    parser.add_argument("--dry-run", action="store_true", help="show planned actions only")
    args = parser.parse_args()

    gtk_cache = shutil.which("gtk-update-icon-cache")
    if not gtk_cache:
        print("icon-cache-rebuild: gtk-update-icon-cache not found on PATH", file=sys.stderr)
        return 1

    if args.roots:
        roots = [Path(r).expanduser() for r in args.roots.split(",") if r]
    else:
        roots = list(DEFAULT_ROOTS)

    themes = find_theme_dirs(roots)
    to_rebuild = [t for t in themes if args.force or not has_cache(t)]
    skipped = len(themes) - len(to_rebuild)

    if not to_rebuild:
        print(f"icon-cache-rebuild: scanned={len(themes)} already_cached={skipped} rebuilt=0")
        return 0

    writable = [t for t in to_rebuild if os.access(t, os.W_OK)]
    needs_sudo = [t for t in to_rebuild if t not in writable]

    for theme in writable:
        if args.dry_run:
            print(f"plan cache {theme}")
            continue
        subprocess.run([gtk_cache, "-f", "-t", str(theme)], check=False)

    if needs_sudo:
        if args.dry_run:
            for theme in needs_sudo:
                print(f"plan cache (sudo) {theme}")
        else:
            print(f"icon-cache-rebuild: {len(needs_sudo)} root-owned theme(s) need sudo — you may be prompted once")
            cmd = ["sudo", "bash", "-c", 'for t in "$@"; do "$0" -f -t "$t"; done', gtk_cache]
            cmd += [str(t) for t in needs_sudo]
            subprocess.run(cmd, check=False)

    mode = "planned" if args.dry_run else "rebuilt"
    print(
        f"icon-cache-rebuild: scanned={len(themes)} already_cached={skipped} "
        f"{mode}={len(to_rebuild)} (user={len(writable)} sudo={len(needs_sudo)})"
    )
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

* App icon workflow notes

** Repo-managed icon theme model
- Source payloads live under =all/.local/share/icons= and =linux/.local/share/icons=.
- Live home icon roots are real directories; =icons-home-sync= links repo-backed
  theme entries into =~/.local/share/icons= and removes repo-side cache files.
- Do not commit =icon-theme.cache= or =.icon-theme.cache=.
- Some remixed themes have invalid cache metadata but still resolve icon files
  correctly. Treat cache failures as a theme-health issue, not as proof the icon
  aliases failed.

** Naming rules
- Read the application's desktop file and use its exact =Icon== value first.
- Add app-id aliases when launchers or Flatpaks may resolve those instead
  (example: =firedragon=, =org.garudalinux.firedragon=,
  =garudalinux-firedragon=).
- Put the preferred real icon name first in =--names=. Additional names become
  relative symlinks by default; use =--copy-aliases= only when a consumer cannot
  resolve aliases.
- Keep icons in the =apps= context unless the desktop file points elsewhere.

** Pixel art handling
- For pixel-art source, scale with nearest-neighbor and inspect small sizes.
- For fake-alpha PNGs, run with =--magic-pink= or
  =--transparent-color '#ff00ff'= before resizing.
- For source art with too much matte space, add =--trim-transparent= after
  color-keying. It crops transparent borders, then pads back to square so icons
  do not distort.
- For antialiased/painterly source, use =--resample lanczos=.
- Generate real size-specific PNGs; do not place a 1024px image in 16px slots.
- For Plasma/SonicDE, include high-resolution app sizes such as 64, 128, and
  256px so panels and switchers do not bilinear-upscale a 48px icon.
- Good future tools to declare in package manifests: =imagemagick= for format
  conversion/sprite sheets, =oxipng= for lossless PNG optimization,
  =pngquant= for lossy palette compression when file size matters, and
  =libresprite= or =aseprite= for manual pixel-art editing.
