def load_border_builder(repo: Path):
    path = repo / BORDER_BUILDER
    if not path.is_file():
        raise SystemExit(f"tangle the border builder first: {path}")
    loader = importlib.machinery.SourceFileLoader("redstone_nextstep_borders", str(path))
    spec = importlib.util.spec_from_loader(loader.name, loader)
    if spec is None or spec.loader is None:
        raise SystemExit(f"cannot load border builder: {path}")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def rows(repo: Path, border) -> list[dict[str, str]]:
    data = json.loads((repo / PALETTES).read_text(encoding="utf-8"))
    return border.variant_palette_rows(data)


def prepare_generated_dir(path: Path) -> None:
    if not path.exists() and not path.is_symlink():
        path.mkdir(parents=True)
        return
    if path.is_symlink() or not path.is_dir():
        raise SystemExit(f"refuse to replace non-directory generated target: {path}")
    provenance = path / "PROVENANCE"
    if not provenance.is_file() or MARKER not in provenance.read_text(encoding="utf-8"):
        raise SystemExit(f"refuse to replace unmarked theme directory: {path}")
    shutil.rmtree(path)
    path.mkdir(parents=True)


def remove_marked_generated_dir(path: Path) -> None:
    if not path.exists() and not path.is_symlink():
        return
    if path.is_symlink() or not path.is_dir():
        raise SystemExit(f"refuse to retire non-directory generated target: {path}")
    provenance = path / "PROVENANCE"
    if not provenance.is_file() or MARKER not in provenance.read_text(encoding="utf-8"):
        raise SystemExit(f"refuse to retire unmarked generated directory: {path}")
    shutil.rmtree(path)


def retire_legacy_public_names(repo: Path, palette_rows: list[dict[str, str]]) -> None:
    """Remove the rejected public family name after proving generator ownership."""
    for row in palette_rows:
        old_theme = repo / THEMES / f"NextStep-Platinum-{row['token']}"
        remove_marked_generated_dir(old_theme)
        old_scheme = repo / SCHEMES / f"nextstep-platinum-{row['slug']}.scheme"
        if old_scheme.is_file():
            text = old_scheme.read_text(encoding="utf-8")
            expected_theme = f'ICEWM_THEME="NextStep-Platinum-{row["token"]}"'
            if expected_theme not in text or "Redstone 9X Desktop Scheme" not in text:
                raise SystemExit(f"refuse to retire unrecognized legacy scheme: {old_scheme}")
            old_scheme.unlink()
    remove_marked_generated_dir(repo / ICONS / "Redstone-Platinum")


def rhapsody_active(row: dict[str, str]) -> str:
    if row["source"].startswith("one_step_back."):
        return row["base_color"]
    return row["active_title"]


def rhapsody_colour(border, old: str | None, row: dict[str, str],
                    face: str) -> str | None:
    del border
    if old is None:
        return None
    value = old.upper()
    if value in {"#000000", "#303030"}:
        return row["black_color"]
    if value == "#FFFFFF":
        return row["white_color"]
    if value in {"#555555", "#808080", "#888888"}:
        return row["darker_color"]
    if value in {"#999999", "#AAAAAA", "#BBBBBB", "#CCCCCC"}:
        return face
    if value in {"#CFD4DC", "#DDDDDD", "#EEEEEE", "#F8F8F8"}:
        return row["lighter_color"]
    raise SystemExit(f"unmapped Tosh donor colour for Rhapsody: {value}")


def recolour_rhapsody(border, source: Path, destination: Path,
                      row: dict[str, str], face: str) -> None:
    image = border.read_xpm(source)
    pixels = [
        [rhapsody_colour(border, image.colours[symbol], row, face)
         for symbol in line]
        for line in image.pixels
    ]
    border.write_xpm(
        destination, [colour for line in pixels for colour in line], pixels,
    )
    emitted = border.read_xpm(destination)
    if (emitted.width, emitted.height) != (image.width, image.height):
        raise SystemExit(f"Rhapsody geometry changed: {source.name}")


def write_rhapsody_pixels(repo: Path, output: Path, row: dict[str, str],
                          border) -> None:
    donor = repo / TOSH
    active = rhapsody_active(row)
    inactive = row["inactive_title"]
    for suffix, face in (("A", row["base_color"]), ("I", inactive)):
        for stem in FRAME_STEMS:
            source = donor / f"frame{suffix}{stem}.xpm"
            if not source.is_file():
                raise SystemExit(f"Tosh frame source missing: {source}")
            recolour_rhapsody(border, source, output / source.name, row, face)
            recolour_rhapsody(
                border, source, output / f"dframe{suffix}{stem}.xpm", row, face,
            )
    for suffix, face in (("A", active), ("I", inactive)):
        for stem in TITLE_STEMS:
            source = donor / f"title{suffix}{stem}.xpm"
            if not source.is_file():
                raise SystemExit(f"Tosh title source missing: {source}")
            recolour_rhapsody(border, source, output / source.name, row, face)
    for suffix, face in (("A", row["base_color"]), ("I", inactive)):
        for stem in BUTTON_STEMS:
            source = donor / f"{stem}{suffix}.xpm"
            if not source.is_file():
                raise SystemExit(f"Tosh button source missing: {source}")
            recolour_rhapsody(border, source, output / source.name, row, face)


def lane_row(row: dict[str, str], lane: str) -> dict[str, str]:
    result = dict(row)
    if lane in {"nextstep", "gnustep"}:
        result["active_title"] = result["black_color"]
    elif lane == "rhapsody":
        result["active_title"] = rhapsody_active(result)
    return result


def default_theme(path: Path, row: dict[str, str], lane: str,
                  output_name: str, border) -> None:
    # IceWM theme files stay Latin-1; normalize display punctuation before
    # writing so an em dash in the human palette label cannot strand a
    # partially generated directory.
    description = f"{LANES[lane]['label']} - {output_name}".replace(" — ", " - ")
    description.encode("latin-1")
    canonical = lane in {"nextstep", "gnustep"}
    colours = border.colour_values(
        row, authentic=lane == "nextstep", canonical=canonical,
        button_style="gray" if lane != "gnustep" else "black",
    )
    if lane == "rhapsody":
        active = rhapsody_active(row)
        colours.update({
            "ColorActiveBorder": row["border_color"],
            "ColorNormalBorder": row["border_color"],
            "ColorNormalTitleButton": row["base_color"],
            "ColorNormalTitleButtonText": border.contrast(row, row["base_color"]),
            "ColorActiveTitleBar": active,
            "ColorNormalTitleBar": row["inactive_title"],
            "ColorActiveTitleBarText": border.contrast(row, active),
            "ColorNormalTitleBarText": border.contrast(row, row["inactive_title"]),
        })
    tosh_keys = border.tosh_colour_keys(path.parent.parent / "Tosh" / "default.theme")
    missing = [key for key in tosh_keys if key not in colours]
    if missing:
        raise SystemExit(f"variant default.theme misses Tosh keys: {missing}")
    lines = [
        "# Generated by redstone-9x-nextstep-variants-build; do not hand-edit.",
        f'ThemeDescription="{description}"',
        'ThemeAuthor="DotCortex; approved NeXTSTEP recovery canon"',
        "Look=pixmap", 'Gradients=""', "TaskBarShowStartMenu=1",
        "ShowMenuButtonIcon=0",
    ]
    if lane == "nextstep":
        lines.extend([
            "TitleBarCentered=1", "TitleBarJustify=50",
            'TitleButtonsLeft="m"', 'TitleButtonsRight="x"',
            'TitleButtonsSupported="mx"', "TitleBarHeight=23",
            "TitleBarHorzOffset=0", "TitleBarVertOffset=0",
            "BorderSizeX=1", "BorderSizeY=1", "CornerSizeX=1", "CornerSizeY=1",
            "DlgBorderSizeX=1", "DlgBorderSizeY=1", "ButtonX=16", "ButtonY=16",
        ])
    elif lane == "gnustep":
        lines.extend([
            "TitleBarCentered=1", "TitleBarJustify=50",
            'TitleButtonsLeft="sr"', 'TitleButtonsRight="xim"',
            'TitleButtonsSupported="xmirhs"', "TitleBarHeight=22",
            "TitleBarHorzOffset=0", "TitleBarVertOffset=0",
            "BorderSizeX=1", "BorderSizeY=1", "CornerSizeX=1", "CornerSizeY=1",
            "DlgBorderSizeX=1", "DlgBorderSizeY=1", "ButtonX=21", "ButtonY=21",
        ])
    else:
        lines.extend([
            "TitleBarCentered=1", "TitleBarJustify=50",
            'TitleButtonsLeft="x"', 'TitleButtonsRight="im"',
            'TitleButtonsSupported="xmris"', "TitleBarHeight=19",
            "TitleBarHorzOffset=0", "TitleBarVertOffset=0",
            "BorderSizeX=6", "BorderSizeY=6", "CornerSizeX=23", "CornerSizeY=23",
            "DlgBorderSizeX=6", "DlgBorderSizeY=6", "ButtonX=22", "ButtonY=22",
        ])
    lines.append(
        'TitleFontNameXft="Charcoal:size=9"'
        if lane == "rhapsody" else 'TitleFontNameXft="Charcoal:size=8:bold"'
    )
    lines.extend(f'{key}="{border.ice_colour(colours[key])}"' for key in tosh_keys)
    path.write_text("\n".join(lines) + "\n", encoding="latin-1")


def build_theme(repo: Path, lane: str, row: dict[str, str], border) -> Path:
    spec = LANES[lane]
    row = lane_row(row, lane)
    name = f"NextStep-{spec['token']}-{row['token']}"
    output = repo / THEMES / name
    prepare_generated_dir(output)
    donor = repo / border.DONOR_RELATIVE
    start = None
    if lane == "nextstep":
        border.write_authentic_titles(donor, output, row)
        border.write_authentic_frames(output, row)
        border.write_authentic_buttons(output, row, "gray", "nextstep-wrap")
        start = border.write_authentic_start(output, repo)
    elif lane == "gnustep":
        border.write_canon_titles(donor, output, row)
        border.write_authentic_frames(output, row)
        border.write_canon_buttons(output, row, "black")
        start = border.write_authentic_start(output, repo)
    else:
        write_rhapsody_pixels(repo, output, row, border)
    default_theme(output / "default.theme", row, lane, row["label"], border)
    (output / "PROVENANCE").write_text(
        f"{spec['label']} approved canon variant\n"
        f"Palette: {row['source']} ({row['label']})\n"
        f"Palette border: {row['border_color']}\n"
        + ("Geometry: authentic NeXTSTEP/OpenStep 23px canon\n"
           if lane == "nextstep" else
           "Geometry: GNUstep 22px pass-9b canon\n"
           if lane == "gnustep" else
           "Geometry: Tosh/X230 donor frame, title, and button XPMs only\n")
        + (f"Taskbar Start: {start['target']} canvas={start['canvas']}\n" if start else "")
        + "Payload: no Tosh menu, logout, switch, taskbar, or dock payload\n"
        + f"{MARKER}\n",
        encoding="utf-8",
    )
    return output

def qt6ct(path: Path, row: dict[str, str], border) -> None:
    text = border.contrast(row, row["base_color"])
    selected = border.contrast(row, row["scale_color"])
    colours = [
        text, row["base_color"], row["lighter_color"], row["lighter_color"],
        row["darker_color"], row["darker_color"], text, "#ffffff", text,
        "#ffffff", row["base_color"], text, row["scale_color"], selected,
        "#0000ff", "#800080", row["lighter_color"], text, "#ffffe1",
        "#000000", row["darker_color"], row["scale_color"],
    ]
    encoded = ", ".join("#ff" + value.lstrip("#") for value in colours)
    path.write_text(
        f"[ColorScheme]\nactive_colors={encoded}\n"
        f"disabled_colors={encoded}\ninactive_colors={encoded}\n",
        encoding="ascii",
    )


def wine(path: Path, row: dict[str, str], border) -> None:
    def decimal(value: str) -> str:
        return " ".join(str(int(value[index:index + 2], 16)) for index in (1, 3, 5))
    values = {
        "ActiveBorder": row["base_color"], "ActiveTitle": row["active_title"],
        "AppWorkspace": row["darker_color"], "Background": row["darker_color"],
        "ButtonDkShadow": "#000000", "ButtonFace": row["base_color"],
        "ButtonHilight": row["lighter_color"], "ButtonLight": row["lighter_color"],
        "ButtonShadow": row["darker_color"],
        "ButtonText": border.contrast(row, row["base_color"]),
        "GradientActiveTitle": row["active_title"],
        "GradientInactiveTitle": row["inactive_title"],
        "GrayText": row["darker_color"], "Hilight": row["scale_color"],
        "HilightText": border.contrast(row, row["scale_color"]),
        "InactiveBorder": row["base_color"], "InactiveTitle": row["inactive_title"],
        "InactiveTitleText": border.contrast(row, row["inactive_title"]),
        "InfoText": "#000000", "InfoWindow": "#ffffe1", "Menu": row["base_color"],
        "MenuText": border.contrast(row, row["base_color"]),
        "Scrollbar": row["base_color"],
        "TitleText": border.contrast(row, row["active_title"]),
        "Window": row["lighter_color"], "WindowFrame": row["border_color"],
        "WindowText": border.contrast(row, row["lighter_color"]),
    }
    lines = ["REGEDIT4", "", r"[HKEY_CURRENT_USER\Control Panel\Colors]"]
    lines.extend(f'"{key}"="{decimal(value)}"' for key, value in values.items())
    path.write_text("\n".join(lines) + "\n", encoding="ascii")


def build_rhapsody_icons(repo: Path) -> None:
    donor = repo / ICONS / "Platinum9/NineIcons"
    output = repo / ICONS / RHAPSODY_ICONS
    if not (donor / "index.theme").is_file():
        raise SystemExit(f"Rhapsody NineIcons donor missing: {donor}")
    prepare_generated_dir(output)
    index = (donor / "index.theme").read_text(encoding="utf-8")
    index = re.sub(r"^Name=.*$", "Name=Redstone Rhapsody", index, flags=re.MULTILINE)
    index = re.sub(
        r"^Comment=.*$", "Comment=Rhapsody projection of Platinum9 NineIcons with Redstone fallbacks",
        index, flags=re.MULTILINE,
    )
    index = re.sub(
        r"^Inherits=.*$", "Inherits=Chicago95,BeOS-r5-Icons,hicolor",
        index, count=1, flags=re.MULTILINE,
    )
    (output / "index.theme").write_text(index, encoding="utf-8")
    for child in sorted(donor.iterdir()):
        if child.name.startswith(".") or not child.is_dir():
            continue
        (output / child.name).symlink_to(Path("../Platinum9/NineIcons") / child.name)
    (output / "PROVENANCE").write_text(
        "Redstone Rhapsody XDG projection\n"
        "Primary: Platinum9/NineIcons (symlinked, not copied)\n"
        "Fallback: Chicago95, BeOS-r5-Icons, hicolor\n"
        "No KDE-1 icon base\n"
        f"{MARKER}\n",
        encoding="utf-8",
    )


def build_all(repo: Path, selected_lane: str | None = None,
              selected_palette: str | None = None) -> None:
    border = load_border_builder(repo)
    palette_rows = rows(repo, border)
    if selected_palette:
        palette_rows = [row for row in palette_rows if row["slug"] == selected_palette]
        if not palette_rows:
            raise SystemExit(f"unknown palette slug: {selected_palette}")
    lane_names = [selected_lane] if selected_lane else list(LANES)
    for directory in (repo / SCHEMES, repo / QT6CT, repo / WINE):
        directory.mkdir(parents=True, exist_ok=True)
    retire_legacy_public_names(repo, palette_rows)
    build_rhapsody_icons(repo)
    for row in palette_rows:
        qt6ct(repo / QT6CT / f"nextstep-{row['slug']}.conf", row, border)
        wine(repo / WINE / f"nextstep-{row['slug']}.reg", row, border)
        for lane in lane_names:
            output = build_theme(repo, lane, row, border)
            scheme(
                repo / SCHEMES / f"nextstep-{lane}-{row['slug']}.scheme",
                lane, row, border,
            )
            print(f"built {lane}:{row['slug']} -> {output.name}")


def digest(paths: list[Path]) -> str:
    checksum = hashlib.sha256()
    for path in paths:
        checksum.update(path.read_bytes())
    return checksum.hexdigest()


def check(repo: Path) -> None:
    border = load_border_builder(repo)
    palette_rows = rows(repo, border)
    all_expected_themes = set()
    required_scheme_fields = (
        "SCHEME_TITLE", "ICEWM_THEME", "GTK_THEME", "ICON_THEME",
        "CURSOR_THEME", "CURSOR_SIZE", "GTK_FONT", "DESKTOP_BG",
        "DESKTOP_ICON_LABEL_MODE", "DESKTOP_ICON_LABEL_FOREGROUND",
        "DESKTOP_ICON_LABEL_SHADOW_COLOR", "SEL_BG", "SEL_FG",
        "QT6CT_COLORS", "WINE_COLORS", "QT_STYLE", "SOUND_SCHEME",
        "FOX_BASE", "FOX_SEL_BG", "FOX_SEL_FG", "FOX_MENU_BG",
        "FOX_MENU_FG", "NEXTSTEP_DOCKAPPS", "ICEWM_PREFS",
    )
    for lane, spec in LANES.items():
        expected_themes = {
            f"NextStep-{spec['token']}-{row['token']}" for row in palette_rows
        }
        expected_schemes = {
            f"nextstep-{lane}-{row['slug']}.scheme" for row in palette_rows
        }
        actual_themes = {
            path.name for path in (repo / THEMES).glob(f"NextStep-{spec['token']}-*")
            if path.is_dir()
        }
        actual_schemes = {
            path.name for path in (repo / SCHEMES).glob(f"nextstep-{lane}-*.scheme")
            if path.is_file()
        }
        if actual_themes != expected_themes or actual_schemes != expected_schemes:
            raise SystemExit(
                f"{lane}: matrix set mismatch "
                f"missing themes={sorted(expected_themes - actual_themes)} "
                f"extra themes={sorted(actual_themes - expected_themes)} "
                f"missing schemes={sorted(expected_schemes - actual_schemes)} "
                f"extra schemes={sorted(actual_schemes - expected_schemes)}"
            )
        all_expected_themes.update(expected_themes)
        themes = [repo / THEMES / name for name in sorted(expected_themes)]
        schemes = [repo / SCHEMES / name for name in sorted(expected_schemes)]
        fingerprints = set()
        for theme in themes:
            palette = next(
                (row for row in palette_rows if theme.name.endswith(row["token"])),
                None,
            )
            if palette is None:
                raise SystemExit(f"cannot match theme to palette row: {theme}")
            provenance = (theme / "PROVENANCE").read_text(encoding="utf-8")
            if MARKER not in provenance:
                raise SystemExit(f"unmarked theme in matrix: {theme}")
            contamination = FORBIDDEN_TOSH.intersection(path.name for path in theme.iterdir())
            if contamination:
                raise SystemExit(f"{theme}: forbidden Tosh payload: {sorted(contamination)}")
            samples = [theme / "default.theme", theme / "titleAM.xpm"]
            if lane == "rhapsody":
                expected_xpms = {
                    *(f"frame{suffix}{stem}.xpm" for suffix in "AI"
                      for stem in FRAME_STEMS),
                    *(f"dframe{suffix}{stem}.xpm" for suffix in "AI"
                      for stem in FRAME_STEMS),
                    *(f"title{suffix}{stem}.xpm" for suffix in "AI"
                      for stem in TITLE_STEMS),
                    *(f"{stem}{suffix}.xpm" for suffix in "AI"
                      for stem in BUTTON_STEMS),
                }
                actual_xpms = {path.name for path in theme.glob("*.xpm")}
                if actual_xpms != expected_xpms:
                    raise SystemExit(
                        f"{theme}: Rhapsody XPM manifest mismatch "
                        f"missing={sorted(expected_xpms - actual_xpms)} "
                        f"extra={sorted(actual_xpms - expected_xpms)}"
                    )
                samples.append(theme / "frameATL.xpm")
                for expected, size in (("titleAM.xpm", (6, 19)),
                                       ("frameATL.xpm", (23, 23)),
                                       ("closeA.xpm", (16, 38))):
                    image = border.read_xpm(theme / expected)
                    if (image.width, image.height) != size:
                        raise SystemExit(f"{theme}/{expected}: expected {size}")
                if (theme / "taskbar/start.xpm").exists():
                    raise SystemExit(f"{theme}: Rhapsody must not carry the NeXT Start image")
                allowed = {
                    palette[key].lower() for key in border.REQUIRED_PALETTE_KEYS
                } | {rhapsody_active(palette).lower(),
                     palette["inactive_title"].lower()}
                emitted = {
                    colour.lower()
                    for path in theme.glob("*.xpm")
                    for colour in border.read_xpm(path).colours.values()
                    if colour is not None
                }
                if not emitted <= allowed:
                    raise SystemExit(
                        f"{theme}: undeclared Rhapsody colours: {sorted(emitted - allowed)}"
                    )
            else:
                samples.append(theme / "closeA.xpm")
                if not (theme / "taskbar/start.xpm").is_file():
                    raise SystemExit(f"{theme}: missing approved Start image")
            fingerprints.add(digest(samples))
        if len(fingerprints) < 24:
            raise SystemExit(f"{lane}: palette response collapsed to {len(fingerprints)} variants")
        for path in schemes:
            text = path.read_text(encoding="utf-8")
            for key in required_scheme_fields:
                if not re.search(rf"^{key}=", text, re.MULTILINE):
                    raise SystemExit(f"{path}: missing {key}")
            slug = path.name.removeprefix(f"nextstep-{lane}-").removesuffix(".scheme")
            palette = next(row for row in palette_rows if row["slug"] == slug)
            expected_theme = f"NextStep-{spec['token']}-{palette['token']}"
            if f'ICEWM_THEME="{expected_theme}"' not in text:
                raise SystemExit(f"{path}: broken ICEWM_THEME cross-link")
            expected_gtk = "PLATINUM" if lane == "rhapsody" else palette["gtk"]
            if f'GTK_THEME="{expected_gtk}"' not in text:
                raise SystemExit(f"{path}: wrong GTK theme")
            if f'ICON_THEME="{spec["icon"]}"' not in text:
                raise SystemExit(f"{path}: wrong icon theme")
            if f'NEXTSTEP_DOCKAPPS="{1 if spec["dock"] else 0}"' not in text:
                raise SystemExit(f"{path}: wrong dock policy")
            if f'QT6CT_COLORS="nextstep-{slug}.conf"' not in text:
                raise SystemExit(f"{path}: broken QT6CT cross-link")
            if f'WINE_COLORS="nextstep-{slug}.reg"' not in text:
                raise SystemExit(f"{path}: broken Wine cross-link")
            if re.search(r"KDE-1|KDE1", text):
                raise SystemExit(f"{path}: forbidden KDE-1 icon reference")
            sourced = subprocess.run(
                ["bash", "-c", 'set -euo pipefail; source "$1"',
                 "redstone-nextstep-check", str(path)],
                check=False, capture_output=True, text=True,
            )
            if sourced.returncode:
                raise SystemExit(f"{path}: bash source failed: {sourced.stderr.strip()}")
    if len(palette_rows) != 27:
        raise SystemExit("palette normalization no longer yields 27 rows")
    matrix_themes = {
        path.name for path in (repo / THEMES).glob("NextStep-*") if path.is_dir()
    }
    if matrix_themes != all_expected_themes or len(matrix_themes) != 81:
        raise SystemExit("matrix theme root contains stale or missing generated names")
    if any((repo / SCHEMES).glob("nextstep-platinum-*.scheme")):
        raise SystemExit("legacy Platinum scheme names remain after the Rhapsody rename")
    if (repo / ICONS / "Redstone-Platinum").exists():
        raise SystemExit("legacy Platinum icon projection remains after the Rhapsody rename")
    expected_qt = {f"nextstep-{row['slug']}.conf" for row in palette_rows}
    expected_wine = {f"nextstep-{row['slug']}.reg" for row in palette_rows}
    actual_qt = {path.name for path in (repo / QT6CT).glob("nextstep-*.conf")}
    actual_wine = {path.name for path in (repo / WINE).glob("nextstep-*.reg")}
    if actual_qt != expected_qt or actual_wine != expected_wine:
        raise SystemExit(
            "toolkit colour set mismatch: "
            f"QT missing={sorted(expected_qt - actual_qt)} extra={sorted(actual_qt - expected_qt)}; "
            f"Wine missing={sorted(expected_wine - actual_wine)} "
            f"extra={sorted(actual_wine - expected_wine)}"
        )
    rhapsody = repo / ICONS / RHAPSODY_ICONS
    index = (rhapsody / "index.theme").read_text(encoding="utf-8")
    if "Inherits=Chicago95,BeOS-r5-Icons,hicolor" not in index:
        raise SystemExit("Rhapsody icon projection lost its approved fallback chain")
    if "KDE" in index:
        raise SystemExit("Rhapsody icon projection contains KDE contamination")
    for name in ("GNUstep", "Irixium"):
        inherited = (repo / ICONS / name / "index.theme").read_text(encoding="utf-8")
        if "Chicago95,BeOS-r5-Icons,hicolor" not in inherited:
            raise SystemExit(f"{name} lost the approved terminal fallback chain")
    print("verified 81 themes, 81 schemes, 27 palette rows, and three icon chains")


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo-root", default=str(Path.home() / "DotCortex"))
    group = parser.add_mutually_exclusive_group(required=True)
    group.add_argument("--all", action="store_true")
    group.add_argument("--check", action="store_true")
    group.add_argument("--lane", choices=sorted(LANES))
    parser.add_argument("--palette", help="one normalized palette slug")
    args = parser.parse_args()
    repo = Path(args.repo_root).expanduser().resolve()
    if args.check:
        if args.palette:
            parser.error("--palette is incompatible with --check")
        check(repo)
    elif args.lane:
        if not args.palette:
            parser.error("--lane requires --palette")
        build_all(repo, args.lane, args.palette)
    else:
        if args.palette:
            parser.error("--palette requires --lane")
        build_all(repo)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
