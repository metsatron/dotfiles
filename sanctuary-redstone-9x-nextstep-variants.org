def one_stepback_gray_role(level: int) -> str:
    """Quantize OneStepBack's antialiased neutral ramp into Rhapsody roles."""
    if level < 32:
        return RHAPSODY_ROLES["black"]
    if level < 72:
        return RHAPSODY_ROLES["contour"]
    if level < 112:
        return RHAPSODY_ROLES["dark_edge_alt"]
    if level < 144:
        return RHAPSODY_ROLES["well_mid"]
    if level < 184:
        return RHAPSODY_ROLES["body"]
    if level < 216:
        return RHAPSODY_ROLES["light"]
    return RHAPSODY_ROLES["highlight"]

def load_border_builder(repo: Path):
    path = repo / BORDER_BUILDER
    if not path.is_file():
        raise SystemExit(f"tangle the border builder first: {path}")
    loader = importlib.machinery.SourceFileLoader("redstone_nextstep_borders", str(path))
    spec = importlib.util.spec_from_loader(loader.name, loader)
    if spec is None or spec.loader is None:
        raise SystemExit(f"cannot load border builder: {path}")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

def prepare_generated_dir(path: Path) -> None:
    if not path.exists() and not path.is_symlink():
        path.mkdir(parents=True)
        return
    if path.is_symlink() or not path.is_dir():
        raise SystemExit(f"refuse to replace non-directory generated target: {path}")
    provenance = path / "PROVENANCE"
    if not provenance.is_file() or MARKER not in provenance.read_text(encoding="utf-8"):
        raise SystemExit(f"refuse to replace unmarked theme directory: {path}")
    shutil.rmtree(path)
    path.mkdir(parents=True)


def remove_marked_generated_dir(path: Path) -> None:
    if not path.exists() and not path.is_symlink():
        return
    if path.is_symlink() or not path.is_dir():
        raise SystemExit(f"refuse to retire non-directory generated target: {path}")
    provenance = path / "PROVENANCE"
    if not provenance.is_file() or MARKER not in provenance.read_text(encoding="utf-8"):
        raise SystemExit(f"refuse to retire unmarked generated directory: {path}")
    shutil.rmtree(path)


def retire_legacy_public_names(repo: Path, palette_rows: list[dict[str, str]]) -> None:
    """Remove the rejected public family name after proving generator ownership."""
    for row in palette_rows:
        old_theme = repo / THEMES / f"NextStep-Platinum-{row['token']}"
        remove_marked_generated_dir(old_theme)
        old_scheme = repo / SCHEMES / f"nextstep-platinum-{row['slug']}.scheme"
        if old_scheme.is_file():
            text = old_scheme.read_text(encoding="utf-8")
            expected_theme = f'ICEWM_THEME="NextStep-Platinum-{row["token"]}"'
            if expected_theme not in text or "Redstone 9X Desktop Scheme" not in text:
                raise SystemExit(f"refuse to retire unrecognized legacy scheme: {old_scheme}")
            old_scheme.unlink()
    remove_marked_generated_dir(repo / ICONS / "Redstone-Platinum")


def rhapsody_active(row: dict[str, str]) -> str:
    if row["source"].startswith("one_step_back."):
        return row["base_color"]
    return row["active_title"]


def rhapsody_colour(border, old: str | None, row: dict[str, str],
                    face: str) -> str | None:
    del border
    if old is None:
        return None
    value = old.upper()
    if value in {"#000000", "#303030"}:
        return row["black_color"]
    if value == "#FFFFFF":
        return row["white_color"]
    if value in {"#555555", "#808080", "#888888"}:
        return row["darker_color"]
    if value in {"#999999", "#AAAAAA", "#BBBBBB", "#CCCCCC"}:
        return face
    if value in {"#CFD4DC", "#DDDDDD", "#EEEEEE", "#F8F8F8"}:
        return row["lighter_color"]
    raise SystemExit(f"unmapped Tosh donor colour for Rhapsody: {value}")


def recolour_rhapsody(border, source: Path, destination: Path,
                      row: dict[str, str], face: str) -> None:
    image = border.read_xpm(source)
    pixels = [
        [rhapsody_colour(border, image.colours[symbol], row, face)
         for symbol in line]
        for line in image.pixels
    ]
    border.write_xpm(
        destination, [colour for line in pixels for colour in line], pixels,
    )
    emitted = border.read_xpm(destination)
    if (emitted.width, emitted.height) != (image.width, image.height):
        raise SystemExit(f"Rhapsody geometry changed: {source.name}")


def write_rhapsody_pixels(repo: Path, output: Path, row: dict[str, str],
                          border) -> None:
    donor = repo / TOSH
    active = rhapsody_active(row)
    inactive = row["inactive_title"]
    for suffix, face in (("A", row["base_color"]), ("I", inactive)):
        for stem in FRAME_STEMS:
            source = donor / f"frame{suffix}{stem}.xpm"
            if not source.is_file():
                raise SystemExit(f"Tosh frame source missing: {source}")
            recolour_rhapsody(border, source, output / source.name, row, face)
            recolour_rhapsody(
                border, source, output / f"dframe{suffix}{stem}.xpm", row, face,
            )
    for suffix, face in (("A", active), ("I", inactive)):
        for stem in TITLE_STEMS:
            source = donor / f"title{suffix}{stem}.xpm"
            if not source.is_file():
                raise SystemExit(f"Tosh title source missing: {source}")
            recolour_rhapsody(border, source, output / source.name, row, face)
    for suffix, face in (("A", row["base_color"]), ("I", inactive)):
        for stem in BUTTON_STEMS:
            source = donor / f"{stem}{suffix}.xpm"
            if not source.is_file():
                raise SystemExit(f"Tosh button source missing: {source}")
            recolour_rhapsody(border, source, output / source.name, row, face)


def lane_row(row: dict[str, str], lane: str) -> dict[str, str]:
    result = dict(row)
    if lane in {"nextstep", "gnustep"}:
        # 2026-08-13 fix: this used to force every skew's active_title to a
        # fixed black_color, collapsing ColorActiveTitleBar and the titleA*/
        # buttonA* pixmaps identically across all 27 skews (GNUstep) or
        # leaving only the bevel palette-responsive (NeXT).  Mirror
        # rhapsody_active's own established fix for the same collapse risk:
        # one_step_back rows already carry a black active_title by design
        # (variant_palette_rows), so fall back to base_color there exactly
        # as Rhapsody does; qt_step rows keep their already palette-varying
        # active_title untouched.
        if result["source"].startswith("one_step_back."):
            result["active_title"] = result["base_color"]
    elif lane == "rhapsody":
        result["active_title"] = rhapsody_active(result)
    return result

def build_theme(repo: Path, lane: str, row: dict[str, str], border) -> Path:
    spec = LANES[lane]
    row = lane_row(row, lane)
    name = f"NextStep-{spec['token']}-{row['token']}"
    output = repo / THEMES / name
    prepare_generated_dir(output)
    donor = repo / border.DONOR_RELATIVE
    start = None
    if lane == "nextstep":
        border.write_authentic_titles(donor, output, row)
        border.write_authentic_frames(output, row)
        border.write_authentic_buttons(output, row, "gray", "nextstep-wrap")
        start = border.write_authentic_start(output, repo)
    elif lane == "gnustep":
        border.write_canon_titles(donor, output, row)
        border.write_authentic_frames(output, row)
        border.write_canon_buttons(output, row, "black")
        start = border.write_authentic_start(output, repo)
    else:
        write_rhapsody_pixels(repo, output, row, border)
    default_theme(output / "default.theme", row, lane, row["label"], border)
    (output / "PROVENANCE").write_text(
        f"{spec['label']} approved canon variant\n"
        f"Palette: {row['source']} ({row['label']})\n"
        f"Palette border: {row['border_color']}\n"
        + ("Geometry: authentic NeXTSTEP/OpenStep 23px canon\n"
           if lane == "nextstep" else
           "Geometry: GNUstep 22px pass-9b canon\n"
           if lane == "gnustep" else
           "Geometry: Tosh/X230 donor frame, title, and button XPMs only\n")
        + (f"Taskbar Start: {start['target']} canvas={start['canvas']}\n" if start else "")
        + "Payload: no Tosh menu, logout, switch, taskbar, or dock payload\n"
        + f"{MARKER}\n",
        encoding="utf-8",
    )
    return output

def trial_prepare_dir(path: Path, marker: str = TRIAL_MARKER) -> None:
    """Recreate only a marker-owned representative directory."""
    if path.exists() or path.is_symlink():
        if path.is_symlink() or not path.is_dir():
            raise SystemExit(f"refuse to replace non-directory trial target: {path}")
        provenance = path / "PROVENANCE"
        if not provenance.is_file() or marker not in provenance.read_text(encoding="utf-8"):
            raise SystemExit(f"refuse to replace unmarked trial target: {path}")
        shutil.rmtree(path)
    path.mkdir(parents=True, exist_ok=True)


def replace_donor_colours(text: str, mapping: dict[str, str]) -> str:
    pattern = re.compile(
        "|".join(re.escape(value) for value in mapping),
        flags=re.IGNORECASE,
    )
    return pattern.sub(
        lambda match: mapping[match.group(0).lower()], text,
    )


def normalize_text(text: str) -> str:
    text = text.replace("\r\n", "\n").rstrip(" \t\n")
    lines = [re.sub(r" +\t", "\t", line.rstrip(" \t"))
             for line in text.split("\n")]
    return "\n".join(lines) + "\n"

def image_converter() -> str:

def recolour_png(source: Path, destination: Path,
                 mapping: dict[str, str] | None = None) -> None:
    """Recolour a donor bitmap while preserving its geometry and alpha."""
    destination.parent.mkdir(parents=True, exist_ok=True)
    mapping = mapping or PLATINUM_COLOUR_MAP
    # Recolour RGB with alpha temporarily disabled, then restore the exact
    # donor alpha mask.  ImageMagick otherwise skips fully transparent RGB
    # samples, leaving hidden donor colours in the generated PNG payload.
    command = [
        image_converter(), str(source), "(", "+clone", "-alpha", "extract",
        "-write", "mpr:rhapsody-alpha", "+delete", ")", "-alpha", "off",
        "-channel", "RGB",
    ]
    for old, new in mapping.items():
        command.extend(["-fuzz", "0%", "-fill", new, "-opaque", old])
    command.extend([
        "+channel", "mpr:rhapsody-alpha", "-compose", "CopyOpacity",
        "-composite", str(destination),
    ])
    subprocess.run(command, check=True, capture_output=True, text=True)


def assert_no_png_colour(root: Path, forbidden: str) -> None:
    needle = forbidden.upper()
    for path in sorted(root.rglob("*.png")):
        result = subprocess.run(
            [image_converter(), str(path), "-depth", "8", "txt:-"],
            check=True, capture_output=True, text=True,
        )
        if needle in result.stdout.upper():
            raise SystemExit(f"forbidden donor pixel {forbidden} remains: {path}")


def assert_no_text_colours(root: Path, forbidden: set[str]) -> None:
    pattern = re.compile(
        "|".join(re.escape(value) for value in sorted(forbidden)),
        flags=re.IGNORECASE,
    )
    for path in sorted(root.rglob("*")):
        if not path.is_file() or path.suffix.lower() == ".png":
            continue
        text = path.read_text(encoding="utf-8", errors="replace")
        match = pattern.search(text)
        if match:
            raise SystemExit(f"forbidden donor text colour {match.group(0)} remains: {path}")


def assert_png_palette(root: Path, allowed: set[str]) -> None:
    for path in sorted(root.rglob("*.png")):
        result = subprocess.run(
            [image_converter(), str(path), "-depth", "8", "txt:-"],
            check=True, capture_output=True, text=True,
        )
        colours = {
            match.group(0).upper()
            for match in re.finditer(r"#[0-9A-Fa-f]{6}", result.stdout)
        }
        unexpected = colours - allowed
        if unexpected:
            raise SystemExit(
                f"unmapped OneStepBack pixels remain in {path}: {sorted(unexpected)}"
            )


def write_rgb_png(path: Path, width: int, height: int,
                  pixels: list[list[tuple[int, int, int]]]) -> None:
    """Write a deterministic, filter-zero RGB PNG without external modules."""
    def chunk(kind: bytes, payload: bytes) -> bytes:
        return (
            struct.pack(">I", len(payload)) + kind + payload
            + struct.pack(">I", zlib.crc32(kind + payload) & 0xFFFFFFFF)
        )

    raw = b"".join(
        b"\x00" + b"".join(bytes(pixel) for pixel in row)
        for row in pixels
    )
    payload = (
        b"\x89PNG\r\n\x1a\n"
        + chunk(b"IHDR", struct.pack(">IIBBBBB", width, height, 8, 2, 0, 0, 0))
        + chunk(b"IDAT", zlib.compress(raw, 9))
        + chunk(b"IEND", b"")
    )
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(payload)


def hex_rgb(value: str) -> tuple[int, int, int]:
    return tuple(int(value[index:index + 2], 16) for index in (1, 3, 5))

def build_rhapsody_gtk(repo: Path, output_name: Path = RHAPSODY_GTK,
                       donor_name: Path = PLATINUM,
                       lane_label: str = "Rhapsody") -> None:
    donor = repo / donor_name
    if not (donor / "LICENSE").is_file():
        raise SystemExit(f"GPL-3 GTK donor missing: {donor}")
    marker = (TRIAL_MARKER if output_name == RHAPSODY_GTK
              else f"{TRIAL_MARKER} --gtk={lane_label.lower()}")
    one_stepback = donor_name.name == "OneStepBack"
    text_mapping = (ONE_STEPBACK_TEXT_COLOUR_MAP
                    if one_stepback else PLATINUM_COLOUR_MAP)
    trial_prepare_dir(repo / output_name, marker)
    output = repo / output_name
    for source_root in ("gtk-2.0", "gtk-3.0", "gtk-4.0"):
        root = donor / source_root
        for source in sorted(root.rglob("*")):
            if source.is_dir() or source.suffix.lower() == ".xcf":
                continue
            target = output / source.relative_to(donor)
            target.parent.mkdir(parents=True, exist_ok=True)
            if source.suffix.lower() == ".png":
                mapping = (dict(ONE_STEPBACK_BITMAP_COLOUR_MAP)
                           if one_stepback else dict(PLATINUM_COLOUR_MAP))
                if "scroll" in str(source).lower() or "slider" in source.name.lower():
                    mapping.update(ONE_STEPBACK_SCROLLBAR_COLOUR_MAP
                                   if one_stepback else PLATINUM_SCROLLBAR_COLOUR_MAP)
                recolour_png(source, target, mapping)
            elif source.name == "gtkrc" or source.suffix.lower() in {
                ".css", ".rc", ".theme", ".txt", ".index", ".xpm",
            }:
                text = replace_donor_colours(
                    source.read_text(encoding="utf-8", errors="replace"),
                    text_mapping,
                )
                if source.name == "gtkrc":
                    lines = [
                        gtk2_scheme_line() if "gtk_color_scheme" in line else line
                        for line in text.splitlines()
                    ]
                    text = "\n".join(lines) + "\n"
                    text += (
                        "\n# Rhapsody Canonical 16px deep-well scrollbar.\n"
                        'style "rhapsody-scrollbar" {\n'
                        "  GtkRange::slider-width = 16\n"
                        "  GtkRange::stepper-size = 16\n"
                        "  GtkRange::trough-border = 1\n"
                        "}\n"
                        'class "GtkScrollbar" style "rhapsody-scrollbar"\n'
                    )
                elif source.name == "scrollbar.rc":
                    text += (
                        "\n# Rhapsody Canonical deep neutral well.\n"
                        'style "rhapsody-scrollbar-well" {\n'
                        '  bg[NORMAL] = "#535353"\n'
                        '  base[NORMAL] = "#808080"\n'
                        "}\n"
                    )
                elif source.name == "gtk.css":
                    version = "gtk3" if source_root == "gtk-3.0" else "gtk4"
                    if version == "gtk4":
                        text = re.sub(
                            r"^@define-color .*;$",
                            lambda match: "/* " + match.group(0) + " */",
                            text, flags=re.MULTILINE,
                        )
                    text += rhapsody_gtk_css(version)
                if donor_name.name == "PLATINUM":
                    text = text.replace("box-shadow: inset 1 0px", "box-shadow: inset 1px 0px")
                    text = text.replace(", inset 0 1px", ", inset 0px 1px")
                    text = text.replace(", inset -2 0px", ", inset -2px 0px")
                    text = text.replace('"Frame-Gap/frame-gap.png"', '"Frame-Gap/shadow-gap.png"')
                    text = text.replace('"Frame-Gap/frame1.png"', '"Frame-Gap/shadow-gap.png"')
                    text = text.replace('"Others/bg-tiled.png"', '"Others/bg-tiled.xpm"')
                    text = text.replace('"Others/window-bg.png"', '"Others/metal.png"')
                    text = text.replace('"Shadows/text-entry-focus.png"', '"Shadows/focus-entry.png"')
                    text = text.replace('"Spin/spin-transparent.png"', '"Spin/spin-bg.png"')
                    text = text.replace('"Scrollbars/slider-horizontal-overlay-insensitive.png"', '"Scrollbars/h-overlay.png"')
                    text = text.replace('"Scrollbars/slider-vertical-overlay-insensitive.png"', '"Scrollbars/v-overlay.png"')
                target.write_text(normalize_text(text), encoding="utf-8")
            else:
                shutil.copy2(source, target)
    shared_images = donor / "img"
    for source in sorted(shared_images.rglob("*")):
        if source.is_dir() or source.suffix.lower() != ".png":
            continue
        target = output / source.relative_to(donor)
        mapping = (dict(ONE_STEPBACK_BITMAP_COLOUR_MAP)
                   if one_stepback else dict(PLATINUM_COLOUR_MAP))
        if "scrollbar" in source.name.lower():
            mapping.update(ONE_STEPBACK_SCROLLBAR_COLOUR_MAP
                           if one_stepback else PLATINUM_SCROLLBAR_COLOUR_MAP)
        recolour_png(source, target, mapping)
    # Preserve the donor geometry while repairing its historical filename
    # typo, because PLATINUM's CSS refers to the corrected public name.
    typo = shared_images / "menuitem-arrow-right-diasbled.png"
    if typo.is_file():
        recolour_png(typo, output / "img/menuitem-arrow-right-disabled.png")
    # GTK3 and GTK4 both reference a 16px coloured thumb and well asset;
    # these preserve the selected donor's geometry rather than inventing art.
    if one_stepback:
        source_thumb = donor / "img/scrollbar_thumb.png"
        source_well = donor / "img/scrollbar_vertical.png"
    else:
        source_thumb = donor / "gtk-2.0/Scrollbars/slider-vertical.png"
        source_well = donor / "gtk-2.0/Scrollbars/trough-scrollbar-vert.png"
    for target_name, source in (
        ("rhapsody-scrollbar-thumb.png", source_thumb),
        ("rhapsody-scrollbar-well.png", source_well),
    ):
        mapping = (dict(ONE_STEPBACK_BITMAP_COLOUR_MAP)
                   if one_stepback else dict(PLATINUM_COLOUR_MAP))
        mapping.update(ONE_STEPBACK_SCROLLBAR_COLOUR_MAP
                       if one_stepback else PLATINUM_SCROLLBAR_COLOUR_MAP)
        recolour_png(source, output / "img" / target_name, mapping)
    (output / "index.theme").write_text(
        "[Desktop Entry]\nType=X-GNOME-Metatheme\n"
        f"Name={output.name}\n"
        f"Comment=Screenshot-matched Rhapsody GTK {lane_label} representative\n\n"
        f"[X-GNOME-Metatheme]\nGtkTheme={output.name}\n",
        encoding="utf-8",
    )
    license_name = (
        "LICENSE-PLATINUM-GPL-3" if donor_name.name == "PLATINUM"
        else "LICENSE-ONESTEPBACK-GPL-3"
    )
    (output / license_name).write_text(
        (donor / "LICENSE").read_text(encoding="utf-8"), encoding="utf-8",
    )
    (output / "PROVENANCE").write_text(
        f"{marker}\n"
        f"GTK geometry donor: {donor_name}\n"
        f"GTK donor license: GPL-3, copied as {license_name}\n"
        "Palette: screenshot-derived Rhapsody roles; donor static role colours replaced\n"
        "Scrollbar: 16px thumb, deep neutral well, wallpaper-responsive lavender highlight\n"
        + ("Alert glyph: donor #D74343 retained for the deliberate warning state\n"
           if one_stepback else "")
        + f"Reference screenshot SHA-256: {RHAPSODY_SCREENSHOT_SHA256}\n",
        encoding="utf-8",
    )
    assert_no_png_colour(output, "#637FFF")
    if one_stepback:
        assert_no_text_colours(output, set(ONE_STEPBACK_TEXT_COLOUR_MAP))
        assert_png_palette(output, ONE_STEPBACK_ALLOWED_PNG_COLOURS)

def build_rhapsody_canon(repo: Path, screenshot: Path) -> None:
    if not screenshot.is_file():
        raise SystemExit(f"Rhapsody reference screenshot missing: {screenshot}")
    digest_value = hashlib.sha256(screenshot.read_bytes()).hexdigest()
    if digest_value != RHAPSODY_SCREENSHOT_SHA256:
        raise SystemExit(
            f"Rhapsody reference screenshot SHA-256 changed: {digest_value}"
        )
    build_rhapsody_gtk(repo)
    build_rhapsody_border(repo)
    build_rhapsody_wallpaper(repo)
    build_rhapsody_menu_button(repo)
    build_rhapsody_toolkit_colours(repo)
    build_rhapsody_scheme(repo)
    print(f"built rhapsody-canon -> {RHAPSODY_TRIAL}")
    print(f"reference_sha256={digest_value}")
    print(f"desktop_bg={RHAPSODY_DESKTOP_BG} dither_tile=4x4")
    print("roles=" + ",".join(f"{key}:{value}" for key, value in RHAPSODY_ROLES.items()))
    print(f"gtk_theme={RHAPSODY_GTK}")
    print(f"scheme={RHAPSODY_SCHEME}")
    print(f"menu_button={RHAPSODY_MENU_IMAGE}")

def build_all(repo: Path, selected_lane: str | None = None,
              selected_palette: str | None = None) -> None:
    border = load_border_builder(repo)
    palette_rows = rows(repo, border)
    if selected_palette:
        palette_rows = [row for row in palette_rows if row["slug"] == selected_palette]
        if not palette_rows:
            raise SystemExit(f"unknown palette slug: {selected_palette}")
    lane_names = [selected_lane] if selected_lane else list(LANES)
    for directory in (repo / SCHEMES, repo / QT6CT, repo / WINE):
        directory.mkdir(parents=True, exist_ok=True)
    retire_legacy_public_names(repo, palette_rows)
    build_rhapsody_icons(repo)
    for row in palette_rows:
        qt6ct(repo / QT6CT / f"nextstep-{row['slug']}.conf", row, border)
        wine(repo / WINE / f"nextstep-{row['slug']}.reg", row, border)
        for lane in lane_names:
            output = build_theme(repo, lane, row, border)
            scheme(
                repo / SCHEMES / f"nextstep-{lane}-{row['slug']}.scheme",
                lane, row, border,
            )
            print(f"built {lane}:{row['slug']} -> {output.name}")


def digest(paths: list[Path]) -> str:
    checksum = hashlib.sha256()
    for path in paths:
        checksum.update(path.read_bytes())
    return checksum.hexdigest()


def check(repo: Path) -> None:
    border = load_border_builder(repo)
    palette_rows = rows(repo, border)
    all_expected_themes = set()
    required_scheme_fields = (
        "SCHEME_TITLE", "ICEWM_THEME", "GTK_THEME", "ICON_THEME",
        "CURSOR_THEME", "CURSOR_SIZE", "GTK_FONT", "DESKTOP_BG",
        "DESKTOP_ICON_LABEL_MODE", "DESKTOP_ICON_LABEL_FOREGROUND",
        "DESKTOP_ICON_LABEL_SHADOW_COLOR", "SEL_BG", "SEL_FG",
        "QT6CT_COLORS", "WINE_COLORS", "QT_STYLE", "SOUND_SCHEME",
        "FOX_BASE", "FOX_SEL_BG", "FOX_SEL_FG", "FOX_MENU_BG",
        "FOX_MENU_FG", "NEXTSTEP_DOCKAPPS", "ICEWM_PREFS",
    )
    for lane, spec in LANES.items():
        expected_themes = {
            f"NextStep-{spec['token']}-{row['token']}" for row in palette_rows
        }
        expected_schemes = {
            f"nextstep-{lane}-{row['slug']}.scheme" for row in palette_rows
        }
        actual_themes = {
            path.name for path in (repo / THEMES).glob(f"NextStep-{spec['token']}-*")
            if path.is_dir() and path.name not in RHAPSODY_CROSS_THEME_NAMES
        }
        actual_schemes = {
            path.name for path in (repo / SCHEMES).glob(f"nextstep-{lane}-*.scheme")
            if path.is_file() and path.name not in RHAPSODY_CROSS_SCHEME_NAMES
        }
        if actual_themes != expected_themes or actual_schemes != expected_schemes:
            raise SystemExit(
                f"{lane}: matrix set mismatch "
                f"missing themes={sorted(expected_themes - actual_themes)} "
                f"extra themes={sorted(actual_themes - expected_themes)} "
                f"missing schemes={sorted(expected_schemes - actual_schemes)} "
                f"extra schemes={sorted(actual_schemes - expected_schemes)}"
            )
        all_expected_themes.update(expected_themes)
        themes = [repo / THEMES / name for name in sorted(expected_themes)]
        schemes = [repo / SCHEMES / name for name in sorted(expected_schemes)]
        fingerprints = set()
        for theme in themes:
            palette = next(
                (row for row in palette_rows if theme.name.endswith(row["token"])),
                None,
            )
            if palette is None:
                raise SystemExit(f"cannot match theme to palette row: {theme}")
            provenance = (theme / "PROVENANCE").read_text(encoding="utf-8")
            if MARKER not in provenance:
                raise SystemExit(f"unmarked theme in matrix: {theme}")
            contamination = FORBIDDEN_TOSH.intersection(path.name for path in theme.iterdir())
            if contamination:
                raise SystemExit(f"{theme}: forbidden Tosh payload: {sorted(contamination)}")
            samples = [theme / "default.theme", theme / "titleAM.xpm"]
            if lane == "rhapsody":
                expected_xpms = {
                    *(f"frame{suffix}{stem}.xpm" for suffix in "AI"
                      for stem in FRAME_STEMS),
                    *(f"dframe{suffix}{stem}.xpm" for suffix in "AI"
                      for stem in FRAME_STEMS),
                    *(f"title{suffix}{stem}.xpm" for suffix in "AI"
                      for stem in TITLE_STEMS),
                    *(f"{stem}{suffix}.xpm" for suffix in "AI"
                      for stem in BUTTON_STEMS),
                }
                actual_xpms = {path.name for path in theme.glob("*.xpm")}
                if actual_xpms != expected_xpms:
                    raise SystemExit(
                        f"{theme}: Rhapsody XPM manifest mismatch "
                        f"missing={sorted(expected_xpms - actual_xpms)} "
                        f"extra={sorted(actual_xpms - expected_xpms)}"
                    )
                samples.append(theme / "frameATL.xpm")
                for expected, size in (("titleAM.xpm", (6, 19)),
                                       ("frameATL.xpm", (23, 23)),
                                       ("closeA.xpm", (16, 38))):
                    image = border.read_xpm(theme / expected)
                    if (image.width, image.height) != size:
                        raise SystemExit(f"{theme}/{expected}: expected {size}")
                if (theme / "taskbar/start.xpm").exists():
                    raise SystemExit(f"{theme}: Rhapsody must not carry the NeXT Start image")
                allowed = {
                    palette[key].lower() for key in border.REQUIRED_PALETTE_KEYS
                } | {rhapsody_active(palette).lower(),
                     palette["inactive_title"].lower()}
                emitted = {
                    colour.lower()
                    for path in theme.glob("*.xpm")
                    for colour in border.read_xpm(path).colours.values()
                    if colour is not None
                }
                if not emitted <= allowed:
                    raise SystemExit(
                        f"{theme}: undeclared Rhapsody colours: {sorted(emitted - allowed)}"
                    )
            else:
                samples.append(theme / "closeA.xpm")
                if not (theme / "taskbar/start.xpm").is_file():
                    raise SystemExit(f"{theme}: missing approved Start image")
            fingerprints.add(digest(samples))
        if len(fingerprints) < 24:
            raise SystemExit(f"{lane}: palette response collapsed to {len(fingerprints)} variants")
        for path in schemes:
            text = path.read_text(encoding="utf-8")
            for key in required_scheme_fields:
                if not re.search(rf"^{key}=", text, re.MULTILINE):
                    raise SystemExit(f"{path}: missing {key}")
            slug = path.name.removeprefix(f"nextstep-{lane}-").removesuffix(".scheme")
            palette = next(row for row in palette_rows if row["slug"] == slug)
            expected_theme = f"NextStep-{spec['token']}-{palette['token']}"
            if f'ICEWM_THEME="{expected_theme}"' not in text:
                raise SystemExit(f"{path}: broken ICEWM_THEME cross-link")
            expected_gtk = "PLATINUM" if lane == "rhapsody" else palette["gtk"]
            if f'GTK_THEME="{expected_gtk}"' not in text:
                raise SystemExit(f"{path}: wrong GTK theme")
            if f'ICON_THEME="{spec["icon"]}"' not in text:
                raise SystemExit(f"{path}: wrong icon theme")
            if f'NEXTSTEP_DOCKAPPS="{1 if spec["dock"] else 0}"' not in text:
                raise SystemExit(f"{path}: wrong dock policy")
            if f'QT6CT_COLORS="nextstep-{slug}.conf"' not in text:
                raise SystemExit(f"{path}: broken QT6CT cross-link")
            if f'WINE_COLORS="nextstep-{slug}.reg"' not in text:
                raise SystemExit(f"{path}: broken Wine cross-link")
            if re.search(r"KDE-1|KDE1", text):
                raise SystemExit(f"{path}: forbidden KDE-1 icon reference")
            sourced = subprocess.run(
                ["bash", "-c", 'set -euo pipefail; source "$1"',
                 "redstone-nextstep-check", str(path)],
                check=False, capture_output=True, text=True,
            )
            if sourced.returncode:
                raise SystemExit(f"{path}: bash source failed: {sourced.stderr.strip()}")
    if len(palette_rows) != 27:
        raise SystemExit("palette normalization no longer yields 27 rows")
    matrix_themes = {
        path.name for path in (repo / THEMES).glob("NextStep-*")
        if path.is_dir() and path.name not in RHAPSODY_CROSS_THEME_NAMES
    }
    if matrix_themes != all_expected_themes or len(matrix_themes) != 81:
        raise SystemExit("matrix theme root contains stale or missing generated names")
    if any((repo / SCHEMES).glob("nextstep-platinum-*.scheme")):
        raise SystemExit("legacy Platinum scheme names remain after the Rhapsody rename")
    if (repo / ICONS / "Redstone-Platinum").exists():
        raise SystemExit("legacy Platinum icon projection remains after the Rhapsody rename")
    expected_qt = {f"nextstep-{row['slug']}.conf" for row in palette_rows}
    expected_wine = {f"nextstep-{row['slug']}.reg" for row in palette_rows}
    actual_qt = {
        path.name for path in (repo / QT6CT).glob("nextstep-*.conf")
        if path.name != RHAPSODY_QT.name
    }
    actual_wine = {
        path.name for path in (repo / WINE).glob("nextstep-*.reg")
        if path.name != RHAPSODY_WINE.name
    }
    if actual_qt != expected_qt or actual_wine != expected_wine:
        raise SystemExit(
            "toolkit colour set mismatch: "
            f"QT missing={sorted(expected_qt - actual_qt)} extra={sorted(actual_qt - expected_qt)}; "
            f"Wine missing={sorted(expected_wine - actual_wine)} "
            f"extra={sorted(actual_wine - expected_wine)}"
        )
    rhapsody = repo / ICONS / RHAPSODY_ICONS
    index = (rhapsody / "index.theme").read_text(encoding="utf-8")
    if "Inherits=Chicago95,BeOS-r5-Icons,hicolor" not in index:
        raise SystemExit("Rhapsody icon projection lost its approved fallback chain")
    if "KDE" in index:
        raise SystemExit("Rhapsody icon projection contains KDE contamination")
    for name in ("GNUstep", "Irixium"):
        inherited = (repo / ICONS / name / "index.theme").read_text(encoding="utf-8")
        if "Chicago95,BeOS-r5-Icons,hicolor" not in inherited:
            raise SystemExit(f"{name} lost the approved terminal fallback chain")
    print("verified 81 themes, 81 schemes, 27 palette rows, and three icon chains")


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo-root", default=str(Path.home() / "DotCortex"))
    group = parser.add_mutually_exclusive_group(required=True)
    group.add_argument("--all", action="store_true")
    group.add_argument("--check", action="store_true")
    group.add_argument("--lane", choices=sorted(LANES))
    group.add_argument(
        "--rhapsody-canon", action="store_true",
        help="build exactly one screenshot-matched Rhapsody representative",
    )
    group.add_argument(
        "--rhapsody-crosslane", choices=("both", "gnustep"),
        help="build both cross-lane representatives, or only the GNUstep representative",
    )
    parser.add_argument(
        "--rhapsody-reference", type=Path,
        help="required reference screenshot for bounded Rhapsody builds",
    )
    parser.add_argument("--palette", help="one normalized palette slug")
    args = parser.parse_args()
    repo = Path(args.repo_root).expanduser().resolve()
    if args.rhapsody_canon or args.rhapsody_crosslane:
        if args.palette:
            parser.error("--palette is incompatible with bounded Rhapsody builds")
        if args.rhapsody_reference is None:
            parser.error("--rhapsody-reference is required for bounded Rhapsody builds")
        reference = args.rhapsody_reference.expanduser().resolve()
        if args.rhapsody_canon:
            build_rhapsody_canon(repo, reference)
        if args.rhapsody_crosslane:
            build_rhapsody_crosslanes(repo, reference, args.rhapsody_crosslane)
    elif args.check:
        if args.palette:
            parser.error("--palette is incompatible with --check")
        check(repo)
    elif args.lane:
        if not args.palette:
            parser.error("--lane requires --palette")
        build_all(repo, args.lane, args.palette)
    else:
        if args.palette:
            parser.error("--palette requires --lane")
        build_all(repo)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
