* Purpose

QEMU/KVM launch renderers for all KVM-tier Virtual Habitat guests. Each VM section
declares its disk, display, network, and virtiofs home mount. The canonical guest identity
(arch, phase, residency tier) lives in =guests.org=.

libvirt integration is an optional layer — scripts here work with plain =qemu-system-x86_64=
and can be adapted to =virsh= / =virt-manager= later.

* Azure Neptune Enterprise Server

KVM VM — active parallel build track, independent of the three Distrobox rooms.
Azure Linux 4 kernel + native RPM base + desktop overlay under validation (Fedora 43
IceWM candidate — see =Azure-Neptune.md= for validation tasks).

#+BEGIN_SRC conf :tangle linux/.config/virtualisation/azure-neptune.conf
# Azure Neptune Enterprise Server — VM configuration stub
# Filled in during KVM parallel build track
#
# DISK_IMAGE=~/vms/azure-neptune.qcow2
# GUEST_HOME=${XDG_DATA_HOME}/dotcortex/guests/azure-neptune/home
#
# CPU:    host-passthrough, scheduler-shared (no CPU pinning)
# RAM:    start=4G  max=16G  balloon=yes
# DISK:   QCOW2 image (OS + RPM base)
# DISP:   SPICE + virtio-gpu
# NET:    host-only bridge, loopback SSH forward 127.0.0.1:2260 → guest:22
# MOUNT:  virtiofs → GUEST_HOME (host rw, guest rw)
#+END_SRC

#+BEGIN_SRC bash :tangle linux/.local/bin/azure-neptune-launch :shebang "#!/usr/bin/env bash"
# Azure Neptune Enterprise Server — launch stub
# Fill in disk image path and virtiofs socket before use.
set -euo pipefail

DISK_IMAGE="${VM_DISK:-$HOME/vms/azure-neptune.qcow2}"
GUEST_HOME="${XDG_DATA_HOME:-$HOME/.local/share}/dotcortex/guests/azure-neptune/home"
SSH_PORT=2260
SPICE_PORT=5960

if [[ ! -f "$DISK_IMAGE" ]]; then
  echo "azure-neptune: disk image not found: $DISK_IMAGE"
  echo "Create with: qemu-img create -f qcow2 \$DISK_IMAGE 40G"
  exit 1
fi

echo "azure-neptune: launching (SSH → 127.0.0.1:${SSH_PORT})"

qemu-system-x86_64 \
  -enable-kvm \
  -cpu host \
  -m 4G,maxmem=16G \
  -balloon virtio \
  -drive file="$DISK_IMAGE",if=virtio,format=qcow2 \
  -device virtio-gpu \
  -display spice-app \
  -spice port="${SPICE_PORT}",addr=127.0.0.1,disable-ticketing=on \
  -net nic,model=virtio \
  -net user,hostfwd=tcp:127.0.0.1:"${SSH_PORT}"-:22 \
  -daemonize \
  "$@"
#+END_SRC

* Shared VM Utilities

#+BEGIN_SRC bash :tangle linux/.local/bin/vm-list :shebang "#!/usr/bin/env bash"
# List known VMs and their current QEMU process status.
set -euo pipefail

vms=(azure-neptune)

echo "Virtual Habitat — KVM VM status"
echo "--------------------------------"
for vm in "${vms[@]}"; do
  if pgrep -f "qemu.*${vm}" >/dev/null 2>&1; then
    echo "  [running] $vm"
  else
    echo "  [stopped] $vm"
  fi
done
#+END_SRC
