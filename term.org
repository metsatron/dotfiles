# --- terminal font fallback (env → env → fc-match) ---
if [[ -z ${term_font-} ]]; then
  if [[ -n ${TERM_FONT-} ]]; then
    term_font="$TERM_FONT"
  elif [[ -n ${WEZTERM_FONT-} ]]; then
    term_font="$WEZTERM_FONT"
  elif command -v fc-match >/dev/null 2>&1; then
    term_font="$(fc-match -f '%{family}\n' 2>/dev/null | head -n1)"
  fi
fi

** Ranger Colorscheme (LainCore)
Custom ranger colorscheme aligned to the LainCore/DreamWeave palette (style.org's ~lc/base16 'dreamweaver-cc2017~, the same slots already wired into ~all/.config/kitty/kitty.conf~ color0-15 and ~all/.config/rofi/colors/laincore-dreamweave.rasi~). Modeled structurally on ranger's shipped ~jungle~ (subclasses Default, tweaks directory/titlebar) and ~snow~ (subclasses ColorScheme directly, minimal reverse/bold) schemes, but follows ~solarized~'s approach of raw 256-color integers throughout since ranger/curses renders 256-color pairs rather than 24-bit truecolor. Hex values are mapped to the nearest xterm-256 index (216-color cube + greyscale ramp); ~base08~/~base09~ round to the same coarse cube cell so ~base08~ (urgent/red) is nudged to 160 by hand to keep it visually distinct from ~base09~'s 166 (orange) -- noted in the file. The three LainCore rasi tokens map straight across: ~selected~ (base0C, cyan) drives the browser selection bar / links / permissions-good / progress bar; ~active~ (base0B, green) drives executables / good hostname / active tab / vcs-sync; ~urgent~ (base08, red) drives errors / bad links / vcs-conflict states. Selection uses ranger's native ~reverse~ attribute (as jungle/default/snow do) rather than a hardcoded bg, so a selected directory keeps rendering as a blue bar and a selected executable as a green bar -- type identity survives the highlight instead of being flattened to one color.
#+BEGIN_SRC python :tangle all/.config/ranger/colorschemes/laincore.py :tangle-mode (identity #o644) :mkdirp yes
# -*- coding: utf-8 -*-
# This file is part of ranger, the console file manager.
# License: GNU GPL version 3, see the file "AUTHORS" for details.
#
# LainCore -- DreamWeave colorscheme for ranger.
# Source of truth: ~/DotCortex/term.org ("Ranger Colorscheme (LainCore)" block).
# Do NOT edit this file directly -- edit term.org and re-tangle with tangle-one.
#
# Palette lineage: style.org's lc/base16 'dreamweaver-cc2017 -- the same base16
# slots already wired into all/.config/kitty/kitty.conf (color0-15) and
# all/.config/rofi/colors/laincore-dreamweave.rasi. ranger/curses addresses
# colors as 256-color pairs, not 24-bit truecolor, so each hex below is mapped
# to the nearest xterm-256 index (216-color cube + greyscale ramp).
#
#   slot    hex        256-idx  role here
#   base00  #1f1f1f    234      bg (left as terminal default in most contexts)
#   base01  #242424    235      panel / alt background (reserved)
#   base02  #454545    238      inactive tab background
#   base03  #6a6a6a    242      muted: plain filenames, inactive pane, cut marker
#   base04  #cbcbcb    251      dim foreground (reserved)
#   base05  #d2d4d4    252      inactive tab foreground
#   base06  #f0f0f0    255      bright foreground (reserved)
#   base07  #ffffff    231      white: tag marker, active tab foreground
#   base08  #dc322f    160      rasi 'urgent': errors, bad links, vcs conflict
#                                (hand-nudged from the raw-nearest 166 so it
#                                 doesn't collide with base09's orange -- the
#                                 256 cube is too coarse to separate them)
#   base09  #cb4b16    166      archives / containers
#   base0A  #b58900    136      marked rows, fifo, device, vcs commit
#   base0B  #4caf50    71       rasi 'active': executables, good hostname/tab,
#                                vcs staged/sync
#   base0C  #15b8cf    38       rasi 'selected': selection bar, links,
#                                permissions-good, progress bar, vcs date
#   base0D  #268bd2    32       directories, vcsinfo, vcs-ahead
#   base0E  #a681d6    140      sockets, other media, vcs-diverged, badinfo
#   base0F  #4b6983    60       reserved (kept for palette completeness)

from __future__ import (absolute_import, division, print_function)

from ranger.gui.colorscheme import ColorScheme
from ranger.gui.color import default, normal, bold, reverse, dim, default_colors

MUTED    = 242  # base03 -- plain filenames / inactive pane / cut marker
WHITE    = 231  # base07
URGENT   = 160  # base08 (hand-nudged, see header)
ORANGE   = 166  # base09
WARN     = 136  # base0A
ACTIVE   = 71   # base0B
ACCENT   = 38   # base0C
DIR_BLUE = 32   # base0D — kept for vcsinfo/titlebar roles
DIR_YELLOW = 221  # folders: canonical #ffcb63 per Mètsàtron's ruling 2026-07-10 (already live in nvim); 221=#ffd75f is the nearest cell ranger's 256-color curses palette can reach
VIOLET   = 140  # base0E
LIGHT_BG = 238  # base02 -- inactive tab background
LIGHT_FG = 252  # base05 -- inactive tab foreground


class Scheme(ColorScheme):
    progress_bar_color = ACCENT

    def use(self, context):  # pylint: disable=too-many-branches,too-many-statements
        fg, bg, attr = default_colors

        if context.reset:
            return default_colors

        elif context.in_browser:
            if context.selected:
                attr |= reverse
                fg = ACCENT
            else:
                attr = normal
                fg = MUTED
            if context.empty or context.error:
                bg = URGENT
            if context.border:
                fg = default
            if context.media:
                fg = WARN if context.image else VIOLET
            if context.container:
                fg = ORANGE
            if context.directory:
                attr |= bold
                fg = DIR_YELLOW
            elif context.executable and not \
                    any((context.media, context.container,
                         context.fifo, context.socket)):
                attr |= bold
                fg = ACTIVE
            if context.socket:
                attr |= bold
                fg = VIOLET
            if context.fifo or context.device:
                fg = WARN
                if context.device:
                    attr |= bold
            if context.link:
                fg = ACCENT if context.good else URGENT
            if context.tag_marker and not context.selected:
                attr |= bold
                fg = WHITE if fg in (URGENT, VIOLET) else URGENT
            if not context.selected:
                if context.copied:
                    attr |= bold
                    fg = WARN
                elif context.cut:
                    attr |= dim
                    fg = MUTED
            if context.main_column:
                if context.selected:
                    attr |= bold
                if context.marked:
                    attr |= bold
                    fg = WARN
            if context.badinfo:
                if attr & reverse:
                    bg = VIOLET
                else:
                    fg = VIOLET
            if context.inactive_pane:
                fg = MUTED

        elif context.in_titlebar:
            if context.hostname:
                fg = URGENT if context.bad else ACTIVE
            elif context.directory:
                fg = DIR_BLUE
            elif context.tab:
                if context.good:
                    bg = ACTIVE
            elif context.link:
                fg = ACCENT
            attr |= bold
            if context.tab and not context.good:
                bg = LIGHT_BG
                fg = LIGHT_FG

        elif context.in_statusbar:
            if context.permissions:
                if context.good:
                    fg = ACCENT
                elif context.bad:
                    fg = URGENT
            if context.marked:
                attr |= bold | reverse
                fg = WARN
            if context.frozen:
                attr |= bold | reverse
                fg = ACCENT
            if context.message:
                if context.bad:
                    attr |= bold
                    fg = URGENT
            if context.loaded:
                bg = self.progress_bar_color
            if context.vcsinfo:
                fg = DIR_BLUE
                attr &= ~bold
            if context.vcscommit:
                fg = WARN
                attr &= ~bold
            if context.vcsdate:
                fg = ACCENT
                attr &= ~bold

        if context.text:
            if context.highlight:
                attr |= reverse

        if context.in_taskview:
            if context.title:
                fg = DIR_BLUE

            if context.selected:
                attr |= reverse

            if context.loaded:
                if context.selected:
                    fg = self.progress_bar_color
                else:
                    bg = self.progress_bar_color

        if context.vcsfile and not context.selected:
            attr &= ~bold
            if context.vcsconflict:
                fg = VIOLET
            elif context.vcsuntracked:
                fg = ACCENT
            elif context.vcschanged:
                fg = URGENT
            elif context.vcsunknown:
                fg = URGENT
            elif context.vcsstaged:
                fg = ACTIVE
            elif context.vcssync:
                fg = ACTIVE
            elif context.vcsignored:
                fg = default

        elif context.vcsremote and not context.selected:
            attr &= ~bold
            if context.vcssync or context.vcsnone:
                fg = ACTIVE
            elif context.vcsbehind:
                fg = URGENT
            elif context.vcsahead:
                fg = DIR_BLUE
            elif context.vcsdiverged:
                fg = VIOLET
            elif context.vcsunknown:
                fg = URGENT

        return fg, bg, attr
#+END_SRC

#+BEGIN_SRC bash :tangle all/.config/ranger/scope.sh :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755) :mkdirp yes
set -euo pipefail
set -o noclobber

FILE_PATH="${1:-}"
PV_WIDTH="${2:-${COLUMNS:-80}}"
PV_HEIGHT="${3:-${LINES:-24}}"
IMAGE_CACHE_PATH="${4:-}"
PV_IMAGE_ENABLED="${5:-False}"

FILE_EXTENSION="${FILE_PATH##*.}"
FILE_EXTENSION_LOWER="$(printf '%s' "${FILE_EXTENSION:-}" | tr '[:upper:]' '[:lower:]')"
MIMETYPE="$(file --dereference --brief --mime-type -- "${FILE_PATH}")"

preview_text() {
  if command -v /home/metsatron/.cargo/bin/bat >/dev/null 2>&1; then
    # --theme=base16: bat's "base16" theme maps syntax roles onto the 16 base
    # ANSI color slots (0-15) rather than hardcoding hex, and those exact 16
    # slots are already wired to the LainCore/DreamWeave base16 palette in
    # kitty.conf/WezTerm (color0-15, see style.org emit-kitty). That makes the
    # preview pane inherit LainCore automatically and stay correct if the
    # palette is ever retuned in style.org, with no duplicated hex here.
    # base16-256 was rejected: its non-syntax roles fall back to the fixed
    # xterm 256-cube, which our terminal configs do not remap, so it would
    # drift from the theme. Plain "ansi" was rejected too: it drops down to
    # bare SGR with far fewer distinguished syntax roles than base16 offers
    # while inheriting the same terminal palette.
    /home/metsatron/.cargo/bin/bat \
      --color=always \
      --style=plain \
      --paging=never \
      --theme=base16 \
      --terminal-width="${PV_WIDTH:-80}" \
      -- "${FILE_PATH}" && exit 5
  fi

  if command -v pygmentize >/dev/null 2>&1; then
    pygmentize -f terminal256 -- "${FILE_PATH}" && exit 5
  fi

  cat -- "${FILE_PATH}" && exit 5
  exit 1
}

preview_pdf() {
  if command -v pdftotext >/dev/null 2>&1; then
    pdftotext -l 10 -nopgbrk -q -- "${FILE_PATH}" - | awk 'NR <= 1000 { print }' && exit 0
  fi
  exit 1
}

preview_image() {
  if [[ "${PV_IMAGE_ENABLED:-False}" == "True" ]]; then
    exit 7
  fi

  if command -v chafa >/dev/null 2>&1; then
    chafa --size "${PV_WIDTH:-80}x${PV_HEIGHT:-24}" -- "${FILE_PATH}" && exit 0
  fi

  file --brief -- "${FILE_PATH}"
  exit 0
}

preview_media_info() {
  file --brief -- "${FILE_PATH}" || true
python3 -c 'import os, sys, time
path = sys.argv[1]
st = os.stat(path)
print(f"Size: {st.st_size} bytes")
print("Modified: " + time.strftime("%Y-%m-%d %H:%M:%S", time.localtime(st.st_mtime)))' "${FILE_PATH}" || true
  exit 0
}

preview_archive() {
  if command -v bsdtar >/dev/null 2>&1; then
    bsdtar --list --file "${FILE_PATH}" && exit 0
  fi

  case "${FILE_EXTENSION_LOWER:-}" in
    zip)
      python3 -m zipfile -l "${FILE_PATH}" && exit 0
      ;;
    tar)
      python3 -m tarfile -l "${FILE_PATH}" && exit 0
      ;;
    tgz|gz|tbz|tbz2|txz|xz)
      python3 -m tarfile -l "${FILE_PATH}" && exit 0
      ;;
  esac

  file --brief -- "${FILE_PATH}"
  exit 0
}

case "${FILE_EXTENSION_LOWER:-}" in
  bash|bat|c|cc|cfg|conf|cpp|css|el|go|h|hpp|html|ini|java|js|json|lua|md|org|pl|py|rb|rs|scm|sh|toml|ts|xml|yaml|yml|zsh)
    preview_text
    ;;
  pdf)
    preview_pdf
    ;;
  zip|tar|tgz|gz|tbz|tbz2|txz|xz)
    preview_archive
    ;;
esac

case "${MIMETYPE:-}" in
  text/*|*/xml|application/json)
    preview_text
    ;;
  application/pdf)
    preview_pdf
    ;;
  image/*)
    preview_image
    ;;
  video/*|audio/*)
    preview_media_info
    ;;
  application/zip|application/x-tar|application/gzip|application/x-gzip|application/x-bzip2|application/x-xz)
    preview_archive
    ;;
esac

printf 'MIME: %s\n' "${MIMETYPE:-unknown}"
file --dereference --brief --mime -- "${FILE_PATH}"
exit 0

* Yazi Config

Yazi terminal file manager — ranger's sibling, so it lives here in =term.org= with the rest of the terminal stack (absorbed from a stray single-package =yazi.org= on 2026-07-10; package manifests group by category, never one org per package). * Superfile (spf) hotkeys — Colemak-NEIO
Superfile (installed via =app.org=) ships Vim =h/j/k/l= movement by default, which collides with the Colemak-NEIO interaction law. This remaps movement to the =n/e/i/o= contract (left/down/up/right), keeping the arrow keys as aliases and dropping the Vim letters entirely. Superfile hotkeys are case-sensitive, so lowercase =e= (down) and uppercase =E= (open-dir-in-editor) coexist cleanly; the three displaced non-movement actions land on mnemonic keys — =S= (sort), =O= (open file in editor, pairs with =o= = open/enter), and =t= (new file panel). Every field from superfile's default is reproduced because =config.toml= sets =ignore_missing_fields = false=, so a partial file would be rejected. Caveat: on a superfile upgrade that introduces new hotkey fields, =spf --fix-hotkeys= would try to write through the stow symlink into the overlay (generated output) — never accept that; add the new field to this block and re-tangle instead.
