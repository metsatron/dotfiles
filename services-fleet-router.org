* Contract

This is the one public, declarative machine × script registry for the Fleet Nervous System; machine roles and hostnames remain in =all/.config/agent-session/version-registry.json=, owned by =services-agent-sessiond.org=, while this file owns tool classes and argument-aware route rules.

The only classes are =hub=, =client-fanout-safe=, and =client-fanout-reconcile=; v1 transport is bounded SSH exec, hub routes run locally only on kikin and otherwise bridge to kikin, and =client-fanout-reconcile= is refused because the P4 staging/reconciler engine remains unimplemented.

Unlisted tools default to =client-fanout-safe= for the compatibility path; this still loads FORGE code over NFS when invoked, but its router-side reachability and executable guards retain the bounded =timeout -k 1 2= failure. Canonical tools are explicitly classified as =hub=, and a failed hub SSH bridge never falls back to local NFS execution.

The hub alias is explicit because the existing machine registry intentionally uses a null SSH alias for the hub's own local execution; it is not a second machine registry.

#+BEGIN_SRC bash :tangle all/.local/bin/hx-router-test :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -Eeuo pipefail

SCRIPT_PATH="$(readlink -f -- "${BASH_SOURCE[0]}")"
ROOT="$(cd -- "$(dirname -- "${SCRIPT_PATH}")/../../.." && pwd)"
tmp="$(mktemp -d)"
trap 'rm -rf "${tmp}"' EXIT
export HOME="${tmp}/home" XDG_CONFIG_HOME="${HOME}/.config"
mkdir -p "${HOME}/.config/agent-session" "${HOME}/.local/libexec" "${HOME}/HelmCortex/FORGE/bin" "${tmp}/bin"
cp "${ROOT}/all/.config/agent-session/fleet-router.json" "${HOME}/.config/agent-session/"
cp "${ROOT}/all/.config/agent-session/version-registry.json" "${HOME}/.config/agent-session/"
cp "${ROOT}/all/.local/libexec/hx-router" "${HOME}/.local/libexec/"
chmod 755 "${HOME}/.local/libexec/hx-router"

cat > "${HOME}/HelmCortex/FORGE/bin/hx-test" <<'EOF'
#!/usr/bin/env bash
printf 'local:%s\n' "$*"
EOF
chmod 755 "${HOME}/HelmCortex/FORGE/bin/hx-test"

cat > "${HOME}/HelmCortex/FORGE/bin/hx-remote" <<'EOF'
#!/usr/bin/env bash
printf 'remote-stdout:%s\n' "$1"
printf 'remote-stderr:%s\n' "$2" >&2
printf 'remote-caller:%s\n' "$HX_CALLER_PWD"
printf 'remote-argv:%s\n' "$@"
exit 37
EOF
chmod 755 "${HOME}/HelmCortex/FORGE/bin/hx-remote"

cat > "${tmp}/bin/ssh" <<'EOF'
#!/usr/bin/env bash
printf '%s\n' "$*" > "${HX_TEST_SSH_LOG}"
cat >/dev/null
command="${!#}"
exec /bin/bash -c "${command}"
EOF
chmod 755 "${tmp}/bin/ssh"
export PATH="${tmp}/bin:${PATH}" HX_TEST_SSH_LOG="${tmp}/ssh.log" HX_ROUTER_TEST=1 HX_SSH_BIN="${tmp}/bin/ssh"

local_out="${tmp}/local.out"
HX_ROUTER_HOSTNAME="kikin-kushi" HELM_ROUTER_MACHINE_REGISTRY="${HOME}/.config/agent-session/version-registry.json" \
  "${HOME}/.local/libexec/hx-router" "${HOME}/.config/agent-session/fleet-router.json" hx-test \
  "${HOME}/HelmCortex/FORGE/bin/hx-test" '$HOME/HelmCortex/FORGE/bin/hx-test' 'a b' 'x;y' >"${local_out}"
grep -F 'local:a b x;y' "${local_out}" >/dev/null

set +e
HX_ROUTER_HOSTNAME="unknown-host" HX_CALLER_PWD="${tmp}/caller dir" \
  "${HOME}/.local/libexec/hx-router" "${HOME}/.config/agent-session/fleet-router.json" recall \
  "${HOME}/missing-local/hx-remote" "${HOME}/HelmCortex/FORGE/bin/hx-remote" \
  'space value' 'semi;colon' "apost'rophe" 'snow☃' $'line\nbreak' >"${tmp}/remote.out" 2>"${tmp}/remote.err"
status=$?
set -e
[[ "${status}" -eq 37 ]] || { echo "remote status ${status}" >&2; exit 1; }
grep -F 'remote-stdout:space value' "${tmp}/remote.out" >/dev/null
grep -F 'remote-stderr:semi;colon' "${tmp}/remote.err" >/dev/null
grep -F "remote-caller:${tmp}/caller dir" "${tmp}/remote.out" >/dev/null
grep -F "remote-argv:apost'rophe" "${tmp}/remote.out" >/dev/null
grep -F $'remote-argv:line\nbreak' "${tmp}/remote.out" >/dev/null
grep -F 'remote-argv:snow☃' "${tmp}/remote.out" >/dev/null
grep -F -- '-o BatchMode=yes' "${HX_TEST_SSH_LOG}" >/dev/null
grep -F -- 'ConnectTimeout=10' "${HX_TEST_SSH_LOG}" >/dev/null
grep -F -- 'kikin' "${HX_TEST_SSH_LOG}" >/dev/null

stdin_out="${tmp}/stdin.out"
{
  if HX_ROUTER_HOSTNAME="unknown-host" "${HOME}/.local/libexec/hx-router" \
    "${HOME}/.config/agent-session/fleet-router.json" recall \
    "${HOME}/missing-local/hx-remote" "${HOME}/HelmCortex/FORGE/bin/hx-remote" \
    stdin-test >/dev/null 2>"${tmp}/stdin.err"; then
    :
  fi
  following="<missing>"
  IFS= read -r following || true
  printf 'following:%s\n' "${following}"
} <<< 'following-command-input' >"${stdin_out}"
grep -F 'following:following-command-input' "${stdin_out}" >/dev/null

set +e

  "${HOME}/.config/agent-session/fleet-router.json" telegram-export-pipeline \
