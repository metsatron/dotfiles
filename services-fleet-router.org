* Contract

This is the one public, declarative machine × script registry for the Fleet Nervous System; machine roles and hostnames remain in =all/.config/agent-session/version-registry.json=, owned by =services-agent-sessiond.org=, while this file owns tool classes and argument-aware route rules.

The only classes are =hub=, =client-fanout-safe=, and =client-fanout-reconcile=. Registry v2 admits the versioned DotCortex root-Org snapshot protocol; undeclared reconcile protocols remain refused. Hub routes run locally only on the declared hub and otherwise bridge there over SSH. Old routers reject the v2 registry before dispatch.

Unlisted tools default to =client-fanout-safe= for the compatibility path; this still loads FORGE code over NFS when invoked, but its router-side reachability and executable guards retain the bounded =timeout -k 1 2= failure. Canonical tools are explicitly classified as =hub=, and a failed hub SSH bridge never falls back to local NFS execution.

The hub alias is explicit because the existing machine registry intentionally uses a null SSH alias for the hub's own local execution; it is not a second machine registry.

Capture policy v2 separates required hub finalization from optional source freshness. The machine selected by =transport.hub_machine_key= owns required finalization; every source-capture row, including that hub, is optional. An unavailable source, absent store or failed capture falls back to already-reconciled canonical LOGS. Missing stores are normal. Absence must never delete or shrink history. Product =complete= is separate from =freshness_complete=: finalization may succeed with incomplete freshness, which is reported as =freshness_complete=false=.

The six harness entries on each source row declare supported local probes, not an assertion that all six stores exist. Per-host statuses mean: =fresh= — source checked and current; =carry-forward= — prior canonical LOGS retained because source/store is not current; =unavailable= — no prior canonical data and source unavailable; =failed= — invalid source response/capture discarded. A failed capture does not replace prior valid content. Capture-v1 consumers must reject the v2 policy before capture or publication; the outer router-v2 and DotCortex root-Org snapshot contracts are unchanged.

import copy
import json
from pathlib import Path
import re
import unittest


class CapturePolicyTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        source = Path("services-fleet-router.org").read_text()
        cls.body = re.search(
            r'^#\+BEGIN_SRC json :tangle all/\.config/agent-session/fleet-router\.json :comments no\n(.*?)^#\+END_SRC',
            source, re.M | re.S).group(1)
        cls.router = json.loads(cls.body)
        cls.capture = cls.router["tools"]["helmcortex-export"]["capture"]

    def assert_policy(self, capture):
        self.assertEqual(set(capture), {"schema", "protocol", "finalization", "freshness",
                                        "harnesses_semantics", "limits", "machines"})
        self.assertEqual(capture["schema"], "helmcortex.fleet-capture.v2")
        self.assertEqual(capture["protocol"], "ephemeral-ssh-v1")
        self.assertEqual(capture["finalization"], {"owner": "hub_machine_key", "policy": "required"})
        self.assertEqual(capture["freshness"], {
            "on_source_unavailable": "carry-forward-canonical-logs",
            "on_store_absent": "carry-forward-canonical-logs",
            "on_capture_failure": "carry-forward-canonical-logs",
            "allow_incomplete": True,
            "host_statuses": ["fresh", "carry-forward", "unavailable", "failed"],
            "incomplete_field": "freshness_complete", "preserve_history": True,
        })
        self.assertIs(capture["freshness"]["allow_incomplete"], True)
        self.assertIs(capture["freshness"]["preserve_history"], True)
        self.assertEqual(capture["harnesses_semantics"], "supported-local-probes")
        self.assertEqual(capture["limits"], {
            "payload_bytes": 4194304, "result_bytes": 67108864,
            "expanded_bytes": 268435456, "file_bytes": 33554432,
            "files": 20000, "seconds": 600, "pages": 1000,
        })
        self.assertEqual(set(capture["machines"]), {"x230", "t480s", "t480", "kikin-kushi", "s24"})
        for machine in capture["machines"].values():
            self.assertEqual(machine, {"policy": "optional", "harnesses": [
                "claude-code", "codex", "opencode", "hermes", "pi-agent", "antigravity",
            ]})

    def test_exact_approved_capture_contract(self):
        self.assert_policy(self.capture)

    def test_source_and_generated_registry_parity(self):
        self.assertEqual(self.body, Path("all/.config/agent-session/fleet-router.json").read_text())

    def test_required_owner_is_separate_from_optional_hub_capture(self):
        transport = self.router["transport"]
        owner = transport[self.capture["finalization"]["owner"]]
        self.assertEqual(owner, "kikin-kushi")
        self.assertEqual(self.capture["machines"][owner]["policy"], "optional")
        roster = json.loads(Path("all/.config/agent-session/version-registry.json").read_text())
        self.assertEqual(set(self.capture["machines"]), {m["key"] for m in roster["machines"]})
        self.assertEqual([m["key"] for m in roster["machines"] if m["role"] == "hub"], [owner])

    def test_router_and_dot_snapshot_contracts_preserved(self):
        self.assertEqual(self.router["schema"], "dotcortex.fleet-router.v2")
        self.assertEqual(self.router["tools"]["dotcortex-export"], {
            "class": "client-fanout-reconcile", "reconcile": {"protocol": "dotcortex-root-org-v1"},
        })

    def test_assertions_detect_policy_regressions(self):
        changes = [lambda c: c.update(schema="helmcortex.fleet-capture.v1"),
                   lambda c: c["finalization"].update(policy="optional"),
                   lambda c: c.update(harnesses_semantics="required-stores"),
                   lambda c: c["freshness"].update(preserve_history=False),
                   lambda c: c["freshness"].update(allow_incomplete=False),
                   lambda c: c["freshness"].update(incomplete_field="complete"),
                   lambda c: c["freshness"].update(host_statuses=["fresh", "carry-forward", "unavailable"])]
        for field in ("on_source_unavailable", "on_store_absent", "on_capture_failure"):
            changes.append(lambda c, key=field: c["freshness"].update({key: "discard-history"}))
        for key in self.capture["machines"]:
            changes.append(lambda c, host=key: c["machines"][host].update(policy="required"))
        for change in changes:
            candidate = copy.deepcopy(self.capture)
            change(candidate)
            with self.subTest(candidate=candidate), self.assertRaises(AssertionError):
                self.assert_policy(candidate)


unittest.main(argv=["fleet-capture-policy-tests"])

    def decode(self, frame):
        return rc.decode_frame(io.BytesIO(frame))

    def changed_frame(self, frame, change):
        manifest, contents, _ = self.decode(frame)
        change(manifest)
        header = rc.canonical(manifest)
        body = rc.MAGIC + len(header).to_bytes(8, "big") + header + b"".join(b for _, b in contents)
        return body + hashlib.sha256(body).digest()

    def test_exact_one_content_read_and_determinism(self):
        calls = []
        original = os.fdopen
        class Reader:
            def __init__(self, stream): self.stream = stream
            def __enter__(self): return self
            def __exit__(self, *args): self.stream.close()
            def fileno(self): return self.stream.fileno()
            def read(self, count):
                calls.append((self.fileno(), count))
                return self.stream.read(count)
        with mock.patch.object(rc.os, "fdopen", side_effect=lambda *a, **k: Reader(original(*a, **k))):
            first = rc.snapshot(str(self.root), str(self.caller))
        self.assertEqual(len(calls), 2)
        self.assertEqual([count for _, count in calls], [rc.MAX_FILE + 1] * 2)
        self.assertEqual(first, rc.snapshot(str(self.root), str(self.caller)))
        (self.root / "a.org").write_bytes(b"edited after capture")
        self.assertEqual(dict((r["path"], b) for r,b in self.decode(first)[1])["a.org"], b"caller dirty\n")
