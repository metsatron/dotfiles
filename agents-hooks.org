* Claude settings

The tracked JSON below is the SHARED settings baseline for every machine. It tangles to =all/.claude/settings.shared.json= (generated, git-ignored), NOT to the live =settings.json=. The live file is produced by =claude-settings-apply= (next section), which overlays this baseline onto the machine's existing =settings.json= while preserving the machine-local keys the harness writes — =model= (=/model=) and =effortLevel= (=/effort=). Those keys are therefore deliberately ABSENT here — they are per-machine state, not shared. =theme= and everything else are shared: edit them here, then run =loom claude:settings= (which tangles this block and re-merges). This is the "merge" tier of the pipeline: stow → tangle → merge. To make another key machine-local, delete it from this block and add it to the preserve-list in =claude-settings-apply=. One key is deep-merged rather than replaced: =enabledPlugins= — shared entries here win per key, but machine-local additions survive the merge, because plugin enablement is per-machine state (the telegram channel plugin runs on exactly ONE machine at a time — one getUpdates consumer per bot token — so it must never be enabled fleet-wide from this baseline; it is enabled locally on the bot machine and the merge preserves it).

* Claude settings merge (apply shared baseline, preserve machine-local model)

=claude-settings-apply= is the deploy step that lets machine-local and shared Claude settings coexist in the single =settings.json= that Claude Code mandates (it has no user-level =settings.local.json=). It reads the tangled shared baseline (=settings.shared.json=) and the machine's current live =settings.json=, then writes a merged =settings.json= where every shared key wins EXCEPT the machine-local ones (=model=, =effortLevel=), which are preserved from the live file (what =/model= and =/effort= write). Bootstrap-safe: with no live file yet, the shared baseline is used verbatim (no =model= → Claude Code's default) until =/model= runs. The old tangle-straight-to-settings.json path is gone — always go tangle → merge.

#+BEGIN_SRC bash :tangle all/.local/bin/claude-settings-apply :shebang "#!/usr/bin/env bash" :comments no
# ~/DotCortex/all/.local/bin/claude-settings-apply
# Merge the shared Claude settings baseline (tangled from agents-hooks.org) onto the
# machine's live settings.json, preserving the machine-local `model` key that /model
# writes. `enabledPlugins` deep-merges: shared entries win per key, machine-local
# additions (e.g. the telegram channel plugin on the bot machine) survive.
# The "merge" tier of the pipeline: tangle -> settings.shared.json, then this
# script -> settings.json. Never tangle directly to settings.json again.
set -euo pipefail

SHARED="$HOME/DotCortex/all/.claude/settings.shared.json"
LIVE="$HOME/DotCortex/all/.claude/settings.json"

if [ ! -f "$SHARED" ]; then
  echo "!! claude-settings-apply: shared baseline missing at $SHARED" >&2
  echo "   run 'tangle-one agents-hooks.org' first (or 'loom claude:settings')" >&2
  exit 1
fi
command -v jq >/dev/null 2>&1 || { echo "!! claude-settings-apply: jq not on PATH" >&2; exit 1; }

# On a fresh machine the live file may not exist; feed jq /dev/null so $live[0] is
# null and the shared baseline is used verbatim (no model -> Claude Code default).
live_input="$LIVE"
[ -f "$LIVE" ] || live_input=/dev/null

tmp="$(mktemp)"
if jq -n \
     --slurpfile shared "$SHARED" \
     --slurpfile live "$live_input" \
     '($shared[0] // {}) as $s
      | ($live[0]  // {}) as $l
      | $s + ( ($l | {model, effortLevel}) | with_entries(select(.value != null)) )
      | .enabledPlugins = (($l.enabledPlugins // {}) + ($s.enabledPlugins // {}))' > "$tmp" \
   && jq empty "$tmp" 2>/dev/null; then
  mv -f "$tmp" "$LIVE"
  printf '>> claude settings merged: model=%s effort=%s theme=%s\n' \
    "$(jq -r '.model // "(unset -> Claude default)"' "$LIVE")" \
    "$(jq -r '.effortLevel // "(unset)"' "$LIVE")" \
    "$(jq -r '.theme // "(unset)"' "$LIVE")"
else
  rm -f "$tmp"
  echo "!! claude-settings-apply: merge produced invalid JSON; left $LIVE untouched" >&2
  exit 1
fi
#+END_SRC

* Claude statusline
#+BEGIN_SRC bash :tangle all/.local/bin/claude-statusline :shebang "#!/usr/bin/env bash" :comments no
set -u
input=$(cat)

# User and host
user=$(whoami)
host=$(hostname -s)

# Directory — basename only
cwd=$(echo "$input" | jq -r '.workspace.current_dir // .cwd')
short_cwd=$(basename "$cwd")

# Git branch from worktree or workspace
git_branch=$(echo "$input" | jq -r '.worktree.branch // .workspace.git_worktree // empty')
if [ -z "$git_branch" ] && command -v git >/dev/null 2>&1; then
  git_branch=$(git -C "$cwd" symbolic-ref --short HEAD 2>/dev/null || git -C "$cwd" rev-parse --short HEAD 2>/dev/null)
fi

# Time
time_str=$(date +%H:%M)

# Model short name
model=$(echo "$input" | jq -r '.model.display_name // empty' | sed 's/Claude //')

# Context USED (not remaining)
used=$(echo "$input" | jq -r '.context_window.used_percentage // empty')

# Claude's documented status-line total_input_tokens is the canonical current
# input-context value.  The fallback mirrors its documented current_usage
# components and excludes output tokens.  current_usage is null immediately
# after compaction; PostCompact clears the owning governor state then.
tokens=$(echo "$input" | jq -r '
  if .context_window.total_input_tokens != null
  then .context_window.total_input_tokens
  elif .context_window.current_usage != null
  then ((.context_window.current_usage.input_tokens // 0) +
        (.context_window.current_usage.cache_creation_input_tokens // 0) +
        (.context_window.current_usage.cache_read_input_tokens // 0))
  else 0
  end')

# Persist only the small, non-secret status snapshot used by the owner.
# Emacs-owned sessions use a channel file under XDG_RUNTIME_DIR; PTY-supervised
# sessions use their private runtime directory.  Both writes are atomic.
owner="${CLAUDE_IDLE_OWNER:-}"
channel="${CLAUDE_IDLE_CHANNEL:-}"
session_id=$(echo "$input" | jq -r '.session_id // empty')
runtime_dir=""
snapshot_path=""
if [ "$owner" = "pty-supervisor" ] || [ -z "$owner" ]; then
  runtime_dir="${CLAUDE_IDLE_RUNTIME_DIR:-}"
  snapshot_path="$runtime_dir/status.json"
  # Claude 2.1.220 may launch statusLine commands without propagating the
  # supervisor's custom CLAUDE_IDLE_* variables.  Recover the owner from the
  # private per-session registry using the validated session ID.  The registry
  # entry must still name a live PID with an owned PTY; the supervisor repeats
  # channel/session validation at the compaction gate.
  if [ -z "$channel" ] || [ -z "$runtime_dir" ]; then
    registry_root="${XDG_RUNTIME_DIR:-}/claude-idle-compaction"
    for state_file in "$registry_root"/claude-*/state.json; do
      [ -r "$state_file" ] || continue
      candidate_session=$(jq -r '.session_id // empty' "$state_file" 2>/dev/null || true)
      candidate_pid=$(jq -r '.pid // empty' "$state_file" 2>/dev/null || true)
      candidate_owned=$(jq -r '.pty_master_owned // false' "$state_file" 2>/dev/null || true)
      candidate_channel=$(basename "$(dirname "$state_file")")
      if [ "$candidate_session" = "$session_id" ] &&
         [[ "$candidate_channel" =~ ^claude-[0-9a-f]{32}$ ]] &&
         [[ "$candidate_pid" =~ ^[0-9]+$ ]] &&
         [ "$candidate_owned" = "true" ] &&
         kill -0 "$candidate_pid" 2>/dev/null; then
        channel="$candidate_channel"
        runtime_dir="$(dirname "$state_file")"
        snapshot_path="$runtime_dir/status.json"
        break
      fi
    done
  fi
fi
if [ -z "$channel" ] || [ -z "$runtime_dir" ]; then
  # An explicitly Emacs-owned session keeps the original channel-file path.
  if [ "$owner" != "pty-supervisor" ] && [ -n "$channel" ]; then
    runtime_dir="${XDG_RUNTIME_DIR:-}"
    snapshot_path="$runtime_dir/claude-idle-compact-${channel}.json"
  fi
fi
if [ "$owner" = "emacs" ]; then
  runtime_dir="${XDG_RUNTIME_DIR:-}"
  snapshot_path="$runtime_dir/claude-idle-compact-${channel}.json"
fi
if [[ "$channel" =~ ^claude-[0-9a-f]+$ && -n "$session_id" &&
      "$session_id" =~ ^[A-Za-z0-9._:-]+$ && "$tokens" =~ ^[0-9]+$ &&
      -n "$snapshot_path" && -d "$runtime_dir" ]]; then
  snapshot_tmp=$(mktemp "$runtime_dir/.claude-idle-compact.XXXXXX" 2>/dev/null || true)
  if [ -n "${snapshot_tmp:-}" ]; then
    if jq -cn --arg channel "$channel" --arg session_id "$session_id" \
         --argjson tokens "$tokens" --argjson timestamp "$(date +%s)" \
         '{channel:$channel, session_id:$session_id,
           current_input_context_tokens:$tokens, update_timestamp:$timestamp}' \
         >"$snapshot_tmp" 2>/dev/null; then
      chmod 600 "$snapshot_tmp" 2>/dev/null || true
      mv -f "$snapshot_tmp" "$snapshot_path"
    else
      rm -f "$snapshot_tmp"
    fi
  fi
fi
if [ -n "$tokens" ] && [ "$tokens" != "null" ] && [ "$tokens" -gt 0 ] 2>/dev/null; then
  tk=$(echo "$tokens" | awk '{if($1>=1000) printf "%.1fK",$1/1000; else printf "%d",$1}')
else
  tk=""
fi

# 5-hour rolling window — remaining (prefer dedicated field, fall back to 100-used)
five_hr=$(echo "$input" | jq -r '
  if .rate_limits.five_hour.remaining_percentage != null
  then .rate_limits.five_hour.remaining_percentage
  elif .rate_limits.five_hour.used_percentage != null
  then (100 - .rate_limits.five_hour.used_percentage)
  else empty
  end')

# Weekly quota — remaining % and days to reset
seven_day_used=$(echo "$input" | jq -r '.rate_limits.seven_day.used_percentage // empty')
seven_day_resets=$(echo "$input" | jq -r '.rate_limits.seven_day.resets_at // empty')
week_pct=""
week_days=""
if [ -n "$seven_day_used" ] && [ "$seven_day_used" != "null" ]; then
  week_pct=$(echo "$seven_day_used" | awk '{printf "%.0f", 100-$1}')
fi
if [ -n "$seven_day_resets" ] && [ "$seven_day_resets" != "null" ] && [ "$seven_day_resets" -gt 0 ] 2>/dev/null; then
  now=$(date +%s)
  secs=$((seven_day_resets - now))
  [ $secs -lt 0 ] && secs=0
  week_days=$(awk "BEGIN{printf \"%.1f\", $secs/86400}")
fi

printf "\033[38;5;166m %s\033[0m" "$user@$host"
printf " \033[33m %s\033[0m" "$short_cwd"
if [ -n "$git_branch" ]; then
  printf " \033[34m %s\033[0m" "$git_branch"
fi
printf "\033[38;5;59m  %s\033[0m" "$time_str"
# The Caddy creature — this session's Pokémon from the sealed assignment ledger
# (HelmCortex FORGE/bin/pokedex statusline; gazetteer sealed 2026-08-24, msg 708).
# Silent when HelmCortex or the ledger is absent — never breaks the statusline.
creature=""
pokedex_bin="$HOME/HelmCortex/FORGE/bin/pokedex"
if [ -x "$pokedex_bin" ]; then
  creature=$(printf '%s' "$input" | "$pokedex_bin" statusline 2>/dev/null)
fi
# Consolidate "partner of Fable" with model "Fable 5": row 2 renders the partner
# as the model chip and row 1 drops the model segment (his word, msg 717).
if [ -n "$creature" ] && [ -n "$model" ]; then
  partner_name="${creature##*partner of }"
  if [ "$partner_name" != "$creature" ]; then
    case "$model" in
      "$partner_name"*)
        creature="${creature%partner of *}partner of $(printf '\033[36m✦ %s\033[0m' "$model")"
        model=""
        ;;
    esac
  fi
fi
if [ -n "$model" ]; then
  printf " \033[36m✦ %s\033[0m" "$model"
fi
if [ -n "$tk" ] && [ -n "$used" ]; then
  printf " \033[38;5;59m%s (%s%%)\033[0m" "$tk" "$(printf '%.0f' "$used")"
elif [ -n "$used" ]; then
  printf " \033[38;5;59m(%s%%)\033[0m" "$(printf '%.0f' "$used")"
fi
# Row 2: creature + the quota stats that overflowed row 1 on portrait panes (msg 717).
row2=""
[ -n "$creature" ] && row2="$creature"
if [ -n "$five_hr" ]; then
  row2="${row2}${row2:+ }$(printf '\033[38;5;59m5h %s%%\033[0m' "$(printf '%.0f' "$five_hr")")"
fi
if [ -n "$week_pct" ] || [ -n "$week_days" ]; then
  out="wk"
  [ -n "$week_pct" ] && out="${out} ${week_pct}%"
  [ -n "$week_days" ] && out="${out} ${week_days}d"
  row2="${row2}${row2:+ }$(printf '\033[38;5;59m%s\033[0m' "$out")"
fi
[ -n "$row2" ] && printf "\n%s" "$row2"
#+END_SRC

    "tui.prompt.append": async () => {

* Git commit guard
Block the exact failure mode that caused the recent regression:

- refuse commits from a dirty tracked worktree
- reject backup/runtime spillover files
- reject generated output commits that do not include canonical Org source
- reject giant sweep commits unless explicitly overridden
- refuse commits while any stash entry exists (rule 12e — a stash is transport, never storage; the 2026-08-06 unpopped pre-pull stash silently reverted four days of work)

Escape hatches exist for intentional exceptional cases:

- =DOTCORTEX_GUARD_ALLOW_DIRTY=1=
- =DOTCORTEX_GUARD_ALLOW_BULK=1=
- =DOTCORTEX_GUARD_ALLOW_JUNK=1=
- =DOTCORTEX_GUARD_ALLOW_GENERATED=1=
- =DOTCORTEX_GUARD_ALLOW_STASH=1=

#+BEGIN_SRC bash :tangle all/.local/bin/dotcortex-git-guard :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/dotcortex-git-guard
set -euo pipefail

note() {
  printf 'dotcortex-guard: %s\n' "$*" >&2
}

die() {
  note "$1"
  exit 1
}

repo="$(git rev-parse --show-toplevel 2>/dev/null || true)"
[ -n "$repo" ] || exit 0
cd "$repo"

allow_dirty="${DOTCORTEX_GUARD_ALLOW_DIRTY:-0}"
allow_bulk="${DOTCORTEX_GUARD_ALLOW_BULK:-0}"
allow_junk="${DOTCORTEX_GUARD_ALLOW_JUNK:-0}"
allow_generated="${DOTCORTEX_GUARD_ALLOW_GENERATED:-0}"
allow_stash="${DOTCORTEX_GUARD_ALLOW_STASH:-0}"

staged=()
while IFS= read -r -d '' path; do
  staged+=("$path")
done < <(git diff --cached --name-only -z --diff-filter=ACMRD)

[ "${#staged[@]}" -gt 0 ] || exit 0

unstaged=()
while IFS= read -r -d '' path; do
  unstaged+=("$path")
done < <(git diff --name-only -z --diff-filter=ACMRD)

if [ "${#unstaged[@]}" -gt 0 ] && [ "$allow_dirty" != "1" ]; then
  note "refusing commit from a dirty tracked worktree"
  printf '%s\n' "${unstaged[@]:0:20}" | sed 's/^/  - /' >&2
  if [ "${#unstaged[@]}" -gt 20 ]; then
    note "... and $((${#unstaged[@]} - 20)) more"
  fi
  note "stage intentionally, or commit the in-flight work as its own wip checkpoint — never stash it away"
  note "override once with: DOTCORTEX_GUARD_ALLOW_DIRTY=1 git commit ..."
  exit 1
fi

stash_count="$(git stash list | wc -l)"
if [ "$stash_count" -gt 0 ] && [ "$allow_stash" != "1" ]; then
  note "refusing commit while $stash_count stash entries exist — a stash is transport, never storage (rule 12e)"
  git stash list --format='  - %gd  %cr  %s' | head -12 >&2
  if [ "$stash_count" -gt 12 ]; then
    note "... and $((stash_count - 12)) more"
  fi
  note "resolve each: git stash pop/apply it, or preserve it as a real ref:"
  note "  git tag rescue/stash-<name> \$(git rev-parse 'stash@{N}') && git stash drop 'stash@{N}'"
  note "override once (triage in progress) with: DOTCORTEX_GUARD_ALLOW_STASH=1 git commit ..."
  exit 1
fi

declare -A roots=()
bad_paths=()
has_org_source=0
generated_paths=0

for path in "${staged[@]}"; do
  root="${path%%/*}"
  roots["$root"]=1

  case "$path" in
    *.org)
      has_org_source=1
      ;;
  esac

  case "$path" in
    all/*|linux/*|debian/*|devuan/*|x230/*|t480s/*|arch/*|osx/*|be/*|navi/*|think/*|.claude/*|.opencode/*|.agents/*|AGENTS.md|CLAUDE.md|Makefile)
      generated_paths=$((generated_paths + 1))
      ;;
  esac

  case "$path" in
    *.bak|*.bak.*|*.backup-*|*~|*.orig|*.rej|*.swp|*.swo|.#*|[#]*[#]|.vorg.db|.claude/stop_count|.codex|all/.hermes/channel_directory.json|all/.hermes/cron/*)
      bad_paths+=("$path")
      ;;
  esac
done

if [ "${#bad_paths[@]}" -gt 0 ] && [ "$allow_junk" != "1" ]; then
  note "refusing commit with backup/runtime spillover paths"
  printf '%s\n' "${bad_paths[@]}" | sed 's/^/  - /' >&2
  note "clean or ignore them first"
  note "override once with: DOTCORTEX_GUARD_ALLOW_JUNK=1 git commit ..."
  exit 1
fi

if [ "$generated_paths" -gt 0 ] && [ "$has_org_source" -eq 0 ] && [ "$allow_generated" != "1" ]; then
  note "generated output is staged without a canonical .org source change"
  note "override once with: DOTCORTEX_GUARD_ALLOW_GENERATED=1 git commit ..."
  exit 1
fi

root_count="${#roots[@]}"
staged_count="${#staged[@]}"
if [ "$staged_count" -gt 80 ] && [ "$root_count" -gt 10 ] && [ "$allow_bulk" != "1" ]; then
  note "refusing giant sweep commit ($staged_count staged paths across $root_count top-level roots)"
  note "split it into reviewed chunks"
  note "override once with: DOTCORTEX_GUARD_ALLOW_BULK=1 git commit ..."
  exit 1
fi

if [ "$has_org_source" -eq 1 ] && [ "$generated_paths" -eq 0 ]; then
  note "source-only commit detected; if this change should affect generated output, run tangle first"
fi

exit 0

* Voice reply relay (outbound — pvox speaks what the agent sends)

The other half of the voice loop, per Mètsàtron's ruling (2026-08-16): every Telegram reply an agent sends is ALSO spoken aloud through pvox, automatically, with the voice chosen by the agent's identity — no extra agent action, no extra tokens. A =PostToolUse= hook fires on the Telegram plugin's =reply= tool, lifts the already-generated =text= from the tool input, resolves the speaking identity, and fire-and-forgets it to pvox on the speaker host over SSH. The hook returns immediately (playback is fully detached), so replies are never delayed by synthesis.

Identity resolution, simplest-first: =CLAUDE_WARM_HERDR_AGENT= (set by every claude-* launcher: Fable, Haiku, Auryn, Opus) → the basename suffix of =TELEGRAM_STATE_DIR= (=telegram-fable= → fable) → =Default=. The identity maps onto the pvox agent roster (=pvox agents=); identities without their own pvox voice yet (Fable, Haiku, Opus) speak as =Claude= until Mètsàtron seals dedicated voices in VoxForge. Delivery is split-lane (ruling 2026-08-17, the night the phone earned its voice): synthesis runs on the hook's own machine via pvox (HelmCortex is NFS-shared, so pvox and the conda engines are everywhere), and playback ships as a *headered WAV* to =PVOX_REPLY_HOST= (default =s24= — the phone, always in his pocket), falling back to =PVOX_REPLY_FALLBACK_HOST= (default =t480s=) when the phone is unreachable. The old pvox-over-SSH law forbade piping *raw* PCM across SSH because the player cannot know the rate; WAV carries its header, so that law is honored, not repealed. Disable per-session with =PVOX_REPLY_OFF=1=.

Delivery is *push-then-play* (fix 2026-08-17, the "stopped halfway" incident): the original design streamed the WAV over SSH straight into =sox='s stdin, which made the entire playback depend on the SSH connection staying alive — when the phone dozed mid-play (screen off, in pocket), the stream stalled and the audio truncated halfway. Now the play host receives the whole WAV into a temp file first, then playback detaches from the SSH session (=nohup=, backgrounded) with a Termux wake-lock held for its duration, so the network matters only during transfer and a doze can no longer cut the voice off. The SSH exit code therefore certifies *delivery*, not playback — the log field says =delivered=, honestly. Every skip and failure path logs one line to =~/.cache/pvox-reply.log= (empty-text skip with stdin byte count, render failure, per-host delivery failure, successful delivery with host + sizes) — the hook's original total silence meant every incident cost a fresh archaeology dig, against the silent-skip law.

Kitten seats render *on the play host itself* (his ruling 2026-08-17: the S24 out-muscles the fleet for small models, Q8-verified ~0.4–0.9 RTF): the hook resolves the seat via =pvox voicespec= and cleans the text via =pvox clean= (both fleet-side — pvox's config and markdown cleaner stay the single source of truth), then ships CLEANED TEXT instead of a WAV to =termux-kitten-say= on the play host, detached with a wake lock, sharing the same play queue lock as the ship lane. The fleet render+ship path remains the lane for non-kitten seats, fallback hosts, and a dead enclave — on-device failure logs =ON-DEVICE-FAIL= and falls through. Kill switch: =PVOX_REPLY_NO_ONDEVICE=1=.

* Voice transcript relay

Auto-transcribes Telegram voice/audio notes and relays the transcript both ways. The official Telegram plugin delivers voice notes as an =attachment_file_id= in the =<channel>= tag without downloading them — without this hook the agent must notice, call =download_attachment=, and transcribe by hand. This hook makes the flow structural, per Mètsàtron's ruling (2026-08-16): the same architecture must hold wherever the agent runs — kikin today, Termux on the phone later — so it lives here in the hooks source, not in any one session's habits.

Flow: =UserPromptSubmit= fires on the inbound channel message → the hook parses =attachment_kind="voice|audio"= + =attachment_file_id= from the tag attributes (plugin-authored; body text could forge them, but senders are allowlisted and a forged file_id can only reach files this bot already holds) → downloads via the Bot API using the token in =TELEGRAM_STATE_DIR/.env= → runs =whisper-transcribe --model small= (found on PATH or at the HelmCortex FORGE fallback) → injects the transcript as =additionalContext= so the agent reads it as if typed → and for provenance parity with the Hermes agent, replies the transcript into the Telegram chat itself, quoted, threaded to the voice message, so the chat log carries what the machine heard. Idempotent per =file_id= (download, transcript, and chat echo each fire once); every failure degrades to a one-line context note rather than blocking the prompt; the token never appears in any emitted text (exception class names only).

            '%s/bot%s/getFile?file_id=%s' % (api, token,

        'transcribed; use download_attachment + whisper-transcribe manually'
        % (e.__class__.__name__, file_id))

        # needs ~175s of compute (timed out at 150 live, 2026-08-16). 480s covers

    except Exception as e:
        with open(os.path.join(state, 'echo.log'), 'a') as lg:
            lg.write('%s -> FAILED %s\n' % (safe_id, e.__class__.__name__))
        # echo is best-effort; the context injection below still lands

    % (os.path.basename(audio), text))

* File edit rollback guard
Stages exact-path rollback anchors around Claude file-edit tools in DotCortex:

- before =Edit=/=MultiEdit=/=Write=, an already-dirty tracked file is staged so its pre-edit state remains recoverable from the index
- before editing an existing untracked file, that file is staged first
- after =Write= creates a new file, the exact new path is staged immediately

The hook does not auto-stage clean tracked files after editing; doing that would overwrite the index and erase the previous rollback point.

#+BEGIN_SRC bash :tangle all/.local/bin/claude-hook-pre-tool-use-file-edit :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/claude-hook-pre-tool-use-file-edit

# are already dirty before the next edit touches them.

import json
import os
import subprocess
import sys
from pathlib import Path


def tool_paths(payload):
    tool_input = payload.get("tool_input", {}) if isinstance(payload, dict) else {}
    candidates = []
    for key in ("file_path", "path"):
        value = tool_input.get(key)
        if isinstance(value, str) and value:
            candidates.append(value)
    return candidates


def run(cmd, cwd):
    return subprocess.run(cmd, cwd=cwd, text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)


def repo_root_for(path):
    probe = path if path.is_dir() else path.parent
    result = run(["git", "rev-parse", "--show-toplevel"], str(probe))
    if result.returncode != 0:
        return None
    repo = Path(result.stdout.strip()).resolve()
    if not (repo / "agents.org").is_file() or repo.name != "DotCortex":
        return None
    return repo


def relpath(repo, path):
    try:
        return str(path.resolve(strict=False).relative_to(repo))
    except ValueError:
        return None


def is_tracked(repo, rel):
    return run(["git", "ls-files", "--error-unmatch", "--", rel], str(repo)).returncode == 0


def has_unstaged(repo, rel):
    return run(["git", "diff", "--quiet", "--", rel], str(repo)).returncode == 1


def is_untracked_existing(repo, rel):
    result = run(["git", "ls-files", "--others", "--exclude-standard", "--", rel], str(repo))
    return result.returncode == 0 and any(line == rel for line in result.stdout.splitlines())


try:
    payload = json.loads(sys.argv[1]) if len(sys.argv) > 1 and sys.argv[1].strip() else {}
except json.JSONDecodeError:
    sys.exit(0)

cwd = Path(os.getcwd()).resolve()
staged = []

for raw in tool_paths(payload):
    path = Path(raw)
    if not path.is_absolute():
        path = cwd / path
    path = path.resolve(strict=False)
    repo = repo_root_for(path)
    if repo is None:
        continue
    rel = relpath(repo, path)
    if not rel or rel == ".git" or rel.startswith(".git/"):
        continue
    if path.exists() and ((is_tracked(repo, rel) and has_unstaged(repo, rel)) or is_untracked_existing(repo, rel)):
        result = run(["git", "add", "--", rel], str(repo))
        if result.returncode != 0:
            print("BLOCKED: could not create DotCortex rollback stage before file edit.")
            print()
            print(f"Path: {path}")
            print(result.stderr.strip())
            sys.exit(2)
        staged.append(rel)

if staged:
    print("DotCortex rollback stage created before edit:")
    for item in staged:
        print(f"  - {item}")

sys.exit(0)
PYEOF
case "$?" in
  0) exit 0 ;;
  2) exit 2 ;;
  *) exit 0 ;;
esac

* Remote git safety guard
Intercepts Bash tool calls that attempt remote git operations without proper precautions.
Blocks =ssh -A= (agent forwarding) and raw remote =git pull= without dirty-state checks.
Also blocks raw =git pull --ff-only= on remote machines when =dotcortex-pull= should be used instead.
Blocks DotCortex stow verbs whose machine target does not match the machine they would run on — the live host for local commands, or the ssh destination for ssh-wrapped commands. Block reasons go to stderr, which is what Claude Code shows the agent on a PreToolUse exit-2.

* Cross-harness stow target guard

The machine-target check is shared by Claude Code, Codex, and =make safe-stow=.
The harness hooks are useful early warnings, but =safe-stow= is the enforcement
boundary: a direct terminal invocation must not be able to apply an X230 or
T480s overlay on the wrong host. Shared overlays (=all linux debian devuan
termux=) are valid everywhere. Exactly one machine-specific overlay may appear
in a package set, and it must match the live hostname. Unknown package names
remain outside this guard so future non-fleet overlays are not silently
reclassified as laptop targets.

The command mode accepts the Claude/Codex shell command text and understands
=loom stow:<machine>=, direct =stow <machine>=, =make safe-stow= with
=STOW_PKGS=, and the legacy =make x11-apply= X230 path. The package mode is
what the Make recipe calls after expanding =STOW_PKGS=.

#+BEGIN_SRC python :tangle all/.local/bin/dotcortex-stow-target-guard :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
#!/usr/bin/env python3
"""Fail-closed host/overlay validation for every DotCortex stow path."""

import argparse
import json
import os
import re
import shlex
import socket
import sys


SHARED_PACKAGES = {"all", "linux", "debian", "devuan", "termux"}
HOST_ALIASES = {
    "ThinkPad-T480s": "t480s",
    "t480s": "t480s",
    "ThinkPad-X230": "x230",
    "x230": "x230",
    "ThinkPad-T480": "t480",
    "t480": "t480",
    "ThinkPad-T1700": "t1700",
    "t1700": "t1700",
    "kikin-kushi": "t1700",
    "OpenMandriva": "openmandriva",
    "openmandriva": "openmandriva",
    "s24": "s24",
}
EXPECTED_HOSTS = {
    "x230": {"x230", "ThinkPad-X230"},
    "t480s": {"t480s", "ThinkPad-T480s"},
    "t480": {"t480", "ThinkPad-T480"},
    "t1700": {"t1700", "ThinkPad-T1700", "kikin-kushi"},
    "openmandriva": {"openmandriva", "OpenMandriva"},
    "s24": {"s24"},
}
MACHINE_PACKAGES = set(EXPECTED_HOSTS)


def sanctuary_context_active():
    """Allow the X230 public layer only inside its named guest home."""
    # The named guest is the only deliberate cross-host stow exception: the
    # Cortex migration projects the X230-shaped public layer into T1700.
    context = os.environ.get("DOTCORTEX_STOW_CONTEXT", "")
    home = os.path.realpath(os.environ.get("HOME", ""))
    guest_suffix = "/.local/share/dotcortex/guests/sanctuary-cortex/home"
    return context == "sanctuary-cortex" and home.endswith(guest_suffix)


def short_host(raw_host):
    return HOST_ALIASES.get(raw_host, raw_host)


def block(reason, target, raw_host, where, codex):
    short = short_host(raw_host)
    message = (
        "DotCortex stow target does not match the machine it would run on. "
        f"Requested target: {target}. Would run on: {where} "
        f"(alias: {short}). {reason}"
    )
    if codex:
        print(json.dumps({
            "hookSpecificOutput": {
                "hookEventName": "PreToolUse",
                "permissionDecision": "deny",
                "permissionDecisionReason": message,
            }
        }))
        return 0
    print("BLOCKED: " + message, file=sys.stderr)
    print(f"Use the stow verb for '{short}', or run this on the matching machine.", file=sys.stderr)
    return 2


def validate(packages, raw_host, where, codex=False):
    targets = [pkg for pkg in packages if pkg in MACHINE_PACKAGES]
    unique_targets = list(dict.fromkeys(targets))
    if len(unique_targets) > 1:
        return block(
            "A package set may contain only one machine-specific overlay.",
            ", ".join(unique_targets),
            raw_host,
            where,
            codex,
        )
    if not unique_targets:
        return 0
    target = unique_targets[0]
    if target == "x230" and sanctuary_context_active():
        return 0
    expected = EXPECTED_HOSTS[target]
    if raw_host not in expected and short_host(raw_host) not in expected:
        return block(
            f"Expected host: {', '.join(sorted(expected))}.",
            target,
            raw_host,
            where,
            codex,
        )
    return 0


def ssh_destination(command):
    try:
        words = shlex.split(command)
    except ValueError:
        words = command.split()
    if not words or words[0] != "ssh":
        return None
    consuming_flags = {"-p", "-i", "-o", "-l", "-F", "-J", "-b", "-e", "-c", "-L", "-R", "-D", "-W", "-S"}
    index = 1
    while index < len(words):
        word = words[index]
        if word in consuming_flags:
            index += 2
            continue
        if word.startswith("-"):
            index += 1
            continue

def simple_commands(text):
    """Split into simple commands, honouring quotes, keeping redirect operators.

    shlex with punctuation_chars emits ; & | < > as their own tokens and leaves
    quoted spans intact as single tokens, so a `>` inside "a -> b" never appears
    as an operator. That is the whole reason this is tokenised rather than
    regexed: two false positives on 2026-07-26 came from matching shell syntax
    inside quoted strings.
    """
    lexer = shlex.shlex(text, posix=True, punctuation_chars=True)
    lexer.whitespace_split = True
    try:
        tokens = list(lexer)
    except ValueError:
        return None  # unbalanced quotes / heredoc: caller falls back

    groups, current = [], []
    for tok in tokens:
        if tok in (";", "&&", "||", "|", "&", "\n"):
            groups.append(current)
            current = []
        else:
            current.append(tok)
    groups.append(current)
    return groups

# starting at `do`, so without this a `for ...; do cp ...; done` writes straight
# through the gate — found by the corpus on 2026-07-26.

                  "if", "until", "case", "esac", "in", "{", "}", "(", ")", "!",
                  "select", "function"}

    """First real word, skipping env assignments, prefixes and shell keywords."""

            continue

            continue
        if w in SHELL_KEYWORDS:
            continue
        return w.rsplit("/", 1)[-1], words[i + 1:]
    return None, []

* Pre-compact
Memory is destructive to truncate and must only be cleared after the Obsidian
append reports success.  The old unconditional sequence lost the live copy when
an archive command failed: the T480 facts survived elsewhere, but the promised
=DotCortex.md= dump did not exist.  A failed append now returns its status and
leaves =MEMORY.md= byte-for-byte untouched.

The hook is model-aware.  A workspace whose =.claude/MEMORY.md= is a *projection*
of a HelmCortex harness =MEMORYS.md= (detected by the =LOCAL AGENT NOTES=
markers) keeps its durable memory version-tracked in the harness — agents write
there, never in the projection — so on compaction we archive and reset only the
per-machine local-notes block and leave the durable projection intact (it
regenerates from the harness on the next compile).  A legacy workspace whose
=MEMORY.md= *is* the memory (=## Session Facts= / =## User Model=) keeps the
original dump-whole-then-truncate behaviour.

#+BEGIN_SRC bash :tangle all/.local/bin/claude-hook-pre-compact :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/claude-hook-pre-compact
WORKSPACE=$(basename "$PWD")
MEMFILE="$PWD/.claude/MEMORY.md"
[ -f "$MEMFILE" ] || { echo 'No MEMORY.md, nothing to save.'; exit 0; }

if grep -q '<!-- BEGIN LOCAL AGENT NOTES -->' "$MEMFILE"; then
  # Harness-projected workspace: durable memory is version-tracked in the
  # HelmCortex harness MEMORYS.md; only the per-machine local-notes block is
  # scratch worth archiving. Reset just that block, leave the durable projection.
  CONTENT=$(awk '/<!-- BEGIN LOCAL AGENT NOTES -->/{f=1;next} /<!-- END LOCAL AGENT NOTES -->/{f=0} f' "$MEMFILE" | grep -v '^[[:space:]]*<!--' | sed '/^[[:space:]]*$/d')
  if [ -n "$CONTENT" ]; then
    DUMP=$(printf '## Memory Dump %s (%s local scratch)\n\n%s' "$(date +%Y-%m-%d)" "$WORKSPACE" "$CONTENT")
    if flatpak run md.obsidian.Obsidian vault=HelmCortex append path="LOGS/TODO/$WORKSPACE.md" content="$DUMP" 2>/dev/null; then
      awk '/<!-- BEGIN LOCAL AGENT NOTES -->/{print; print "<!-- Per-machine scratch only. Promote durable notes into the harness MEMORYS.md and commit. -->"; f=1; next} /<!-- END LOCAL AGENT NOTES -->/{f=0} !f' "$MEMFILE" > "$MEMFILE.tmp" && mv "$MEMFILE.tmp" "$MEMFILE"
      echo "Local scratch saved to $WORKSPACE TODO; durable memory lives in the harness MEMORYS.md (write there, not the projection)."
    else
      status=$?
      echo "Memory archive failed for $WORKSPACE; local scratch preserved." >&2
      exit "$status"
    fi
  else
    echo "No local scratch; durable memory is in the harness MEMORYS.md."
  fi
else
  # Legacy workspace: MEMORY.md itself IS the memory (## Session Facts / ## User Model).
  CONTENT=$(tail -n +2 "$MEMFILE" 2>/dev/null | sed '/^$/d' | sed '/^## Session Facts$/d' | sed '/^## User Model.*$/d')
  if [ -n "$CONTENT" ]; then
    DUMP=$(printf '## Memory Dump %s\n\n%s' "$(date +%Y-%m-%d)" "$CONTENT")
    if flatpak run md.obsidian.Obsidian vault=HelmCortex append path="LOGS/TODO/$WORKSPACE.md" content="$DUMP" 2>/dev/null; then
      printf '# Session Memory (auto-updated by Opus, max 2000 chars)\n\n## Session Facts\n\n\n## User Model (max 1500 chars for this section)\n\n' > "$MEMFILE"
      echo "Memory saved to $WORKSPACE TODO via Obsidian CLI and MEMORY.md truncated."
    else
      status=$?
      echo "Memory archive failed for $WORKSPACE; MEMORY.md preserved." >&2
      exit "$status"
    fi
  else
    echo 'MEMORY.md empty, nothing to save.'
  fi
fi
