* Scheme builders, converters & importers

** Windows Classic theme builder — generalised

One builder for the whole classic dynasty: pristine upstream Windows-95 base + any titlebar gradient endpoints. Me/2000/XP-classic genuinely shared ONE "Windows Standard" scheme (#0A246A→#A6CAF0) — one built theme serves all three, distinctness living in icons/sounds/desktop exactly as real Windows did it.

from PIL import Image
g4, face, sel, seltxt, bhi, bsh, bdk, fgc = sys.argv[1:9]
def rgb(h): return tuple(int(h.lstrip('#')[i:i+2], 16) for i in (0, 2, 4))

def lum(c): return 0.2126*c[0] + 0.7152*c[1] + 0.0722*c[2]
def mixw(f): return '#%02x%02x%02x' % tuple(round(c + (255 - c) * f) for c in F)

# bevel-drawing declarations (box-shadow / border) — the same hexes elsewhere
# are backgrounds and data whites that must not turn face-coloured.

        if fn.endswith('.css'):

            for ln in open(p, encoding='latin-1').read().splitlines(True):
                if 'box-shadow' in ln or 'border' in ln:
                    for k, v in BEVLINE.items():
                        ln = re.sub(re.escape(k) + r'(?![0-9a-fA-F])', v, ln, flags=re.I)
                out.append(ln)
            t = ''.join(out)
            t = re.sub(r'#c0c0c0', face, t, flags=re.I)
            t = re.sub(r'\b192,\s*192,\s*192\b', '%d, %d, %d' % F, t)
            t = re.sub(r'#000080', sel, t, flags=re.I)
            t = re.sub(r'\b0,\s*0,\s*128\b', '%d, %d, %d' % S, t)
            t = re.sub(r'(@define-color theme(?:_unfocused)?_selected_fg_color) #FFFFFF',
                       r'\1 ' + seltxt, t, flags=re.I)
            # Notebook page + levelbar trough fixes carried over from the CDE
            # lane (the pavucontrol white-page / flat-meter defects).
            t = t.replace(
                'notebook > stack:not(:only-child) {\n  background-color: #FFFFFF; }',
                'notebook > stack:not(:only-child) {\n  background-color: ' + face + '; }')
            t = t.replace(
                'notebook > stack:not(:only-child):backdrop {\n    background-color: #FFFFFF; }',
                'notebook > stack:not(:only-child):backdrop {\n    background-color: ' + face + '; }')

            # buttons — the "hard highlight on tables" (ruling 2026-08-07).
            # A 95-style header is a raised face button; give it the unified
            # bevel ramp and the theme fg.

                'columnview.view > header > button, treeview.view > header > button {\n    color: gray;\n    background-color: #FFFFFF;\n    font-weight: bold;\n    text-shadow: none;\n    box-shadow: none; }',

            # Horizontal rules / separators: hardcoded mid-grey — follow the
            # bevel shadow tone so rules sit softly on any face.
            t = t.replace(
                'separator {\n  -gtk-icon-source: none;\n  min-width: 1px;\n  min-height: 1px;\n  background-color: #7d7d7d;',
                'separator {\n  -gtk-icon-source: none;\n  min-width: 1px;\n  min-height: 1px;\n  background-color: ' + bsh + ';')
            t = t.replace(
                'levelbar trough {\n  padding: 1px;\n  border-radius: 0; }',
                'levelbar trough {\n  padding: 1px;\n  border-radius: 0;\n'
                '  border: 1px solid ' + bsh + ';\n  border-bottom-color: ' + bhi + ';\n'
                '  border-right-color: ' + bhi + ';\n  background-color: ' + face + '; }')
            if fgc.lower() != '#000000':
                t = re.sub(r'(?<![-\w])color:\s*#000000', 'color: ' + fgc, t)
                t = re.sub(r'(@define-color theme_(?:fg|text|unfocused_fg|unfocused_text)_color) #000000',
                           r'\1 ' + fgc, t)
            if lum(F) < 140:
                # Dark face: mix the base's light neutral backgrounds toward the
                # face (backdrop/insensitive tones) — same dict as the CDE lane.
                BRIGHT = {'#ececec': mixw(0.10), '#e0e0e0': mixw(0.08),
                          '#dedede': mixw(0.08), '#d5d5d5': mixw(0.06),
                          '#cccccc': mixw(0.06), '#c5c5c5': mixw(0.04),
                          '#b6b6b6': mixw(0.02), '#adadad': mixw(0.02),

            # are black rasters that must follow the theme fg on dark faces.

** SerenityOS Start button per-variant recolour

The =serenity= Start button (=menu-buttons/serenity/start.png=, the LadyBall + "Start" wordmark on a Win95-style raised grey =#d4d0c8= face) is one static raster. Copied verbatim to all 32 SerenityOS variants it leaves a fixed grey button clashing with every recoloured palette (Nord, Catppuccin, Dark…) — exactly the mismatch the caption glyphs avoid by recolouring per-variant. This tool is that missing recolour pass for the Start button: it tints the master's grey bevel/face ramp to each variant's own face colour while leaving the red LadyBall and the dark "Start" wordmark alone, inverting the wordmark to a light tone only on dark faces (the same luminance<96 rule the rest of the pipeline uses).

Why build-time, not apply-time: scheme switching runs =redstone-9x-scheme-apply= INSIDE the guest (the IceWM Settings menu calls it directly), and the guest has no python3/Pillow — so any Pillow recolour must happen on the host at build time and land as a pre-baked asset that =scheme-apply= merely =cp='s. Delivery rides the =menu-buttons/= tree, which the refresh re-projects wholesale every run (unlike theme dirs, which are gap-fill-only and would silently never update a live guest). Each variant keeps =MENU_BUTTON="serenity"=; the per-scheme file =menu-buttons/serenity/start-<scheme-id>.png= is what =scheme-apply= prefers over the grey =start.png= master (legacy =serenity=, whose face IS =#d4d0c8=, has no per-scheme file and correctly keeps the master).

The tint is luminance-preserving against a fixed anchor (the master's own =#d4d0c8= face luminance): each greyish pixel is re-expressed as a blend fraction toward white (highlights) or black (shadows) applied to the target face, so the 3D bevel structure survives the recolour into any base colour. Saturated pixels (the ball) and near-black art (the wordmark) are classified out by saturation and luminance and never tinted. Rerun =--all= after any palette edit.

Usage:
  redstone-9x-start-recolor --all                 regenerate all serenity-*.scheme buttons
  redstone-9x-start-recolor <face_hex> <out_png>  recolour one button to <face_hex>

MENUDIR  = os.path.join(ROOT, "linux/.icewm/menu-buttons/serenity")
MASTER   = os.path.join(MENUDIR, "start.png")
SCHEMES  = os.path.join(ROOT, "linux/.config/redstone-9x/schemes")

MASTER_FACE = (212, 208, 200)          # #d4d0c8 — the master button's flat face
SAT_MIN     = 28                        # >this chroma => mascot colour, never tinted
ART_L_MAX   = 40                        # <this luma  => dark art (wordmark/outline)
DARK_FACE_L = 96                        # face luma <this => light art on dark button

def lum(c):   return 0.299*c[0] + 0.587*c[1] + 0.114*c[2]
def blend(a, b, t): return tuple(round(a[i] + (b[i]-a[i])*t) for i in range(3))
def hx(s):
    s = s.lstrip("#")
    return (int(s[0:2],16), int(s[2:4],16), int(s[4:6],16))

LFACE = lum(MASTER_FACE)

def recolour(face, bhi=None, bsh=None):
    """Return a recoloured copy of the master button for the given face RGB.

    Bevel blend TARGETS are the scheme's unified BEVEL_HIGHLIGHT/BEVEL_SHADOW
    (ruling 2026-08-08: blending to pure white/black reproduced the master's
    harsh ramp on every face — the Start button never got the soft-bevel
    treatment its theme chrome did). Fallback mirrors the converter's law."""
    if bhi is None or bsh is None:
        dark = lum(face) < 140
        f_hi, f_sh = (0.25, 0.5) if dark else (0.6, 0.4)
        bhi = bhi or blend(face, (255, 255, 255), f_hi)
        bsh = bsh or tuple(round(c * (1 - f_sh)) for c in face)

            if a == 0:

            sat = max(r, g, b) - min(r, g, b)
            if sat > SAT_MIN:                      # LadyBall colour: leave it
                continue
            L = lum((r, g, b))
            if L < ART_L_MAX:                      # dark art: invert only on dark faces
                if dark:
                    px[x, y] = (art[0], art[1], art[2], a)
                continue
            if L >= LFACE:                         # highlight side of the bevel
                t  = (L - LFACE) / (255 - LFACE) if LFACE < 255 else 0.0
                nc = blend(face, bhi, t)
            else:                                  # shadow side of the bevel
                t  = (LFACE - L) / LFACE
                nc = blend(face, bsh, t)
            px[x, y] = (nc[0], nc[1], nc[2], a)
    return im

def hx(c):
    return '#%02x%02x%02x' % tuple(max(0, min(255, round(v))) for v in c)

def unhex(h):
    h = h.lstrip('#')
    return tuple(int(h[i:i+2], 16) for i in (0, 2, 4))

def lum(c):
    return 0.299 * c[0] + 0.587 * c[1] + 0.114 * c[2]

def hsv(c):
    r, g, b = (v / 255.0 for v in c)
    h, s, v = colorsys.rgb_to_hsv(r, g, b)
    return h * 360.0, s, v

def contrast(bar_rgb):
    return (0, 0, 0) if lum(bar_rgb) >= 128 else (255, 255, 255)

def blend_white(c, f):
    return tuple(x + (255 - x) * f for x in c)

def blend_black(c, f):
    return tuple(x * (1 - f) for x in c)

def parse_dp(path):
    sets = []
    with open(path, encoding='latin-1') as f:
        for line in f:
            line = line.strip()
            if not line.startswith('#'):
                continue
            h = line.lstrip('#')
            groups = [h[i:i+4] for i in (0, 4, 8)]
            rgb = tuple(int(g, 16) >> 8 for g in groups)
            sets.append(rgb)

def scale_to(rgb, target, ref):
    """Reproduce this pixel's tonal position relative to its file's own
    reference tone (blend toward white for highlights, toward black for
    shadows) on the new target anchor -- not a multiplicative luminance
    ratio, which clamps/blows out highlights and drifts hue at extremes
    (rejected at the mandatory ground-truth gate, 2026-07-20)."""
    lp, lr = lum(rgb), lum(ref)
    if abs(lp - lr) < 1e-6:
        return tuple(round(c) for c in target)
    if lp > lr:
        frac = (lp - lr) / max(1e-6, 255 - lr)
        out = blend_white(target, frac)
    else:
        frac = (lr - lp) / max(1e-6, lr)
        out = blend_black(target, frac)
    return tuple(max(0, min(255, round(c))) for c in out)

def classify_family(rgb, base1, base2, hue_band=22.0, sat_neutral=0.06):
    h, s, v = hsv(rgb)
    if s < sat_neutral:
        return None
    h1, _, _ = hsv(base1)
    h2, _, _ = hsv(base2)
    def hd(a, b):
        d = abs(a - b) % 360
        return min(d, 360 - d)
    d1, d2 = hd(h, h1), hd(h, h2)
    if min(d1, d2) > hue_band:
        return None
    return 1 if d1 <= d2 else 2

def parse_xbm(path):

def render_backdrop(bm_path, out_png, set8):

    w, h, bits = parse_xbm(bm_path)
    fg = tuple(round(c * 0.72) for c in set8)
    img = Image.new('RGB', (w, h), set8)
    px = img.load()
    row_bytes = (w + 7) // 8
    for y in range(h):
        for xb in range(row_bytes):
            idx = y * row_bytes + xb
            if idx >= len(bits):
                continue
            byte = bits[idx]
            for bit in range(8):
                x = xb * 8 + bit
                if x >= w:
                    continue
                if (byte >> bit) & 1:
                    px[x, y] = fg
    # Pre-tile the pattern onto a large canvas. pcmanfm's own wallpaper_mode=
    # "tile" is broken in the sanctuary (it uses the flaky XRender tiling path
    # and paints white — same RENDER fragility that crashed pcmanfm on the

    # ship a full image the scheme displays with wallpaper_mode=center: pixel-
    # perfect, no scaling, no border on any sanctuary-sized display.
    CW, CH = 2560, 1600
    canvas = Image.new('RGB', (CW, CH), set8)
    for ty in range(0, CH, h):
        for tx in range(0, CW, w):
            canvas.paste(img, (tx, ty))
    os.makedirs(os.path.dirname(out_png), exist_ok=True)
    canvas.save(out_png)

def write_workspace_buttons(tb_dir, face, active):
    """Emit BeOS-style workspace-button tiles so the pager lines up with the bar.

    IceWM TILES taskbar button pixmaps rather than stretching them, so the only
    seam-free shape is a strip a few pixels wide at EXACTLY the taskbar's height
    — which is why the BeOS-r5 theme ships every one of its taskbar pixmaps as
    3x24. A fixed-width bevel pixmap instead repeats "[ ][ ]" across any button
    wider than one tile; that is the defect that forced the CDE task-button
    bevel to be reverted (8ef9ba7f0), and it is the trap to avoid here.

    Height is read from taskbarbg.xpm rather than hardcoded, so the tiles track
    the taskbar the palette actually produced. Only the top and bottom bevel
    rows can be expressed — a horizontally tiled strip has no left/right edge —
    which matches BeOS's own vertical-strip construction.
    """
    height = 28
    bg_path = os.path.join(tb_dir, 'taskbarbg.xpm')
    if os.path.exists(bg_path):
        with open(bg_path) as fh:
            for line in fh:
                m = re.match(r'"\s*(\d+)\s+(\d+)\s+(\d+)\s+(\d+)', line)
                if m:
                    height = int(m.group(2))
                    break
    for fn, base in (('workspacebuttonbg.xpm', face),
                     ('workspacebuttonactive.xpm', active)):
        light = hx(blend_white(base, 0.45))
        dark = hx(blend_black(base, 0.45))
        body = hx(base)
        # Raised while idle, sunken while current — Motif's own convention, and
        # the way a real CDE front-panel switcher reads.
        hi, lo = (light, dark) if fn.endswith('bg.xpm') else (dark, light)
        rows = [hi if y < 2 else lo if y >= height - 2 else body
                for y in range(height)]
        palette, chars = [], {}
        for colour in rows:
            if colour not in chars:
                chars[colour] = chr(ord('a') + len(chars))
                palette.append((chars[colour], colour))
        name = os.path.splitext(fn)[0]
        out = ['/* XPM */', f'static char * {name}_xpm[] = {{',
               f'"3 {height} {len(palette)} 1",']
        out += [f'"{ch} c {colour}",' for ch, colour in palette]
        out += [f'"{chars[colour] * 3}",' for colour in rows]
        out[-1] = out[-1][:-1]
        out.append('};')
        with open(os.path.join(tb_dir, fn), 'w') as fh:
            fh.write('\n'.join(out) + '\n')

** GTK → Wine colour converter — derive a scheme's Wine palette from its GTK theme

The BeOS/Haiku analogue of the CDE =.dp= → wine derivation. Hand-made GTK themes (Bluecurve-BeOS, BeOS-r5, Haiku, and any future recolour) already carry the canonical widget palette as GTK-3.0 =@define-color= keys; this converter reads those, plus the scheme's IceWM theme titlebar colours, and emits a Wine =[Control Panel\Colors]= =.reg= that lines up exactly with the GTK/Qt/FOX layers. Make a colour variant by recolouring the GTK theme and setting the IceWM =ColorActiveTitleBar=, run this once, and the Wine palette follows — no hand-tuned registry guesses.

Role mapping is adapted from =endolith/KillerKiwi='s =gtk2winecolors= (2009): =ButtonFace/ActiveBorder/Menu= ← =theme_bg_color=, =Window= ← =theme_base_color=, =Hilight/HilightText= ← =theme_selected_{bg,fg}_color=, =GrayText= ← =insensitive_fg_color=, button hilight/shadow/scrollbar are luminance blends of the face (the CDE converter's =blend_white=/=blend_black= math). Their script has to *guess* titlebars — PyGTK cannot read them — so those guesses are replaced here with the real per-variant IceWM =default.theme= =ColorActiveTitleBar=/=ColorNormalTitleBar= (+ their text colours). BeOS's yellow tab is pixmap art with no flat colour key, so the honest flat titlebar value the file itself declares (grey for R5/Max/Haiku, terracotta =#be816c= for Personal) is exactly right. The 27-key set matches =write_wine_reg= so the whole scheme fleet stays consistent.

** IceWM theme deriver — minimal-touch, exact-ramp

The first derivations mangled tested themes: a broad "is it bluish?" pixel test chewed antialiased button and bevel pixels, and broken-symlink pruning dropped pixmaps IceWM then faked badly. This tool is the correction: the authoritative gradient ramp is collected from the =title*.xpm= palettes ONLY, and ONLY those exact colours are remapped anywhere else — every other pixel and file stays byte-identical to the source theme the user already approved.

=--warm= extends the same exact-ramp discipline to the BeOS-tab dynasty. There the tab and its caption buttons share one warm ramp (icedano's yellow: highlight =#FFFF52=, body =#FECA00=, shadow =#B17C00= — and the richer 8-stop yellow of the grafted BeOS-r5 buttons), so the ramp is scanned from =title*.xpm= AND the button pixmaps and remapped together, keeping the tab and buttons in one colour family (yellow → terracotta for Personal). The warm predicate is saturation-gated (=max-min > 40=, red and green each ≥ blue), so neutral bevels, black outlines, white glyph strokes, and the grey past-tab strip are never touched — they read as structure, not colour. This is how a new BeOS colourway lines up from a single dark/light pair, exactly like the CDE palette variants.

#+BEGIN_SRC bash :tangle linux/.local/bin/redstone-9x-icewm-derive :shebang "#!/bin/bash" :tangle-mode (identity #o755)
# redstone-9x-icewm-derive <src> <out> <dark-hex> <light-hex> [--flat-menusel HEX] [--drop-icons] [--warm]
# Clone an IceWM theme and remap ONLY its titlebar gradient ramp onto
# dark->light. Nothing else changes: buttons, bevels, taskbar art stay
# byte-identical; valid symlinks are preserved; only BROKEN links are pruned.
#
# Ramp modes:
#   (default) blue  — the ramp is the BLUE palette of title*.xpm only. For
#                     Win9x/CDE-style navy titlebars where the caption art is a
#                     separate byte-frozen layer.
#   --warm          — the ramp is the WARM/saturated (yellow/gold/terracotta)
#                     palette, scanned from title*.xpm AND the caption BUTTONS
#                     (close/minimize/maximize/restore/menuButton/depth). For
#                     BeOS-tab themes where the tab AND its buttons share one
#                     warm ramp and must recolour together (yellow->terracotta).
#                     Neutral greys/blacks/whites (bevels, outlines, text) are
#                     never warm, so they stay byte-identical either way.
set -euo pipefail
SRC="${1:?src theme dir}"; OUT="${2:?out dir}"; DARK="${3:?dark hex}"; LIGHT="${4:?light hex}"
shift 4
FLAT=""; DROP_ICONS=0; MODE=blue
while [[ $# -gt 0 ]]; do
  case "$1" in
    --flat-menusel) FLAT="$2"; shift 2 ;;
    --drop-icons) DROP_ICONS=1; shift ;;
    --warm) MODE=warm; shift ;;
    *) echo "unknown arg: $1" >&2; exit 1 ;;
  esac
done
rm -rf "${OUT}"
cp -a "${SRC}" "${OUT}"
find "${OUT}" -xtype l -delete
rm -f "${OUT}/default.theme.custom" "${OUT}"/default.theme.save "${OUT}"/*.save "${OUT}/results.txt" 2>/dev/null || true
[[ "${DROP_ICONS}" -eq 1 ]] && rm -rf "${OUT}/icons"

python3 - "${OUT}" "${DARK}" "${LIGHT}" "${FLAT}" "${MODE}" <<'PYINNER'
import sys, os, re
out, dark, light, flat, mode = sys.argv[1], sys.argv[2], sys.argv[3], sys.argv[4], sys.argv[5]
DARK = tuple(int(dark.lstrip('#')[i:i+2],16) for i in (0,2,4))
LIGHT = tuple(int(light.lstrip('#')[i:i+2],16) for i in (0,2,4))
FLAT = tuple(int(flat.lstrip('#')[i:i+2],16) for i in (0,2,4)) if flat else None
HEXRE = re.compile(r'c #([0-9a-fA-F]{6})')
def lum(c): return 0.299*c[0] + 0.587*c[1] + 0.114*c[2]
def is_blue(c): return c[2] > c[0] + 24 and c[2] > c[1] + 16
# Warm = saturated, red/green both at least as strong as blue (yellow/gold/
# orange/terracotta). Neutral greys have (max-min)==0 so they never qualify.
def is_warm(c): return (max(c)-min(c)) > 40 and c[0] >= c[2] and c[1] >= c[2]

warm = (mode == 'warm')
pred = is_warm if warm else is_blue
# In warm mode the tab and its caption buttons share one ramp, so scan both;
# in blue mode the caption art is a frozen layer, so scan title*.xpm ONLY.
BTN = ('close', 'minimize', 'maximize', 'restore', 'menuButton', 'depth')
def in_ramp_scan(fn):
    if not fn.endswith('.xpm'): return False
    if fn.startswith('title'): return True
    return warm and fn.startswith(BTN)

# The ramp = predicate-matching palette entries of the scanned pixmaps.
ramp = set()
for fn in os.listdir(out):
    if in_ramp_scan(fn):
        for m in HEXRE.finditer(open(os.path.join(out, fn), encoding='latin-1').read()):
            c = tuple(int(m.group(1)[i:i+2],16) for i in (0,2,4))
            if pred(c): ramp.add(c)
if not ramp:
    print("no titlebar ramp found — nothing remapped"); sys.exit(0)
LO, HI = min(map(lum, ramp)), max(map(lum, ramp))
def mapped(c):
    t = 0.0 if HI == LO else (lum(c)-LO)/(HI-LO)
    return tuple(round(d + (l-d)*t) for d, l in zip(DARK, LIGHT))
n = 0
try:
    from PIL import Image
    HAVE_PIL = True
except ImportError:
    HAVE_PIL = False
for dp, _, fs in os.walk(out):
    for fn in fs:
        p = os.path.join(dp, fn)
        if not os.path.isfile(p) or os.path.islink(p):
            continue
        flat_this = FLAT and fn.startswith('menusel')
        if fn.endswith('.xpm'):
            t = open(p, encoding='latin-1').read()
            def sub(m):
                c = tuple(int(m.group(1)[i:i+2],16) for i in (0,2,4))
                if c not in ramp: return m.group(0)
                R,G,B = FLAT if flat_this else mapped(c)
                return f'c #{R:02X}{G:02X}{B:02X}'
            t2 = HEXRE.sub(sub, t)
            if t2 != t: open(p, 'w', encoding='latin-1').write(t2); n += 1
        elif fn.endswith('.png') and HAVE_PIL:
            im = Image.open(p).convert('RGBA'); px = im.load(); ch = False
            for y in range(im.height):
                for x in range(im.width):
                    r,g,b,a = px[x,y]
                    if (r,g,b) in ramp:
                        R,G,B = FLAT if flat_this else mapped((r,g,b))
                        px[x,y] = (R,G,B,a); ch = True
            if ch: im.save(p); n += 1
print(f"exact-ramp remap: {len(ramp)} ramp colours, {n} files touched")

def emit(imgs, cat, name):
    # imgs: {16: Image|None, 32: Image|None}; nearest-neighbour resample for
    # the in-between/large sizes IceWM's graft wants (pixel-art rule).
    global count
    base32 = imgs.get(32) or (imgs.get(16).resize((32, 32), Image.NEAREST) if imgs.get(16) else None)
    base16 = imgs.get(16) or (imgs.get(32).resize((16, 16), Image.NEAREST) if imgs.get(32) else None)
    if base32 is None and base16 is None:
        return
    for size in SIZES:
        if size == 16:
            img = base16
        elif size == 32:
            img = base32
        else:
            src = base32 if base32 is not None else base16
            img = src.resize((size, size), Image.NEAREST)
        d = os.path.join(out, cat, str(size))
        os.makedirs(d, exist_ok=True)
        img.save(os.path.join(d, name + ".png"))
        count += 1

import sys
from PIL import Image

theme_out = sys.argv[1]
BLACK = (0, 0, 0, 255)
FACE = (212, 208, 200, 255)
# (center_x, top_y) per frame: frame0=normal (y0-17), frame1=pressed (y18-35)
FRAMES = [(8, 6), (9, 24)]

def down_triangle_pixels(cx, top):
    pts = []
    for i, w in enumerate((7, 5, 3, 1)):
        y = top + i
        x0 = cx - w // 2
        pts += [(x, y) for x in range(x0, x0 + w)]
    return pts

def up_triangle_pixels(cx, top):
    pts = []
    for i, w in enumerate((1, 3, 5, 7)):
        y = top + i
        x0 = cx - w // 2
        pts += [(x, y) for x in range(x0, x0 + w)]
    return pts

def redraw(fname, glyph_fn):
    path = f'{theme_out}/{fname}'
    im = Image.open(path).convert('RGBA')
    px = im.load()
    W, H = im.size
    for y in range(H):
        for x in range(W):
            if px[x, y] == BLACK:
                px[x, y] = FACE
    for cx, top in FRAMES:
        # top+1: centre the 4-row triangle on the 18px frame (same fix as the
        # shared title-gradient path, pixel-measured 2026-08-08)
        for x, y in glyph_fn(cx, top + 1):
            if 0 <= x < W and 0 <= y < H:
                px[x, y] = BLACK
    im.save(path)

redraw('minimizeA.png', down_triangle_pixels)
redraw('minimizeI.png', down_triangle_pixels)
redraw('maximizeA.png', up_triangle_pixels)
redraw('maximizeI.png', up_triangle_pixels)

            for y in range(img.height):
                for x in range(img.width):
                    r, g, b, a = px[x, y]
                    m = COLORMAP.get((r, g, b))
                    if m:
                        px[x, y] = (*m, a)
                        changed = True
            if changed:
                img.save(p)
                n += 1

** Windows classic colour schemes — 95 / 98 / 2000 default scheme import

One importer for the three eras, replacing the win95-only entrypoint shipped earlier the same day (reconciliation: identical decode core, generalized configs; the old =redstone-9x-win95-schemes-import= artifact is removed). Sources are archived in NEXUS with checksums, each extracted from real install media: Win95 =shell2.inf= (precopy2.cab, OSR2.5 ISO, 25 trailing COLORREF dwords per blob), Win98 SE =shell2.inf= (PRECOPY1.CAB, 29 dwords — the extended SYSCOLOR set whose last two are the DECLARED caption-gradient endpoints; most 98 schemes duplicate the solid caption there, so they authentically render solid, while e.g. Rainy Day declares a genuinely distinct endpoint), and Windows 2000 =HIVEDEF.INF= (i386, UTF-16LE, fixed 712-byte blobs, colour array always the last 116 bytes; sourced from the Simplified Chinese edition ISO — display strings are localized but tokens are canonical English and colour values are locale-independent, verified against the known #0A246A→#A6CAF0 Standard chrome). Win95 runs =FLAT_TITLE=1= (no caption gradients existed); 98/2000 emit their declared Gradient* keys verbatim — the no-made-up-values law end to end. Icons per era: Chicago95 / Win98SE / SE98.

#+BEGIN_SRC bash :tangle linux/.local/bin/redstone-9x-classic-schemes-import :shebang "#!/bin/bash" :mkdirp yes
# redstone-9x-classic-schemes-import <win95|win98|w2k>
# Decode Windows default Appearance scheme blobs -> .theme files -> theme-convert.
set -euo pipefail
MODE="${1:?mode: win95 | win98 | w2k}"
NEX="${HOME}/HelmCortex/NEXUS/sanctuaries/sanctuary-redstone-9x/themes"
TMP="$(mktemp -d)"; trap 'rm -rf "${TMP}"' EXIT
python3 - "${MODE}" "${NEX}" "${TMP}" <<'PYWCL'

mode, nex, out = sys.argv[1], sys.argv[2], sys.argv[3]
C9X = {  # shared 9x/NT curated name set, tokens differ per INF
    'BRICK_256': 'Brick', 'DESERT_256': 'Desert', 'EGGPLANT_256': 'Eggplant',
    'LILAC_256': 'Lilac', 'MAPLE_256': 'Maple', 'MARINE_HC': 'Marine',
    'PLUM_HC': 'Plum', 'PUMPKIN_EL_256': 'Pumpkin', 'RAINY_DAY_256': 'Rainy Day',
    'ROSE_256': 'Rose', 'SLATE': 'Slate', 'SPRUCE_256': 'Spruce',
    'STARS_STRIPES_16': 'Red White and Blue', 'STORM_16': 'Storm',
    'TEAL_16': 'Teal', 'WHEAT_256': 'Wheat',
}
CNT = {
    'BRICK': 'Brick', 'DESERT': 'Desert', 'EGGPLANT': 'Eggplant',
    'LILA': 'Lilac', 'MAPLE': 'Maple', 'MARINE_HIGH_COLOR': 'Marine',
    'PLUM_HIGH_COLOR_': 'Plum', 'PUMPKIN_LARGE': 'Pumpkin',
    'RAINY_DAY': 'Rainy Day', 'ROSE': 'Rose', 'SLATE': 'Slate',
    'SPRUCE': 'Spruce', 'RED_WHITE_AND_BLUE_VGA': 'Red White and Blue',
    'STORM_VGA': 'Storm', 'TEAL_VGA': 'Teal', 'WHEAT': 'Wheat',
    'WINDOWS_CLASSIC': 'Windows Classic',

CFG = {
    'win95': dict(inf='win95-schemes/shell2.inf', enc='latin-1', nc=25,
                  prefix='w95', tpre='Win95', curated=C9X, nt=False, fixlen=None),
    'win98': dict(inf='win98-schemes/shell2.inf', enc='latin-1', nc=29,
                  prefix='w98', tpre='Win98', curated=C9X, nt=False, fixlen=None),
    'w2k':   dict(inf='win2000-schemes/HIVEDEF.INF', enc='utf-16', nc=29,
                  prefix='w2k', tpre='W2K', curated=CNT, nt=True, fixlen=712),
}[mode]
KEYS = ['Scrollbar', 'Background', 'ActiveTitle', 'InactiveTitle', 'Menu',
        'Window', 'WindowFrame', 'MenuText', 'WindowText', 'TitleText',
        'ActiveBorder', 'InactiveBorder', 'AppWorkspace', 'Hilight',
        'HilightText', 'ButtonFace', 'ButtonShadow', 'GrayText', 'ButtonText',
        'InactiveTitleText', 'ButtonHilight', 'ButtonDkShadow', 'ButtonLight',
        'InfoText', 'InfoWindow',
        None, 'HotTrackingColor', 'GradientActiveTitle', 'GradientInactiveTitle']
path = os.path.join(nex, CFG['inf'])
if not os.path.exists(path):
    raise SystemExit(f"ERROR: {path} missing — NEXUS source archive not reachable")
# Win98 SE's shell2.inf terminates lines \r\r\n — strip CRs, then collapse the
# doubled newlines so continuation blocks parse as contiguous lines.
text = open(path, encoding=CFG['enc']).read().replace('\r', '')
text = re.sub(r'\n+', '\n', text)
if CFG['nt']:
    pat = re.compile(r'HKCU,"Control Panel\\Appearance\\Schemes","%([A-Z0-9_]+)%",0x00030001,\\\n((?:[ \t]*[0-9a-fA-F, \t]+,?\\?\n)+)')
else:
    pat = re.compile(r'HKCU,"[^"]*",%([A-Z0-9_]+)%,1,\\\n((?:[ \t]*[0-9a-fA-F, \t]+,?\\?\n)+)')
entries = {}
for m in pat.finditer(text):
    hexbytes = re.findall(r'[0-9a-fA-F]{2}', m.group(2))
    entries[m.group(1)] = bytes(int(h, 16) for h in hexbytes)

    if blob is None:

    if CFG['fixlen'] and len(blob) != CFG['fixlen']:
        raise SystemExit(f"FATAL {token}: blob {len(blob)} bytes, expected {CFG['fixlen']}")
    if len(blob) < nc * 4:

    slug = name.lower().replace(' ', '-').replace("'", '')
    p = os.path.join(out, slug + '.theme')
    with open(p, 'w') as f:
        f.write('[Control Panel\\Colors]\n')
        for k, (r, g, b) in zip(KEYS[:nc], colors):

                f.write(f'{k}={r} {g} {b}\n')
    made.append((p, f"{CFG['tpre']} {name}", f"{CFG['prefix']}-{slug}"))
with open(os.path.join(out, 'manifest.tsv'), 'w') as f:
    for p, tn, sid in made:
        f.write(f'{p}\t{tn}\t{sid}\n')
print(f"decoded {len(made)} {mode} schemes")
PYWCL
case "${MODE}" in
  win95) export FLAT_TITLE=1 REDSTONE_ICON_THEME=Chicago95 ;;
  win98) export REDSTONE_ICON_THEME=Win98SE ;;
  w2k)   export REDSTONE_ICON_THEME=SE98 ;;
