  echo "No Termux package manifest at $MANIFEST"

awk -v FS='[ \t]+' '
  function dequote(s){ return (s=="\"\"" ? "" : s) }
  $0 ~ /^[[:space:]]*#/ || NF==0 { next }
  {
    n1=$1; n2=$2; n3=$3; n4=$4;
    printf "%s|%s|%s|%s\n",
      dequote(n1),dequote(n2),dequote(n3),dequote(n4)
  }
' "$MANIFEST" | while IFS='|' read -r pkg _ver _scope _extra; do
  [ -n "$pkg" ] || continue
  if command -v "$pkg" >/dev/null 2>&1; then
    echo "[$pkg] already installed"
  else
    echo ">> [$pkg] installing via pkg"
    if pkg install -y "$pkg"; then
      echo "[$pkg] installed"
    else
      echo "!! [$pkg] install failed"
    fi
  fi
done

echo "Done."

* Piper TTS on the phone (Q1 verdict: NATIVE-ARM64)
The ruled recipe: piper via pip on Termux's own python, with Termux's Bionic =python-onnxruntime= symlinked over the glibc wheel pip drags in. Idempotent; safe to re-run after piper upgrades (re-asserts the symlink).

#+BEGIN_SRC bash :tangle termux/.local/bin/termux-piper-apply :shebang "#!/data/data/com.termux/files/usr/bin/bash"
# ~/DotCortex/termux/.local/bin/termux-piper-apply
set -euo pipefail

echo "== termux piper apply =="

python -c "import onnxruntime" 2>/dev/null || {
  echo "!! Termux python-onnxruntime not importable — run termux-pkg-apply first"
  exit 1
}

# python -m pip, never the bare pip shim: Termux python major upgrades leave
# /usr/bin/pip pointing at the removed interpreter (bad shebang), while the
# module lane always matches the live python (bitten live, 2026-08-16)
#
# piper 1.7 ships no Android wheel — AND the PyPI sdist ships NO C SOURCES
# at all (no CMakeLists.txt, no espeakbridge.c): building from PyPI yields a
# py3-none-any wheel whose synthesis dies on `ImportError: espeakbridge`,
# while sox plays the zero bytes with a clean exit (silent phone, 2026-08-16).
# The only true source is the git repo, which carries the CMake build: it
# vendors espeak-ng statically (ExternalProject clone, needs network + git)
# and compiles espeakbridge on-device into a cp39-abi3 android_arm64 wheel.
# Proven recipe (S24, 2026-08-16): Termux toolchain from pkg (ninja/cmake/
# clang/git, in the manifest), the build backends piper imports (setuptools +
# skbuild), and --no-build-isolation so the build uses them instead of an
# isolated venv that tries to pip-compile ninja and dies.
python -m pip install setuptools wheel scikit-build
python -m pip install --no-cache-dir --no-deps --no-build-isolation --upgrade \
  "piper-tts @ git+https://github.com/OHF-voice/piper1-gpl@v1.7.0"
python -m pip install --no-build-isolation "pathvalidate>=3,<4"

# pip resolves an onnxruntime wheel linked against glibc; it cannot dlopen on
# Bionic. Point pip's install at Termux's own native module instead
# (documented workaround, rhasspy/piper#814).
SITE="$(python -c 'import site; print(site.getsitepackages()[0])')"
NATIVE="$(python - <<'PY'
import onnxruntime, os, sys
d = os.path.dirname(onnxruntime.__file__)
print(d)
PY
)"
PIP_ORT="$SITE/onnxruntime"
if [ -d "$PIP_ORT" ] && [ "$PIP_ORT" != "$NATIVE" ]; then
  rm -rf "$PIP_ORT"
  ln -s "$NATIVE" "$PIP_ORT"
  echo "symlinked Bionic onnxruntime over pip's glibc wheel"
fi

# The artifact is the evidence: a pure-python install passes the import above
# and still cannot speak. Assert the compiled bridge is physically on disk.
python - <<'PY'
import os, piper
d = os.path.dirname(piper.__file__)
so = [f for f in os.listdir(d) if f.startswith("espeakbridge") and f.endswith(".so")]
assert so, f"espeakbridge.so MISSING from {d} — pure-python wheel installed; rebuild from git"
assert os.path.isdir(os.path.join(d, "espeak-ng-data")), "espeak-ng-data missing"
print(f"piper native OK: {so[0]}")
PY
python -c "from piper import PiperVoice; print('piper importable')"
echo "Done. Voices download to ~/.local/share/piper on first use (or fetch manually)."
#+END_SRC

* Speak on the phone (working primitive)
=termux-tts-say= is the phone's speaking primitive: text on stdin or argv → piper → phone speakers through PulseAudio. Voice model path defaults to the fleet-standard Amy but any =.onnx= piper voice works. This is the substrate the generic =tts-synth=/=tts-play= contract will formalize once the HelmCortex identity membrane lands; until then this one script is the honest whole of it.

#+BEGIN_SRC bash :tangle termux/.local/bin/termux-tts-say :shebang "#!/data/data/com.termux/files/usr/bin/bash"
# ~/DotCortex/termux/.local/bin/termux-tts-say
set -euo pipefail

MODEL="${TERMUX_TTS_MODEL:-$HOME/.local/share/piper/en_US-amy-medium.onnx}"
[ -f "$MODEL" ] || {
  echo "termux-tts-say: no voice model at $MODEL — set TERMUX_TTS_MODEL or place a piper .onnx there" >&2
  exit 1
}

# pulseaudio server: start on demand (Termux has no service manager by default)
pulseaudio --check 2>/dev/null || pulseaudio --start --exit-idle-time=300 2>/dev/null || true

if [ $# -gt 0 ]; then TEXT="$*"; else TEXT="$(cat)"; fi
# piper 1.7 CLI: -m + --output-raw (the 1.4-era --model/--output_raw silently
# produced zero bytes while sox exited 0 — the first phone speech was silence
# with a clean exit code, 2026-08-16). pipefail + visible stderr so a dead
# stage can never again masquerade as success.
set -o pipefail
printf '%s' "$TEXT" | python -m piper -m "$MODEL" --output-raw \
  | sox -t raw -r 22050 -e signed -b 16 -c 1 - -d
#+END_SRC

* Kitten TTS in the enclave (on-device consort voices)
The consort voice family (Bella, Luna, Kiki, Jasper, Rosie — the pvox kitten bank) rendered fleet-side only, because kittentts pulls =misaki[en]= which pins Python ≤3.12 and bare Termux/Bionic cannot hold that stack. The enclave can: inside the Debian proot, stock manylinux aarch64 wheels (onnxruntime included) just work. This chapter, ruled 2026-08-17 ("my phone has more horsepower for rendering on device than the rest of the fleet" — and Q8 agrees: kitten ~0.4–0.9 RTF on the S24), installs a uv-managed Python 3.12 venv with the SAME kittentts 0.8.1 wheel the HelmCortex conda env pins, plus a one-shot synth script. The Termux-side speaker below splits the lanes on-device: render in the enclave (glibc), play through Termux's pulseaudio (sox) — same WAV-with-header law as the fleet.

#+BEGIN_SRC bash :tangle termux/.local/bin/termux-kitten-apply :shebang "#!/data/data/com.termux/files/usr/bin/bash"
# ~/DotCortex/termux/.local/bin/termux-kitten-apply
set -euo pipefail

echo "== termux kitten apply (enclave) =="
command -v proot-distro >/dev/null || { echo "!! proot-distro missing — run termux-pkg-apply"; exit 1; }

proot-distro login debian -- bash -lc '
  set -e
  export PATH="$HOME/.local/bin:$PATH"
  command -v uv >/dev/null || { curl -LsSf https://astral.sh/uv/install.sh | sh; }
  export PATH="$HOME/.local/bin:$PATH"
  uv python install 3.12
  [ -x "$HOME/kitten-venv/bin/python" ] || uv venv --python 3.12 "$HOME/kitten-venv"
  uv pip install --python "$HOME/kitten-venv/bin/python" \
    "kittentts @ https://github.com/KittenML/KittenTTS/releases/download/0.8.1/kittentts-0.8.1-py3-none-any.whl"

  # one-shot synth: text on stdin, headered WAV on stdout (never raw PCM)
  cat > "$HOME/kitten-say.py" <<"PY"
import argparse, io, sys, wave
import numpy as np

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--voice", default="Bella")
    ap.add_argument("--speed", type=float, default=1.0)
    args = ap.parse_args()
    text = sys.stdin.read().strip()
    if not text:
        return
    from kittentts import KittenTTS
    m = KittenTTS("KittenML/kitten-tts-nano-0.8-int8")
    try:
        audio = m.generate(text, voice=args.voice, speed=args.speed)
    except TypeError:
        audio = m.generate(text, voice=args.voice)
    pcm = np.clip(audio, -1.0, 1.0)
    pcm16 = (pcm * 32767).astype("<i2").tobytes()
    out = io.BytesIO()
    with wave.open(out, "wb") as w:
        w.setnchannels(1); w.setsampwidth(2); w.setframerate(24000)
        w.writeframes(pcm16)
    sys.stdout.buffer.write(out.getvalue())

if __name__ == "__main__":
    main()
PY
  "$HOME/kitten-venv/bin/python" -c "import kittentts; print(\"kittentts importable\")"
'
echo "Done. Model downloads from HuggingFace on first synth. Test: echo hello | termux-kitten-say"
#+END_SRC

The speaker. Chunking, cleanup, and voice/speed policy stay the CALLER's business (the fleet hook resolves agent → voice/speed from the pvox config — one source of truth; this script is dumb machinery). First synth downloads the ~25MB model into the enclave's HuggingFace cache.

#+BEGIN_SRC bash :tangle termux/.local/bin/termux-kitten-say :shebang "#!/data/data/com.termux/files/usr/bin/bash"
# ~/DotCortex/termux/.local/bin/termux-kitten-say
# text (stdin or argv) -> enclave kitten render -> phone speakers
set -euo pipefail

VOICE="${KITTEN_VOICE:-Bella}"
SPEED="${KITTEN_SPEED:-1.0}"
while [ $# -gt 0 ]; do
  case "$1" in
    --voice) VOICE="$2"; shift 2 ;;
    --speed) SPEED="$2"; shift 2 ;;
    --wav-out) WAV_OUT="$2"; shift 2 ;;
    -h|--help) echo "usage: termux-kitten-say [--voice V] [--speed S] [--wav-out F] [text]"; exit 0 ;;
    *) break ;;
  esac
done

if [ $# -gt 0 ]; then TEXT="$*"; else TEXT="$(cat)"; fi
[ -n "$TEXT" ] || exit 0

W="${WAV_OUT:-$(mktemp --suffix=.wav)}"
printf '%s' "$TEXT" | proot-distro login debian -- bash -lc \
  "\$HOME/kitten-venv/bin/python \$HOME/kitten-say.py --voice '$VOICE' --speed '$SPEED'" > "$W"
[ -s "$W" ] || { echo "termux-kitten-say: empty render" >&2; rm -f "$W"; exit 1; }

if [ -n "${WAV_OUT:-}" ]; then
  echo "$W"
else
  pulseaudio --check 2>/dev/null || pulseaudio --start --exit-idle-time=300 2>/dev/null || true
  set -o pipefail
  sox -q "$W" -d
  rm -f "$W"
fi
#+END_SRC

* Android kill-vector check (Q5 requirement)
Android will kill long-running relays unless defused: the phantom process killer, battery optimization, and Doze. The battery exemption stays a Settings-app toggle (Unrestricted for Termux, Termux:Boot, AND Shizuku — a battery-frozen Shizuku binder times out every rish call, bitten live 2026-08-17). The phantom killer, though, IS now scriptable from unrooted Termux through the rish lane below — the original "needs adb from a PC" claim is retired. This check REPORTS the state honestly (through rish when available) so a silent future kill is never a mystery.

#+BEGIN_SRC bash :tangle termux/.local/bin/termux-defuse-check :shebang "#!/data/data/com.termux/files/usr/bin/bash"
# ~/DotCortex/termux/.local/bin/termux-defuse-check
set -euo pipefail

echo "== android kill-vector check =="
if command -v rish >/dev/null 2>&1 && rish -c id 2>/dev/null | grep -q uid=; then
  v="$(rish -c 'settings get global settings_enable_monitor_phantom_procs' 2>/dev/null || echo unreadable)"
  p="$(rish -c 'device_config get activity_manager max_phantom_processes' 2>/dev/null || echo unreadable)"
  echo "phantom process killer: $v (want: false)"
  echo "max_phantom_processes: $p (want: 2147483647)"
  [ "$v" = "false" ] || echo "!! run termux-defuse-apply"
elif command -v settings >/dev/null 2>&1; then
  v="$(settings get global settings_enable_monitor_phantom_procs 2>/dev/null || echo unreadable)"
  echo "phantom process killer: $v (want: false — set via termux-defuse-apply once rish is wired)"
else
  echo "phantom process killer: unreadable — rish not wired (run termux-rish-apply) and no adb lane"
fi
if command -v termux-battery-status >/dev/null 2>&1; then
  termux-battery-status 2>/dev/null | head -5 || true
fi
echo "battery optimization: Unrestricted for Termux, Termux:Boot, Shizuku (Settings > Apps > ... > Battery)"
echo "wake lock: termux-wake-lock while a relay runs; termux-wake-unlock to release"
#+END_SRC

* Shizuku rish lane + phantom-killer defusal
Shizuku (started by Mètsàtron at every boot, alongside the tailnet) exposes ADB-shell power to authorized apps; =rish= is its shell client. Prerequisites, each bitten live 2026-08-17: the Shizuku SERVICE must be running (the app's front card says so), Shizuku must be battery-Unrestricted (else the binder freezes and every rish call times out), and Termux must be authorized in Shizuku (the approval dialog fires on first use). The rish files come from the Shizuku app's "Use Shizuku in terminal apps" export to shared storage; the apply below installs them properly: dex beside the script in app-private storage with the write bit removed (Android 14+ refuses writable dex — and shared storage cannot drop the write bit, which is why the export must be COPIED in, never run in place), =PKG= patched to =com.termux=, and a PATH wrapper that execs the real script so =dirname $0= still finds the dex.

* Service supervision (runit, Termux-side)
The night handoff's open item: runit is installed but nothing starts it, so sshd/pulseaudio/herdr come up by hand and die with the session. The wiring, ruled by the Q5 verdict (supervision stays Termux-side — proot has no init): =termux-services= provides the svlogger the runit package's shipped service dirs already symlink to (broken until installed) plus =sv-enable= tooling; the apply script below asserts service dirs for the phone's standing services directly in =$PREFIX/var/service= (matching where the runit package put sshd's); and a Termux:Boot script holds a wake lock and starts =runsvdir= at device boot. Boot-transition note (2026-08-17): a manually-started sshd or daemonized pulseaudio conflicts with the supervised instance on the same port/socket — the supervised one flaps until the manual one exits. This resolves itself at the first reboot after wiring; do not run the apply while depending on the manual sshd without knowing that.

Run scripts export HOME/PATH explicitly: runsvdir inherits whatever environment started it, and Termux:Boot's is minimal — an implicit-env service that works from an interactive test then dies at boot is exactly the silent failure this section exists to end.

#+BEGIN_SRC bash :tangle termux/.local/bin/termux-services-apply :shebang "#!/data/data/com.termux/files/usr/bin/bash"
# ~/DotCortex/termux/.local/bin/termux-services-apply
set -euo pipefail

echo "== termux services apply =="
SV="$PREFIX/var/service"
LOGGER="$PREFIX/share/termux-services/svlogger"
[ -e "$LOGGER" ] || echo "!! svlogger missing — termux-services not installed yet (run termux-pkg-apply); log/run symlinks will heal once it lands"

TERMUX_HOME=/data/data/com.termux/files/home
TERMUX_BIN=/data/data/com.termux/files/usr/bin

mksv() {
  local name="$1"; shift
  mkdir -p "$SV/$name/log"
  printf '%s\n' "#!$TERMUX_BIN/sh" \
    "export HOME=$TERMUX_HOME" \
    "export PATH=$TERMUX_BIN:$TERMUX_HOME/.local/bin" \
    "$@" > "$SV/$name/run"
  chmod +x "$SV/$name/run"
  ln -sf "$LOGGER" "$SV/$name/log/run"
  echo "[$name] service dir asserted"
}

mksv pulseaudio 'exec pulseaudio --daemonize=no --exit-idle-time=-1 2>&1'
mksv herdr "exec $TERMUX_HOME/.local/bin/herdr server 2>&1"

# sshd's dir ships with the packages; enable it (runit skips down'd services)
rm -f "$SV/sshd/down"
echo "service dirs: $(ls "$SV" | tr '\n' ' ')"
pgrep -x runsvdir >/dev/null && echo "runsvdir: running" || echo "runsvdir: NOT running (starts at boot via Termux:Boot, or start now: setsid runsvdir \$PREFIX/var/service &)"
#+END_SRC

Termux:Boot (the =com.termux.boot= app, installed 2026-08-16) runs every script in =~/.termux/boot/= at device boot once Termux has been opened after install. This is the reboot-survival half of the defusal story: wake lock first, then supervision.

#+BEGIN_SRC sh :tangle termux/.termux/boot/00-services :shebang "#!/data/data/com.termux/files/usr/bin/sh"
