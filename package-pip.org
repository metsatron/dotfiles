set -euo pipefail

touch "$MAN"

tmp_have="$(mktemp)"
tmp_new="$(mktemp)"

awk '!/^[[:space:]]*#/ && NF { print $1 }' "$MAN" | sort -u >"$tmp_existing"

# Currently installed (user + system, skip pip/setuptools/wheel)

  | awk -F== '{ print $1 }' \
  | grep -vEi '^(pip|setuptools|wheel|pkg.resources)$' \
  | sort -u >"$tmp_have"

# new = have - existing
comm -13 "$tmp_existing" "$tmp_have" >"$tmp_new"

rm -f "$tmp_existing" "$tmp_have"

if [ ! -s "$tmp_new" ]; then
  rm -f "$tmp_new"
  echo "No new pip packages to capture into $MAN"
  exit 0
fi

while read -r pkg; do
  [ -n "$pkg" ] || continue
  printf '%s "" ""\n' "$pkg" >>"$MAN"
done <"$tmp_new"

rm -f "$tmp_new"
echo "Updated $MAN"

#+BEGIN_SRC bash :tangle all/.local/bin/pip-apply :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/pip-apply
set -euo pipefail

MAN="$HOME/DotCortex/all/.pip/manifest/packages.ssv"
failed=()

echo "== pip apply =="

# Bootstrap Guix pip if broken (24.x has resolvelib bug)
GUIX_PYTHON="$HOME/.guix-extra-profiles/core/core/bin/python3"
if [ -x "$GUIX_PYTHON" ]; then
  GUIX_PIP_VER=$("$GUIX_PYTHON" -m pip --version 2>/dev/null | awk '{print $2}' || echo "0")
  GUIX_PIP_MAJOR=$(echo "$GUIX_PIP_VER" | cut -d. -f1)
  if [ "${GUIX_PIP_MAJOR:-0}" -lt 25 ] 2>/dev/null; then
    echo "  bootstrapping pip (Guix pip $GUIX_PIP_VER is broken)..."
    curl -sS https://bootstrap.pypa.io/get-pip.py | "$GUIX_PYTHON" - --user \
      && echo "  pip bootstrapped to 26+" \
      || echo "  WARNING: get-pip.py failed"
  fi
fi

while IFS= read -r line; do
  # Skip comments and blanks
  [[ "$line" =~ ^[[:space:]]*# ]] && continue
  [[ -z "${line// /}" ]] && continue

  pkg="$(echo "$line" | awk '{ print $1 }')"
  version="$(echo "$line" | awk '{ print $2 }' | tr -d '"')"
  # Third field is the SOURCE. It was parsed by nothing until 2026-07-13, so a
  # manifest line like `antigravity_decryptor "" "git+https://…"` was installed
  # as the bare name from PyPI, where it does not exist — the declared git
  # source had never once been honoured, and the package failed on every fresh
  # machine while looking "ok" on any host that already had it.
  source="$(echo "$line" | awk '{ print $3 }' | tr -d '"')"

  [ -z "$pkg" ] && continue

  if [ -n "$source" ]; then
    spec="$source"
  else
    spec="$pkg"
    [ -n "$version" ] && spec="$pkg==$version"
  fi

  if python3 -m pip show "$pkg" &>/dev/null; then
    echo "  ok: $pkg"
  else
    echo "  installing: $spec"
    # One unbuildable package must not abort the rest of the manifest.
    if python3 -m pip install --user --break-system-packages "$spec"; then
      echo "  installed: $pkg"
    else
      echo "  FAILED: $pkg ($spec)"
      failed+=("$pkg")
    fi
  fi
done <"$MAN"

if [ "${#failed[@]}" -gt 0 ]; then
  echo "FAILED (${#failed[@]}): ${failed[*]}"
  exit 1
fi

echo "Done."

* Nanobot Telegram runtime (Linux, uv-managed)

Nanobot is a fleet bot, not a user-site Python library.  Keep its dependency
graph isolated inside the HelmCortex Nano workspace and pin the upstream release
here.  Re-running =nanobot-apply= upgrades or repairs the runtime to the declared
version without touching private credentials, OAuth state, sessions, or memory.
The public non-secret template is recursively merged into an existing private
config on every apply, with the template winning only for managed keys. This
keeps access policy, MCP servers, model defaults, and gateway settings
reproducible while preserving pairing state, WebUI secrets, OAuth metadata, and
Nanobot-added defaults that the template does not own.

The managed seam below keeps Nanobot's native locking and consolidation path,
but replaces its fixed idle threshold with the fleet's measured endpoint/model
policy. Upstream auto-compaction remains explicitly disabled in config; the
scanner remains disabled whenever policy is missing, stale, unsupported, or
unreadable. Nano explicitly accepts either the fully supported tier or a
repeatable observed endpoint/model transition; weaker evidence cannot arm it.

"""Keep filesystem search from blocking Nanobot's gateway event loop."""

from __future__ import annotations

import asyncio
import multiprocessing
import os
from pathlib import Path
import signal
import time
from typing import Any

from nanobot.agent.tools.base import ToolResult
from nanobot.security.workspace_access import (
    bind_workspace_scope,
    current_workspace_scope,
    reset_workspace_scope,
)

_MAX_CHILDREN = 4
_TIMEOUT_SECONDS = 60.0


def _child(send: Any, tool: Any, params: dict[str, Any], scope: Any) -> None:
    if os.name == "posix":
        os.setsid()
        try:
            Path("/proc/self/comm").write_text("nanobot-search\n", encoding="utf-8")
        except OSError:
            pass
    token = bind_workspace_scope(scope) if scope is not None else None
    try:
        result = asyncio.run(tool.execute(**params))
        send.send((bool(getattr(result, "is_error", False)), str(result)))
    except BaseException as exc:
        send.send((True, f"Error: {type(exc).__name__}: {exc}"))
    finally:
        if token is not None:
            reset_workspace_scope(token)
        send.close()


async def isolated_search(tool: Any, params: dict[str, Any], name: str,
                          timeout: float = _TIMEOUT_SECONDS) -> str:
    """Run native search in a spawned process; return a bounded tool result."""
    if name not in {"grep", "find_files"}:
        return ToolResult.error(f"Error: unsupported isolated tool {name}")
    active = [child for child in multiprocessing.active_children()
              if child.name == "nanobot-search"]
    if len(active) >= _MAX_CHILDREN:
        return ToolResult.error("Error: search workers are still blocked; retry later")
    ctx = multiprocessing.get_context("spawn")
    receive, send = ctx.Pipe(duplex=False)
    child = ctx.Process(target=_child, args=(send, tool, params,
                                             current_workspace_scope()),
                        name="nanobot-search", daemon=True)
    started = False
    try:
        child.start()
        started = True
        send.close()
        deadline = time.monotonic() + timeout
        while time.monotonic() < deadline:
            if receive.poll(0):
                try:
                    is_error, result = receive.recv()
                except EOFError:
                    return ToolResult.error(f"Error: {name} search worker exited without a result")
                return ToolResult.error(result) if is_error else result
            if not child.is_alive():
                return ToolResult.error(f"Error: {name} search worker exited without a result")
            await asyncio.sleep(0.05)
        return ToolResult.error(
            f"Error: {name} search worker timed out after {timeout:g} seconds; "
            "narrow the path and retry"
        )
    except asyncio.CancelledError:
        raise
    except (OSError, TypeError, ValueError) as exc:
        return ToolResult.error(f"Error: cannot start {name} search worker: {exc}")
    finally:
        receive.close()
        send.close()
        if started:
            child.join(timeout=0.1)
            if child.is_alive():
                try:
                    if os.name == "posix" and os.getpgid(child.pid) == child.pid:
                        os.killpg(child.pid, signal.SIGKILL)
                    else:
                        child.kill()
                except (OSError, ProcessLookupError):
                    pass
                child.join(timeout=0.1)

kittentts pulls =misaki[en]=, which pins Python ≤3.12 while the fleet's bare python3 is 3.13 — so the pvox kitten engine cannot join piper/kokoro in the Lane 1 local-pip elevation (see the exclusion note in the pip manifest above). On an NFS client the old fallback was the Lane 2 conda env over NFS, and off-LAN (a travelling T480s over a DERP-relayed tailnet) that stalls: the Super+V clipboard reader fires its notification but the kitten synth subprocess hangs reaching =~/HelmCortex/CORE/anaconda3= across the wire, so nothing speaks.

This is the glibc sibling of the S24 enclave (=termux-kitten-apply=, sealed 2026-08-17, "my phone has more horsepower for rendering on device than the rest of the fleet"): a uv-managed Python 3.12 venv at =~/kitten-venv= holding the SAME kittentts 0.8.1 wheel and =misaki[en]==0.9.4= the HelmCortex conda env pins, with torch stripped. It also carries =piper-tts= + =kokoro-onnx= so it is a full THREE-engine interpreter — because kittentts pins Python ≤3.12, it can never live in the fleet's bare python3 (3.13), so the local warm daemon must run under this venv or not at all. pvox prefers it two ways: =kitten_python()= uses it for the cold subprocess path, and =daemon_python_candidates()= prepends it so the resident voice daemon runs under it and holds ALL THREE engines warm on-device (the three-engine seat) — instead of the bare-python3 daemon refusing kitten and cold-subprocessing it (~6-8s) every call. Absent on the hub and un-provisioned clients, so pvox falls back to conda / bare-python3 there unchanged.
