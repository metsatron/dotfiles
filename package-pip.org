* Capture current pip → manifest
#+BEGIN_SRC bash :tangle all/.local/bin/pip-capture :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/pip-capture
set -euo pipefail

MAN="$HOME/DotCortex/all/.pip/manifest/packages.ssv"

mkdir -p "$(dirname "$MAN")"
touch "$MAN"

tmp_existing="$(mktemp)"
tmp_have="$(mktemp)"
tmp_new="$(mktemp)"

# Already tracked
awk '!/^[[:space:]]*#/ && NF { print $1 }' "$MAN" | sort -u >"$tmp_existing"

# Currently installed (user + system, skip pip/setuptools/wheel)
python3 -m pip list --format=freeze 2>/dev/null \
  | awk -F== '{ print $1 }' \
  | grep -vEi '^(pip|setuptools|wheel|pkg.resources)$' \
  | sort -u >"$tmp_have"

# new = have - existing
comm -13 "$tmp_existing" "$tmp_have" >"$tmp_new"

rm -f "$tmp_existing" "$tmp_have"

if [ ! -s "$tmp_new" ]; then
  rm -f "$tmp_new"
  echo "No new pip packages to capture into $MAN"
  exit 0
fi

while read -r pkg; do
  [ -n "$pkg" ] || continue
  printf '%s "" ""\n' "$pkg" >>"$MAN"
done <"$tmp_new"

rm -f "$tmp_new"
echo "Updated $MAN"
#+END_SRC

  | awk -F== '{ print $1 }' \
  | sort -u >"$tmp_have"

echo "== pip plan =="
echo "[to install]"
comm -23 "$tmp_want" "$tmp_have" | sed 's/^/  + /'

echo "[extra - not in manifest]"
comm -13 "$tmp_want" "$tmp_have" \
  | grep -vEi '^(pip|setuptools|wheel|pkg.resources)$' \
  | sed 's/^/  - /'

rm -f "$tmp_want" "$tmp_have"

* Apply manifest
#+BEGIN_SRC bash :tangle all/.local/bin/pip-apply :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/pip-apply
set -euo pipefail

MAN="$HOME/DotCortex/all/.pip/manifest/packages.ssv"
failed=()

echo "== pip apply =="

# Bootstrap Guix pip if broken (24.x has resolvelib bug)
GUIX_PYTHON="$HOME/.guix-extra-profiles/core/core/bin/python3"
if [ -x "$GUIX_PYTHON" ]; then
  GUIX_PIP_VER=$("$GUIX_PYTHON" -m pip --version 2>/dev/null | awk '{print $2}' || echo "0")
  GUIX_PIP_MAJOR=$(echo "$GUIX_PIP_VER" | cut -d. -f1)
  if [ "${GUIX_PIP_MAJOR:-0}" -lt 25 ] 2>/dev/null; then
    echo "  bootstrapping pip (Guix pip $GUIX_PIP_VER is broken)..."
    curl -sS https://bootstrap.pypa.io/get-pip.py | "$GUIX_PYTHON" - --user \
      && echo "  pip bootstrapped to 26+" \
      || echo "  WARNING: get-pip.py failed"
  fi
fi

while IFS= read -r line; do
  # Skip comments and blanks
  [[ "$line" =~ ^[[:space:]]*# ]] && continue
  [[ -z "${line// /}" ]] && continue

  pkg="$(echo "$line" | awk '{ print $1 }')"
  version="$(echo "$line" | awk '{ print $2 }' | tr -d '"')"
  # Third field is the SOURCE. It was parsed by nothing until 2026-07-13, so a
  # manifest line like `antigravity_decryptor "" "git+https://…"` was installed
  # as the bare name from PyPI, where it does not exist — the declared git
  # source had never once been honoured, and the package failed on every fresh
  # machine while looking "ok" on any host that already had it.
  source="$(echo "$line" | awk '{ print $3 }' | tr -d '"')"

  [ -z "$pkg" ] && continue

  if [ -n "$source" ]; then
    spec="$source"
  else
    spec="$pkg"
    [ -n "$version" ] && spec="$pkg==$version"
  fi

  if python3 -m pip show "$pkg" &>/dev/null; then
    echo "  ok: $pkg"
  else
    echo "  installing: $spec"
    # One unbuildable package must not abort the rest of the manifest.
    if python3 -m pip install --user --break-system-packages "$spec"; then
      echo "  installed: $pkg"
    else
      echo "  FAILED: $pkg ($spec)"
      failed+=("$pkg")
    fi
  fi
done <"$MAN"

if [ "${#failed[@]}" -gt 0 ]; then
  echo "FAILED (${#failed[@]}): ${failed[*]}"
  exit 1
fi

echo "Done."
#+END_SRC

* Apply pipx manifest
#+BEGIN_SRC bash :tangle all/.local/bin/pipx-apply :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/pipx-apply
set -euo pipefail

MAN="$HOME/DotCortex/all/.pipx/manifest/packages.ssv"

echo "== pipx apply =="

if ! command -v pipx &>/dev/null; then
  echo "  ERROR: pipx not found — install via nala:apply (pipx is in the nala manifest)"
  exit 1
fi

pipx ensurepath &>/dev/null || true

while IFS= read -r line; do
  [[ "$line" =~ ^[[:space:]]*# ]] && continue
  [[ -z "${line// /}" ]] && continue

  pkg="$(echo "$line" | awk '{ print $1 }')"
  version="$(echo "$line" | awk '{ print $2 }' | tr -d '"')"

  [ -z "$pkg" ] && continue

  spec="$pkg"
  [ -n "$version" ] && spec="$pkg==$version"

  if pipx list --short 2>/dev/null | awk '{print $1}' | grep -qx "$pkg"; then
    echo "  ok: $pkg"
  else
    echo "  installing: $spec"
    pipx install "$spec"
  fi
done <"$MAN"

echo "Done."
#+END_SRC

* Local kitten voice venv (Linux clients, uv-managed)

kittentts pulls =misaki[en]=, which pins Python ≤3.12 while the fleet's bare python3 is 3.13 — so the pvox kitten engine cannot join piper/kokoro in the Lane 1 local-pip elevation (see the exclusion note in the pip manifest above). On an NFS client the old fallback was the Lane 2 conda env over NFS, and off-LAN (a travelling T480s over a DERP-relayed tailnet) that stalls: the Super+V clipboard reader fires its notification but the kitten synth subprocess hangs reaching =~/HelmCortex/CORE/anaconda3= across the wire, so nothing speaks.

This is the glibc sibling of the S24 enclave (=termux-kitten-apply=, sealed 2026-08-17, "my phone has more horsepower for rendering on device than the rest of the fleet"): a uv-managed Python 3.12 venv at =~/kitten-venv= holding the SAME kittentts 0.8.1 wheel and =misaki[en]==0.9.4= the HelmCortex conda env pins, with torch stripped. It also carries =piper-tts= + =kokoro-onnx= so it is a full THREE-engine interpreter — because kittentts pins Python ≤3.12, it can never live in the fleet's bare python3 (3.13), so the local warm daemon must run under this venv or not at all. pvox prefers it two ways: =kitten_python()= uses it for the cold subprocess path, and =daemon_python_candidates()= prepends it so the resident voice daemon runs under it and holds ALL THREE engines warm on-device (the three-engine seat) — instead of the bare-python3 daemon refusing kitten and cold-subprocessing it (~6-8s) every call. Absent on the hub and un-provisioned clients, so pvox falls back to conda / bare-python3 there unchanged.

Linux-scoped (Guix/Devuan fleet), not =all=: Termux has its own enclave lane, and the hub reaches conda locally. Run =kitten-apply= once per client after stow; the ~25MB kitten model downloads from HuggingFace into the local cache on the first pvox kitten synth.

#+BEGIN_SRC bash :tangle linux/.local/bin/kitten-apply :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
# ~/DotCortex/linux/.local/bin/kitten-apply
set -euo pipefail

echo "== kitten venv apply (uv, ~/kitten-venv) =="
export PATH="$HOME/.local/bin:$PATH"
command -v uv >/dev/null || { echo "!! uv missing — apply via pip-apply (uv is in the pip manifest)"; exit 1; }

uv python install 3.12
[ -x "$HOME/kitten-venv/bin/python" ] || uv venv --python 3.12 "$HOME/kitten-venv"

# misaki pinned to the fleet-proven 0.9.4 (NO torch): unpinned, uv resolves a
# newer misaki whose chain drags torch + nvidia wheels onto the box (bitten live
# on the S24 2026-08-17, 107-package resolution for a 25MB model). The conda env
# runs kitten with misaki 0.9.4 and no torch — mirror that truth.
# --torch-backend cpu: misaki's curated chain hard-requires torch, and on x86_64 the
# default torch wheel drags the entire ~4.8GB nvidia CUDA stack (18 wheels) that CPU-only
# onnx synthesis never touches — on a slow link that download blew the provision timeout
# (t480, 2026-09-01). The CPU backend fetches torch==X+cpu (~190MB) and ZERO nvidia wheels
# (verified via `uv pip install --dry-run` 2026-09-01: 18 cuda/nvidia pkgs -> 0). torch is
# still stripped below (the kitten synth path never imports it); this just stops us paying
# 4.6GB of CUDA to throw it away. (aarch64/Termux gets CPU wheels natively — its enclave
# recipe in package-termux.org is separate and unaffected.)
uv pip install --python "$HOME/kitten-venv/bin/python" --torch-backend cpu \
  "misaki[en]==0.9.4" \
  "kittentts @ https://github.com/KittenML/KittenTTS/releases/download/0.8.1/kittentts-0.8.1-py3-none-any.whl" \
  piper-tts kokoro-onnx

# The curated branch (misaki[en] -> spacy-curated-transformers -> curated-transformers)
# DECLARES torch, but the kitten synth path never imports it — the fleet conda env runs
# exactly this shape: curated pair present, torch absent. Strip the dead weight.
uv pip uninstall --python "$HOME/kitten-venv/bin/python" \
  torch triton $(uv pip list --python "$HOME/kitten-venv/bin/python" 2>/dev/null | awk '/^nvidia-/ {print $1}') 2>/dev/null || true
uv cache prune >/dev/null 2>&1 || true

# torch in this venv is always wrong (fleet parity: kitten runs without it) — its
# presence means resolver drift or a polluted venv; fail loud instead of shipping half.
"$HOME/kitten-venv/bin/python" -c '
import importlib.util, sys
if importlib.util.find_spec("torch"):
    sys.exit("!! torch present in kitten-venv — polluted or drifted; rm -rf ~/kitten-venv and re-run")
'
"$HOME/kitten-venv/bin/python" -c 'import kittentts, piper, kokoro_onnx; print("three-engine venv OK: kittentts + piper + kokoro_onnx importable")'

# Localize the kokoro model + voices to local disk too (hot-path-local, sibling of
# the fonts) so off-LAN kokoro never pulls its ~340MB over the tailnet. Delegated to
# pvox, which owns the kokoro asset paths and no-ops when the source is already local;
# best-effort so a missing pvox / unmounted HelmCortex never fails the venv setup.
if command -v pvox >/dev/null 2>&1; then
  pvox assets-localize || echo "kitten-apply: pvox assets-localize failed (non-fatal); run it manually once HelmCortex is reachable" >&2
else
  echo "kitten-apply: pvox not on PATH — run 'pvox assets-localize' after HelmCortex is mounted" >&2
fi

echo "Done. pvox prefers this venv as the daemon's three-engine warm seat; kitten model downloads from HuggingFace on first synth. Test: pclip Default"
#+END_SRC
