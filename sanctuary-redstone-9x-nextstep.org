def extend_gnustep_glyph(kind: str, glyph: tuple[str, ...],
                         pressed: bool) -> tuple[str, ...]:
    """Return the lattice-faithful pass-9b mask in the 15x15 canvas."""
    del glyph  # The filed normal mask is the centering reference below.
    compact = GNUSTEP_LATTICE_GLYPHS[kind]
    compact_points = {(x, y) for y, line in enumerate(compact)
                      for x, mark in enumerate(line) if mark == "#"}
    donor = GNUSTEP_DONOR_GLYPHS[kind]["normal"]
    donor_points = {(x, y) for y, line in enumerate(donor)
                    for x, mark in enumerate(line) if mark == "#"}
    donor_center = (
        (min(x for x, y in donor_points) + max(x for x, y in donor_points)) / 2,
        (min(y for x, y in donor_points) + max(y for x, y in donor_points)) / 2,
    )
    compact_center = (
        (min(x for x, y in compact_points) + max(x for x, y in compact_points)) / 2,
        (min(y for x, y in compact_points) + max(y for x, y in compact_points)) / 2,
    )
    # Nearest integer preserves the filed parity when an enlarged mask has a
    # half-pixel center; ties intentionally resolve down, as the donor does.
    anchor = tuple(
        int(value // 1) if value >= 0 else int(-((-value) // 1))
        for value in (donor_center[0] - compact_center[0],
                      donor_center[1] - compact_center[1])
    )
    # Per-glyph optical nudge, ruled by eye on the live bar: maximize sits one
    # pixel high relative to minimize after the pass-9b extrusion.
    nudge = GNUSTEP_GLYPH_NUDGE.get(kind, (0, 0))
    points = {(x + anchor[0] + nudge[0], y + anchor[1] + nudge[1])
              for x, y in compact_points}
    if pressed and kind != "rolldown":
        points = {(x + 1, y + 1) for x, y in points}
    if any(x < 0 or x >= 15 or y < 0 or y >= 15 for x, y in points):
        raise SystemExit(f"pass-9b lattice extrusion escaped 15x15 canvas: {kind}")
    return tuple("".join("#" if (x, y) in points else "."
                          for x in range(15)) for y in range(15))

class Xpm:
    def __init__(self, width: int, height: int, colours: dict[str, str | None],
                 pixels: list[str]) -> None:
        self.width = width
        self.height = height
        self.colours = colours
        self.pixels = pixels


def quoted(line: str) -> str:
    start = line.find('"')
    end = line.rfind('"')
    if start < 0 or end <= start:
        raise ValueError(f"not an XPM string: {line!r}")
    return line[start + 1:end]


def read_xpm(path: Path) -> Xpm:
    lines = path.read_text(encoding="latin-1").splitlines()
    header_index = next(
        (index for index, line in enumerate(lines) if XPM_HEADER.match(line)),
        None,
    )
    if header_index is None:
        raise ValueError(f"XPM header missing: {path}")
    match = XPM_HEADER.match(lines[header_index])
    assert match is not None
    width, height, colour_count, cpp = map(int, match.groups())
    if cpp != 1:
        raise ValueError(f"unsupported XPM chars-per-pixel in {path}: {cpp}")
    colours: dict[str, str | None] = {}
    for line in lines[header_index + 1:header_index + 1 + colour_count]:
        value = quoted(line)
        symbol = value[:cpp]
        colour_match = COLOUR_LINE.search(value[cpp:])
        if colour_match is None:
            raise ValueError(f"XPM colour missing in {path}: {line}")
        colour = colour_match.group(1).split()[0]
        if colour.lower() == "none":
            colours[symbol] = None
        elif HEX.fullmatch(colour):
            colours[symbol] = colour.upper()
        else:
            raise ValueError(f"unsupported XPM colour in {path}: {colour}")
    pixel_start = header_index + 1 + colour_count
    pixels = [quoted(line) for line in lines[pixel_start:pixel_start + height]]
    if len(pixels) != height or any(len(row) != width for row in pixels):
        raise ValueError(f"incomplete XPM pixels: {path}")
    if any(symbol not in colours for row in pixels for symbol in row):
        raise ValueError(f"XPM pixel refers to an undeclared colour: {path}")
    return Xpm(width, height, colours, pixels)


def write_xpm(path: Path, colours: list[str | None], pixels: list[list[str | None]]) -> None:
    unique = list(dict.fromkeys(colours))
    symbols = ".+@#$%&*=-;:!?ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789"
    if None in unique:
        symbols = " " + symbols
    if len(unique) > len(symbols):
        raise ValueError(f"too many output colours for {path}: {len(unique)}")
    symbol_for = {colour: symbols[index] for index, colour in enumerate(unique)}
    width = len(pixels[0])
    height = len(pixels)
    lines = [
        "/* XPM */",
        f"static char *{path.stem.replace('-', '_')}_xpm[] = {{",
        f'"{width} {height} {len(unique)} 1",',
    ]
    for colour in unique:
        symbol = symbol_for[colour]
        encoded = "None" if colour is None else colour
        lines.append(f'"{symbol} c {encoded}",')
    for row in pixels[:-1]:
        lines.append('"' + "".join(symbol_for[colour] for colour in row) + '",')
    lines.append('"' + "".join(symbol_for[colour] for colour in pixels[-1]) + '"};')
    path.write_text("\n".join(lines) + "\n", encoding="latin-1")


def rgb_luma(colour: str) -> int:
    red, green, blue = (int(colour[index:index + 2], 16) for index in (1, 3, 5))
    return (299 * red + 587 * green + 114 * blue) // 1000


def ice_colour(colour: str) -> str:
    return f"rgb:{colour[1:3]}/{colour[3:5]}/{colour[5:7]}"


def dark_chrome(colour: str) -> str:
    # 2026-08-11 live verdict: a literal black outline was rejected.  The
    # single outer contour is dark chrome of the skew's face family: 50%
    # darker_color mixed toward black, never a fixed #000000 frame band.
    channels = [int(colour[index:index + 2], 16) // 2 for index in (1, 3, 5)]
    return "#" + "".join(f"{channel:02X}" for channel in channels)


def mix_colour(left: str, right: str, amount: float) -> str:
    values = []
    for index in (1, 3, 5):
        start = int(left[index:index + 2], 16)
        end = int(right[index:index + 2], 16)
        values.append(round(start + (end - start) * amount))
    return "#" + "".join(f"{max(0, min(255, value)):02x}" for value in values)


def safe_name(value: str) -> str:

            )
        row = {name: str(raw[name]).lower() for name in REQUIRED_PALETTE_KEYS}
        row.update({
            "slug": f"onestepback-{key}",
            "token": f"OneStepBack-{safe_name(label)}",
            "label": f"OneStepBack — {label}",
            "gtk": ONE_STEP_GTK[key],
            "source": f"one_step_back.{key}",
            "active_title": row["black_color"],
            "inactive_title": row["base_color"],
            "bright": (row["lighter_color"]
                       if key.startswith("darkadobe") else row["white_color"]),
        })
        rows.append(row)
    for key, label in QT_STEP_LABELS.items():
        raw = data.get("qt_step", {}).get(key)
        if not isinstance(raw, dict):
            raise SystemExit(f"palette row is missing: qt_step.{key}")
        required = ("accent", "active_title", "border", "inactive_title")
        missing = [name for name in required if name not in raw]
        if missing:
            raise SystemExit(f"palette row qt_step.{key} is missing: {', '.join(missing)}")
        values = {name: str(raw[name]).lower() for name in required}
        for name, value in values.items():
            if not HEX.fullmatch(value):
                raise SystemExit(f"palette row qt_step.{key} has invalid {name}: {value}")
        face = values["inactive_title"]
        rows.append({
            "slug": f"qtstep-{safe_name(label).lower()}",
            "token": f"QTStep-{safe_name(label)}",
            "label": f"QTStep — {label}",
            "gtk": f"NextStep-QTStep-{safe_name(label)}",
            "source": f"qt_step.{key}",
            "base_color": face,
            "black_color": "#000000",
            "darker_color": mix_colour(face, "#000000", 0.35),
            "lighter_color": mix_colour(face, "#ffffff", 0.30),
            "scale_color": values["accent"],
            "white_color": "#ffffff",
            "active_title": values["active_title"],
            "inactive_title": face,
            "bright": "#ffffff",
        })
    if len(rows) != 27 or len({row["slug"] for row in rows}) != 27:
        raise SystemExit(f"expected 27 unique palette rows, found {len(rows)}")
    for row in rows:
        for name in REQUIRED_PALETTE_KEYS:
            if not HEX.fullmatch(row[name]):
                raise SystemExit(f"palette row {row['source']} has invalid {name}: {row[name]}")
    return rows


def palette_row(data: dict, skew: str) -> dict[str, str]:
    # The authentic default NeXT trial is black-focused, but its unfocused
    # face and bevel remain the normal OneStepBack gray family.  Keep one
    # validated palette source and make the focused-title exception explicit.
    palette_skew = "base" if skew == "default" else skew
    raw = data.get("one_step_back", {}).get(palette_skew)
    if not isinstance(raw, dict):
        raise SystemExit(f"palette row is missing or ambiguous: one_step_back.{palette_skew}")
    missing = [key for key in REQUIRED_PALETTE_KEYS if key not in raw]
    if missing:
        raise SystemExit(f"palette row one_step_back.{palette_skew} is missing: {', '.join(missing)}")
    row = {key: str(raw[key]).lower() for key in REQUIRED_PALETTE_KEYS}
    for key, value in row.items():
        if not HEX.fullmatch(value):
            raise SystemExit(f"palette row one_step_back.{palette_skew} has invalid {key}: {value}")
    row.update({
        # WMakerDefault's focused body is a dark slab.  The differing
        # darker_color role is the deliberate palette-responsive title face.
        "active_title": row["darker_color"],
        "inactive_title": row["base_color"],
        "bright": row["lighter_color"] if skew == "darkadobe" else row["white_color"],
    })
    if skew != "default":
        row["active_title"] = row["black_color"]
    return row

def recolour_xpm(source: Path, row: dict[str, str], state: str, context: str,
                 fallbacks: set[tuple[str, str]]) -> tuple[int, int, list[list[str | None]]]:
    image = read_xpm(source)
    mapping = donor_colour_map(row)
    body_symbol: str | None = None
    if context == "title":
        interior = [
            symbol for line in image.pixels[2:-2] for symbol in line
            if image.colours[symbol] is not None
        ]
        body_symbol = collections.Counter(interior).most_common(1)[0][0]
    output: list[list[str | None]] = []
    for line in image.pixels:
        output_line: list[str | None] = []
        for symbol in line:
            old = image.colours[symbol]
            if old is None:
                output_line.append(None)
                continue
            if context == "title" and symbol == body_symbol:
                new = row["active_title"] if state == "active" else row["inactive_title"]
            else:
                new = mapping.get(old.upper())
                if new is None:
                    fallbacks.add((source.name, old.upper()))
                    new = old.upper()
            output_line.append(new)
        output.append(output_line)
    return image.width, image.height, output


def write_recoloured(source: Path, destination: Path, row: dict[str, str], state: str,
                     context: str, fallbacks: set[tuple[str, str]]) -> None:
    width, height, pixels = recolour_xpm(source, row, state, context, fallbacks)
    write_xpm(destination, [colour for line in pixels for colour in line], pixels)
    if (width, height) != (len(pixels[0]), len(pixels)):
        raise AssertionError(f"XPM size changed during recolour: {source}")


def title_sources() -> dict[str, str]:
    return {
        "B": "title-5", "T": "title-3", "S": "title-1",
        "L": "title-1", "R": "title-5", "M": "title-2", "P": "title-4",
    }


def write_titles(donor: Path, output: Path, row: dict[str, str],
                 fallbacks: set[tuple[str, str]]) -> None:
    for state, suffix in (("active", "A"), ("inactive", "I")):
        for stem, donor_stem in title_sources().items():
            source = donor / f"{donor_stem}-{state}.xpm"
            write_recoloured(source, output / f"title{suffix}{stem}.xpm", row,
                             state, "title", fallbacks)


def write_wrapped_titles(donor: Path, output: Path, row: dict[str, str],
                         title_bar_height: int,
                         title_slab_height: int,
                         inner_divider: bool = False,
                         black_active: bool = False) -> None:
    """Emit a titlebar whose bevel wraps the assembled bar boundary."""
    for state, suffix in (("active", "A"), ("inactive", "I")):
        title = row["active_title"] if state == "active" else row["inactive_title"]
        highlight = row["lighter_color"]
        shadow = row["darker_color"]
        separator = row["black_color"]
        for stem, donor_stem in title_sources().items():
            source = donor / f"{donor_stem}-{state}.xpm"
            image = read_xpm(source)
            # IceWM sizes the title buttons from their loaded pixmap width.
            # Match the L/R title segments to the 23px button frame; the donor
            # L/R width is 23px and otherwise leaves a one-pixel strip before
            # each outermost button.
            segment_width = (title_bar_height
                             if stem in {"L", "R"} else image.width)
            # Authentic NeXT wraps the assembled bar: highlight first, then
            # the 19px slab, shadow last, and a black separator.  The J/Q
            # segments are the frame-to-button boundaries; L/R are the
            # button-to-pane segments and remain horizontal-only here.
            pixels = [[title for _ in range(segment_width)]
                      for _ in range(title_bar_height)]
            if black_active and suffix == "A":
                pixels = [[separator for _ in range(segment_width)]
                          for _ in range(title_bar_height)]
                # Pass 5: the black active bar still has its single top
                # highlight; rows 1..20 are black and row 21 is separator.
                pixels[0] = [highlight] * segment_width
            else:
                pixels[0] = [highlight] * segment_width
                pixels[title_slab_height + 1] = [shadow] * segment_width
                pixels[title_slab_height + 2] = [separator] * segment_width
            # titleL is drawn immediately after the left button cluster and
            # titleR immediately before the right cluster.  Their first/last
            # two columns are therefore the inner groove; J/Q are frame-side.
            if inner_divider and stem == "L":
                for y in range(1, title_bar_height - 1):
                    pixels[y][0] = shadow
                    pixels[y][1] = highlight
            elif inner_divider and stem == "R":
                for y in range(1, title_bar_height - 1):
                    pixels[y][-2] = highlight
                    pixels[y][-1] = shadow
            write_xpm(output / f"title{suffix}{stem}.xpm",
                      [colour for line in pixels for colour in line], pixels)


def write_authentic_titles(donor: Path, output: Path,
                           row: dict[str, str]) -> None:
    """Emit the frozen Default titlebar anatomy."""
    write_wrapped_titles(donor, output, row, AUTHENTIC_TITLE_BAR_HEIGHT,
                         AUTHENTIC_TITLE_SLAB_HEIGHT)


def write_canon_titles(donor: Path, output: Path,
                       row: dict[str, str]) -> None:
    """Emit pass-6's flush, button-matched 21px panel plus separator.

    IceWM tiles these resources around the button widgets.  L/R therefore
    carry the one-pixel black cluster separator at their outer column and the
    panel's matching vertical bevel immediately inside it; all other title
    segments carry only the continuous horizontal bevel rows.  J/Q remain
    absent so the button clusters stay flush to the one-pixel frame.
    """
    highlight = GNUSTEP_BUTTON_HIGHLIGHT
    shadow = GNUSTEP_BUTTON_SHADOW
    black = row["black_color"]
    for state, suffix in (("active", "A"), ("inactive", "I")):
        face = row["active_title"] if state == "active" else row["inactive_title"]
        for stem, donor_stem in title_sources().items():
            source = donor / f"{donor_stem}-{state}.xpm"
            width = CANON_TITLE_BAR_HEIGHT if stem in {"L", "R"} else read_xpm(source).width
            pixels = [[face for _ in range(width)] for _ in range(CANON_TITLE_BAR_HEIGHT)]
            pixels[0] = [highlight] * width
            for y in range(1, CANON_TITLE_BAR_HEIGHT - 2):
                pixels[y] = [face] * width
            pixels[CANON_TITLE_BAR_HEIGHT - 2] = [shadow] * width
            pixels[CANON_TITLE_BAR_HEIGHT - 1] = [black] * width
            if stem == "L":
                for y in range(CANON_TITLE_BAR_HEIGHT - 1):
                    pixels[y][0] = black
                    if y > 0:
                        pixels[y][1] = highlight
                pixels[CANON_TITLE_BAR_HEIGHT - 2][0] = black
            elif stem == "R":
                last = width - 1
                for y in range(CANON_TITLE_BAR_HEIGHT - 1):
                    pixels[y][last] = black
                    if y > 0:
                        pixels[y][last - 1] = shadow
                pixels[CANON_TITLE_BAR_HEIGHT - 2][last] = black
            write_xpm(output / f"title{suffix}{stem}.xpm",
                      [colour for line in pixels for colour in line], pixels)


def write_gnustep_cluster_borders(output: Path, row: dict[str, str]) -> None:
    """Emit black beveled frame-to-button segments (IceWM J and Q).

    IceWM's titleJ/titleQ resources are explicitly the frame-to-left-buttons
    and right-buttons-to-frame boundaries.  They are distinct from titleL/R,
    which vendor/engine source comments identify as pane-adjacent segments.
    Pass 4 moves the relief away from these frame-side segments: active J/Q
    are flat black, while inactive J/Q retain the gray titlebar wrap.  The
    actual inner groove is carried by the edge columns of titleL/titleR.
    """
    height = CANON_TITLE_BAR_HEIGHT
    width = 3
    black = row["black_color"]
    for suffix in ("A", "I"):
        face = black if suffix == "A" else row["inactive_title"]
        pixels = [[face for _ in range(width)] for _ in range(height)]
        if suffix == "I":
            pixels[0] = [row["lighter_color"]] * width
            pixels[CANON_TITLE_SLAB_HEIGHT + 1] = [row["darker_color"]] * width
            pixels[CANON_TITLE_SLAB_HEIGHT + 2] = [black] * width
        else:
            pixels[CANON_TITLE_SLAB_HEIGHT + 2] = [black] * width
        for side, stem in (("left", "J"), ("right", "Q")):
            write_xpm(output / f"title{suffix}{stem}.xpm",
                      [colour for line in pixels for colour in line], pixels)


def write_frames(output: Path, row: dict[str, str]) -> None:
    contour = dark_chrome(row["darker_color"])
    face = row["base_color"]
    highlight = row["lighter_color"]
    shadow = row["darker_color"]
    pixels = {
        # Cross-sections run from outside toward the window interior.
        "T": [[contour], [highlight], [face]],
        "L": [[contour, highlight, face]],
        "R": [[face, shadow, contour]],
        "B": [[face], [shadow], [contour]],
        "TL": [
            [contour, contour, contour],
            [contour, highlight, highlight],
            [contour, highlight, face],
        ],
        "TR": [
            [contour, contour, contour],
            [highlight, highlight, shadow],
            [face, face, contour],
        ],
        "BL": [
            [contour, highlight, face],
            [contour, shadow, shadow],
            [contour, contour, contour],
        ],
        "BR": [
            [face, face, contour],
            [shadow, shadow, contour],
            [contour, contour, contour],
        ],
    }
    for suffix in ("A", "I"):
        for prefix in ("frame", "dframe"):
            for stem, frame in pixels.items():
                write_xpm(
                    output / f"{prefix}{suffix}{stem}.xpm",
                    [colour for line in frame for colour in line],
                    frame,
                )


def write_authentic_frames(output: Path, row: dict[str, str]) -> None:
    """Emit the one-pixel black outer window frame for authentic NeXT."""
    black = row["black_color"]
    for suffix in ("A", "I"):
        for prefix in ("frame", "dframe"):
            for stem in ("T", "L", "R", "B", "TL", "TR", "BL", "BR"):
                write_xpm(output / f"{prefix}{suffix}{stem}.xpm",
                          [black], [[black]])


def authentic_button_pixels(row: dict[str, str], kind: str,
                            pressed: bool,
                            button_style: str,
                            bar_color: str,
                            margin_style: str,
                            edge: str | None,
                            title_bar_height: int = AUTHENTIC_TITLE_BAR_HEIGHT,
                            title_slab_height: int = AUTHENTIC_TITLE_SLAB_HEIGHT
                            ) -> list[list[str]]:
    """Build one wrapped IceWM frame with a 16px NeXT button inset.

    IceWM copies half of a stacked button pixmap into a widget exactly
    TitleBarHeight high.  The 16px button therefore sits at x/y=3..18 inside
    this 23px frame: the bevel is row/column 0, two full slab rows/columns
    1..2 remain clear, and the three terminal rows/columns 19..21 reach the
    shadow/separator boundary, with the black terminal pixel serving as the
    phantom shadow.  Its bevel occupies the first 15px; the final
    row/column are a distinct outer black shadow.  At button-local (14, 0),
    the one-pixel bevel-shadow corner is diagonal to that outer shadow and
    deliberately remains a single square.
    """
    style = AUTHENTIC_BUTTON_STYLES.get(button_style)
    if style is None:
        raise SystemExit(f"unknown authentic button style: {button_style}")
    size = AUTHENTIC_BUTTON_ART_SIZE
    black = row["black_color"]
    highlight = row["white_color"]
    shadow = row["darker_color"]
    if pressed:
        # Depressed state: the bevel inverts (shadow where light was) so the
        # button reads sunken, matching the glyph's one-pixel travel.
        highlight, shadow = shadow, highlight
    face = row[style["face"]]
    glyph = row[style["glyph"]]
    pixels = [[face for _ in range(size)] for _ in range(size)]
    bevel_edge = size - 2
    for x in range(bevel_edge):
        pixels[0][x] = highlight
    pixels[0][bevel_edge] = shadow
    for y in range(1, bevel_edge):
        pixels[y][0] = highlight
        pixels[y][bevel_edge] = shadow
    for x in range(1, bevel_edge + 1):
        pixels[bevel_edge][x] = shadow
    # The final row/column are the extra black outer shadow, distinct from
    # the bevel shadow.  They are part of the visible 16px construction.
    for y in range(size):
        pixels[y][size - 1] = black
    for x in range(size):
        pixels[size - 1][x] = black

    offset = 1 if pressed else 0
    if kind == "maximize":
        glyph_left = 2 + offset
        glyph_right = min(size - 4, size - 3 + offset)
        glyph_top = 2 + offset
        glyph_bottom = min(size - 4, size - 3 + offset)
        for y in range(glyph_top, glyph_top + 3):
            if y < size - 1:
                for x in range(glyph_left, glyph_right + 1):
                    pixels[y][x] = glyph
        for x in (glyph_left, glyph_right):
            for y in range(glyph_top + 3, glyph_bottom + 1):
                if y < size - 1:
                    pixels[y][x] = glyph
        for x in range(glyph_left, glyph_right + 1):
            if x < size - 1:
                pixels[glyph_bottom][x] = glyph
    elif kind == "close":
        glyph_left = 2 + offset
        for y, pattern in enumerate(AUTHENTIC_CLOSE_GLYPHS[pressed],
                                    start=glyph_left):
            for x, bead in enumerate(pattern, start=glyph_left):
                if bead == "X":
                    pixels[y][x] = glyph
    frame_size = title_bar_height
    if margin_style not in BUTTON_MARGIN_STYLES:
        raise SystemExit(f"unknown button margin style: {margin_style}")
    # IceWM copies the button pixmap over the title pixmap.  The named
    # GNUstep-flat mode deliberately leaves every margin at the slab colour;
    # authentic NeXT wraps the full bar pattern through the margins.
    frame = [[bar_color for _ in range(frame_size)] for _ in range(frame_size)]
    if margin_style == "nextstep-wrap":
        frame[0] = [row["lighter_color"] for _ in range(frame_size)]
        frame[title_slab_height + 1] = [row["darker_color"] for _ in range(frame_size)]
        frame[title_slab_height + 2] = [row["black_color"] for _ in range(frame_size)]
    inset_y = AUTHENTIC_BUTTON_ART_TOP
    # The close button moves one pixel right inside its frame; its right edge
    # remains the bar-right shadow column. Every other button keeps x=3..18.
    inset_x = (AUTHENTIC_BUTTON_ART_TOP + 1
               if kind == "close" else AUTHENTIC_BUTTON_ART_TOP)
    if inset_y + size > title_slab_height + 1:
        raise SystemExit("authentic button art does not fit inside title slab")
    if inset_x + size > frame_size:
        raise SystemExit("authentic button art does not fit inside button frame")
    for y, row_pixels in enumerate(pixels, start=inset_y):
        frame[y][inset_x:inset_x + size] = row_pixels
    # In this theme TitleButtonsLeft="m" is maximize and
    # TitleButtonsRight="x" is close.  Preserve the L/R vertical bevel in
    # only those two outer edge columns after art insertion; title L/R and all
    # other generated button assets have no edge claim and retain horizontal
    # rows only.
    if margin_style == "nextstep-wrap" and edge == "left":
        for y in range(1, title_slab_height + 1):
            frame[y][0] = row["lighter_color"]
    elif margin_style == "nextstep-wrap" and edge == "right":
        for y in range(1, title_slab_height + 1):
            frame[y][-1] = row["darker_color"]
    return frame


def write_authentic_buttons(output: Path, row: dict[str, str],
                            button_style: str,
                            margin_style: str,
                            title_bar_height: int = AUTHENTIC_TITLE_BAR_HEIGHT,
                            title_slab_height: int = AUTHENTIC_TITLE_SLAB_HEIGHT,
                            edge_by_target: dict[str, str] | None = None
                            ) -> None:
    """Emit styled 16px NeXT buttons as stacked title-bar frames."""
    targets = {
        "close": "close", "minimize": "minimize", "maximize": "maximize",
        "restore": "restore", "rollup": "rollup", "rolldown": "rolldown",
        "menuButton": "menuButton",
    }
    edge_by_target = edge_by_target or {"maximize": "left", "close": "right"}
    for target, kind in targets.items():
        glyph_kind = kind if kind in {"close", "maximize"} else ""
        for suffix, bar_color in (("A", row["active_title"]),
                                  ("I", row["inactive_title"])):
            normal = authentic_button_pixels(
                row, glyph_kind, pressed=False, button_style=button_style,
                bar_color=bar_color, margin_style=margin_style,
                edge=edge_by_target.get(target),
                title_bar_height=title_bar_height,
                title_slab_height=title_slab_height)
            pressed = authentic_button_pixels(
                row, glyph_kind, pressed=True, button_style=button_style,
                bar_color=bar_color, margin_style=margin_style,
                edge=edge_by_target.get(target),
                title_bar_height=title_bar_height,
                title_slab_height=title_slab_height)
            write_xpm(
                output / f"{target}{suffix}.xpm",
                [colour for line in normal + pressed for colour in line],
                normal + pressed)


def write_canon_buttons(output: Path, row: dict[str, str],
                        button_style: str) -> None:
    """Emit pass-8's full 21x21 bevels with owned separator columns.

    wmbutton.cc uses the pixmap width as the widget width and copies one
    half-height frame into a TitleBarHeight=22 widget.  Each half therefore
    contains the 21x21 art plus its black separator row.  Separators are
    extra owned columns, never replacements for a square's shadow edge.
    """
    if button_style != "black":
        raise SystemExit("GNUstep canon buttons require --button-style black")
    targets = {
        "close": "close", "minimize": "minimize", "maximize": "maximize",
        "restore": "restore", "rollup": "rollup", "rolldown": "rolldown",
        "menuButton": "menuButton",
    }
    # Pass-8 ownership map, with the configured strings documented by
    # IceWM's themable.h: left="sr" is menu, rollup; right="xim" is
    # maximize, minimize, close from left to right.  The menu owns the black
    # column before rollup; max/min (and restore for the Maxi state swap) own
    # the black columns before their right-hand neighbour.  The titleAL/R
    # separator columns own the two cluster-to-title junctions, while the
    # window frame owns the two outermost borders.
    separator_right = {"menuButton", "maximize", "restore", "minimize"}
    for target, kind in targets.items():
        owns_separator = target in separator_right
        for suffix in ("A", "I"):
            active = suffix == "A"
            normal = gnustep_donor_button_pixels(
                row, kind, pressed=False, active=active)
            pressed = gnustep_donor_button_pixels(
                row, kind, pressed=True, active=active)
            if owns_separator:
                normal = [line + [row["black_color"]] for line in normal]
                pressed = [line + [row["black_color"]] for line in pressed]
            normal_with_separator = normal + [[row["black_color"]] * len(normal[0])]
            pressed_with_separator = pressed + [[row["black_color"]] * len(pressed[0])]
            write_xpm(
                output / f"{target}{suffix}.xpm",
                [colour for line in normal_with_separator + pressed_with_separator
                 for colour in line],
                normal_with_separator + pressed_with_separator,
            )


def gnustep_donor_button_pixels(row: dict[str, str], kind: str,
                                pressed: bool, active: bool,
                                ) -> list[list[str]]:
    """Build one 21x21 button half with the filed donor glyph.

    The bevel is exactly one pixel on each side.  Cluster separators are
    appended by the caller as owned columns, so the square's shadow edge is
    never consumed by separator paint.
    """
    try:
        glyph = GNUSTEP_DONOR_GLYPHS[kind]["pressed" if pressed else "normal"]
    except KeyError as exc:
        raise SystemExit(f"missing GNUstep donor glyph: {kind}") from exc
    if len(glyph) != 15 or any(len(line) != 15 for line in glyph):
        raise SystemExit(f"invalid GNUstep donor glyph dimensions: {kind}")
    glyph = extend_gnustep_glyph(kind, glyph, pressed)
    black = row["black_color"]
    face = black if active else row["inactive_title"]
    highlight = GNUSTEP_BUTTON_HIGHLIGHT
    shadow = GNUSTEP_BUTTON_SHADOW
    if pressed:
        highlight, shadow = shadow, highlight
    size = GNUSTEP_BUTTON_SIZE
    pixels = [[face for _ in range(size)] for _ in range(size)]
    glyph_colour = row["white_color"] if active else contrast(row, face)
    pixels[0] = [highlight] * size
    for y in range(1, size - 1):
        pixels[y][0] = highlight
        pixels[y][-1] = shadow
    pixels[size - 1] = [shadow] * size
    # Center the 15px donor mask in the 21px face; the enclosing bevel is
    # therefore two pixels away on every side of the glyph's 15px canvas.
    offset_x = (size - 15) // 2
    offset_y = 3
    # Rolldown's filed normal and pressed masks are identical, so its new
    # lattice remains anchored while the other glyphs receive the donor's
    # established one-pixel pressed travel in extend_gnustep_glyph().
    travel_x = 0
    travel_y = 0
    for dy, line in enumerate(glyph):
        for dx, mark in enumerate(line):
            if mark == "#":
                pixels[offset_y + dy + travel_y][offset_x + dx + travel_x] = glyph_colour
    return pixels


def write_authentic_start(output: Path, repo: Path) -> dict[str, str]:
    """Emit the NeXT taskbar Start image through the established raster path.

    Recover the approved cheerleader construction: rasterize the canonical
    cube directly, render SerenitySans once per letter, threshold the colored
    face at alpha 192, then surround it with a two-pixel black dilation.
    """
    from PIL import Image, ImageDraw, ImageFont

    svg = repo / NEXT_LOGO_SVG_RELATIVE
    top_font_path = repo / NEXT_START_TOP_FONT_RELATIVE
    png2xpm = repo / "linux/.local/bin/png2xpm.py"
    if not svg.is_file():
        raise SystemExit(f"NeXT Start SVG missing: {svg}")
    if not top_font_path.is_file():
        raise SystemExit(f"NeXT Start font missing: {top_font_path}")
    if not png2xpm.is_file():
        raise SystemExit(f"PNG-to-XPM helper missing: {png2xpm}")
    svg_text = svg.read_text(encoding="utf-8")
    svg_fills = re.findall(r'fill="(#[0-9a-fA-F]{6})"', svg_text)
    if len(svg_fills) != len(AUTHENTIC_START_TEXT):
        raise SystemExit(
            "NeXT Start SVG must expose exactly four coloured facets"
        )
    if re.search(r"<text\b|glyph-name=|aria-label=", svg_text):
        raise SystemExit("NeXT Start SVG unexpectedly contains letter glyph data")
    # The SVG is facet-only: its fills are used for the unhaloed cube.  The
    # final wordmark colours are the explicit cheerleader ruling, not a
    # positional guess from the facet order.
    letter_colours = {
        "N": "#fa1516", "e": "#fcfc17",
        "X": "#16d41a", "T": "#fb16bb",
    }
    destination = output / "taskbar/start.xpm"
    destination.parent.mkdir(parents=True, exist_ok=True)
    with tempfile.TemporaryDirectory(prefix="nextstep-start-") as temp_dir:
        icon_png = Path(temp_dir) / "next-logo.png"
        start_png = Path(temp_dir) / "start.png"
        rsvg = shutil.which("rsvg-convert")
        if rsvg is None:
            guix_rsvg = Path.home() / ".guix-extra-profiles/core/core/bin/rsvg-convert"
            if guix_rsvg.is_file():
                rsvg = str(guix_rsvg)
        if rsvg is None:
            raise SystemExit("rsvg-convert is required to rasterize the NeXT Start SVG")
        try:
            subprocess.run(
                [rsvg, "-w", str(AUTHENTIC_START_ICON_WIDTH),
                 "-h", str(AUTHENTIC_START_ICON_HEIGHT), str(svg),
                 "-o", str(icon_png)],
                check=True, stdout=subprocess.DEVNULL,
                stderr=subprocess.PIPE, text=True,
            )
        except subprocess.CalledProcessError as error:
            detail = error.stderr.strip() or "no rsvg-convert diagnostics"
            raise SystemExit(f"NeXT Start SVG rasterization failed: {detail}") from error

        icon = Image.open(icon_png).convert("RGBA")
        if icon.size != (AUTHENTIC_START_ICON_WIDTH, AUTHENTIC_START_ICON_HEIGHT):
            raise SystemExit(f"unexpected NeXT SVG raster size: {icon.size}")
        metrics = ImageDraw.Draw(Image.new("RGBA", (1, 1)))
        canvas = Image.new("RGBA", (256, AUTHENTIC_START_HEIGHT),
                           (0, 0, 0, 0))
        # The cube remains a plain rsvg raster: no threshold or dilation.
        canvas.alpha_composite(icon, (AUTHENTIC_START_PAD, 2))
        text_left = AUTHENTIC_START_PAD + icon.width + AUTHENTIC_START_GAP
        opaque_alpha = lambda image: image.getchannel("A").point(
            lambda pixel: 255 if pixel >= 128 else 0
        ).getbbox()
        top_font = ImageFont.truetype(str(top_font_path), AUTHENTIC_START_TEXT_SIZE)
        top_bbox = metrics.textbbox(
            (0, 0), AUTHENTIC_START_TEXT, font=top_font, stroke_width=0,
        )
        top_text_height = top_bbox[3] - top_bbox[1]
        top_text_y = ((AUTHENTIC_START_HEIGHT - top_text_height) // 2
                      - top_bbox[1] + AUTHENTIC_START_TOP_OFFSET_Y)
        top_layer = Image.new("RGBA", (131, AUTHENTIC_START_HEIGHT),
                              (0, 0, 0, 0))
        top_draw = ImageDraw.Draw(top_layer)
        cursor = 0.0
        for letter in AUTHENTIC_START_TEXT:
            top_draw.text((2 + round(cursor), top_text_y), letter, font=top_font,
                          fill=letter_colours[letter])
            cursor += top_font.getlength(letter)

        core_mask = top_layer.getchannel("A").point(
            lambda pixel: 255 if pixel >= AUTHENTIC_START_CORE_ALPHA else 0
        )
        core_layer = top_layer.copy()
        core_layer.putalpha(core_mask)
        core_bbox = opaque_alpha(core_layer)
        if core_bbox is None:
            raise SystemExit("NeXT Start cheerleader core unexpectedly has no content")
        top_x = (text_left - core_bbox[0] + AUTHENTIC_START_TOP_OFFSET_X)

        outline_layer = Image.new("RGBA", core_layer.size, (0, 0, 0, 0))
        outline_pixels = outline_layer.load()
        mask_pixels = core_mask.load()
        for y in range(core_layer.height):
            for x in range(core_layer.width):
                if not mask_pixels[x, y]:
                    continue
                for dy in range(-AUTHENTIC_START_OUTLINE_RADIUS,
                                AUTHENTIC_START_OUTLINE_RADIUS + 1):
                    for dx in range(-AUTHENTIC_START_OUTLINE_RADIUS,
                                    AUTHENTIC_START_OUTLINE_RADIUS + 1):
                        nx, ny = x + dx, y + dy
                        if (0 <= nx < core_layer.width and
                                0 <= ny < core_layer.height and
                                not mask_pixels[nx, ny]):
                            outline_pixels[nx, ny] = (0, 0, 0, 255)
        outline_bbox = opaque_alpha(outline_layer)
        if outline_bbox is None:
            raise SystemExit("NeXT Start outline unexpectedly has no content")
        canvas.alpha_composite(outline_layer, (top_x, 0))
        canvas.alpha_composite(core_layer, (top_x, 0))
        alpha_bbox = opaque_alpha(canvas)
        if alpha_bbox is None:
            raise SystemExit("NeXT Start canvas unexpectedly has no content")
        core_bbox_global = tuple(
            coordinate + (top_x if index % 2 == 0 else 0)
            for index, coordinate in enumerate(core_bbox)
        )
        outline_bbox_global = tuple(
            coordinate + (top_x if index % 2 == 0 else 0)
            for index, coordinate in enumerate(outline_bbox)
        )
        width = alpha_bbox[2] + AUTHENTIC_START_PAD
        canvas = canvas.crop((0, 0, width, AUTHENTIC_START_HEIGHT))
        canvas.save(start_png)
        converted = subprocess.run(
            [sys.executable, str(png2xpm), str(start_png), str(destination),
             "start_xpm"], check=True, capture_output=True, text=True,
        )
        if converted.stdout.strip():
            print(converted.stdout.strip())

    return {
        "canvas": f"{width}x{AUTHENTIC_START_HEIGHT}",
        "icon": f"{AUTHENTIC_START_ICON_WIDTH}x{AUTHENTIC_START_ICON_HEIGHT}",
        "icon_source": str(NEXT_LOGO_SVG_RELATIVE),
        "text": AUTHENTIC_START_TEXT,
        "font": "SerenitySans-Regular.ttf:size=12:regular (single pass; no Charcoal bed; core alpha threshold=192/255)",
        "text_bbox": f"core=x={core_bbox_global[0]}..{core_bbox_global[2] - 1},y={core_bbox_global[1]}..{core_bbox_global[3] - 1},height={core_bbox_global[3] - core_bbox_global[1]}; outline=x={outline_bbox_global[0]}..{outline_bbox_global[2] - 1},y={outline_bbox_global[1]}..{outline_bbox_global[3] - 1},height={outline_bbox_global[3] - outline_bbox_global[1]}",
        "layer_offset": f"face=({AUTHENTIC_START_TOP_OFFSET_X},{AUTHENTIC_START_TOP_OFFSET_Y}) from natural SerenitySans origin",
        "outline": "pure #000000 two-pixel 8-neighbor dilation; colored pixels win",
        "letter_colours": ",".join(f"{letter}={colour}" for letter, colour in letter_colours.items()),
        "svg_finding": "facet-only SVG: four filled paths, no letter glyphs; cube=rsvg 15x18 plain raster; label colours explicit",
        "target": "taskbar/start.xpm",
    }


def write_buttons(donor: Path, output: Path, row: dict[str, str],
                  fallbacks: set[tuple[str, str]]) -> None:
    """Keep the donor's flat-margin construction for GNUstep reuse."""
    sources = {
        "close": "close", "minimize": "hide", "maximize": "maximize",
        "restore": "maximize-toggled", "rollup": "shade",
        "rolldown": "shade-toggled", "menuButton": "menu",
    }
    for state, suffix in (("active", "A"), ("inactive", "I")):
        for target, source_stem in sources.items():
            normal = read_xpm(donor / f"{source_stem}-{state}.xpm")
            pressed = read_xpm(donor / f"{source_stem}-pressed.xpm")
            normal_path = donor / f"{source_stem}-{state}.xpm"
            pressed_path = donor / f"{source_stem}-pressed.xpm"
            _, _, top = recolour_xpm(normal_path, row, state, "button", fallbacks)
            _, _, bottom = recolour_xpm(pressed_path, row, state, "button", fallbacks)
            if (normal.width, normal.height) != (19, 19) or (pressed.width, pressed.height) != (19, 19):
                raise SystemExit(f"caption button geometry changed in donor: {source_stem}")
            write_xpm(output / f"{target}{suffix}.xpm", [
                colour for line in top + bottom for colour in line
            ], top + bottom)


def gnustep_button_pixels(source: Xpm, row: dict[str, str], state: str,
                          button_style: str, pressed: bool) -> list[list[str]]:
    """Build one 19px GNUstep button with slab margins and an inverted press."""
    style = AUTHENTIC_BUTTON_STYLES.get(button_style)
    if style is None:
        raise SystemExit(f"unknown button style: {button_style}")
    if (source.width, source.height) != (19, 19):
        raise SystemExit("GNUstep button geometry changed in donor")
    bar = row["active_title"] if state == "active" else row["inactive_title"]
    face = row[style["face"]]
    glyph = row[style["glyph"]]
    highlight = row["lighter_color"]
    shadow = row["darker_color"]
    if pressed:
        highlight, shadow = shadow, highlight
    pixels = [[face for _ in range(19)] for _ in range(19)]
    # GNUstep-flat: the outer one-pixel margin is the titlebar slab, while
    # the 17px inner square retains the bevel and selected face family.
    for x in range(19):
        pixels[0][x] = bar
        pixels[18][x] = bar
    for y in range(19):
        pixels[y][0] = bar
        pixels[y][18] = bar
    for x in range(1, 18):
        pixels[1][x] = highlight
        pixels[17][x] = shadow
    for y in range(1, 18):
        pixels[y][1] = highlight
        pixels[y][17] = shadow
    # The donor's exact white/black pixels inside the 15px face are the glyph
    # mask (inactive donor glyphs are black; bevels are outside this region).
    # Recolour them to the requested role, including the pressed half.
    for y, line in enumerate(source.pixels):
        for x, symbol in enumerate(line):
            if (2 <= x <= 16 and 2 <= y <= 16 and
                    source.colours[symbol] in {"#FFFFFF", "#000000"}):
                pixels[y][x] = glyph
    return pixels


def write_gnustep_buttons(donor: Path, output: Path, row: dict[str, str],
                          button_style: str, margin_style: str) -> None:
    """Emit the named GNUstep button mode, including its pressed inversion."""
    if margin_style != "gnustep-flat":
        write_buttons(donor, output, row, set())
        return
    sources = {
        "close": "close", "minimize": "hide", "maximize": "maximize",
        "restore": "maximize-toggled", "rollup": "shade",
        "rolldown": "shade-toggled", "menuButton": "menu",
    }
    for state, suffix in (("active", "A"), ("inactive", "I")):
        for target, source_stem in sources.items():
            normal = read_xpm(donor / f"{source_stem}-{state}.xpm")
            pressed = read_xpm(donor / f"{source_stem}-pressed.xpm")
            top = gnustep_button_pixels(normal, row, state, button_style, False)
            bottom = gnustep_button_pixels(pressed, row, state, button_style, True)
            write_xpm(output / f"{target}{suffix}.xpm",
                      [colour for line in top + bottom for colour in line],
                      top + bottom)


def tosh_colour_keys(path: Path) -> list[str]:
    keys = []
    for line in path.read_text(encoding="latin-1").splitlines():
        match = re.match(r"^(Color[A-Za-z0-9]+)=", line)
        if match:
            keys.append(match.group(1))
    if not keys:
        raise SystemExit(f"Tosh Color* checklist is empty: {path}")
    return keys


def contrast(row: dict[str, str], background: str) -> str:
    return row["black_color"] if rgb_luma(background) >= 128 else row["white_color"]

def build(repo: Path, skew: str, button_style: str = "gray",
          margin_style: str = "nextstep-wrap") -> Path:
    palette_path = repo / PALETTE_RELATIVE
    donor = repo / DONOR_RELATIVE
    tosh = repo / TOSH_DEFAULT_RELATIVE
    if not palette_path.is_file():
        raise SystemExit(f"palette table missing: {palette_path}")
    if not donor.is_dir():
        raise SystemExit(f"QTStep xfwm4 donor missing: {donor}")
    if not tosh.is_file():
        raise SystemExit(f"Tosh key checklist missing: {tosh}")
    row = palette_row(json.loads(palette_path.read_text(encoding="utf-8")), skew)
    donor_title = read_xpm(donor / "title-1-active.xpm")
    donor_side = read_xpm(donor / "left-active.xpm")
    donor_corner = read_xpm(donor / "top-left-active.xpm")
    donor_button = read_xpm(donor / "close-active.xpm")
    if donor_title.height != 19 or donor_button.width != 19 or donor_button.height != 19:
        raise SystemExit("donor TitleBarHeight/button geometry is not the measured 19px contract")
    if donor_side.width != 1 or donor_side.height != 23 or donor_corner.width != 1 or donor_corner.height != 19:
        raise SystemExit("donor border-strip geometry changed; refuse to guess")
    donor_border = donor_side.colours[donor_side.pixels[0][0]]
    if donor_border is None:
        raise SystemExit("donor border pixel is transparent")
    output = repo / "linux/.icewm/themes" / OUTPUTS[skew]
    if output.exists() or output.is_symlink():
        if output.is_symlink() or not output.is_dir():
            output.unlink()
        else:
            shutil.rmtree(output)
    output.mkdir(parents=True)
    authentic = skew == "default"
    canon = not authentic
    fallbacks: set[tuple[str, str]] = set()
    start_info: dict[str, str] | None = None
    if authentic:
        write_authentic_titles(donor, output, row)
        write_authentic_frames(output, row)
        write_authentic_buttons(output, row, button_style, margin_style)
        start_info = write_authentic_start(output, repo)
    else:
        write_canon_titles(donor, output, row)
        write_authentic_frames(output, row)
        write_canon_buttons(output, row, button_style)
        start_info = write_authentic_start(output, repo)
    tosh_keys = tosh_colour_keys(tosh)
    write_default(output / "default.theme", row, skew, donor, tosh_keys,
                  authentic=authentic, canon=canon, button_style=button_style)
    (output / "PROVENANCE").write_text(
        "NeXTSTEP/OpenStep trial border representative\n"
        f"Palette row: one_step_back.{'base' if skew == 'default' else skew}\n"
        + ("Focused title law: authentic NeXT black slab with white caption text; unfocused remains base face\n"
           if skew == "default" else "")
        + (f"Style mode: authentic-nextstep; 1px black frame; 1px highlight + 19px slab + 1px shadow + 1px separator; centered caption; 16px {button_style}-face buttons; no caption-text bevel\n"
           if authentic else f"Style mode: GNUstep-canon pass9b; black active titlebar with #b6b6b6/#616161 wrap; inactive {row['inactive_title']} face family with the same absolute bevel; 1px black frame; 21px face + 1px separator; centered Charcoal caption; 21x44 stacked buttons with complete 21x21 bevels plus owned separator columns; pass-9b continues each donor 2x2 stroke lattice one pixel per step and preserves patterned-row phase; left menu+rollup and right maximize+minimize+close; no J/Q gap; taskbar Start restored\n")
        + f"xfwm4 donor: {DONOR_RELATIVE}\n"
        + "Donor measures: title 19px source; authentic title 23px; title-1 23px wide; title-2 4px wide; "
        + "side 1x23; corner 1x19; buttons 19x19\n"
        + ("IceWM authentic frame: 1px black window border; titlebar bevel is in title XPMs\n"

* Dockapp lifecycle

The manager records exact PIDs with their display and never kills by name.
That keeps concurrent sanctuary guests in the shared container PID namespace
out of reach.  This lifecycle plumbing survived the 2026-08-10 rejection and
remains inert unless a future trial-gated scheme requests it through
=NEXTSTEP_DOCKAPPS=.

#+BEGIN_SRC bash :tangle linux/.local/bin/redstone-9x-nextstep-dockapps :shebang "#!/bin/bash" :tangle-mode (identity #o755)
set -uo pipefail

state="${HOME}/.cache/redstone-9x/nextstep-dockapps.state"
log="${HOME}/.cache/redstone-9x/nextstep-dockapps.log"
mkdir -p "$(dirname "${state}")"

stop_dockapps() {
  [[ -r "${state}" ]] || return 0
  while IFS='|' read -r recorded_display pid command; do
    [[ "${recorded_display}" == "${DISPLAY:-}" ]] || continue
    [[ "${pid}" =~ ^[0-9]+$ ]] || continue
    kill -0 "${pid}" 2>/dev/null || continue
    current="$(tr '\0' ' ' < "/proc/${pid}/cmdline" 2>/dev/null || true)"
    [[ "${current}" == *"${command}"* ]] && kill "${pid}" 2>/dev/null || true
  done < "${state}"
  rm -f "${state}"
}

wait_for_icedock() {
  local attempt
  for attempt in {1..30}; do
    if xwininfo -root -tree 2>/dev/null | grep -q '"IceDock"'; then
      return 0
    fi
    sleep 0.1
  done
  printf 'IceDock was not ready on %s\n' "${DISPLAY:-}" >> "${log}"
  return 1
}

start_dockapps() {
  stop_dockapps
  : > "${state}"
  : > "${log}"
  wait_for_icedock || return 1
  local -a specs=(
    'nextstep-dock'
    'wmclock|-24|-led|#d6d6d6'
    "wmcpuload|--display|${DISPLAY:-:0}|--light-color|#aeb8bf"
    "wmbattery|-d|${DISPLAY:-:0}|-e"
    "wmnd|-d|${DISPLAY:-:0}|-F|-q|-c|#d6d6d6|-C|#8fa9bd|-L|#000000"
    'wmamixer|-w|-s|-b|#000000|-l|#d6d6d6'
  )
  local spec command binary pid
  local -a argv
  for spec in "${specs[@]}"; do
    IFS='|' read -r -a argv <<< "${spec}"
    command="${argv[0]}"
    binary="$(command -v "${command}" 2>/dev/null || true)"
    if [[ -z "${binary}" ]]; then
      printf 'missing dockapp: %s\n' "${command}" >> "${log}"
      continue
    fi
    "${binary}" "${argv[@]:1}" >> "${log}" 2>&1 &
    pid=$!
    sleep 0.1
    if kill -0 "${pid}" 2>/dev/null; then
      printf '%s|%s|%s\n' "${DISPLAY:-}" "${pid}" "${command}" >> "${state}"
    else
      printf 'dockapp exited during startup: %s\n' "${command}" >> "${log}"
    fi
  done
}

case "${1:-}" in
  start) start_dockapps ;;
  stop) stop_dockapps ;;
  sync)
    if [[ "${2:-0}" == "1" ]]; then start_dockapps; else stop_dockapps; fi
    ;;
  *) printf 'usage: redstone-9x-nextstep-dockapps start|stop|sync 0|1\n' >&2; exit 2 ;;
esac
#+END_SRC

* NeXT logo dock tile

The original GNUstep plan named a custom tile but never contained the claimed C
source.  This is the missing implementation: a conventional withdrawn X11
dockapp with a 64-pixel icon window.  Its embedded 48-pixel artwork is the
=NeXTSTEP-cube.xpm= carried by the GNUstep sanctuary's Window Maker icon pool;
embedding it keeps the Redstone package reproducible and removes any runtime
dependency on the GNUstep guest home.

static unsigned long named_pixel(Display *display, int screen, const char *name) {
    XColor exact;
    XColor allocated;
    Colormap colormap = DefaultColormap(display, screen);
    if (XAllocNamedColor(display, colormap, name, &allocated, &exact))
        return allocated.pixel;
    return BlackPixel(display, screen);
}

static void draw(Display *display, Window window, GC gc, int screen) {
    unsigned long black = BlackPixel(display, screen);
    unsigned long white = WhitePixel(display, screen);
    unsigned long colours[256] = {0};
    colours[(unsigned char)'.'] = named_pixel(display, screen, "#05070A");
    colours[(unsigned char)'+'] = named_pixel(display, screen, "#E6EBEE");
    colours[(unsigned char)'@'] = named_pixel(display, screen, "#AAB5BC");
    colours[(unsigned char)'#'] = named_pixel(display, screen, "#77838C");
    colours[(unsigned char)'$'] = named_pixel(display, screen, "#2C373F");
    colours[(unsigned char)'&'] = white;
    XSetForeground(display, gc, black);
    XFillRectangle(display, window, gc, 0, 0, 64, 64);
    XSetForeground(display, gc, white);
    XDrawRectangle(display, window, gc, 1, 1, 61, 61);
    for (int y = 0; y < 48; ++y) {
        for (int x = 0; x < 48; ++x) {
            unsigned char code = (unsigned char)cube[y][x];
            if (code == ' ') continue;
            XSetForeground(display, gc, colours[code]);
            XDrawPoint(display, window, gc, x + 8, y + 8);
        }
    }
}

int main(void) {
    Display *display = XOpenDisplay(NULL);
    if (!display) return 1;
    int screen = DefaultScreen(display);
    Window root = RootWindow(display, screen);
    Window main_window = XCreateSimpleWindow(display, root, 0, 0, 1, 1, 0,
                                              BlackPixel(display, screen),
                                              BlackPixel(display, screen));
    Window icon_window = XCreateSimpleWindow(display, root, 0, 0, 64, 64, 0,
                                              WhitePixel(display, screen),
                                              BlackPixel(display, screen));
    XClassHint class_hint = {"nextstep-dock", "DockApp"};
    XSetClassHint(display, main_window, &class_hint);
    XStoreName(display, main_window, "NeXT");
    XWMHints hints;
    memset(&hints, 0, sizeof(hints));
    hints.flags = StateHint | IconWindowHint | WindowGroupHint;
    hints.initial_state = WithdrawnState;
    hints.icon_window = icon_window;
    hints.window_group = main_window;
    XSetWMHints(display, main_window, &hints);
    XSelectInput(display, icon_window, ExposureMask | StructureNotifyMask);
    XMapWindow(display, main_window);
    GC gc = XCreateGC(display, icon_window, 0, NULL);
    draw(display, icon_window, gc, screen);
    for (;;) {
        XEvent event;
        XNextEvent(display, &event);
        if (event.type == Expose && event.xexpose.count == 0)
            draw(display, icon_window, gc, screen);
        if (event.type == DestroyNotify) break;
    }
    XFreeGC(display, gc);
    XCloseDisplay(display);
    return 0;
}
