#+AUTHOR: Mètsàtron

This module owns the generated NeXTSTEP Desktop Scheme family.  The source
palettes remain the seven checked-in OneStepBack GTK themes and twenty QTStep
Xfwm4 themes.  QTStep OpenStep supplies the NeXTSTEP pixel geometry; the
Rhapsody and Highlights families share one Platinum-shaped geometry, while
Platinum reuses the existing Tosh-derived border assets.  Menu and session
payloads from donor themes are never copied.  Generated payloads are
reproducible artifacts, never hand-edited.

The complete matrix is four border styles by twenty-seven colour skews:
=NeXTSTEP=, =Rhapsody=, =Highlights=, and =Platinum=.  The first three styles
carry IceWM's right-edge Window Maker dock; Platinum deliberately omits it.

* Theme and scheme generator

Every generated IceWM theme receives a clean declaration assembled by the
builder.  Donor theme declarations, menus, gradients, and session files are
not propagated.

def tiff_library() -> ctypes.CDLL:
    name = ctypes.util.find_library("tiff")
    if not name:
        raise SystemExit("libtiff is required to build the NeXTSTEP icon theme")
    lib = ctypes.CDLL(name)
    lib.TIFFOpen.argtypes = [ctypes.c_char_p, ctypes.c_char_p]
    lib.TIFFOpen.restype = ctypes.c_void_p
    lib.TIFFClose.argtypes = [ctypes.c_void_p]
    lib.TIFFGetField.argtypes = [ctypes.c_void_p, ctypes.c_uint32, ctypes.c_void_p]
    lib.TIFFGetField.restype = ctypes.c_int
    lib.TIFFScanlineSize64.argtypes = [ctypes.c_void_p]
    lib.TIFFScanlineSize64.restype = ctypes.c_uint64
    lib.TIFFReadScanline.argtypes = [ctypes.c_void_p, ctypes.c_void_p,
                                     ctypes.c_uint32, ctypes.c_uint16]
    lib.TIFFReadScanline.restype = ctypes.c_int
    lib.TIFFReadRGBAImageOriented.argtypes = [ctypes.c_void_p, ctypes.c_uint32,
                                              ctypes.c_uint32,
                                              ctypes.POINTER(ctypes.c_uint32),
                                              ctypes.c_int, ctypes.c_int]
    lib.TIFFReadRGBAImageOriented.restype = ctypes.c_int
    return lib


def decode_nextspace_tiff(lib: ctypes.CDLL, path: Path) -> tuple[int, int, bytes]:
    handle = lib.TIFFOpen(str(path).encode(), b"r")
    if not handle:
        raise SystemExit(f"cannot open NextSpace icon: {path}")
    try:
        width = ctypes.c_uint32()
        height = ctypes.c_uint32()
        if not lib.TIFFGetField(handle, 256, ctypes.byref(width)) or not lib.TIFFGetField(
                handle, 257, ctypes.byref(height)):
            raise SystemExit(f"NextSpace icon has no dimensions: {path}")
        w, h = width.value, height.value
        scanline_size = lib.TIFFScanlineSize64(handle)

        # NeXT's original Workspace art stores 2-bit grayscale and 2-bit alpha
        # in separate planes. Generic TIFF RGBA readers reject that historical
        # layout, but libtiff still decompresses each packed scanline correctly.
        if scanline_size == (w + 3) // 4:
            planes: list[list[int]] = []
            row = (ctypes.c_ubyte * scanline_size)()
            for sample in range(2):
                values: list[int] = []
                for y in range(h):
                    if lib.TIFFReadScanline(handle, row, y, sample) < 0:
                        raise SystemExit(f"cannot decode NextSpace icon row: {path}")
                    packed = bytes(row)
                    values.extend((packed[x // 4] >> (6 - 2 * (x % 4))) & 3 for x in range(w))
                planes.append(values)
            rgba = bytearray()
            for grey, alpha in zip(planes[0], planes[1]):
                rgba.extend((grey * 85, grey * 85, grey * 85, alpha * 85))
            return w, h, bytes(rgba)

        raster = (ctypes.c_uint32 * (w * h))()
        if not lib.TIFFReadRGBAImageOriented(handle, w, h, raster, 1, 0):
            raise SystemExit(f"cannot decode NextSpace icon: {path}")
        return w, h, bytes(raster)
    finally:
        lib.TIFFClose(handle)


def nearest_rgba(width: int, height: int, rgba: bytes, size: int) -> bytes:
    output = bytearray(size * size * 4)
    for y in range(size):
        source_y = min(height - 1, y * height // size)
        for x in range(size):
            source_x = min(width - 1, x * width // size)
            source = (source_y * width + source_x) * 4
            target = (y * size + x) * 4
            output[target:target + 4] = rgba[source:source + 4]
    return bytes(output)


def png_chunk(kind: bytes, data: bytes) -> bytes:
    return struct.pack(">I", len(data)) + kind + data + struct.pack(">I", zlib.crc32(kind + data))


def write_png(path: Path, size: int, rgba: bytes) -> None:
    rows = b"".join(b"\0" + rgba[row * size * 4:(row + 1) * size * 4]
                    for row in range(size))
    payload = (b"\x89PNG\r\n\x1a\n"
               + png_chunk(b"IHDR", struct.pack(">IIBBBBB", size, size, 8, 6, 0, 0, 0))
               + png_chunk(b"IDAT", zlib.compress(rows, 9))
               + png_chunk(b"IEND", b""))
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(payload)


def build_icon_themes(repo: Path) -> None:
    archive_root = (Path.home() / "HelmCortex/NEXUS/sanctuaries/"
                    "sanctuary-redstone-9x/themes")
    nextspace_source = archive_root / "nextspace-icons"
    gnustep_source = archive_root / "gnustep-icons"
    for label, source in (("NextSpace", nextspace_source),
                          ("GNUstep", gnustep_source)):
        if not (source / "PROVENANCE.md").is_file():
            raise SystemExit(f"sanctioned {label} icon archive missing: {source}")

    output = repo / "linux/.local/share/icons/NeXTSTEP"
    if output.exists() or output.is_symlink():
        if output.is_dir() and not output.is_symlink():
            shutil.rmtree(output)
        else:
            output.unlink()
    output.mkdir(parents=True)

    lib = tiff_library()
    directories: set[str] = set()
    for relative, contexts in NEXTSPACE_ICON_SOURCES.items():
        icon_source = nextspace_source / relative
        if not icon_source.is_file():
            raise SystemExit(f"archived NextSpace icon missing: {icon_source}")
        width, height, rgba = decode_nextspace_tiff(lib, icon_source)
        for size in ICON_SIZES:
            scaled = nearest_rgba(width, height, rgba, size)
            for context, names in contexts.items():
                directory = f"{size}x{size}/{context}"
                directories.add(directory)
                for name in names:
                    write_png(output / directory / f"{name}.png", size, scaled)

    # The X230's historical GNUstep theme is the broad, already-owned shell
    # vocabulary: applications, categories, devices, MIME types, places, and
    # status icons.  Its original index inherited Breeze and its actions layer
    # contained KDE-specific artwork, so neither is imported.  Copy only the
    # GNUstep-styled contexts and emit our own Irixium-backed index below.
    for context in GNUSTEP_ICON_CONTEXTS:
        context_root = gnustep_source / context
        if not context_root.is_dir():
            raise SystemExit(f"archived GNUstep context missing: {context_root}")
        for size_root in sorted(context_root.iterdir()):
            if not size_root.is_dir() or not size_root.name.isdigit():
                continue
            directory = f"{size_root.name}x{size_root.name}/{context}"
            directories.add(directory)
            destination_dir = output / directory
            destination_dir.mkdir(parents=True, exist_ok=True)
            for icon_source in size_root.glob("*.png"):
                shutil.copy2(icon_source, destination_dir / icon_source.name)

    # The CDE Desktop Schemes already solve Redstone's broad application
    # vocabulary through Irixium.  Preserve that architecture: do not copy its
    # artwork into this theme and never collapse unrelated names onto a generic
    # glyph.  NeXT/GNUstep wins where it has deliberate art; Irixium then fills
    # each missing name with its distinct historical icon.  The existing graft
    # helper supplies Chicago95 as the final IceWM safety net.
    irixium_index = repo / "linux/.local/share/icons/Irixium/index.theme"
    if not irixium_index.is_file():
        raise SystemExit(f"CDE/Irixium fallback theme missing: {irixium_index}")

    ordered = sorted(directories)
    lines = [
        "[Icon Theme]", "Name=NeXTSTEP",
        "Comment=NextSpace and GNUstep icons over the Redstone CDE/Irixium fallback",
        "Inherits=Irixium", f"Directories={','.join(ordered)}", "",
    ]
    contexts = dict(GNUSTEP_ICON_CONTEXTS)
    contexts["actions"] = "Actions"
    for directory in ordered:
        size_text, context = directory.split("/")
        lines.extend([f"[{directory}]", f"Size={size_text.split('x')[0]}",
                      f"Context={contexts[context]}", "Type=Fixed", ""])
    (output / "index.theme").write_text("\n".join(lines), encoding="utf-8")
    shutil.copy2(nextspace_source / "LICENSE", output / "LICENSE")
    (output / "PROVENANCE").write_text(
        "NeXTSTEP XDG icon theme\n"
        f"Primary shell vocabulary: DotCortex GNUstep pack at commit {GNUSTEP_ICON_COMMIT}\n"
        f"Additional original artwork: NextSpace commit {NEXTSPACE_COMMIT}\n"
        "Archives: HelmCortex/NEXUS/sanctuaries/sanctuary-redstone-9x/themes/"
        "{gnustep-icons,nextspace-icons}\n"
        "Fallback: Irixium, the icon theme used by Redstone's CDE schemes\n"
        "Policy: historical GNUstep actions and Breeze inheritance excluded; "
        "missing names remain distinct and resolve through Irixium\n"
        "Scaling: NextSpace TIFF aliases use nearest-neighbour; GNUstep PNGs "
        "retain their original sizes\n",
        encoding="utf-8",
    )

    platinum_source = repo / "linux/.local/share/icons/Platinum9/NineIcons"
    platinum_link = repo / "linux/.local/share/icons/NineIcons"
    if not (platinum_source / "index.theme").is_file():
        raise SystemExit(f"Platinum NineIcons source missing: {platinum_source}")
    if platinum_link.exists() and not platinum_link.is_symlink():
        raise SystemExit(f"refusing to replace non-symlink Platinum projection: {platinum_link}")
    if platinum_link.is_symlink():
        platinum_link.unlink()
    platinum_link.symlink_to("Platinum9/NineIcons", target_is_directory=True)


def rgb(value: str) -> tuple[int, int, int]:
    value = value.lstrip("#")
    return tuple(int(value[index:index + 2], 16) for index in (0, 2, 4))


def colour(values: tuple[int, int, int]) -> str:
    return "#" + "".join(f"{max(0, min(255, round(value))):02x}" for value in values)


def mix(left: str, right: str, amount: float) -> str:
    a, b = rgb(left), rgb(right)
    return colour(tuple(x + (y - x) * amount for x, y in zip(a, b)))


def readable(background: str) -> str:
    red, green, blue = rgb(background)
    return "#000000" if red * 299 + green * 587 + blue * 114 >= 128000 else "#ffffff"


def ice(value: str) -> str:
    value = value.lstrip("#")
    return f"rgb:{value[0:2]}/{value[2:4]}/{value[4:6]}"


def safe_name(value: str) -> str:
    return re.sub(r"[^A-Za-z0-9]+", "-", value).strip("-")

def replace_key(text: str, key: str, value: str) -> str:
    line = f'{key}="{value}"'
    pattern = re.compile(rf"^{re.escape(key)}=.*$", re.MULTILINE)
    return pattern.sub(line, text) if pattern.search(text) else text.rstrip() + "\n" + line + "\n"


def recolour_xpm(path: Path, darkest: str, face: str, lightest: str) -> None:
    text = path.read_text(encoding="latin-1")
    lines = text.splitlines()
    header = next((index for index, line in enumerate(lines) if XPM_HEADER.match(line)), None)
    if header is None:
        return
    match = XPM_HEADER.match(lines[header])
    assert match is not None
    count = int(match.group(3))
    for index in range(header + 1, header + 1 + count):
        found = re.search(r"\bc\s+(#[0-9A-Fa-f]{6})", lines[index])
        if not found:
            continue
        old = found.group(1)
        red, green, blue = rgb(old)
        luma = (red * 299 + green * 587 + blue * 114) / 255000
        new = mix(darkest, face, luma * 2) if luma < 0.5 else mix(face, lightest, (luma - 0.5) * 2)
        lines[index] = lines[index].replace(old, new)
    path.write_text("\n".join(lines) + "\n", encoding="latin-1")


def write_title(path: Path, top: str, bottom: str, accent: str | None = None) -> None:
    shades = [mix(top, bottom, row / 18) for row in range(19)]
    if accent:
        shades[-2:] = [accent, accent]
    unique = list(dict.fromkeys(shades))
    symbols = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
    table = {value: symbols[index] for index, value in enumerate(unique)}
    rows = [table[value] for value in shades]
    body = ["/* XPM */", "static char *title[] = {", f'"1 19 {len(unique)} 1",']
    body.extend(f'"{table[value]} c {value}",' for value in unique)
    body.extend(f'"{row}",' for row in rows[:-1])
    body.append(f'"{rows[-1]}"}};')
    path.write_text("\n".join(body) + "\n", encoding="ascii")


def glyph_points(kind: str, pressed: bool) -> set[tuple[int, int]]:
    offset = 1 if pressed else 0
    points: set[tuple[int, int]] = set()
    def line(x1: int, y1: int, x2: int, y2: int) -> None:
        steps = max(abs(x2 - x1), abs(y2 - y1))
        for step in range(steps + 1):
            x = round(x1 + (x2 - x1) * step / max(steps, 1)) + offset
            y = round(y1 + (y2 - y1) * step / max(steps, 1)) + offset
            points.add((x, y))
    if kind == "close":
        line(5, 5, 10, 10); line(10, 5, 5, 10)
    elif kind == "maximize":
        line(4, 4, 11, 4); line(4, 4, 4, 11); line(11, 4, 11, 11); line(4, 11, 11, 11)
    elif kind == "minimize":
        line(4, 10, 11, 10)
    elif kind == "restore":
        line(3, 6, 9, 6); line(3, 6, 3, 12); line(3, 12, 9, 12)
        line(6, 3, 12, 3); line(12, 3, 12, 9); line(6, 9, 12, 9)
    elif kind == "rollup":
        line(4, 10, 8, 6); line(8, 6, 12, 10)
    elif kind == "rolldown":
        line(4, 6, 8, 10); line(8, 10, 12, 6)
    else:
        line(4, 5, 11, 5); line(4, 8, 11, 8); line(4, 11, 11, 11)
    return points


def write_button(path: Path, face: str, glyph: str, kind: str, flat: bool) -> None:
    shadow = face if flat else mix(face, "#000000", 0.45)
    highlight = face if flat else mix(face, "#ffffff", 0.55)
    symbols = {face: ".", glyph: "#", shadow: "s", highlight: "h"}
    colors = list(dict.fromkeys(symbols))
    body = ["/* XPM */", "static char *button[] = {", f'"16 38 {len(colors)} 1",']
    body.extend(f'"{symbols[value]} c {value}",' for value in colors)
    rows: list[str] = []
    for pressed in (False, True):
        marks = glyph_points(kind, pressed)
        for y in range(19):
            row = []
            for x in range(16):
                if (x, y) in marks:
                    row.append(symbols[glyph])
                elif not flat and (x == 1 or y == 1):
                    row.append(symbols[highlight if not pressed else shadow])
                elif not flat and (x == 14 or y == 17):
                    row.append(symbols[shadow if not pressed else highlight])
                else:
                    row.append(symbols[face])
            rows.append("".join(row))
    body.extend(f'"{row}",' for row in rows[:-1])
    body.append(f'"{rows[-1]}"}};')
    path.write_text("\n".join(body) + "\n", encoding="ascii")

def duplicate_xpm_state(source: Path, destination: Path) -> None:
    """Translate one 19px Xfwm4 button into IceWM's normal/pressed stack."""
    lines = source.read_text(encoding="latin-1").splitlines()
    header = next((index for index, line in enumerate(lines) if XPM_HEADER.match(line)), None)
    if header is None:
        raise SystemExit(f"Xfwm4 button has no XPM header: {source}")
    match = XPM_HEADER.match(lines[header])
    assert match is not None
    width, height, colours, cpp = (int(match.group(index)) for index in range(1, 5))
    palette = lines[header + 1:header + 1 + colours]
    pixels = lines[header + 1 + colours:header + 1 + colours + height]
    if len(pixels) != height:
        raise SystemExit(f"Xfwm4 button has incomplete pixels: {source}")
    body = ["/* XPM */", "static char *icewm_button[] = {",
            f'"{width} {height * 2} {colours} {cpp}",']
    body.extend(line.rstrip(",") + "," for line in palette)
    doubled = pixels + pixels
    body.extend(line.rstrip(",") + "," for line in doubled[:-1])
    body.append(doubled[-1].rstrip(",") + "};")
    destination.write_text("\n".join(body) + "\n", encoding="latin-1")


def write_nextstep_pixels(repo: Path, output: Path) -> None:

    if not (source / "preview.xpm").is_file():
        raise SystemExit(f"QTStep OpenStep pixel reference missing: {source}")
    for state, suffix in (("active", "A"), ("inactive", "I")):
        title_source = source / f"title-3-{state}.xpm"
        for stem in ("L", "M", "R", "T", "B", "P", "S"):
            shutil.copy2(title_source, output / f"title{suffix}{stem}.xpm")
        border = "#000000"
        for frame in ("T", "TL", "TR", "L", "R", "B", "BL", "BR"):
            write_solid_xpm(output / f"frame{suffix}{frame}.xpm", border)
            write_solid_xpm(output / f"dframe{suffix}{frame}.xpm", border)

    button_sources = {
        "close": "close", "minimize": "hide", "maximize": "maximize",
        "restore": "maximize-toggled", "rollup": "shade",
        "rolldown": "shade-toggled", "menuButton": "menu",
    }
    for stem, source_stem in button_sources.items():
        duplicate_xpm_state(source / f"{source_stem}-active.xpm", output / f"{stem}A.xpm")
        duplicate_xpm_state(source / f"{source_stem}-inactive.xpm", output / f"{stem}I.xpm")


def write_shaped_pixels(output: Path, style: str, row: dict[str, str]) -> None:
    """Author coherent Rhapsody-shaped chrome without donor menu baggage."""
    active_top, active_bottom = "#bebebe", "#9e9e9e"
    inactive_top = inactive_bottom = "#d8d8d8"
    border, face, light = row["dark"], row["face"], row["light"]
    for state, suffix, colour_value in (
            ("active", "A", border), ("inactive", "I", mix(border, "#ffffff", 0.25))):
        del state
        for frame in ("T", "TL", "TR", "L", "R", "B", "BL", "BR"):
            write_solid_xpm(output / f"frame{suffix}{frame}.xpm", colour_value)
            write_solid_xpm(output / f"dframe{suffix}{frame}.xpm", colour_value)
    for stem in ("L", "M", "R", "T", "B", "P", "S"):
        write_title(output / f"titleA{stem}.xpm", active_top, active_bottom)
        write_title(output / f"titleI{stem}.xpm", inactive_top, inactive_bottom)
    kinds = {
        "close": "close", "maximize": "maximize", "minimize": "minimize",
        "restore": "restore", "rollup": "rollup", "rolldown": "rolldown",
        "menuButton": "menu",
    }
    for stem, kind in kinds.items():
        write_button(output / f"{stem}A.xpm", active_bottom, readable(active_bottom), kind, False)
        write_button(output / f"{stem}I.xpm", inactive_bottom, readable(inactive_bottom), kind, False)

def generate_qt_gtk(repo: Path, row: dict[str, str]) -> None:
    if not row["slug"].startswith("qtstep-"):
        return
    donor = repo / "linux/.themes/OneStepBack"
    output = repo / "linux/.themes" / row["gtk"]
    if output.exists():
        shutil.rmtree(output)
    shutil.copytree(donor, output, symlinks=True, ignore=shutil.ignore_patterns(".git"))
    roles = {
        "white_color": "#ffffff", "lighter_color": row["light"],
        "base_color": row["face"], "darker_color": row["dark"],
        "black_color": "#000000", "scale_color": row["accent"],
    }
    for relative in ("gtk-3.0/gtk.css", "gtk-4.0/gtk.css"):
        path = output / relative
        text = path.read_text(encoding="utf-8")
        for key, value in roles.items():
            text = re.sub(rf"^@define-color\s+{key}\s+[^;]+;", f"@define-color {key} {value};", text, flags=re.MULTILINE)
        path.write_text(text, encoding="utf-8")
    gtkrc = output / "gtk-2.0/gtkrc"
    if gtkrc.is_file():
        text = gtkrc.read_text(encoding="utf-8")
        gtk2_roles = dict(roles)
        gtk2_roles.update({
            "button_selected_bg_color": row["dark"],
            "text_selected_bg_color": row["accent"],
            "text_selected_fg_color": readable(row["accent"]),
            "menuitem_selected_bg_color": row["accent"],
            "menuitem_selected_fg_color": readable(row["accent"]),
            "listitem_selected_bg_color": row["accent"],
            "listitem_selected_fg_color": readable(row["accent"]),
        })
        for key, value in gtk2_roles.items():
            text = re.sub(
                rf"(\b{re.escape(key)}\s*:\s*)#[0-9A-Fa-f]{{6}}",
                rf"\g<1>{value}",
                text,
            )
        gtkrc.write_text(text, encoding="utf-8")
    index = output / "index.theme"
    if index.is_file():
        text = index.read_text(encoding="utf-8")
        text = re.sub(r"^Name=.*$", f"Name={row['gtk']}", text, flags=re.MULTILINE)
        text = re.sub(r"^GtkTheme=.*$", f"GtkTheme={row['gtk']}", text, flags=re.MULTILINE)
        index.write_text(text, encoding="utf-8")


def qt6ct(path: Path, row: dict[str, str]) -> None:
    face, light, dark, accent = row["face"], row["light"], row["dark"], row["accent"]
    text = readable(face)
    selected = readable(accent)
    colors = [text, face, light, light, dark, dark, text, "#ffffff", text, "#ffffff", face,
              text, accent, selected, "#0000ff", "#800080", light, text, "#ffffe1", "#000000",
              dark, accent]
    encoded = ", ".join("#ff" + value.lstrip("#") for value in colors)
    path.write_text(f"[ColorScheme]\nactive_colors={encoded}\ndisabled_colors={encoded}\ninactive_colors={encoded}\n", encoding="ascii")


def wine(path: Path, row: dict[str, str]) -> None:
    def dec(value: str) -> str:
        return " ".join(str(part) for part in rgb(value))
    face, light, dark, accent = row["face"], row["light"], row["dark"], row["accent"]
    values = {
        "ActiveBorder": face, "ActiveTitle": row["active"], "AppWorkspace": dark,
        "Background": dark, "ButtonDkShadow": "#000000", "ButtonFace": face,
        "ButtonHilight": light, "ButtonLight": light, "ButtonShadow": dark,
        "ButtonText": readable(face), "GradientActiveTitle": row["active"],
        "GradientInactiveTitle": row["inactive"], "GrayText": dark, "Hilight": accent,
        "HilightText": readable(accent), "InactiveBorder": face,
        "InactiveTitle": row["inactive"], "InactiveTitleText": readable(row["inactive"]),
        "InfoText": "#000000", "InfoWindow": "#ffffe1", "Menu": face,
        "MenuText": readable(face), "Scrollbar": face, "TitleText": readable(row["active"]),
        "Window": light, "WindowFrame": "#000000", "WindowText": readable(light),
    }
    lines = ["REGEDIT4", "", r"[HKEY_CURRENT_USER\Control Panel\Colors]"]
    lines.extend(f'"{key}"="{dec(value)}"' for key, value in values.items())
    path.write_text("\n".join(lines) + "\n", encoding="ascii")

def missing_icewm_icons(repo: Path) -> list[str]:
    graft_path = repo / "linux/.local/bin/redstone-9x-icons-graft"
    graft = graft_path.read_text(encoding="utf-8")
    names_match = re.search(r'^NAMES="(.*?)"\nSIZES=', graft, re.MULTILINE | re.DOTALL)
    if names_match is None:
        raise SystemExit("cannot read IceWM graft vocabulary")
    required = names_match.group(1).split()
    icon_root = repo / "linux/.local/share/icons"
    with tempfile.TemporaryDirectory(prefix="redstone-nextstep-icons-") as temporary:
        subprocess.run(
            [str(graft_path), str(icon_root), temporary, "NeXTSTEP"],
            check=True,
        )
        destination = Path(temporary)
        return [f"{name}@{size}" for name in required for size in ICON_SIZES
                if not (destination / f"{name}_{size}x{size}.png").is_file()]


def build_all(repo: Path) -> None:
    data = json.loads((repo / "linux/.config/redstone-9x/nextstep-palettes.json").read_text())

    build_icon_themes(repo)
    missing = missing_icewm_icons(repo)
    if missing:
        raise SystemExit(f"NeXTSTEP trial still misses IceWM icon names: {missing}")

    base_row = rows[0]
    for style in STYLES:
        label, token, _ = STYLES[style]
        build_theme(repo, repo / "linux/.icewm/themes" / f"NextStep-{token}-base", style, base_row, f"{label} Template Base")

        qt6ct(qt / f"nextstep-{row['slug']}.conf", row)

            build_theme(repo, repo / "linux/.icewm/themes" / f"NextStep-{token}-{row['token']}", style, row, f"{label} - {row['label']}")

def build_trial(repo: Path, trial_spec: str) -> None:

    try:
        style, slug = trial_spec.split(":", 1)

    if style not in STYLES:
        raise SystemExit(f"unknown trial style: {style}")

        raise SystemExit(f"unknown trial palette slug: {slug}")

    scheme(repo / "linux/.config/redstone-9x/schemes" / f"nextstep-{style}-{row['slug']}.scheme",
           style, row)
    print(f"built representative trial: {output.name}")


def check(repo: Path) -> None:
    schemes = []
    for style in STYLES:
        style_schemes = list((repo / "linux/.config/redstone-9x/schemes").glob(f"nextstep-{style}-*.scheme"))
        if len(style_schemes) != 27:
            raise SystemExit(f"expected 27 {style} schemes, found {len(style_schemes)}")
        schemes.extend(style_schemes)
    generated = [path for path in (repo / "linux/.icewm/themes").glob("NextStep-*") if path.name != "NextStep-NeXT-base" or (path / "PROVENANCE").is_file()]
    expected_themes = 112
    if len(schemes) != 108:
        raise SystemExit(f"expected 108 matrix schemes, found {len(schemes)}")
    if len(generated) != expected_themes:
        raise SystemExit(f"expected {expected_themes} generated/base themes, found {len(generated)}")
    icon_roots = {
        "NeXTSTEP": repo / "linux/.local/share/icons/NeXTSTEP/index.theme",
        "NineIcons": repo / "linux/.local/share/icons/NineIcons/index.theme",
    }
    for name, index in icon_roots.items():
        if not index.is_file():
            raise SystemExit(f"{name} icon theme is not guest-projectable: {index}")
    nextstep_index = icon_roots["NeXTSTEP"].read_text(encoding="utf-8")
    if "Inherits=Irixium" not in nextstep_index:
        raise SystemExit("NeXTSTEP icon theme lost its CDE/Irixium fallback")
    if re.search(r"breeze|kde", nextstep_index, re.IGNORECASE):
        raise SystemExit("NeXTSTEP icon index contains KDE/Breeze contamination")
    nextstep_icons = list(icon_roots["NeXTSTEP"].parent.rglob("*.png"))
    if len(nextstep_icons) < 500:
        raise SystemExit(f"NeXTSTEP icon theme is incomplete: {len(nextstep_icons)} PNGs")
    icon_provenance = (icon_roots["NeXTSTEP"].parent / "PROVENANCE").read_text(encoding="utf-8")
    if "Fallback: Irixium" not in icon_provenance:
        raise SystemExit("NeXTSTEP icon theme lost its CDE/Irixium fallback provenance")
    missing = missing_icewm_icons(repo)
    if missing:
        raise SystemExit(f"NeXTSTEP icon theme misses IceWM graft names: {missing}")
    forbidden_tosh = {"menubg.xpm", "logoutbg.xpm", "switchbg.xpm"}
    for theme in generated:
        contamination = forbidden_tosh.intersection(path.name for path in theme.iterdir())
        if contamination:
            raise SystemExit(f"{theme}: Tosh/Mac payload contamination: {sorted(contamination)}")
        provenance = (theme / "PROVENANCE").read_text(encoding="utf-8")
        if "no donor menus" not in provenance:
            raise SystemExit(f"{theme}: missing clean payload policy provenance")
    for path in schemes:
        text = path.read_text(encoding="utf-8")
        for key in ("SCHEME_TITLE", "ICEWM_THEME", "GTK_THEME", "ICON_THEME", "ICEWM_PREFS"):
            if not re.search(rf"^{key}=", text, re.MULTILINE):
                raise SystemExit(f"{path}: missing {key}")
        sourced = subprocess.run(
            ["bash", "-c", 'set -euo pipefail; source "$1"; printf "%s\\n" "$ICEWM_PREFS"',
             "redstone-nextstep-check", str(path)],
            check=False,
            capture_output=True,
            text=True,
        )
        if sourced.returncode != 0:
            raise SystemExit(f"{path}: cannot be sourced by bash: {sourced.stderr.strip()}")
        if 'NetworkStatusDevice="en* wl* wg* tailscale*"' not in sourced.stdout:
            raise SystemExit(f"{path}: ICEWM_PREFS lost its quoted network-device glob")
        style = path.name.split("-", 2)[1]
        expected_icon = "NeXTSTEP" if style == "nextstep" else "NineIcons"
        if f'ICON_THEME="{expected_icon}"' not in text:
            raise SystemExit(f"{path}: expected ICON_THEME={expected_icon}")
        if "KDE-1-Classic" in text:
            raise SystemExit(f"{path}: KDE icon theme contamination")
    print("verified 108 NeXTSTEP Desktop Schemes, 112 generated/base theme directories, "
          f"{len(nextstep_icons)} curated NeXTSTEP icons with Irixium fallback, "
          "and the NeXTSTEP/NineIcons family split")

* IceWM menu generator

The top-level IceWM menu contains one dynamic NeXTSTEP submenu.  This helper
emits four nested style menus and reads their titles from the deployed scheme
declarations, so the visible menu and the actual 108 runnable schemes cannot
drift apart.

#+BEGIN_SRC bash :tangle linux/.local/bin/redstone-9x-nextstep-menu :shebang "#!/bin/bash" :tangle-mode (identity #o755)
set -uo pipefail

scheme_dir="${HOME}/.config/redstone-9x/schemes"
for spec in "nextstep:NeXTSTEP Style" "rhapsody:Rhapsody Style" \
            "highlights:Highlights Style" "platinum:Platinum Style"; do
  style="${spec%%:*}"
  label="${spec#*:}"
  printf 'menu "%s" preferences-desktop-theme {\n' "${label}"
  while IFS= read -r path; do
    scheme_id="$(basename "${path}" .scheme)"
    scheme_title="$(sed -n 's/^SCHEME_TITLE="\(.*\)"$/\1/p' "${path}" | head -1)"
    entry="${scheme_title#* — }"
    printf '  prog "%s" preferences-desktop-theme redstone-9x-scheme-apply %s\n' "${entry}" "${scheme_id}"
  done < <(find -L "${scheme_dir}" -maxdepth 1 -type f -name "nextstep-${style}-*.scheme" -print | sort)
  printf '}\n'
done

* Dockapp lifecycle

The manager records exact PIDs with their display and never kills by name.
That keeps concurrent sanctuary guests in the shared container PID namespace
out of reach.  Scheme apply calls =sync 1= for NeXTSTEP, Rhapsody, and
Highlights, and =sync 0= for Platinum and every non-NeXT scheme.

#+BEGIN_SRC bash :tangle linux/.local/bin/redstone-9x-nextstep-dockapps :shebang "#!/bin/bash" :tangle-mode (identity #o755)
set -uo pipefail

state="${HOME}/.cache/redstone-9x/nextstep-dockapps.state"
log="${HOME}/.cache/redstone-9x/nextstep-dockapps.log"
mkdir -p "$(dirname "${state}")"

stop_dockapps() {
  [[ -r "${state}" ]] || return 0
  while IFS='|' read -r recorded_display pid command; do
    [[ "${recorded_display}" == "${DISPLAY:-}" ]] || continue
    [[ "${pid}" =~ ^[0-9]+$ ]] || continue
    kill -0 "${pid}" 2>/dev/null || continue
    current="$(tr '\0' ' ' < "/proc/${pid}/cmdline" 2>/dev/null || true)"
    [[ "${current}" == *"${command}"* ]] && kill "${pid}" 2>/dev/null || true
  done < "${state}"
  rm -f "${state}"
}

wait_for_icedock() {
  local attempt
  for attempt in {1..30}; do
    if xwininfo -root -tree 2>/dev/null | grep -q '"IceDock"'; then
      return 0
    fi
    sleep 0.1
  done
  printf 'IceDock was not ready on %s\n' "${DISPLAY:-}" >> "${log}"
  return 1
}

start_dockapps() {
  stop_dockapps
  : > "${state}"
  : > "${log}"
  wait_for_icedock || return 1
  local -a specs=(
    'nextstep-dock'
    'wmclock|-24|-led|#d6d6d6'
    "wmcpuload|--display|${DISPLAY:-:0}|--light-color|#aeb8bf"
    "wmbattery|-d|${DISPLAY:-:0}|-e"
    "wmnd|-d|${DISPLAY:-:0}|-F|-q|-c|#d6d6d6|-C|#8fa9bd|-L|#000000"
    'wmamixer|-w|-s|-b|#000000|-l|#d6d6d6'
  )
  local spec command binary pid
  local -a argv
  for spec in "${specs[@]}"; do
    IFS='|' read -r -a argv <<< "${spec}"
    command="${argv[0]}"
    binary="$(command -v "${command}" 2>/dev/null || true)"
    if [[ -z "${binary}" ]]; then
      printf 'missing dockapp: %s\n' "${command}" >> "${log}"
      continue
    fi
    "${binary}" "${argv[@]:1}" >> "${log}" 2>&1 &
    pid=$!
    sleep 0.1
    if kill -0 "${pid}" 2>/dev/null; then
      printf '%s|%s|%s\n' "${DISPLAY:-}" "${pid}" "${command}" >> "${state}"
    else
      printf 'dockapp exited during startup: %s\n' "${command}" >> "${log}"
    fi
  done
}

case "${1:-}" in
  start) start_dockapps ;;
  stop) stop_dockapps ;;
  sync)
    if [[ "${2:-0}" == "1" ]]; then start_dockapps; else stop_dockapps; fi
    ;;
  *) printf 'usage: redstone-9x-nextstep-dockapps start|stop|sync 0|1\n' >&2; exit 2 ;;
esac

* NeXT logo dock tile

The original GNUstep plan named a custom tile but never contained the claimed C
source.  This is the missing implementation: a conventional withdrawn X11
dockapp with a 64-pixel icon window.  Its embedded 48-pixel artwork is the
=NeXTSTEP-cube.xpm= carried by the GNUstep sanctuary's Window Maker icon pool;
embedding it keeps the Redstone package reproducible and removes any runtime
dependency on the GNUstep guest home.

static unsigned long named_pixel(Display *display, int screen, const char *name) {
    XColor exact;
    XColor allocated;
    Colormap colormap = DefaultColormap(display, screen);
    if (XAllocNamedColor(display, colormap, name, &allocated, &exact))
        return allocated.pixel;
    return BlackPixel(display, screen);
}

static void draw(Display *display, Window window, GC gc, int screen) {
    unsigned long black = BlackPixel(display, screen);
    unsigned long white = WhitePixel(display, screen);
    unsigned long colours[256] = {0};
    colours[(unsigned char)'.'] = named_pixel(display, screen, "#05070A");
    colours[(unsigned char)'+'] = named_pixel(display, screen, "#E6EBEE");
    colours[(unsigned char)'@'] = named_pixel(display, screen, "#AAB5BC");
    colours[(unsigned char)'#'] = named_pixel(display, screen, "#77838C");
    colours[(unsigned char)'$'] = named_pixel(display, screen, "#2C373F");
    colours[(unsigned char)'&'] = white;
    XSetForeground(display, gc, black);
    XFillRectangle(display, window, gc, 0, 0, 64, 64);
    XSetForeground(display, gc, white);
    XDrawRectangle(display, window, gc, 1, 1, 61, 61);
    for (int y = 0; y < 48; ++y) {
        for (int x = 0; x < 48; ++x) {
            unsigned char code = (unsigned char)cube[y][x];
            if (code == ' ') continue;
            XSetForeground(display, gc, colours[code]);
            XDrawPoint(display, window, gc, x + 8, y + 8);
        }
    }
}

int main(void) {
    Display *display = XOpenDisplay(NULL);
    if (!display) return 1;
    int screen = DefaultScreen(display);
    Window root = RootWindow(display, screen);
    Window main_window = XCreateSimpleWindow(display, root, 0, 0, 1, 1, 0,
                                              BlackPixel(display, screen),
                                              BlackPixel(display, screen));
    Window icon_window = XCreateSimpleWindow(display, root, 0, 0, 64, 64, 0,
                                              WhitePixel(display, screen),
                                              BlackPixel(display, screen));
    XClassHint class_hint = {"nextstep-dock", "DockApp"};
    XSetClassHint(display, main_window, &class_hint);
    XStoreName(display, main_window, "NeXT");
    XWMHints hints;
    memset(&hints, 0, sizeof(hints));
    hints.flags = StateHint | IconWindowHint | WindowGroupHint;
    hints.initial_state = WithdrawnState;
    hints.icon_window = icon_window;
    hints.window_group = main_window;
    XSetWMHints(display, main_window, &hints);
    XSelectInput(display, icon_window, ExposureMask | StructureNotifyMask);
    XMapWindow(display, main_window);
    GC gc = XCreateGC(display, icon_window, 0, NULL);
    draw(display, icon_window, gc, screen);
    for (;;) {
        XEvent event;
        XNextEvent(display, &event);
        if (event.type == Expose && event.xexpose.count == 0)
            draw(display, icon_window, gc, screen);
        if (event.type == DestroyNotify) break;
    }
    XFreeGC(display, gc);
    XCloseDisplay(display);
    return 0;
}
