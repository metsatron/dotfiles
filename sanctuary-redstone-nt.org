* Overview

Redstone NT Enterprise Server is the kikin-kushi-resident edition of the Redstone
product family.  It shares the full canonical Redstone desktop base (projected by
=redstone-project-common=) with the 9X Professional Edition but owns its own
display transport (Xpra in Phase 2, Sunshine/Moonlight after GPU swap) and runs
on server hardware.

Two tangled scripts:

| Script                          | Tangle target                                      | Role                                                  |
|---------------------------------|----------------------------------------------------|-------------------------------------------------------|
| =sanctuary-redstone-nt-launch=  | =linux/.local/bin/sanctuary-redstone-nt-launch=    | Projection + Xpra lifecycle (start/stop/status)       |
| =sanctuary-redstone-nt-session= | =linux/.local/bin/sanctuary-redstone-nt-session=   | Xpra =--start-child= target: distrobox → IceWM       |

Prerequisites:

#+BEGIN_EXAMPLE
vm-homes-init
distrobox-sanctuaries-create sanctuary-redstone-nt
loom guix:sanctuary-apply  (desktop-common + room-gaming + room-windows-compat)
#+END_EXAMPLE

Entry points:

- On kikin-kushi: =sanctuary-redstone-nt-launch start-full=
- From T480s/other: =sanctuary-enter= picks up the =xpra-redstone-nt= remote profile

* Launcher — sanctuary-redstone-nt-launch

Runs the shared Redstone projection, then manages the Xpra display lifecycle.
Subcommands: =start-full= (projection + display), =stop=, =status=.

The =--start-child= invocation uses =env -i= to strip the host daemon environment,
preventing API key leaks into =podman exec --env== arguments (see
=Sanctuary-Cortex-Env-Hardening.md=).  NT includes this fix from birth; the
cortex retrofit is separate work.

#+BEGIN_SRC bash :tangle linux/.local/bin/sanctuary-redstone-nt-launch :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

if [[ "${1:-}" == "-h" || "${1:-}" == "--help" ]]; then
  cat <<'EOF'
sanctuary-redstone-nt-launch - Redstone NT Enterprise Server (Xpra + IceWM)

Usage:
  sanctuary-redstone-nt-launch start-full   Run projection, start Xpra display
  sanctuary-redstone-nt-launch stop         Stop the Xpra display
  sanctuary-redstone-nt-launch status       Show Xpra session status
  sanctuary-redstone-nt-launch -h|--help    This help

Prerequisites:
  vm-homes-init
  distrobox-sanctuaries-create sanctuary-redstone-nt
  loom guix:sanctuary-apply

Display: Xpra :102 (override: REDSTONE_NT_DISPLAY)
Desktop: IceWM (override: SANCTUARY_DESKTOP)

Normally invoked via: sanctuary-enter (xpra-redstone-nt remote profile)
                  or: sanctuary-redstone-nt-launch start-full (on kikin-kushi)
EOF
  exit 0
fi

DISPLAY_NUM="${REDSTONE_NT_DISPLAY:-:102}"
GEOMETRY="${REDSTONE_NT_GEOMETRY:-1920x1080}"

DESKTOP="${SANCTUARY_DESKTOP:-icewm}"
SANCTUARY="sanctuary-redstone-nt"
GUESTS_DIR="${XDG_DATA_HOME:-$HOME/.local/share}/dotcortex/guests"
GUEST_HOME="${GUESTS_DIR}/${SANCTUARY}/home"
GUIX_PROFILES="${HOME}/.guix-extra-profiles"
FONT_STAGE="${GUESTS_DIR}/${SANCTUARY}/fonts"
RETROPIE_HOST_ROOT="${HOME}/RetroPie"
RETROPIE_CONTAINER_ROOT="/run/dotcortex/retropie"
GAMES_CONTAINER_ROOT="/run/dotcortex/games"
REPO_ROOT="${DOTCORTEX_ROOT:-${HOME}/DotCortex}"
GAMING_INOTIFY_APPLY="${REPO_ROOT}/linux/.local/bin/sanctuary-gaming-inotify-apply"
RETROPIE_STOW="${REPO_ROOT}/linux/.local/bin/sanctuary-retropie-stow"

export REDSTONE_PRODUCT="Mineshaft Redstone NT Enterprise Server"

SESSION="${HOME}/.local/bin/sanctuary-redstone-nt-session"

command -v xpra >/dev/null 2>&1 || { echo "${SANCTUARY}: xpra is required on the host" >&2; exit 127; }

start_full() {
  # ── Shared Redstone projection ──
  . "${REPO_ROOT}/linux/.local/bin/redstone-project-common"

  [[ -x "${GAMING_INOTIFY_APPLY}" ]] || {
    echo "${SANCTUARY}: ERROR — gaming inotify helper missing: ${GAMING_INOTIFY_APPLY}" >&2
    exit 1
  }
  "${GAMING_INOTIFY_APPLY}"
  redstone_preflight_guix_profiles
  prepare_retropie_tree
  [[ -x "${RETROPIE_STOW}" ]] || {
    echo "${SANCTUARY}: ERROR — RetroPie projection helper missing: ${RETROPIE_STOW}" >&2
    exit 1
  }
  SANCTUARY_GUEST_HOME="${GUEST_HOME}" "${RETROPIE_STOW}" --mapping openra-content --apply
  redstone_check_desktop_prereqs

  redstone_project

  sanctuary-trash-shield "${SANCTUARY}" "${HOME}"

  # If display is already live, projection is refreshed but display stays.
  if xpra list 2>/dev/null | grep -Fq "LIVE session at ${DISPLAY_NUM}"; then
    echo "${SANCTUARY}: display ${DISPLAY_NUM} already running — projection refreshed"
    return 0
  fi

  [[ -x "${SESSION}" ]] || { echo "${SANCTUARY}: session helper not found: ${SESSION}" >&2; exit 1; }

  echo "${SANCTUARY}: starting Xpra display ${DISPLAY_NUM} (${GEOMETRY}, desktop=${DESKTOP})…"

  # Env-hardened --start-child: env -i strips the host daemon environment.
  # Only the variables the session script actually needs are passed through.
  # The picom start-child gives Electron CSD apps (Clippy) ARGB blending —
  # without a compositor their transparent windows render black; xpra sets
  # DISPLAY for children and --exit-with-children owns picom's lifetime.
  export GST_PLUGIN_SYSTEM_PATH=/usr/lib/x86_64-linux-gnu/gstreamer-1.0
  xpra start-desktop "${DISPLAY_NUM}" --daemon=yes --exit-with-children=yes \
    --resize-display="${GEOMETRY}" --mdns=no \
    --min-quality=70 --min-speed=40 \
    --auto-refresh-delay=0.5 \
    --video-scaling=off \
    --av-sync=no \
    --compress=1 \
    --start-child="picom --backend xrender --unredir-if-possible" \
    --start-child="env -i HOME=${HOME} USER=${USER} LOGNAME=${USER} DISPLAY=${DISPLAY_NUM} XAUTHORITY=${HOME}/.Xauthority LANG=${LANG:-en_US.UTF-8} TERM=${TERM:-xterm-256color} PATH=${HOME}/.guix-extra-profiles/core/core/bin:${HOME}/.local/bin:/usr/local/bin:/usr/bin:/bin SANCTUARY_DESKTOP=${DESKTOP} DOTCORTEX_ROOT=${REPO_ROOT} REDSTONE_PRODUCT='${REDSTONE_PRODUCT}' XCURSOR_THEME=${SCHEME_CURSOR_THEME:-Chicago95_Cursor_White} XCURSOR_SIZE=${SCHEME_CURSOR_SIZE:-32} XDG_RUNTIME_DIR=${XDG_RUNTIME_DIR:-/run/user/$(id -u)} ${SESSION}"

  echo "${SANCTUARY}: Xpra display ${DISPLAY_NUM} started"
}

case "${1:-}" in
  start-full) start_full ;;
  stop) xpra stop "${DISPLAY_NUM}" 2>/dev/null || true ;;
  status) xpra list ;;
  *) echo "Usage: sanctuary-redstone-nt-launch start-full|stop|status" >&2; exit 2 ;;
esac
#+END_SRC

* Session — sanctuary-redstone-nt-session

The Xpra =--start-child= target.  Enters the Distrobox container and launches
IceWM with the Redstone environment.  Runs inside the =env -i= sandbox set up
by the launcher, so only allowlisted variables are available.

#+BEGIN_SRC bash :tangle linux/.local/bin/sanctuary-redstone-nt-session :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

CONTAINER="sanctuary-redstone-nt"
DESKTOP="${SANCTUARY_DESKTOP:-icewm}"
REPO_ROOT="${DOTCORTEX_ROOT:-${HOME}/DotCortex}"
GUESTS_DIR="${XDG_DATA_HOME:-$HOME/.local/share}/dotcortex/guests"
GUEST_HOME="${GUESTS_DIR}/${CONTAINER}/home"
GUIX_PROFILES="${HOME}/.guix-extra-profiles"

if [[ "${DESKTOP}" != "icewm" ]]; then
  echo "sanctuary-redstone-nt-session: only icewm is supported in Phase 2 (got '${DESKTOP}')" >&2
  exit 1
fi

exec distrobox enter "${CONTAINER}" -- env DISPLAY="${DISPLAY:?Xpra did not provide DISPLAY}" \
  bash -c "
    set -e
    REDSTONE_GUEST_BIN=${GUEST_HOME}/.local/bin
    GUIX_PROFILE=${GUIX_PROFILES}/desktop-common/desktop-common
    . \${GUIX_PROFILE}/etc/profile
    GUIX_PROFILE=${GUIX_PROFILES}/room-gaming/room-gaming
    . \${GUIX_PROFILE}/etc/profile
    export GTK_OVERLAY_SCROLLING=0
    export GTK_CSD=0
    export GTK_USE_PORTAL=0
    export GIO_USE_PORTAL=0
    export NO_AT_BRIDGE=1
    export XCURSOR_THEME=\${XCURSOR_THEME:-Chicago95_Cursor_White}
    export XCURSOR_SIZE=\${XCURSOR_SIZE:-32}
    export XCURSOR_PATH=\${HOME}/.icons:/usr/share/icons:\${XCURSOR_PATH:-}
    export GDK_PIXBUF_MODULE_FILE=\${GUIX_PROFILE}/lib/gdk-pixbuf-2.0/2.10.0/loaders.cache
    GUIX_PROFILE=${GUIX_PROFILES}/room-windows-compat/room-windows-compat
    . \${GUIX_PROFILE}/etc/profile
    export XDG_DATA_DIRS=${GUIX_PROFILES}/room-windows-compat/room-windows-compat/share:${GUIX_PROFILES}/room-gaming/room-gaming/share:${GUIX_PROFILES}/desktop-common/desktop-common/share:${HOME}/.local/share/flatpak/exports/share:/usr/local/share:/usr/share
    export XDG_CONFIG_DIRS=${GUIX_PROFILES}/room-windows-compat/room-windows-compat/etc/xdg:${GUIX_PROFILES}/room-gaming/room-gaming/etc/xdg:${GUIX_PROFILES}/desktop-common/desktop-common/etc/xdg:/etc/xdg
    export GSETTINGS_SCHEMA_DIR=${GUIX_PROFILES}/room-gaming/room-gaming/share/glib-2.0/schemas
    export WINEPREFIX=${GUEST_HOME}/.wine-redstone
    export WINEARCH=win64
    export WINEDEBUG=-all
    export DOTCORTEX_DIR=${REPO_ROOT}
    export REDSTONE_PRODUCT='Mineshaft Redstone NT Enterprise Server'
    export SHELL=${GUIX_PROFILES}/core/core/bin/zsh
    export PATH=\${REDSTONE_GUEST_BIN}:${GUEST_HOME}/.local/ice2k/bin:\${PATH}
    export LD_LIBRARY_PATH=${GUEST_HOME}/.local/ice2k/lib:${GUIX_PROFILES}/room-ice2k/room-ice2k/lib:\${LD_LIBRARY_PATH:-}
    xrdb -merge ${REPO_ROOT}/linux/.config/redstone-9x/Xresources 2>/dev/null || echo ${CONTAINER}: warning — Redstone Xresources not loaded
    xrandr --dpi 96 2>/dev/null || true
    bash ${REPO_ROOT}/linux/.local/bin/sanctuary-redstone-9x-wineprefix-init
    exec dbus-run-session sh -c 'gamemoded -d >/dev/null 2>&1 || true; exec ${GUEST_HOME}/.local/bin/redstone-9x-flatpak-session icewm-session'
  "
#+END_SRC
