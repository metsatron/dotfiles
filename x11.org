* T480s DeskFlow Moonlight guard
#+BEGIN_SRC bash :tangle t480s/.local/bin/deskflow-moonlight-guard :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

INTERVAL="${DESKFLOW_MOONLIGHT_GUARD_INTERVAL:-0.25}"
RESUME_MISSES="${DESKFLOW_MOONLIGHT_GUARD_RESUME_MISSES:-8}"
LOG_FILE="${DESKFLOW_MOONLIGHT_GUARD_LOG:-$HOME/.cache/deskflow-moonlight-guard.log}"
STATE="resumed"
MISSES=0

usage() {
  cat <<'EOF'
deskflow-moonlight-guard

Pauses DeskFlow while Moonlight is the active X11 foreground window.
Requires several consecutive non-Moonlight samples before resuming, because
Moonlight can briefly expose blank or root-ish active-window data while it owns
the input grab.

Commands:
  --once      Print whether Moonlight is the active window, then exit
  --debug     Print active-window and DeskFlow detection details, then exit
  --resume    Resume any stopped DeskFlow process and exit
  --help      Show this help

Emergency resume:
  deskflow-moonlight-guard --resume
  pkill -CONT -x deskflow
  pkill -CONT -x deskflow-core
  pkill -CONT -x deskflow-server
EOF
}

log() {
  mkdir -p "$(dirname "$LOG_FILE")"
  printf '%s %s\n' "$(date '+%Y-%m-%d %H:%M:%S')" "$*" >>"$LOG_FILE"
}

require_x11() {
  [ -n "${DISPLAY:-}" ] || return 1

  case "${XDG_SESSION_TYPE:-x11}" in
    x11|"") return 0 ;;
    *) return 1 ;;
  esac
}

deskflow_pids() {
  {
    pgrep -u "$USER" -x deskflow || true
    pgrep -u "$USER" -x .deskflow-real || true
    pgrep -u "$USER" -x deskflow-core || true
    pgrep -u "$USER" -x deskflow-server || true
    pgrep -u "$USER" -f '(^|[ /])deskflow($| )' || true
  } | sort -u
}

active_window_details() {
  local wid class title pid args

  if ! require_x11; then
    printf 'x11=unavailable DISPLAY=%s XDG_SESSION_TYPE=%s\n' "${DISPLAY:-}" "${XDG_SESSION_TYPE:-}"
    return 1
  fi

  if ! command -v xdotool >/dev/null 2>&1; then
    printf 'xdotool=missing\n'
    return 1
  fi

  wid="$(xdotool getactivewindow 2>/dev/null || true)"
  if [ -z "$wid" ]; then
    printf 'active_window=unavailable DISPLAY=%s\n' "${DISPLAY:-}"
    return 1
  fi

  class="$(xdotool getwindowclassname "$wid" 2>/dev/null || true)"
  title="$(xdotool getwindowname "$wid" 2>/dev/null || true)"
  pid="$(xdotool getwindowpid "$wid" 2>/dev/null || true)"
  args=""
  if [ -n "$pid" ]; then
    args="$(ps -p "$pid" -o args= 2>/dev/null || true)"
  fi

  printf 'wid=%s\nclass=%s\ntitle=%s\npid=%s\nargs=%s\n' "$wid" "$class" "$title" "$pid" "$args"
}

moonlight_active() {
  local wid class title pid args

  require_x11 || return 1
  command -v xdotool >/dev/null 2>&1 || return 1

  wid="$(xdotool getactivewindow 2>/dev/null || true)"
  [ -n "$wid" ] || return 1

  class="$(xdotool getwindowclassname "$wid" 2>/dev/null || true)"
  title="$(xdotool getwindowname "$wid" 2>/dev/null || true)"
  pid="$(xdotool getwindowpid "$wid" 2>/dev/null || true)"

  args=""
  if [ -n "$pid" ]; then
    args="$(ps -p "$pid" -o args= 2>/dev/null || true)"
  fi

  printf '%s\n%s\n%s\n' "$class" "$title" "$args" \
    | grep -Eiq 'moonlight|com\.moonlight_stream\.Moonlight'
}

pause_deskflow() {
  local pids

  pids="$(deskflow_pids)"
  [ -n "$pids" ] || return 0

  echo "$pids" | xargs -r kill -STOP
  STATE="paused"
  log "paused DeskFlow pids: $(echo "$pids" | tr '\n' ' ')"
}

resume_deskflow() {
  local pids

  pids="$(deskflow_pids)"
  [ -n "$pids" ] || return 0

  echo "$pids" | xargs -r kill -CONT
  STATE="resumed"
  log "resumed DeskFlow pids: $(echo "$pids" | tr '\n' ' ')"
}

case "${1-}" in
  --once)
    if moonlight_active; then
      echo "moonlight-active"
    else
      echo "moonlight-inactive"
    fi
    exit 0
    ;;
  --debug)
    active_window_details || true
    printf 'deskflow_pids=%s\n' "$(deskflow_pids | tr '\n' ' ')"
    if moonlight_active; then
      echo "moonlight-active"
    else
      echo "moonlight-inactive"
    fi
    exit 0
    ;;
  --resume)
    resume_deskflow
    exit 0
    ;;
  --help|-h)
    usage
    exit 0
    ;;
  "")
    ;;
  *)
    usage >&2
    exit 2
    ;;
esac

if ! require_x11; then
  log "X11 DISPLAY not available; exiting DISPLAY=${DISPLAY:-} XDG_SESSION_TYPE=${XDG_SESSION_TYPE:-}"
  exit 0
fi

if ! command -v xdotool >/dev/null 2>&1; then
  log "xdotool not found; exiting"
  exit 0
fi

trap resume_deskflow EXIT
trap 'resume_deskflow; exit 0' INT TERM

log "started DISPLAY=${DISPLAY:-} XDG_SESSION_TYPE=${XDG_SESSION_TYPE:-} interval=$INTERVAL resume_misses=$RESUME_MISSES"

while true; do
  if moonlight_active; then
    MISSES=0
    [ "$STATE" = "paused" ] || pause_deskflow
  else
    if [ "$STATE" = "paused" ]; then
      MISSES=$((MISSES + 1))
      if [ "$MISSES" -ge "$RESUME_MISSES" ]; then
        resume_deskflow
        MISSES=0
      fi
    else
      MISSES=0
    fi
  fi

  sleep "$INTERVAL"
done
#+END_SRC

* T480 Deskflow enter command

Releases all stuck modifier keys when Deskflow focus enters the T480.

Standard =xdotool keyup= only targets the primary keycode per keysym. On T480,
Super_L and Hyper_L have secondary keycodes (206/207) from extra xmodmap entries
that Deskflow can leave stuck. This script:
1. Releases common modifier keysyms via xdotool
2. Finds and releases any stuck keys in the XTEST virtual keyboard via XTestFakeKeyEvent

Point Deskflow's client "enter command" to this script:
Settings → Client → On Screen Enter → =~/.local/bin/deskflow-enter-release=

#+BEGIN_SRC bash :tangle t480/.local/bin/deskflow-enter-release :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail
DISPLAY="${DISPLAY:-:0}"
export DISPLAY

# Release standard modifier keysyms
xdotool keyup alt shift ctrl super Hyper_L ISO_Level3_Shift Return KP_Enter Linefeed 2>/dev/null || true

# Release any keys stuck in the XTEST virtual keyboard device (device named
# "Virtual core XTEST keyboard"). Deskflow sends keys via XTest and can leave
# modifiers down when switching screens if the key-up is lost mid-transfer.
/usr/bin/python3 - <<'PYEOF'
import ctypes, subprocess, os, re

DISPLAY = os.environ.get('DISPLAY', ':0')
env = {**os.environ, 'DISPLAY': DISPLAY}

# Find XTEST keyboard device ID
result = subprocess.run(['xinput', 'list'], env=env, capture_output=True, text=True)
device_id = None
for line in result.stdout.splitlines():
    if 'XTEST keyboard' in line:
        m = re.search(r'id=(\d+)', line)
        if m:
            device_id = int(m.group(1))
            break

if device_id is None:
    raise SystemExit(0)

# Find keycodes that are stuck down
state = subprocess.run(
    ['xinput', 'query-state', str(device_id)],
    env=env, capture_output=True, text=True
)
stuck = []
for line in state.stdout.splitlines():
    line = line.strip()
    if line.endswith('=down'):
        m = re.search(r'key\[(\d+)\]=down', line)
        if m:
            stuck.append(int(m.group(1)))

if not stuck:
    raise SystemExit(0)

from ctypes.util import find_library as _fl
_x11_path = _fl('X11')
_xtst_path = _fl('Xtst')
if not _x11_path or not _xtst_path:
    raise SystemExit(0)
xlib = ctypes.cdll.LoadLibrary(_x11_path)
xtst = ctypes.cdll.LoadLibrary(_xtst_path)
xlib.XOpenDisplay.restype  = ctypes.c_void_p
xlib.XOpenDisplay.argtypes = [ctypes.c_char_p]
xlib.XFlush.argtypes       = [ctypes.c_void_p]
xlib.XCloseDisplay.argtypes= [ctypes.c_void_p]
xtst.XTestFakeKeyEvent.restype  = ctypes.c_int
xtst.XTestFakeKeyEvent.argtypes = [
    ctypes.c_void_p, ctypes.c_uint, ctypes.c_int, ctypes.c_ulong
]

d = xlib.XOpenDisplay(DISPLAY.encode())
if not d:
    raise SystemExit(1)

for kc in stuck:
    xtst.XTestFakeKeyEvent(d, kc, 0, 0)

xlib.XFlush(d)
xlib.XCloseDisplay(d)
PYEOF
#+END_SRC

* T480 Deskflow configure

Patches =~/.config/Deskflow/Deskflow.conf= to restore the enter command after
Deskflow updates or GUI resets wipe it. The full config is not owned by DotCortex
(it contains machine-local IPs and TLS paths), but this script surgically sets just
the two keys that Deskflow needs to run the enter command on screen entry.

Run after any Deskflow update: =deskflow-configure && pkill -x deskflow; deskflow &=

#+BEGIN_SRC bash :tangle t480/.local/bin/deskflow-configure :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

CONF="${HOME}/.config/Deskflow/Deskflow.conf"
ENTER_CMD="${HOME}/.local/bin/deskflow-enter-release"

[ -f "$CONF" ] || { printf 'deskflow-configure: %s not found\n' "$CONF" >&2; exit 1; }

cp "$CONF" "$CONF.backup-$(date +%Y%m%d-%H%M%S)"

python3 - "$CONF" "$ENTER_CMD" <<'PYEOF'
import sys, re

conf_path, enter_cmd = sys.argv[1], sys.argv[2]

with open(conf_path) as f:
    content = f.read()

updates = {
    'cbRunEnterCommand': 'true',
    'lineCommandEnter':  enter_cmd,
}

lines   = content.splitlines(keepends=True)
result  = []
set_keys = set()
in_cfg  = False

for line in lines:
    stripped = line.strip()
    if stripped.startswith('['):
        # If leaving [internalConfig], flush any missing keys first
        if in_cfg and set_keys != set(updates):
            for k, v in updates.items():
                if k not in set_keys:
                    result.append(f'{k}={v}\n')
                    set_keys.add(k)
        in_cfg = (stripped == '[internalConfig]')
    if in_cfg:
        for key, val in updates.items():
            if stripped.startswith(key + '='):
                line = f'{key}={val}\n'
                set_keys.add(key)
                break
    result.append(line)

# Append missing keys if [internalConfig] was the last section
if in_cfg and set_keys != set(updates):
    for k, v in updates.items():
        if k not in set_keys:
            result.append(f'{k}={v}\n')

with open(conf_path, 'w') as f:
    f.writelines(result)

for k in sorted(updates):
    print(f'  {k} = {updates[k]}')
print('deskflow-configure: done — restart Deskflow to apply')
PYEOF

* SDDM theme (T480)
Setup script for the sddmsel login theme on T480.
Works on both OpenMandriva (rpm/dnf) and Vendefoul Wolf/Devuan (dpkg/apt).
Installs to =/usr/share/sddm/themes/sel-shaders/= and writes the SDDM config drop-in.
Run once after =tangle-one x11.org && STOW_PKGS="all linux t480" make safe-stow=.

#+BEGIN_SRC bash :tangle t480/.local/bin/sddm-theme-setup :mkdirp yes :shebang "#!/usr/bin/env bash"
set -euo pipefail

THEME_NAME="sel-shaders"
THEME_DEST="/usr/share/sddm/themes/${THEME_NAME}"
SDDM_CONF="/etc/sddm.conf.d/10-theme.conf"
REPO_URL="https://github.com/leonardochappuis/sddmsel"
CLONE_DIR="$(mktemp -d)"

die() { printf 'sddm-theme-setup: %s\n' "$*" >&2; exit 1; }

echo "[1/4] Qt6 dependencies..."
NEED=()
if command -v rpm &>/dev/null && command -v dnf &>/dev/null; then
  # OpenMandriva / RPM-based
  rpm -q lib64Qt6Multimedia  &>/dev/null || NEED+=(lib64Qt6Multimedia)
  rpm -q lib64Qt6Core5Compat &>/dev/null || NEED+=(lib64Qt6Core5Compat)
  rpm -q lib64Qt6Svg         &>/dev/null || NEED+=(lib64Qt6Svg)
  [ ${#NEED[@]} -eq 0 ] || sudo dnf install -y "${NEED[@]}"
else
  # Debian/Devuan/apt-based (Vendefoul Wolf)
  dpkg -l libqt6multimedia6  &>/dev/null || NEED+=(libqt6multimedia6)
  dpkg -l libqt6core5compat6 &>/dev/null || NEED+=(libqt6core5compat6)
  dpkg -l libqt6svg6         &>/dev/null || NEED+=(libqt6svg6)
  [ ${#NEED[@]} -eq 0 ] || sudo apt-get install -y "${NEED[@]}"
fi

echo "[2/4] Cloning sddmsel..."
git clone --depth=1 "$REPO_URL" "$CLONE_DIR"
[ -d "${CLONE_DIR}/${THEME_NAME}" ] || die "theme variant '${THEME_NAME}' not found in repo"

echo "[3/4] Installing theme to ${THEME_DEST}..."
sudo install -d -m 755 "$THEME_DEST"
sudo cp -r "${CLONE_DIR}/${THEME_NAME}/." "$THEME_DEST/"
rm -rf "$CLONE_DIR"

echo "[4/5] Writing SDDM config..."
sudo install -d -m 755 /etc/sddm.conf.d
sudo tee "$SDDM_CONF" > /dev/null <<EOF
[Theme]
Current=${THEME_NAME}
EOF

echo "[5/6] Patching Xsetup for ALSA volume at login..."
XSETUP="/usr/share/sddm/scripts/Xsetup"
ALSA_MARKER="# sddmsel: ALSA init"
if ! grep -qF "$ALSA_MARKER" "$XSETUP" 2>/dev/null; then
  sudo tee -a "$XSETUP" > /dev/null <<'XEOF'
# sddmsel: ALSA init — ensure audible levels before login screen renders
amixer -c 0 set Master 100% unmute 2>/dev/null || true
amixer -c 0 set Speaker 100% unmute 2>/dev/null || true
amixer -c 0 set PCM 100% 2>/dev/null || true
XEOF
fi
sudo mkdir -p /var/lib/alsa && sudo alsactl store 2>/dev/null || true

echo "[6/7] Granting sddm user ALSA access (Qt6 multimedia fallback)..."
# Qt6 tries PipeWire first; without a session PipeWire it must fall back to
# ALSA — but only if the sddm user is in the audio group.
if ! id sddm | grep -q '(audio)'; then
  sudo usermod -aG audio sddm
fi

echo "[7/8] Configuring ALSA default device for sddm (bypass PulseAudio plugin)..."
# On systems where ALSA 'default' routes through PulseAudio/PipeWire, the
# sddm user has no audio session at login time and Qt6 falls silent. Route
# default directly to hardware for this user only.
SDDM_HOME="$(getent passwd sddm | cut -d: -f6)"
if [ -n "$SDDM_HOME" ] && ! grep -qF 'type plug' "${SDDM_HOME}/.asoundrc" 2>/dev/null; then
  sudo tee "${SDDM_HOME}/.asoundrc" > /dev/null <<'AEOF'
pcm.!default {
  type plug
  slave.pcm "hw:0,0"
}
ctl.!default {
  type hw
  card 0
}
AEOF
  sudo chown sddm:sddm "${SDDM_HOME}/.asoundrc"
fi

echo "[8/8] Overriding PipeWire systemd condition for sddm greeter..."
# Qt6 multimedia uses PipeWire as its sole audio backend. pipewire.service has
# ConditionUser=!@system which blocks it for system users (uid < 1000) like
# sddm. PipeWire socket activation exists but the service never starts.
# Override to remove the condition so socket activation works for sddm.
# Only applies on systemd systems — sysv-init (Devuan) skips this step.
if command -v systemctl &>/dev/null; then
  SDDM_HOME="$(getent passwd sddm | cut -d: -f6)"
  PW_OVERRIDE_DIR="${SDDM_HOME}/.config/systemd/user/pipewire.service.d"
  sudo -u sddm mkdir -p "$PW_OVERRIDE_DIR"
  if ! grep -q 'ConditionUser=' "${PW_OVERRIDE_DIR}/override.conf" 2>/dev/null; then
    sudo tee "${PW_OVERRIDE_DIR}/override.conf" > /dev/null <<'PEOF'
[Unit]
ConditionUser=
PEOF
    sudo chown -R sddm:sddm "${SDDM_HOME}/.config/systemd"
  fi
else
  echo "  (skipped — sysv-init detected, no systemd user units)"
fi

echo
echo "=== SDDM theme ‘${THEME_NAME}’ installed ==="
echo "Test with: sddm-greeter-qt6 --test-mode --theme ${THEME_DEST}"
#+END_SRC
