# Keep the distro gtk3-nocsd hook from exporting session-wide LD_PRELOAD.
# That leaks into Guix and Electron binaries and causes loader warnings.
export GTK3_NOCSD_IGNORE=1
export GTK_CSD=0

# Pull in per-session environment snippets before the desktop starts.
if [ -d "$HOME/.xsessionrc.d" ]; then
  for xsessionrc_snippet in "$HOME"/.xsessionrc.d/*.sh; do
    [ -r "$xsessionrc_snippet" ] || continue
    set +u
    . "$xsessionrc_snippet"
    set +u
  done
  unset xsessionrc_snippet
fi

# Guix GIO modules are built against Guix's GLib and break system binaries
# like xfce4-session that link against the system GLib. Unset them so the
# session manager uses the system GIO module path instead.
unset GIO_EXTRA_MODULES

# AppMenu is enabled through XFCE xsettings by xfce-appmenu-configure.
# Do not export GTK_MODULES session-wide: terminals and agent children inherit
# it, and Electron/Flatpak CLIs log host GTK module warnings.
unset GTK_MODULES
export UBUNTU_MENUPROXY="${UBUNTU_MENUPROXY:-1}"
export QT_QPA_PLATFORMTHEME="${QT_QPA_PLATFORMTHEME:-gtk2}"

* AppMenu registrar bootstrap
#+BEGIN_SRC sh :tangle linux/.xsessionrc.d/30-appmenu-registrar.sh :shebang "#!/usr/bin/env sh"
# Start the registrar inside the real X session so exported menus and the
# panel plugin meet on the same session bus.
registrar_has_owner() {
  command -v gdbus >/dev/null 2>&1 || return 1
  gdbus call --session \
    --dest org.freedesktop.DBus \
    --object-path /org/freedesktop/DBus \
    --method org.freedesktop.DBus.NameHasOwner \
    com.canonical.AppMenu.Registrar 2>/dev/null | grep -q '(true,'
}

APPMENU_REGISTRAR=
if ! registrar_has_owner; then
  pkill -x appmenu-registr >/dev/null 2>&1 || true
  pkill -x appmenu-registrar >/dev/null 2>&1 || true

  if command -v appmenu-registrar >/dev/null 2>&1; then
    APPMENU_REGISTRAR="$(command -v appmenu-registrar)"
  elif [ -x /usr/libexec/vala-panel/appmenu-registrar ]; then
    APPMENU_REGISTRAR=/usr/libexec/vala-panel/appmenu-registrar
  fi

  if [ -n "${APPMENU_REGISTRAR}" ]; then
    # Keep the registrar alive after the helper shell exits.
    nohup "$APPMENU_REGISTRAR" -r >/dev/null 2>&1 &
  fi
fi

unset APPMENU_REGISTRAR
#+END_SRC

* T480s DeskFlow Moonlight guard
#+BEGIN_SRC bash :tangle t480s/.local/bin/deskflow-moonlight-guard :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

INTERVAL="${DESKFLOW_MOONLIGHT_GUARD_INTERVAL:-0.25}"
LOG_FILE="${DESKFLOW_MOONLIGHT_GUARD_LOG:-$HOME/.cache/deskflow-moonlight-guard.log}"
STATE="resumed"

usage() {
  cat <<'EOF'
deskflow-moonlight-guard

Pauses DeskFlow while Moonlight is the active X11 foreground window.

Commands:
  --once      Print whether Moonlight is the active window, then exit
  --debug     Print active-window and DeskFlow detection details, then exit
  --resume    Resume any stopped DeskFlow process and exit
  --help      Show this help

Emergency resume:
  deskflow-moonlight-guard --resume
  pkill -CONT -x deskflow
  pkill -CONT -x deskflow-core
  pkill -CONT -x deskflow-server
EOF
}

log() {
  mkdir -p "$(dirname "$LOG_FILE")"
  printf '%s %s\n' "$(date '+%Y-%m-%d %H:%M:%S')" "$*" >>"$LOG_FILE"
}

require_x11() {
  [ -n "${DISPLAY:-}" ] || return 1

  case "${XDG_SESSION_TYPE:-x11}" in
    x11|"") return 0 ;;
    *) return 1 ;;
  esac
}

deskflow_pids() {
  {
    pgrep -u "$USER" -x deskflow || true
    pgrep -u "$USER" -x .deskflow-real || true
    pgrep -u "$USER" -x deskflow-core || true
    pgrep -u "$USER" -x deskflow-server || true
    pgrep -u "$USER" -f '(^|[ /])deskflow($| )' || true
  } | sort -u
}

active_window_details() {
  local wid class title pid args

  if ! require_x11; then
    printf 'x11=unavailable DISPLAY=%s XDG_SESSION_TYPE=%s\n' "${DISPLAY:-}" "${XDG_SESSION_TYPE:-}"
    return 1
  fi

  if ! command -v xdotool >/dev/null 2>&1; then
    printf 'xdotool=missing\n'
    return 1
  fi

  wid="$(xdotool getactivewindow 2>/dev/null || true)"
  if [ -z "$wid" ]; then
    printf 'active_window=unavailable DISPLAY=%s\n' "${DISPLAY:-}"
    return 1
  fi

  class="$(xdotool getwindowclassname "$wid" 2>/dev/null || true)"
  title="$(xdotool getwindowname "$wid" 2>/dev/null || true)"
  pid="$(xdotool getwindowpid "$wid" 2>/dev/null || true)"
  args=""
  if [ -n "$pid" ]; then
    args="$(ps -p "$pid" -o args= 2>/dev/null || true)"
  fi

  printf 'wid=%s\nclass=%s\ntitle=%s\npid=%s\nargs=%s\n' "$wid" "$class" "$title" "$pid" "$args"
}

moonlight_active() {
  local wid class title pid args

  require_x11 || return 1
  command -v xdotool >/dev/null 2>&1 || return 1

  wid="$(xdotool getactivewindow 2>/dev/null || true)"
  [ -n "$wid" ] || return 1

  class="$(xdotool getwindowclassname "$wid" 2>/dev/null || true)"
  title="$(xdotool getwindowname "$wid" 2>/dev/null || true)"
  pid="$(xdotool getwindowpid "$wid" 2>/dev/null || true)"

  args=""
  if [ -n "$pid" ]; then
    args="$(ps -p "$pid" -o args= 2>/dev/null || true)"
  fi

  printf '%s\n%s\n%s\n' "$class" "$title" "$args" \
    | grep -Eiq 'moonlight|com\.moonlight_stream\.Moonlight'
}

pause_deskflow() {
  local pids

  pids="$(deskflow_pids)"
  [ -n "$pids" ] || return 0

  echo "$pids" | xargs -r kill -STOP
  STATE="paused"
  log "paused DeskFlow pids: $(echo "$pids" | tr '\n' ' ')"
}

resume_deskflow() {
  local pids

  pids="$(deskflow_pids)"
  [ -n "$pids" ] || return 0

  echo "$pids" | xargs -r kill -CONT
  STATE="resumed"
  log "resumed DeskFlow pids: $(echo "$pids" | tr '\n' ' ')"
}

case "${1-}" in
  --once)
    if moonlight_active; then
      echo "moonlight-active"
    else
      echo "moonlight-inactive"
    fi
    exit 0
    ;;
  --debug)
    active_window_details || true
    printf 'deskflow_pids=%s\n' "$(deskflow_pids | tr '\n' ' ')"
    if moonlight_active; then
      echo "moonlight-active"
    else
      echo "moonlight-inactive"
    fi
    exit 0
    ;;
  --resume)
    resume_deskflow
    exit 0
    ;;
  --help|-h)
    usage
    exit 0
    ;;
  "")
    ;;
  *)
    usage >&2
    exit 2
    ;;
esac

if ! require_x11; then
  log "X11 DISPLAY not available; exiting DISPLAY=${DISPLAY:-} XDG_SESSION_TYPE=${XDG_SESSION_TYPE:-}"
  exit 0
fi

if ! command -v xdotool >/dev/null 2>&1; then
  log "xdotool not found; exiting"
  exit 0
fi

trap resume_deskflow EXIT
trap 'resume_deskflow; exit 0' INT TERM

log "started DISPLAY=${DISPLAY:-} XDG_SESSION_TYPE=${XDG_SESSION_TYPE:-} interval=$INTERVAL"

while true; do
  if moonlight_active; then
    [ "$STATE" = "paused" ] || pause_deskflow
  else
    [ "$STATE" = "resumed" ] || resume_deskflow
  fi

  sleep "$INTERVAL"
done
#+END_SRC

* XFCE appmenu xsettings helper
#+BEGIN_SRC bash :tangle linux/.local/bin/xfce-appmenu-configure :shebang "#!/usr/bin/env bash"
set -euo pipefail

if ! command -v xfconf-query >/dev/null 2>&1; then
  echo "xfconf-query not found; XFCE xsettings cannot be configured" >&2
  exit 1
fi

set_bool() {
  local path="$1" value="$2"
  if xfconf-query -c xsettings -p "$path" >/dev/null 2>&1; then
    xfconf-query -c xsettings -p "$path" -t bool -s "$value"
  else
    xfconf-query -c xsettings -p "$path" -n -t bool -s "$value"
  fi
}

set_string() {
  local path="$1" value="$2"
  if xfconf-query -c xsettings -p "$path" >/dev/null 2>&1; then
    xfconf-query -c xsettings -p "$path" -t string -s "$value"
  else
    xfconf-query -c xsettings -p "$path" -n -t string -s "$value"
  fi
}

resolve_registrar() {
  if command -v appmenu-registrar >/dev/null 2>&1; then
    command -v appmenu-registrar
  elif [ -x /usr/libexec/vala-panel/appmenu-registrar ]; then
    printf '%s\n' /usr/libexec/vala-panel/appmenu-registrar
  else
    return 1
  fi
}

set_bool /Gtk/ShellShowsMenubar true
set_bool /Gtk/ShellShowsAppmenu true

gtk_modules="$(xfconf-query -c xsettings -p /Gtk/Modules 2>/dev/null || true)"
case ":${gtk_modules}:" in
  *:appmenu-gtk-module:*) ;;
  *) gtk_modules="${gtk_modules:+$gtk_modules:}appmenu-gtk-module" ;;
esac
set_string /Gtk/Modules "$gtk_modules"

registrar_bin="$(resolve_registrar || true)"

if [[ -n "$registrar_bin" ]]; then
  pkill -x appmenu-registr >/dev/null 2>&1 || true
  pkill -x appmenu-registrar >/dev/null 2>&1 || true
  # Keep exactly one registrar alive after this configuration helper exits.
  nohup "$registrar_bin" -r >/dev/null 2>&1 &
fi

printf 'Gtk/ShellShowsMenubar=%s\n' "$(xfconf-query -c xsettings -p /Gtk/ShellShowsMenubar 2>/dev/null || echo unset)"
printf 'Gtk/ShellShowsAppmenu=%s\n' "$(xfconf-query -c xsettings -p /Gtk/ShellShowsAppmenu 2>/dev/null || echo unset)"
printf 'Gtk/Modules=%s\n' "$(xfconf-query -c xsettings -p /Gtk/Modules 2>/dev/null || echo unset)"
pgrep -af '[a]ppmenu-registrar' || true
#+END_SRC

* SDDM theme (T480)
Setup script for the sddmsel login theme on T480 (OpenMandriva).
Installs to =/usr/share/sddm/themes/sel-shaders/= and writes the SDDM config drop-in.
Run once after =tangle-one x11.org && loom stow:openmandriva=.

#+BEGIN_SRC bash :tangle t480/.local/bin/sddm-theme-setup :mkdirp yes :shebang "#!/usr/bin/env bash"
set -euo pipefail

THEME_NAME="sel-shaders"
THEME_DEST="/usr/share/sddm/themes/${THEME_NAME}"
SDDM_CONF="/etc/sddm.conf.d/10-theme.conf"
REPO_URL="https://github.com/leonardochappuis/sddmsel"
CLONE_DIR="$(mktemp -d)"

die() { printf ‘sddm-theme-setup: %s\n’ "$*" >&2; exit 1; }

echo "[1/4] Qt6 dependencies..."
NEED=()
rpm -q qt6-qtmultimedia &>/dev/null || NEED+=(qt6-qtmultimedia)
rpm -q qt6-qt5compat     &>/dev/null || NEED+=(qt6-qt5compat)
rpm -q qt6-qtsvg         &>/dev/null || NEED+=(qt6-qtsvg)
[ ${#NEED[@]} -eq 0 ] || sudo dnf install -y "${NEED[@]}"

echo "[2/4] Cloning sddmsel..."
git clone --depth=1 "$REPO_URL" "$CLONE_DIR"
[ -d "${CLONE_DIR}/${THEME_NAME}" ] || die "theme variant ‘${THEME_NAME}’ not found in repo"

echo "[3/4] Installing theme to ${THEME_DEST}..."
sudo install -d -m 755 "$THEME_DEST"
sudo cp -r "${CLONE_DIR}/${THEME_NAME}/." "$THEME_DEST/"
rm -rf "$CLONE_DIR"

echo "[4/4] Writing SDDM config..."
sudo install -d -m 755 /etc/sddm.conf.d
sudo tee "$SDDM_CONF" > /dev/null <<EOF
[Theme]
Current=${THEME_NAME}
EOF

echo
echo "=== SDDM theme ‘${THEME_NAME}’ installed ==="
echo "Test with: sddm-greeter-qt6 --test-mode --theme ${THEME_DEST}"
#+END_SRC
