*** Token Conflict / Workaround

The Claude Code plugin currently shares =@dotcortex_opencode_bot='s token with the OpenCode
bot. Telegram enforces one polling consumer per token — the OpenCode bot (running since Apr 30)
holds the slot, causing 409 Conflict for the Claude plugin.

*Workaround (active):* =claude-auryn --telegram= sends SIGSTOP to the opencode-telegram process
before launching Claude, freeing the token. On session exit the wrapper sends SIGCONT to resume it.

*Proper fix (pending):* Create a dedicated bot for Auryn via @BotFather, update
=~/.claude/channels/telegram/.env= and =NEXUS/stow/linux/.claude/channels/telegram/.env=,
run =helmstow=. Then remove the SIGSTOP workaround from =claude-auryn=.

** Codex

Codex still has no native Telegram channel, but HelmCortex now supplies a
private Codex app-server adapter for HelmAstra.

- Wrapper: =codex-helmastra --telegram= ensures the adapter, then continues
  into interactive Codex when attached to a terminal; the host manager uses
  the same flag non-interactively to establish only the background adapter
- Backend: Codex =app-server= JSON-RPC, with persistent Telegram-to-thread mapping
- Startup contract: the wrapper returns success only after bot identity and the
  Codex app-server are verified and the adapter publishes its PID-owned ready marker
- Approval policy: =never= inside the workspace-write sandbox
- Compaction: native =thread/compact/start= plus the 27-minute / 70,000-token
  idle governor inherited from =codex-luna= / =codex-warm=
- Runtime credentials: private =~/.config/codex-telegram-helmastra/.env= only
- Identity safety: the expected bot ID and username are pinned in the private
  env; a token for OpenCode, Claude, Hermes, or any other bot fails closed

The adapter is deliberately separate from Ductor: Ductor keeps its own bot
identity and remains the fleet coordinator, while HelmAstra owns this bot.

* OpenCode ownership helper

#+BEGIN_SRC python :tangle all/.local/bin/opencode-owner :shebang "#!/usr/bin/env python3"
"""Fail-closed lifecycle owner for the isolated Kikin OpenCode pair.

This helper deliberately discovers nothing by process-name search.  The state
file is an ownership claim, and every operation revalidates PID, Linux start
ticks, cwd, and argv against that claim before declaring or stopping a process.
"""

from __future__ import annotations

import argparse
import json
import os
import signal
import subprocess
import sys
import time
from pathlib import Path


OWNER = "telegram-agent-host:opencode"
BOT = "opencode"
DEFAULT_PROJECT = "/home/metsatron/HelmCortex"
DEFAULT_HOST = "127.0.0.1"
DEFAULT_PORT = 4096
PACKAGE_NAME = "@grinev/opencode-telegram-bot"
PACKAGE_VERSION = "0.22.5"


def json_out(value: object) -> None:
    print(json.dumps(value, sort_keys=True))


def read_json(path: Path) -> dict | None:
    try:
        value = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, ValueError):
        return None
    return value if isinstance(value, dict) else None


def pid_info(pid: int) -> dict | None:
    proc = Path("/proc") / str(pid)
    try:
        stat = (proc / "stat").read_text(encoding="utf-8")
        after_comm = stat[stat.rfind(")") + 2 :].split()
        start_ticks = int(after_comm[19])
        argv = [part.decode(errors="replace") for part in (proc / "cmdline").read_bytes().split(b"\0") if part]
        cwd = os.path.realpath(proc / "cwd")
    except (OSError, ValueError, IndexError, UnicodeError):
        return None
    if not argv:
        return None
    return {"pid": pid, "start_ticks": start_ticks, "cwd": cwd, "argv": argv}


def state_path(value: str | None) -> Path:
    return Path(value or os.environ.get("OPENCODE_STATE_FILE", str(Path.home() / ".local/state/telegram-agents/opencode.json")))


def project_path(value: str | None) -> str:
    return os.path.realpath(value or os.environ.get("OPENCODE_PROJECT_ROOT", DEFAULT_PROJECT))


def port_value(value: str | None) -> int:
    return int(value or os.environ.get("OPENCODE_PORT", str(DEFAULT_PORT)))


def expected_backend_argv(host: str, port: int) -> list[str]:
    return ["opencode", "serve", "--hostname", host, "--port", str(port)]


def argv_has_backend_contract(argv: list[str], host: str, port: int) -> bool:
    if len(argv) < 6:
        return False
    # The executable may be absolute after a launcher resolves it.  Everything
    # after it must be the exact serve contract, with no broad substring match.
    return os.path.basename(argv[0]) == "opencode" and argv[1:] == expected_backend_argv(host, port)[1:]


def argv_has_poller_contract(argv: list[str], entrypoint: str | None) -> bool:
    if not argv or not entrypoint:
        return False
    expected = os.path.realpath(entrypoint)
    for index, item in enumerate(argv):
        if os.path.realpath(item) == expected:
            return argv[index + 1 :] == ["start", "--mode", "installed"]
    return False


def check_record(state: dict, role: str, expected_project: str, host: str, port: int) -> tuple[dict | None, list[str]]:
    pid_key = f"{role}_pid"
    ticks_key = f"{role}_start_ticks"
    argv_key = f"{role}_argv"
    cwd_key = f"{role}_cwd"
    reasons: list[str] = []
    try:
        pid = int(state.get(pid_key))
    except (TypeError, ValueError):
        return None, [f"{role}-pid-missing"]
    current = pid_info(pid)
    if current is None:
        return None, [f"{role}-down"]
    if str(state.get(ticks_key)) != str(current["start_ticks"]):
        reasons.append(f"{role}-start-ticks-mismatch")
    if os.path.realpath(str(state.get(cwd_key) or "")) != expected_project or current["cwd"] != expected_project:
        reasons.append(f"{role}-wrong-cwd")
    recorded_argv = state.get(argv_key)
    if not isinstance(recorded_argv, list) or [str(x) for x in recorded_argv] != current["argv"]:
        reasons.append(f"{role}-argv-mismatch")
    if role == "backend" and not argv_has_backend_contract(current["argv"], host, port):
        reasons.append("backend-argv-contract-mismatch")
    if role == "poller" and not argv_has_poller_contract(current["argv"], state.get("poller_entrypoint")):
        reasons.append("poller-identity-mismatch")
    return current, reasons


def native_poller_status(entrypoint: str | None) -> dict:
    if not entrypoint:
        return {"status": "unavailable", "reason": "poller-entrypoint-unknown"}
    try:
        result = subprocess.run(
            [entrypoint, "status"], capture_output=True, text=True, timeout=5, check=False
        )
    except (OSError, subprocess.TimeoutExpired) as exc:
        return {"status": "unavailable", "reason": type(exc).__name__}
    text = (result.stdout + result.stderr).strip()
    status = "running" if "Service status: running" in text else "stopped" if "Service status: stopped" in text else "unknown"
    return {"status": status, "exit": result.returncode, "output": text[-500:]}


def native_service_state() -> dict | None:
    home = Path(os.environ.get("OPENCODE_TELEGRAM_HOME", str(Path.home() / ".config/opencode-telegram-bot")))
    return read_json(home / "run/bot-service.json")


def settings_binding(project: str) -> dict:
    home = Path(os.environ.get("OPENCODE_TELEGRAM_HOME", str(Path.home() / ".config/opencode-telegram-bot")))
    settings_file = home / "settings.json"
    settings = read_json(settings_file)
    if not settings:
        return {"status": "session-unbound", "source": str(settings_file), "reason": "settings-missing"}
    current_project = settings.get("currentProject")
    current_session = settings.get("currentSession")
    worktree = current_project.get("worktree") if isinstance(current_project, dict) else None
    session_id = current_session.get("id") if isinstance(current_session, dict) else None
    directory = current_session.get("directory") if isinstance(current_session, dict) else None
    if worktree != project or directory != project or not session_id:
        return {
            "status": "session-unbound",
            "source": str(settings_file),
            "project_worktree": worktree,
            "session_directory": directory,
            "reason": "missing-or-project-mismatch",
        }
    return {
        "status": "bound",
        "source": str(settings_file),
        "project_worktree": worktree,
        "session_directory": directory,
        "session_id": session_id,
        "session_title": current_session.get("title"),
    }


def inspect(args: argparse.Namespace) -> int:
    state = read_json(state_path(args.state))
    result = {"status": "stopped", "backend": None, "poller": None, "reasons": []}
    if not state:
        result["reasons"] = ["owner-state-missing"]
        json_out(result)
        return 1
    project = project_path(args.project or state.get("project"))
    host = args.host or str(state.get("host") or DEFAULT_HOST)
    port = port_value(args.port or str(state.get("port") or DEFAULT_PORT))
    backend, backend_reasons = check_record(state, "backend", project, host, port)
    poller, poller_reasons = check_record(state, "poller", project, host, port)
    if backend is not None:
        backend["valid"] = not backend_reasons
        backend["reasons"] = backend_reasons
    if poller is not None:
        poller["valid"] = not poller_reasons
        poller["reasons"] = poller_reasons
    result["backend"] = backend
    result["poller"] = poller
    result["reasons"] = backend_reasons + poller_reasons
    if state.get("owner") != OWNER:
        result["reasons"].append("foreign-owner")
    if state.get("bot") != BOT:
        result["reasons"].append("bot-identity-mismatch")
    if project != os.path.realpath(DEFAULT_PROJECT):
        result["reasons"].append("wrong-project")
    if host != DEFAULT_HOST or port != DEFAULT_PORT:
        result["reasons"].append("wrong-listen-boundary")
    result["native_poller"] = native_poller_status(state.get("poller_entrypoint"))
    binding = settings_binding(project)
    result["session"] = binding
    if state.get("session_binding") != binding:
        # Session/project metadata is safe to refresh.  Credentials and package
        # settings remain outside this owner file.
        state["session_binding"] = binding
        path = state_path(args.state)
        temp = path.with_name(f"{path.name}.{os.getpid()}.tmp")
        temp.write_text(json.dumps(state, indent=2) + "\n", encoding="utf-8")
        os.replace(temp, path)
    result["project"] = project
    result["host"] = host
    result["port"] = port
    result["status"] = "running" if not result["reasons"] else "degraded"
    json_out(result)
    return 0 if result["status"] == "running" else 1


def capture(args: argparse.Namespace) -> int:
    project = project_path(args.project)
    host = args.host or DEFAULT_HOST
    port = port_value(args.port)
    backend = pid_info(int(args.backend_pid))
    poller = pid_info(int(args.poller_pid))
    if not backend or not poller:
        print("cannot capture missing backend/poller", file=sys.stderr)
        return 1
    if backend["cwd"] != project or not argv_has_backend_contract(backend["argv"], host, port):
        print("backend does not satisfy exact owner contract", file=sys.stderr)
        return 1
    if poller["cwd"] != project or not argv_has_poller_contract(poller["argv"], args.poller_entrypoint):
        print("poller does not satisfy exact owner contract", file=sys.stderr)
        return 1
    service_state = read_json(Path(args.poller_service_state)) if args.poller_service_state else None
    if args.poller_service_state and (not service_state or int(service_state.get("pid", -1)) != poller["pid"]):
        print("poller PID is not the package-native daemon PID", file=sys.stderr)
        return 1
    binding = settings_binding(project)
    state = {
        "schema": "helmcortex.opencode-owner.v2",
        "owner": OWNER,
        "bot": BOT,
        "project": project,
        "host": host,
        "port": port,
        "backend_pid": backend["pid"],
        "backend_start_ticks": backend["start_ticks"],
        "backend_cwd": backend["cwd"],
        "backend_argv": backend["argv"],
        "poller_pid": poller["pid"],
        "poller_start_ticks": poller["start_ticks"],
        "poller_cwd": poller["cwd"],
        "poller_argv": poller["argv"],
        "poller_entrypoint": os.path.realpath(args.poller_entrypoint),
        "poller_package": PACKAGE_NAME,
        "poller_package_version": PACKAGE_VERSION,
        "poller_runtime": "daemon" if args.poller_service_state else "unknown",
        "poller_service_state": args.poller_service_state,
        "poller_native_status": native_poller_status(args.poller_entrypoint),
        "session_binding": binding,
    }
    path = state_path(args.state)
    path.parent.mkdir(parents=True, exist_ok=True)
    temp = path.with_name(f"{path.name}.{os.getpid()}.tmp")
    temp.write_text(json.dumps(state, indent=2) + "\n", encoding="utf-8")
    os.replace(temp, path)
    json_out(state)
    return 0


def stop(args: argparse.Namespace) -> int:
    state = read_json(state_path(args.state))
    if not state:
        print("opencode: STOPPED (owner state absent)")
        return 0
    project = project_path(args.project or state.get("project"))
    host = args.host or str(state.get("host") or DEFAULT_HOST)
    port = port_value(args.port or str(state.get("port") or DEFAULT_PORT))
    owned: list[int] = []
    refused = 0
    for role in ("poller", "backend"):
        current, reasons = check_record(state, role, project, host, port)
        if reasons:
            refused += 1
            print(f"refusing to stop {role}: {','.join(reasons)}", file=sys.stderr)
            continue
        if current:
            owned.append(current["pid"])
    for pid in owned:
        try:
            os.kill(pid, signal.SIGTERM)
        except ProcessLookupError:
            pass
    deadline = time.monotonic() + 5
    while owned and time.monotonic() < deadline:
        if all(pid_info(pid) is None for pid in owned):
            break
        time.sleep(0.1)
    remaining = [pid for pid in owned if pid_info(pid) is not None]
    if remaining:
        for pid in remaining:
            try:
                os.kill(pid, signal.SIGKILL)
            except ProcessLookupError:
                pass
    if not remaining and refused == 0:
        try:
            state_path(args.state).unlink()
        except FileNotFoundError:
            pass
    print(f"opencode: stopped_owned={len(owned)} refused={refused}")
    return 0 if not remaining and refused == 0 else 1


def parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser()
    sub = p.add_subparsers(dest="command", required=True)
    for name in ("inspect", "stop"):
        q = sub.add_parser(name)
        q.add_argument("--state")
        q.add_argument("--project")
        q.add_argument("--host")
        q.add_argument("--port")
    q = sub.add_parser("capture")
    q.add_argument("--state")
    q.add_argument("--project", default=DEFAULT_PROJECT)
    q.add_argument("--host", default=DEFAULT_HOST)
    q.add_argument("--port", default=str(DEFAULT_PORT))
    q.add_argument("--backend-pid", required=True)
    q.add_argument("--poller-pid", required=True)
    q.add_argument("--poller-entrypoint", required=True)
    q.add_argument("--poller-service-state")
    return p


def main() -> int:
    args = parser().parse_args()
    return {"inspect": inspect, "capture": capture, "stop": stop}[args.command](args)


if __name__ == "__main__":
    raise SystemExit(main())

check_opencode() {
    local status_json status_code

        echo "opencode: STOPPED (owner helper unavailable)"
        return 1

    set +e
    status_json=$("$OPENCODE_OWNER_HELPER" inspect \
        --state "$OPENCODE_STATE_FILE" \
        --project "$OPENCODE_PROJECT_ROOT" \
        --host 127.0.0.1 --port "$OPENCODE_PORT")
    status_code=$?
    set -e
    python3 - "$status_json" <<'PY'
import json
import sys

payload = json.loads(sys.argv[1])
print(f"opencode: {payload.get('status', 'unknown').upper()}")
for role in ("backend", "poller"):
    row = payload.get(role) or {}
    state = "RUNNING" if row.get("valid") else ("STOPPED" if not row else "INVALID")
    print(f"  opencode {role}: {state} pid={row.get('pid', '?') if row else '?'}")
native = payload.get("native_poller") or {}
session = payload.get("session") or {}
print(f"  opencode native poller status: {native.get('status', 'unknown')}")
print(f"  opencode session binding: {session.get('status', 'session-unbound')}")
print(f"  opencode project: {payload.get('project', 'see owner state')}")
print(f"  opencode port: {payload.get('host', '127.0.0.1')}:{payload.get('port', '?')}")
PY
    return "$status_code"

enable_agent() {
    local agent="$1"
    validate_agent "$agent"
    local host
    host=$(get_hostname)

    if [ ! -f "$HOSTS_FILE" ]; then
        echo "Error: hosts.conf not found at $HOSTS_FILE"
        exit 1
    fi

    local line
    line=$(grep -i "^${host}|" "$HOSTS_FILE" 2>/dev/null || true)
    if [ -z "$line" ]; then
        echo "Error: host '$host' not found in hosts.conf"
        exit 1
    fi

    local current_agents
    current_agents=$(echo "$line" | cut -d'|' -f2)

    if echo "$current_agents" | grep -qw "$agent"; then
        echo "$agent is already enabled on $host"
        return
    fi

    local new_agents
    if [ -z "$current_agents" ]; then
        new_agents="$agent"
    else
        new_agents="$current_agents,$agent"
    fi

    local tmp_hosts
    tmp_hosts="${HOSTS_FILE}.tmp.$$"
    awk -F'|' -v host="$host" -v agents="$new_agents" '
        BEGIN { OFS = FS }
        tolower($1) == tolower(host) { $2 = agents }
        { print }
    ' "$HOSTS_FILE" > "$tmp_hosts"
    mv "$tmp_hosts" "$HOSTS_FILE"
    echo "Enabled $agent on $host"
    echo "Run 'telegram-agent-host sync' to apply, then 'telegram-agent-host start'"
}

disable_agent() {
    local agent="$1"
    validate_agent "$agent"
    local host
    host=$(get_hostname)

    if [ ! -f "$HOSTS_FILE" ]; then
        echo "Error: hosts.conf not found at $HOSTS_FILE"
        exit 1
    fi

    local line
    line=$(grep -i "^${host}|" "$HOSTS_FILE" 2>/dev/null || true)
    if [ -z "$line" ]; then
        echo "Error: host '$host' not found in hosts.conf"

    local current_agents
    current_agents=$(echo "$line" | cut -d'|' -f2)

    if ! echo "$current_agents" | grep -qw "$agent"; then
        echo "$agent is not enabled on $host"
        return
    fi

    local new_agents
    new_agents=$(echo "$current_agents" | tr ',' '\n' | grep -v "^${agent}$" | tr '\n' ',' | sed 's/,$//' | sed 's/^,//')

    local tmp_hosts
    tmp_hosts="${HOSTS_FILE}.tmp.$$"
    awk -F'|' -v host="$host" -v agents="$new_agents" '
        BEGIN { OFS = FS }
        tolower($1) == tolower(host) { $2 = agents }
        { print }
    ' "$HOSTS_FILE" > "$tmp_hosts"
    mv "$tmp_hosts" "$HOSTS_FILE"
    echo "Disabled $agent on $host"
    echo "Run 'telegram-agent-host stop $agent' to stop it"
}

start_agent() {
    local agent="$1"
    validate_agent "$agent"
    case "$agent" in
        opencode)
            if [ ! -d "$OPENCODE_PROJECT_ROOT" ]; then
                echo "OpenCode project root not found: $OPENCODE_PROJECT_ROOT" >&2
                return 1
            fi
            if [ -r "$OPENCODE_STATE_FILE" ]; then
                if check_opencode >/dev/null 2>&1; then
                    echo "opencode is already running"
                    return 0
                fi
                echo "Existing OpenCode owner state is not healthy; validating and stopping only its owned processes..."
                "$OPENCODE_OWNER_HELPER" stop --state "$OPENCODE_STATE_FILE" --project "$OPENCODE_PROJECT_ROOT" --host 127.0.0.1 --port "$OPENCODE_PORT" || true
            fi
            if command -v ss >/dev/null 2>&1 && ss -Hln "sport = :$OPENCODE_PORT" 2>/dev/null | grep -q .; then
                echo "Refusing OpenCode start: port $OPENCODE_PORT is already occupied by an unowned listener" >&2
                return 1
            fi
            echo "Starting opencode serve in $OPENCODE_PROJECT_ROOT..."
            (
                cd "$OPENCODE_PROJECT_ROOT"
                exec nohup opencode serve --hostname 127.0.0.1 --port "$OPENCODE_PORT"
            ) > "$STATE_DIR/opencode-serve.log" 2>&1 </dev/null &
            printf '%s\n' "$!" > "$OPENCODE_BACKEND_PID_FILE"
            sleep 2
            if [ ! -s "$OPENCODE_BACKEND_PID_FILE" ]; then
                echo "OpenCode backend did not publish an owned PID" >&2
                return 1
            fi
            echo "Starting opencode-telegram in $OPENCODE_PROJECT_ROOT..."
            if ! (umask 077; cd "$OPENCODE_PROJECT_ROOT" && env NODE_OPTIONS="$OPENCODE_NODE_OPTIONS" "$OPENCODE_POLLER_BIN" start --daemon --mode installed > "$STATE_DIR/opencode-telegram.log" 2>&1 </dev/null); then
                backend_pid=$(cat "$OPENCODE_BACKEND_PID_FILE")
                kill -TERM "$backend_pid" 2>/dev/null || true
                rm -f "$OPENCODE_BACKEND_PID_FILE"
                echo "OpenCode Telegram daemon failed to start; rolled back owned backend" >&2
                return 1
            fi
            for _ in {1..15}; do
                [ -s "$OPENCODE_TELEGRAM_HOME/run/bot-service.json" ] && break
                sleep 1
            done
            write_opencode_state || {
                "$OPENCODE_POLLER_BIN" stop >/dev/null 2>&1 || true
                backend_pid=$(cat "$OPENCODE_BACKEND_PID_FILE")
                kill -TERM "$backend_pid" 2>/dev/null || true
                rm -f "$OPENCODE_BACKEND_PID_FILE" "$OPENCODE_POLLER_PID_FILE"
                echo "OpenCode backend/poller did not produce a complete owner state" >&2
                return 1
            }
            python3 -c 'import json,sys; print(json.load(open(sys.argv[1]))["pid"])' "$OPENCODE_TELEGRAM_HOME/run/bot-service.json" > "$OPENCODE_POLLER_PID_FILE"
            ;;
        ductor)
            if command -v ductor >/dev/null 2>&1; then
                if ductor_running; then
                    echo "ductor is already running"
                else
                    echo "Starting ductor (supervised)..."
                    setsid env DUCTOR_HOME="$DUCTOR_HOME_DIR" ductor-supervise > "$STATE_DIR/ductor-supervise.log" 2>&1 </dev/null &
                    sleep 8
                fi
            else
                echo "ductor not found in PATH"
            fi
            ;;
        claudecode)
            echo "claudecode Telegram runs inline via the PTY-supervised claude-auryn launcher."
            echo "Launch Claude with: claude-auryn --telegram"
            echo "Access control: ~/.claude/channels/telegram/access.json"
            ;;
        hermes)
            start_hermes
            ;;
        codex)
            local wrapper="${CODEX_HELMASTRA_WRAPPER:-$HOME/HelmCortex/FORGE/bin/codex-helmastra}"
            if [ ! -x "$wrapper" ]; then
                echo "codex-helmastra wrapper not found at $wrapper"
                return 1
            fi
            if codex_telegram_running; then
                echo "HelmAstra Telegram is already running"
            else
                echo "Starting HelmAstra Codex Telegram adapter..."
                if ! "$wrapper" --telegram \
                    >> "$STATE_DIR/codex-helmastra-telegram.log" 2>&1 </dev/null; then
                    echo "HelmAstra Telegram adapter failed readiness; inspect $STATE_DIR/codex-helmastra-telegram.log"
                    return 1
                fi
                if ! codex_telegram_running; then
                    echo "HelmAstra Telegram adapter reported ready but is not running"
                    return 1
                fi
            fi
            ;;
        pi-agent)
            echo "pi-agent has no Telegram bot — skipping"
            ;;
        *)
            echo "Unknown agent: $agent"
            exit 1
            ;;
    esac
    echo "$agent started"
}

stop_agent() {
    local agent="$1"
    validate_agent "$agent"
    case "$agent" in
        opencode)
            if [ -x "$OPENCODE_OWNER_HELPER" ]; then
                "$OPENCODE_OWNER_HELPER" stop \
                    --state "$OPENCODE_STATE_FILE" \
                    --project "$OPENCODE_PROJECT_ROOT" \
                    --host 127.0.0.1 --port "$OPENCODE_PORT"
            else
                echo "OpenCode owner helper unavailable; refusing unscoped stop" >&2
                return 1
            fi
            ;;
        ductor)
            # Kill the supervisor first — its SIGTERM trap stops the daemon child
            # with no relaunch — then sweep any bare pre-supervisor daemon.
            pkill -TERM -f "ductor-supervise" 2>/dev/null || true
            sleep 1
            pkill -f "python.*$HOME/.local/bin/ductor" 2>/dev/null || true
            ;;
        claudecode)
            pkill -f "claude.*channels.*telegram" 2>/dev/null || true
            ;;
        hermes)
            stop_hermes
            ;;
        codex)
            local codex_pid codex_pgid
            if codex_telegram_running; then
                codex_pid=$(cat "$STATE_DIR/codex-helmastra-telegram.pid")
                codex_pgid=$(ps -o pgid= -p "$codex_pid" 2>/dev/null | tr -d ' ' || true)
                if [ "$codex_pgid" = "$codex_pid" ]; then
                    kill -TERM -- "-$codex_pgid" 2>/dev/null || true
                else
                    kill -TERM "$codex_pid" 2>/dev/null || true
                fi
                for _ in {1..20}; do
                    codex_telegram_running || break
                    sleep 0.25
                done
                if codex_telegram_running; then
                    echo "HelmAstra Telegram did not stop gracefully; forcing its verified process group" >&2
                    if [ "$codex_pgid" = "$codex_pid" ]; then
                        kill -KILL -- "-$codex_pgid" 2>/dev/null || true
                    else
                        kill -KILL "$codex_pid" 2>/dev/null || true
                    fi
                    for _ in {1..20}; do
                        codex_telegram_running || break
                        sleep 0.25
                    done
                fi
            fi
            if codex_telegram_running; then
                echo "HelmAstra Telegram failed to stop" >&2
                return 1
            fi
            ;;
        pi-agent)
            pkill -f "pi-agent\|pi-coding-agent" 2>/dev/null || true
            ;;
        *)
            echo "Unknown agent: $agent"
            exit 1
            ;;
    esac
    echo "$agent stopped"
}

switch_to_machine() {
    local target="$1"
    local current_host
    current_host=$(get_hostname)

    if [ "$current_host" = "$target" ]; then
        echo "Already on $target"
        return
    fi

    local target_agents
    target_agents=$(get_enabled_agents "$target")

    if [ -z "$target_agents" ]; then
        echo "No agents configured for $target"
        exit 1
    fi

    local target_ssh
    target_ssh=$(grep -i "^${target}|" "$HOSTS_FILE" 2>/dev/null | cut -d'|' -f3)

    echo "Switching from $current_host to $target"

    echo "Stopping all agents on $current_host..."
    stop_all

    echo "Stopping agents on $target before restart..."
    if [ -n "$target_ssh" ]; then
        ssh "$target_ssh" "systemctl --user stop hermes-gateway.service 2>/dev/null || pkill -x hermes 2>/dev/null || true; pkill -f 'ductor' 2>/dev/null || true" 2>/dev/null || true
        sleep 1
    fi

    echo "Starting agents on $target via SSH..."
    if [ -n "$target_ssh" ]; then
        ssh "$target_ssh" "\$HOME/.local/bin/telegram-agent-remote-start $target_agents"
        echo "Switched to $target"
    else
        echo "No SSH config for $target. Manually start on target machine:"
        echo " ssh $target"
        echo " telegram-agent-host start"
    fi
    echo ""
    echo "Enabled on $target: $target_agents"
}

bring_here() {
    local current_host
    current_host=$(get_hostname)
    local agents
    agents=$(get_enabled_agents "$current_host")

    if [ -z "$agents" ]; then
        echo "No agents configured for $current_host"
        exit 1
    fi

    echo "Bringing agents to $current_host..."

    for host_line in $(grep -v '^#' "$HOSTS_FILE" | grep -v '^$'); do
        local other_host
        other_host=$(echo "$host_line" | cut -d'|' -f1)
        if [ "$other_host" != "$current_host" ]; then
            local other_ssh
            other_ssh=$(echo "$host_line" | cut -d'|' -f3)
            if [ -n "$other_ssh" ]; then
                echo "Stopping agents on $other_host..."
                ssh "$other_ssh" "systemctl --user stop hermes-gateway.service 2>/dev/null || pkill -x hermes 2>/dev/null || true; pkill -f 'ductor' 2>/dev/null || true" 2>/dev/null || true
            fi
        fi
    done

    sleep 1
    echo "Starting agents locally..."
    start_all
}
