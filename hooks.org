* Claude statusline
#+BEGIN_SRC bash :tangle all/.local/bin/claude-statusline :shebang "#!/usr/bin/env bash" :comments no
input=$(cat)

# User and host
user=$(whoami)
host=$(hostname -s)

# Directory — basename only
cwd=$(echo "$input" | jq -r '.workspace.current_dir // .cwd')
short_cwd=$(basename "$cwd")

# Git branch from worktree or workspace
git_branch=$(echo "$input" | jq -r '.worktree.branch // .workspace.git_worktree // empty')
if [ -z "$git_branch" ] && command -v git >/dev/null 2>&1; then
  git_branch=$(git -C "$cwd" symbolic-ref --short HEAD 2>/dev/null || git -C "$cwd" rev-parse --short HEAD 2>/dev/null)
fi

# Time
time_str=$(date +%H:%M)

# Model short name
model=$(echo "$input" | jq -r '.model.display_name // empty' | sed 's/Claude //')

# Context USED (not remaining)
used=$(echo "$input" | jq -r '.context_window.used_percentage // empty')

# Token count as XK or X.XK
tokens=$(echo "$input" | jq -r '(.context_window.total_input_tokens // 0) + (.context_window.total_output_tokens // 0)')
if [ -n "$tokens" ] && [ "$tokens" != "null" ] && [ "$tokens" -gt 0 ] 2>/dev/null; then
  tk=$(echo "$tokens" | awk '{if($1>=1000) printf "%.1fK",$1/1000; else printf "%d",$1}')
else
  tk=""
fi

# 5-hour rolling window — remaining (prefer dedicated field, fall back to 100-used)
five_hr=$(echo "$input" | jq -r '
  if .rate_limits.five_hour.remaining_percentage != null
  then .rate_limits.five_hour.remaining_percentage
  elif .rate_limits.five_hour.used_percentage != null
  then (100 - .rate_limits.five_hour.used_percentage)
  else empty
  end')

# Weekly quota — remaining % and days to reset
seven_day_used=$(echo "$input" | jq -r '.rate_limits.seven_day.used_percentage // empty')
seven_day_resets=$(echo "$input" | jq -r '.rate_limits.seven_day.resets_at // empty')
week_pct=""
week_days=""
if [ -n "$seven_day_used" ] && [ "$seven_day_used" != "null" ]; then
  week_pct=$(echo "$seven_day_used" | awk '{printf "%.0f", 100-$1}')
fi
if [ -n "$seven_day_resets" ] && [ "$seven_day_resets" != "null" ] && [ "$seven_day_resets" -gt 0 ] 2>/dev/null; then
  now=$(date +%s)
  secs=$((seven_day_resets - now))
  [ $secs -lt 0 ] && secs=0
  week_days=$(awk "BEGIN{printf \"%.1f\", $secs/86400}")
fi

printf "\033[38;5;166m %s\033[0m" "$user@$host"
printf " \033[33m %s\033[0m" "$short_cwd"
if [ -n "$git_branch" ]; then
  printf " \033[34m %s\033[0m" "$git_branch"
fi
printf "\033[38;5;59m  %s\033[0m" "$time_str"
if [ -n "$model" ]; then
  printf " \033[36m✦ %s\033[0m" "$model"
fi
if [ -n "$tk" ] && [ -n "$used" ]; then
  printf " \033[38;5;59m%s (%s%%)\033[0m" "$tk" "$(printf '%.0f' "$used")"
elif [ -n "$used" ]; then
  printf " \033[38;5;59m(%s%%)\033[0m" "$(printf '%.0f' "$used")"
fi
if [ -n "$five_hr" ]; then
  printf " \033[38;5;59m5h %s%%\033[0m" "$(printf '%.0f' "$five_hr")"
fi
if [ -n "$week_pct" ] || [ -n "$week_days" ]; then
  out="wk"
  [ -n "$week_pct" ] && out="${out} ${week_pct}%"
  [ -n "$week_days" ] && out="${out} ${week_days}d"
  printf " \033[38;5;59m%s\033[0m" "$out"
fi

#+BEGIN_SRC bash :tangle all/.local/bin/dotcortex-opencode-config-merge-guard :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/dotcortex-opencode-config-merge-guard
set -euo pipefail

mode="${1:-check}"
if [ "$mode" != "check" ] && [ "$mode" != "apply" ]; then
  echo "usage: dotcortex-opencode-config-merge-guard [check|apply]" >&2
  exit 2
fi

repo="${DOTCORTEX_REPO:-$HOME/DotCortex}"
desired="${DOTCORTEX_OPENCODE_DESIRED:-$repo/all/.config/opencode/opencode.json}"
live="${DOTCORTEX_OPENCODE_LIVE:-$HOME/.config/opencode/opencode.json}"
state_home="${XDG_STATE_HOME:-$HOME/.local/state}"
marker="${DOTCORTEX_OPENCODE_MERGE_MARKER:-$state_home/dotcortex/opencode-config-merge.json}"

python3 - "$mode" "$repo" "$desired" "$live" "$marker" <<'PYEOF'
import json
import os
import shutil
import subprocess
import sys
from datetime import datetime

mode, repo, desired_path, live_path, marker_path = sys.argv[1:6]


def same_path(a, b):
    try:
        return os.path.samefile(a, b)
    except OSError:
        return False


def path_join(base, key):
    return f"{base}.{key}" if base else key


def live_only_paths(live_value, desired_value, path=""):
    if isinstance(live_value, dict):
        if not isinstance(desired_value, dict):
            return [path or "$"]
        paths = []
        for key, value in live_value.items():
            child = path_join(path, key)
            if key not in desired_value:
                paths.append(child)
                continue
            paths.extend(live_only_paths(value, desired_value[key], child))
        return paths

    if isinstance(live_value, list):
        if not isinstance(desired_value, list):
            return [path or "$"]
        return [path or "$"] if any(item not in desired_value for item in live_value) else []

    return [path or "$"] if live_value != desired_value else []


def safe_remove(path):
    if os.path.islink(path) or os.path.isfile(path):
        os.remove(path)
        return
    if os.path.isdir(path):
        shutil.rmtree(path)


def clear_marker():
    if os.path.exists(marker_path):
        os.remove(marker_path)


if not os.path.isfile(desired_path):
    print(f"error: desired config missing: {desired_path}", file=sys.stderr)
    sys.exit(1)

with open(desired_path, "r", encoding="utf-8") as handle:
    try:
        desired_value = json.load(handle)
    except json.JSONDecodeError as exc:
        print(f"error: desired config is invalid JSON: {exc}", file=sys.stderr)
        sys.exit(1)

if same_path(desired_path, live_path):
    clear_marker()
    print("status=managed")
    sys.exit(0)

if not (os.path.exists(live_path) or os.path.islink(live_path)):
    clear_marker()
    print("status=missing")
    sys.exit(0)

extra_paths = []
try:
    with open(live_path, "r", encoding="utf-8") as handle:
        live_value = json.load(handle)
    extra_paths = sorted(set(live_only_paths(live_value, desired_value)))
except json.JSONDecodeError as exc:
    extra_paths = [f"$ (invalid JSON: {exc})"]

if mode == "check":
    if extra_paths:
        print("status=backup")
        for item in extra_paths:
            print(f"extra={item}")
    else:
        print("status=overwrite")
    sys.exit(0)

os.makedirs(os.path.dirname(marker_path), exist_ok=True)

if extra_paths:
    timestamp = datetime.now().strftime("%Y%m%d-%H%M%S")
    backup_path = f"{live_path}.bak.{timestamp}"
    subprocess.run(["cp", "-a", live_path, backup_path], check=True)
    safe_remove(live_path)
    payload = {
        "repo": os.path.realpath(repo),
        "live": live_path,
        "desired": desired_path,
        "backup": backup_path,
        "created_at": datetime.now().isoformat(timespec="seconds"),
        "extra_paths": extra_paths,
    }
    with open(marker_path, "w", encoding="utf-8") as handle:
        json.dump(payload, handle, indent=2)
        handle.write("\n")
    print(f"status=backup backup={backup_path}")
    for item in extra_paths:
        print(f"extra={item}")
    sys.exit(0)

clear_marker()
safe_remove(live_path)
print("status=overwrite")
PYEOF

* Git commit guard
Block the exact failure mode that caused the recent regression:

- refuse commits from a dirty tracked worktree
- reject backup/runtime spillover files
- reject generated output commits that do not include canonical Org source
- reject giant sweep commits unless explicitly overridden

Escape hatches exist for intentional exceptional cases:

- =DOTCORTEX_GUARD_ALLOW_DIRTY=1=
- =DOTCORTEX_GUARD_ALLOW_BULK=1=
- =DOTCORTEX_GUARD_ALLOW_JUNK=1=
- =DOTCORTEX_GUARD_ALLOW_GENERATED=1=

#+BEGIN_SRC bash :tangle all/.local/bin/dotcortex-git-guard :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/dotcortex-git-guard
set -euo pipefail

note() {
  printf 'dotcortex-guard: %s\n' "$*" >&2
}

die() {
  note "$1"
  exit 1
}

repo="$(git rev-parse --show-toplevel 2>/dev/null || true)"
[ -n "$repo" ] || exit 0
cd "$repo"

allow_dirty="${DOTCORTEX_GUARD_ALLOW_DIRTY:-0}"
allow_bulk="${DOTCORTEX_GUARD_ALLOW_BULK:-0}"
allow_junk="${DOTCORTEX_GUARD_ALLOW_JUNK:-0}"
allow_generated="${DOTCORTEX_GUARD_ALLOW_GENERATED:-0}"

staged=()
while IFS= read -r -d '' path; do
  staged+=("$path")
done < <(git diff --cached --name-only -z --diff-filter=ACMRD)

[ "${#staged[@]}" -gt 0 ] || exit 0

unstaged=()
while IFS= read -r -d '' path; do
  unstaged+=("$path")
done < <(git diff --name-only -z --diff-filter=ACMRD)

if [ "${#unstaged[@]}" -gt 0 ] && [ "$allow_dirty" != "1" ]; then
  note "refusing commit from a dirty tracked worktree"
  printf '%s\n' "${unstaged[@]:0:20}" | sed 's/^/  - /' >&2
  if [ "${#unstaged[@]}" -gt 20 ]; then
    note "... and $((${#unstaged[@]} - 20)) more"
  fi
  note "stash or stage intentionally, then retry"
  note "override once with: DOTCORTEX_GUARD_ALLOW_DIRTY=1 git commit ..."
  exit 1
fi

declare -A roots=()
bad_paths=()
has_org_source=0
generated_paths=0

for path in "${staged[@]}"; do
  root="${path%%/*}"
  roots["$root"]=1

  case "$path" in
    *.org)
      has_org_source=1
      ;;
  esac

  case "$path" in
    all/*|linux/*|debian/*|devuan/*|x230/*|t480s/*|arch/*|osx/*|be/*|navi/*|think/*|.claude/*|.opencode/*|.agents/*|AGENTS.md|CLAUDE.md|Makefile)
      generated_paths=$((generated_paths + 1))
      ;;
  esac

  case "$path" in
    *.bak|*.bak.*|*.backup-*|*~|*.orig|*.rej|*.swp|*.swo|.#*|[#]*[#]|.vorg.db|.claude/stop_count|.codex|all/.hermes/channel_directory.json|all/.hermes/cron/*)
      bad_paths+=("$path")
      ;;
  esac
done

if [ "${#bad_paths[@]}" -gt 0 ] && [ "$allow_junk" != "1" ]; then
  note "refusing commit with backup/runtime spillover paths"
  printf '%s\n' "${bad_paths[@]}" | sed 's/^/  - /' >&2
  note "clean or ignore them first"
  note "override once with: DOTCORTEX_GUARD_ALLOW_JUNK=1 git commit ..."
  exit 1
fi

if [ "$generated_paths" -gt 0 ] && [ "$has_org_source" -eq 0 ] && [ "$allow_generated" != "1" ]; then
  note "generated output is staged without a canonical .org source change"
  note "override once with: DOTCORTEX_GUARD_ALLOW_GENERATED=1 git commit ..."
  exit 1
fi

root_count="${#roots[@]}"
staged_count="${#staged[@]}"
if [ "$staged_count" -gt 80 ] && [ "$root_count" -gt 10 ] && [ "$allow_bulk" != "1" ]; then
  note "refusing giant sweep commit ($staged_count staged paths across $root_count top-level roots)"
  note "split it into reviewed chunks"
  note "override once with: DOTCORTEX_GUARD_ALLOW_BULK=1 git commit ..."
  exit 1
fi

if [ "$has_org_source" -eq 1 ] && [ "$generated_paths" -eq 0 ]; then
  note "source-only commit detected; if this change should affect generated output, run tangle first"
fi

exit 0
#+END_SRC

* Safe pull guard
Wrapper around =git pull --ff-only= that refuses if the working tree has any
dirty tracked files. Prevents the exact failure mode where a fast-forward pull
silently succeeds even with uncommitted local changes that don't conflict with
the incoming commits, destroying the ability to audit what was in-flight.

Use =dotcortex-pull= instead of =git pull --ff-only= for any automated or
remote-triggered pull. The override =DOTCORTEX_PULL_ALLOW_DIRTY=1= exists for
intentional exceptions (e.g. pulling while stash is present and you know it).

#+BEGIN_SRC bash :tangle all/.local/bin/dotcortex-pull :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/dotcortex-pull
set -euo pipefail

die() {
  printf 'dotcortex-pull: %s\n' "$*" >&2
  exit 1
}

repo="$(git rev-parse --show-toplevel 2>/dev/null || true)"
[ -n "$repo" ] || die "not inside a git repo"
cd "$repo"

allow_dirty="${DOTCORTEX_PULL_ALLOW_DIRTY:-0}"

unstaged=()
while IFS= read -r -d '' path; do
  unstaged+=("$path")
done < <(git diff --name-only -z --diff-filter=ACMRD)

if [ "${#unstaged[@]}" -gt 0 ] && [ "$allow_dirty" != "1" ]; then
  printf 'dotcortex-pull: refusing pull from a dirty tracked worktree\n' >&2
  printf '%s\n' "${unstaged[@]:0:20}" | sed 's/^/  - /' >&2
  [ "${#unstaged[@]}" -gt 20 ] && printf 'dotcortex-pull: ... and %d more\n' "$((${#unstaged[@]} - 20))" >&2
  printf 'dotcortex-pull: stash or commit first\n' >&2
  printf 'dotcortex-pull: override once with: DOTCORTEX_PULL_ALLOW_DIRTY=1 dotcortex-pull\n' >&2
  exit 1
fi

exec git pull --ff-only "$@"
#+END_SRC

set -euo pipefail

echo "== claude plugins apply =="

declare -A seen_marketplaces

while IFS=' ' read -r pkg marketplace_name marketplace_repo; do
  [[ "$pkg" =~ ^[[:space:]]*# ]] && continue
  [ -z "${pkg:-}" ] && continue

  if [ -z "${seen_marketplaces[$marketplace_name]:-}" ]; then
    echo "marketplace: $marketplace_name ($marketplace_repo)"
    claude plugin marketplace add "$marketplace_repo" 2>/dev/null || true
    seen_marketplaces["$marketplace_name"]=1
  fi

  installed_json="$HOME/.claude/plugins/installed_plugins.json"
  key="${pkg}@${marketplace_name}"

  if [ -f "$installed_json" ] && python3 -c "
import json, sys
data = json.load(open('$installed_json'))
sys.exit(0 if '$key' in data.get('plugins', {}) else 1)
" 2>/dev/null; then
    echo "ok: $key"
  else
    echo "install: $key"
    claude plugin install "${pkg}@${marketplace_name}"
  fi

done < <(awk '!/^[[:space:]]*#/ && NF' "$MAN")

echo "Done. Restart Claude Code or run /reload-plugins to activate."

extract_lightcone() {
  local content="$1"
  local extracted
  extracted=$(printf '%s' "$content" | awk '/^## LightCone/{p=1} p && /^## / && !/^## LightCone/{exit} p')
  if [ -n "$extracted" ]; then
    printf '%s\n' "$extracted"
  else
    printf '%s\n' "$content"
  fi
}

# --- Machine Identity (always first — survives compaction) ---

esac

echo "--- Machine Identity ---"

# --- LightCone ---

Smart-gated skill evaluation: surface the YES/NO skill block only when the user's prompt
keyword-matches a skill, so unrelated prompts cost zero skill-overhead tokens. Falls back
to the full skill list only if the prompt text can't be parsed (non-Claude-Code callers).

#+BEGIN_SRC bash :tangle all/.local/bin/claude-hook-user-prompt-submit :shebang "#!/usr/bin/env bash"

# Fires for Claude Code (UserPromptSubmit) and OpenCode.

    if marker and mrepo and (cwd == mrepo or cwd.startswith(mrepo + os.sep)):
        extra_paths = marker.get('extra_paths', [])

* Remote git safety guard
Intercepts Bash tool calls that attempt remote git operations without proper precautions.
Blocks =ssh -A= (agent forwarding) and raw remote =git pull= without dirty-state checks.
Also blocks raw =git pull --ff-only= on remote machines when =dotcortex-pull= should be used instead.
Blocks DotCortex stow/apply mistakes unless the agent has made a recent explicit
machine-identity check and the requested stow target matches the live host.

#+BEGIN_SRC bash :tangle all/.local/bin/claude-hook-pre-tool-use-bash :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/claude-hook-pre-tool-use-bash
# PreToolUse hook for Bash — blocks unsafe remote git/stow operations.
# Exit 2 = block the tool call; stdout is shown to Claude as the reason.
# No set -euo pipefail: hooks must not abort on grep non-match (exit 1).

input=$(cat)
cmd=$(printf '%s' "$input" | python3 -c "
import sys, json
try:
    d = json.load(sys.stdin)
    print(d.get('tool_input', {}).get('command', ''))
except Exception:
    pass
" 2>/dev/null) || true

if [ -z "$cmd" ]; then
  exit 0
fi

python3 - "$cmd" <<'PYEOF'
import json
import os
import re
import shlex
import socket
import sys
import time

cmd = sys.argv[1]

HOST_ALIASES = {
    "ThinkPad-T480s": "t480s",
    "t480s": "t480s",
    "x230": "x230",
    "ThinkPad-T480": "t480",
    "t480": "t480",
    "s24": "s24",
}

EXPECTED_HOSTS = {
    "x230": {"x230"},
    "t480s": {"t480s", "ThinkPad-T480s"},
    "t480": {"t480", "ThinkPad-T480"},
    "s24": {"s24"},
}

CHECK_RE = re.compile(r"^\s*(hostname\s+-s|uname\s+-n|hostnamectl\s+hostname)\s*$")
STOW_RE = re.compile(
    r"(^|[;&|]\s*)("
    r"(?:\S*/)?loom\s+stow(?::[A-Za-z0-9_-]+)?"
    r"|(?:\S*/)?make(?:\s+-f\s+\S+)?\s+(?:safe-stow|stow|x11-apply)"
    r"|(?:\S*/)?stow\s+(?:all|linux|debian|devuan|x230|t480s|t480|s24|openmandriva)\b"
    r")"
)


def state_path():
    user = os.environ.get("USER") or str(os.getuid())
    filename = f"dotcortex-agent-host-check-{user}.json"
    for base in (os.environ.get("XDG_RUNTIME_DIR"), os.environ.get("TMPDIR"), "/tmp"):
        if not base:
            continue
        try:
            os.makedirs(base, exist_ok=True)
            if os.access(base, os.W_OK):
                return os.path.join(base, filename)
        except OSError:
            continue
    return os.path.join("/tmp", filename)


def current_host():
    raw = socket.gethostname().split(".", 1)[0]
    return raw, HOST_ALIASES.get(raw, raw)


def write_check_marker(raw, short):
    path = state_path()
    payload = {
        "host": raw,
        "short": short,
        "cwd": os.getcwd(),
        "checked_at": int(time.time()),
    }
    try:
        os.makedirs(os.path.dirname(path), exist_ok=True)
        with open(path, "w", encoding="utf-8") as handle:
            json.dump(payload, handle)
            handle.write("\n")
    except OSError:
        pass


def fresh_check(raw, short, max_age=300):
    if CHECK_RE.search(cmd):
        write_check_marker(raw, short)
        return True, "same command"

    path = state_path()
    try:
        with open(path, encoding="utf-8") as handle:
            payload = json.load(handle)
    except (OSError, json.JSONDecodeError):
        return False, "missing"

    if payload.get("host") != raw or payload.get("short") != short:
        return False, "host changed"
    age = int(time.time()) - int(payload.get("checked_at", 0))
    if age > max_age:
        return False, f"stale ({age}s)"
    return True, f"fresh ({age}s)"


def shell_words(text):
    try:
        return shlex.split(text)
    except ValueError:
        return text.split()


def stow_targets(text):
    targets = []

    for match in re.finditer(r"(?:^|[;&|]\s*)(?:\S*/)?loom\s+stow(?::([A-Za-z0-9_-]+))?", text):
        targets.append(match.group(1) or "shared")

    for match in re.finditer(r"(?:^|[;&|]\s*)(?:\S*/)?stow\s+([A-Za-z0-9_-]+)", text):
        targets.append(match.group(1))

    if re.search(r"(?:^|[;&|]\s*)(?:\S*/)?make(?:\s+-f\s+\S+)?\s+safe-stow\b", text):
        words = shell_words(text)
        pkgs = []
        for word in words:
            if word.startswith("STOW_PKGS="):
                pkgs.extend(word.split("=", 1)[1].split())
        for candidate in ("x230", "t480s", "t480", "s24"):
            if candidate in pkgs:
                targets.append(candidate)
        if not pkgs:
            targets.append("shared")

    return targets


if not STOW_RE.search(cmd):
    if CHECK_RE.search(cmd):
        raw_host, short_host = current_host()
        write_check_marker(raw_host, short_host)
    sys.exit(0)

raw_host, short_host = current_host()
checked, check_reason = fresh_check(raw_host, short_host)
targets = stow_targets(cmd)

if not checked:
    print("BLOCKED: DotCortex stow command without a fresh machine check.")
    print()
    print(f"Hook-detected machine: {raw_host} ({short_host})")
    print(f"Check state: {check_reason}")
    print()
    print("Required before stow:")
    print("  1. Run `hostname -s` or `uname -n` as its own tool call.")
    print("  2. Only then run the stow verb matching that machine.")
    print()
    print("This prevents an agent from stowing x230/t480s/s24 overlays on the wrong host.")
    sys.exit(2)

for target in targets:
    expected = EXPECTED_HOSTS.get(target)
    if expected and short_host not in expected and raw_host not in expected:
        print("BLOCKED: DotCortex stow target does not match the current machine.")
        print()
        print(f"Hook-detected machine: {raw_host} ({short_host})")
        print(f"Requested stow target: {target}")
        print()
        print("Run `hostname -s` again, then use the stow verb for the detected machine.")
        sys.exit(2)

sys.exit(0)
PYEOF

# Extract only the first token/command from the string to avoid matching
# inside heredoc bodies or commit messages. We check the first shell word
# on each line that starts an actual command (after &&, ;, or at line start).
# This avoids false-positive matches on git commit -m bodies that discuss ssh.

# Block ssh -A (agent forwarding): match ssh immediately followed by -A flag
# anywhere a ssh command would actually start (line start, after &&, after ;)
if printf '%s' "$cmd" | grep -qE '(^|&&|;)[[:space:]]*(ssh)[[:space:]]+-A([[:space:]]|$)'; then
  printf 'BLOCKED: ssh -A (agent forwarding) exposes your local SSH keys to the remote machine.\n'
  printf 'Confirm with the user that agent forwarding is acceptable before proceeding.\n'
  printf 'Consider whether the remote machine has its own key instead.\n'
  exit 2
fi

# Block remote git pull without dotcortex-pull:
# Match ssh as an actual command start, with a hostname, containing git pull on same line.
# The double-negative (no dotcortex-pull) is a guard for legitimate syncs.
if printf '%s' "$cmd" | grep -qP '(^|&&|;)\s*ssh\s+\S+.*git pull' 2>/dev/null && \
   ! printf '%s' "$cmd" | grep -qF 'dotcortex-pull'; then
  printf 'BLOCKED: remote git pull without dirty-state guard.\n\n'
  printf 'Required steps before pulling on a remote machine:\n'
  printf '  1. ssh <host> "cd ~/DotCortex && git status --short"\n'
  printf '  2. Confirm no dirty tracked files (or get explicit user approval)\n'
  printf '  3. Use dotcortex-pull instead of git pull --ff-only\n\n'
  printf 'Correct pattern:\n'
  printf '  ssh <host> "cd ~/DotCortex && dotcortex-pull && loom stow:<machine>"\n'
  exit 2
fi

exit 0
