* DNF manifest
Columns: PKG, VERSION, CATEGORY, SCOPE, EXTRA
- VERSION: empty means latest from enabled DNF repos
- CATEGORY: migration, bootstrap, cli, desktop, fonts, media, dev, system, network
- SCOPE: =openmandriva= for OpenMandriva-family hosts
- EXTRA: reserved

The T480 display migration is intentionally first. OpenMandriva's SonicDE testing note says
=task-sonicde= conflicts with Plasma and needs =--allowerasing= when Plasma is already present.
XLibre package names are OpenMandriva Cooker/RPM names; validate live with =dnf repoquery= before
running the migration on a non-Cooker/Rome image.

* Apply script
#+BEGIN_SRC bash :tangle openmandriva/.local/bin/dnf-apply :shebang "#!/usr/bin/env bash"
set -euo pipefail

MAN="${DNF_SSV:-$HOME/.dnf/manifest/packages.ssv}"
SCOPE="${DNF_SCOPE:-openmandriva}"
UPDATE_FIRST="${UPDATE_FIRST:-1}"
CATEGORIES="${DNF_CATEGORIES:-}"
ONLY_PACKAGES="${DNF_PACKAGES:-}"

if [ ! -f "$MAN" ]; then
  echo "No manifest at $MAN" >&2
  exit 1
fi

if ! command -v dnf >/dev/null 2>&1; then
  echo "dnf not found; this manifest is for OpenMandriva T480" >&2
  exit 1
fi

scope_matches() {
  local scope="$1"
  case "$scope" in
    "$SCOPE"|shared|"") return 0 ;;
    *) return 1 ;;
  esac
}

category_matches() {
  local cat="$1"
  [ -z "$CATEGORIES" ] && return 0
  case " $CATEGORIES " in
    *" $cat "*) return 0 ;;
    *) return 1 ;;
  esac
}

package_matches() {
  local pkg="$1"
  [ -z "$ONLY_PACKAGES" ] && return 0
  case " $ONLY_PACKAGES " in
    *" $pkg "*) return 0 ;;
    *) return 1 ;;
  esac
}

pkg_available() {
  local pkg="$1"
  [ -n "$(dnf -q repoquery "$pkg" 2>/dev/null | head -n 1)" ] ||
    [ -n "$(dnf -q repoquery --whatprovides "$pkg" 2>/dev/null | head -n 1)" ]
}

pkg_installed() {
  local pkg="$1"
  rpm -q "$pkg" >/dev/null 2>&1 ||
    [ -n "$(dnf -q repoquery --installed "$pkg" 2>/dev/null | head -n 1)" ] ||
    [ -n "$(dnf -q repoquery --installed --whatprovides "$pkg" 2>/dev/null | head -n 1)" ]
}

INSTALL_LIST=()
SKIPPED=()

while IFS='|' read -r pkg _ver cat scope _extra; do
  [ -n "$pkg" ] || continue
  if [ "$cat" = "migration" ]; then
    echo "Migration package $pkg is handled by openmandriva-sonicde-xlibre; skipping in dnf-apply."
    continue
  fi
  package_matches "$pkg" || continue
  category_matches "$cat" || continue
  scope_matches "$scope" || continue
  pkg_installed "$pkg" && continue
  if pkg_available "$pkg"; then
    INSTALL_LIST+=("$pkg")
  else
    SKIPPED+=("$pkg")
  fi
done < <(
  awk -v FS='[ \t]+' '
    function dq(s){ return (s == "\"\"" ? "" : s) }
    /^[ \t]*#/ || NF==0 { next }
    {
      printf "%s|%s|%s|%s|%s\n",
        dq($1), dq($2), dq($3), dq($4), dq($5)
    }
  ' "$MAN"
)

if [ ${#SKIPPED[@]} -gt 0 ]; then
  echo "Unavailable package names skipped:"
  printf "  ! %s\n" "${SKIPPED[@]}"
fi

if [ ${#INSTALL_LIST[@]} -eq 0 ]; then
  echo "All available DNF manifest packages are installed."
  exit 0
fi

if [ "$UPDATE_FIRST" = "1" ]; then
  sudo dnf --refresh distro-sync -y
fi

echo "Installing ${#INSTALL_LIST[@]} DNF packages..."
printf "  + %s\n" "${INSTALL_LIST[@]}"
sudo dnf --refresh install -y --allowerasing "${INSTALL_LIST[@]}"
#+END_SRC

* Diff script
#+BEGIN_SRC bash :tangle openmandriva/.local/bin/dnf-diff :shebang "#!/usr/bin/env bash"
set -euo pipefail

MAN="${DNF_SSV:-$HOME/.dnf/manifest/packages.ssv}"
SCOPE="${DNF_SCOPE:-openmandriva}"

if [ ! -f "$MAN" ]; then
  echo "No manifest at $MAN" >&2
  exit 1
fi

if ! command -v rpm >/dev/null 2>&1; then
  echo "rpm not found; this manifest is for OpenMandriva T480" >&2
  exit 1
fi

wanted_tmp="$(mktemp)"
have_tmp="$(mktemp)"
trap 'rm -f "$wanted_tmp" "$have_tmp"' EXIT

awk -v FS='[ \t]+' -v scope_name="$SCOPE" '
  function dq(s){ return (s == "\"\"" ? "" : s) }
  /^[ \t]*#/ || NF==0 { next }
  {
    pkg = dq($1)
    scope = dq($4)
    if (scope == scope_name || scope == "shared" || scope == "") print pkg
  }
' "$MAN" | sort -u > "$wanted_tmp"

rpm -qa --qf '%{NAME}\n' | sort -u > "$have_tmp"

to_install="$(comm -23 "$wanted_tmp" "$have_tmp")"

echo "=== dnf:diff ==="
echo
if [ -n "$to_install" ]; then
  count=$(echo "$to_install" | wc -l)
  echo "[to install] ($count packages before repo availability filtering)"
  echo "$to_install" | sed 's/^/  + /'
else
  echo "[to install] (none)"
fi

echo
echo "Manifest: $(wc -l < "$wanted_tmp") packages for scope=$SCOPE"
echo "Installed: $(wc -l < "$have_tmp") rpm packages"
#+END_SRC

* Capture script
#+BEGIN_SRC bash :tangle openmandriva/.local/bin/dnf-capture :shebang "#!/usr/bin/env bash"
set -euo pipefail

MAN="${DNF_SSV:-$HOME/DotCortex/openmandriva/.dnf/manifest/packages.ssv}"

if [ ! -f "$MAN" ]; then
  echo "No manifest at $MAN; creating with header" >&2
  mkdir -p "$(dirname "$MAN")"
  echo '# PKG VERSION CATEGORY SCOPE EXTRA' > "$MAN"
fi

existing_tmp="$(mktemp)"
manual_tmp="$(mktemp)"
trap 'rm -f "$existing_tmp" "$manual_tmp"' EXIT

awk -v FS='[ \t]+' '!/^[ \t]*#/ && NF { print $1 }' "$MAN" | sort -u > "$existing_tmp"
dnf repoquery --userinstalled --qf '%{name}' | sort -u > "$manual_tmp"

new_pkgs="$(comm -23 "$manual_tmp" "$existing_tmp")"

if [ -z "$new_pkgs" ]; then
  echo "No new user-installed DNF packages to capture."
  exit 0
fi

count=$(echo "$new_pkgs" | wc -l)
echo "Found $count new user-installed packages:"
echo "$new_pkgs" | sed 's/^/  /'
echo

echo "Appending to $MAN with category=uncategorized scope=openmandriva"
echo "" >> "$MAN"
echo "# --- Captured $(date +%Y-%m-%d) ---" >> "$MAN"
while IFS= read -r pkg; do
  printf '%s "" uncategorized openmandriva ""\n' "$pkg" >> "$MAN"
done <<< "$new_pkgs"
#+END_SRC

* SonicDE/XLibre migration
#+BEGIN_SRC bash :tangle openmandriva/.local/bin/openmandriva-sonicde-xlibre :shebang "#!/usr/bin/env bash"
set -euo pipefail

MAN="${DNF_SSV:-$HOME/.dnf/manifest/packages.ssv}"

if [ ! -f "$MAN" ]; then
  echo "No manifest at $MAN" >&2
  exit 1
fi

if ! command -v dnf >/dev/null 2>&1; then
  echo "dnf not found; this migration is for OpenMandriva T480" >&2
  exit 1
fi

mapfile -t MIGRATION_PKGS < <(
  awk -v FS='[ \t]+' '
    function dq(s){ return (s == "\"\"" ? "" : s) }
    /^[ \t]*#/ || NF==0 { next }
    dq($3) == "migration" && dq($4) == "openmandriva" { print dq($1) }
  ' "$MAN"
)

SONICDE_CONFLICTS=(
  lib64kworkspace6
  qml-org.kde.plasma.private.sessions
  qml-org.kde.plasma.workspace
  qml-org.kde.breeze.components
)

if [ ${#MIGRATION_PKGS[@]} -eq 0 ]; then
  echo "No migration packages found in $MAN" >&2
  exit 1
fi

echo "SonicDE/XLibre package set:"
printf "  + %s\n" "${MIGRATION_PKGS[@]}"
echo
echo "This will allow DNF to erase conflicting Plasma/Xorg packages."

sudo dnf --refresh distro-sync -y --allowerasing

mapfile -t INSTALLED_CONFLICTS < <(
  for pkg in "${SONICDE_CONFLICTS[@]}"; do
    rpm -q "$pkg" >/dev/null 2>&1 && printf '%s\n' "$pkg"
  done
)

if [ ${#INSTALLED_CONFLICTS[@]} -gt 0 ]; then
  echo
  echo "Removing Plasma packages that conflict at the RPM file level:"
  printf "  - %s\n" "${INSTALLED_CONFLICTS[@]}"
  sudo dnf remove -y "${INSTALLED_CONFLICTS[@]}"
fi

sudo dnf --refresh install -y --allowerasing "${MIGRATION_PKGS[@]}"
#+END_SRC
