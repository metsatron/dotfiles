* Install termux packages
#+BEGIN_SRC bash :tangle termux/.local/bin/termux-pkg-apply :shebang "#!/usr/bin/env bash"
# ~/DotCortex/termux/.local/bin/termux-pkg-apply
set -euo pipefail

MANIFEST="$HOME/DotCortex/termux/.termux-pkg/manifest/packages.ssv"

if [ ! -f "$MANIFEST" ]; then
  echo "No Termux package manifest at $MANIFEST"
  exit 0
fi

echo "== termux pkg apply =="

awk -v FS='[ \t]+' '
  function dequote(s){ return (s=="\"\"" ? "" : s) }
  $0 ~ /^[[:space:]]*#/ || NF==0 { next }
  {
    n1=$1; n2=$2; n3=$3; n4=$4;
    printf "%s|%s|%s|%s\n",
      dequote(n1),dequote(n2),dequote(n3),dequote(n4)
  }
' "$MANIFEST" | while IFS='|' read -r pkg _ver _scope _extra; do
  [ -n "$pkg" ] || continue
  if command -v "$pkg" >/dev/null 2>&1; then
    echo "[$pkg] already installed"
  else
    echo ">> [$pkg] installing via pkg"
    if pkg install -y "$pkg"; then
      echo "[$pkg] installed"
    else
      echo "!! [$pkg] install failed"
    fi
  fi
done

echo "Done."
#+END_SRC

* Termux health check
#+BEGIN_SRC bash :tangle termux/.local/bin/termux-pkg-health :shebang "#!/usr/bin/env bash"
# ~/DotCortex/termux/.local/bin/termux-pkg-health
set -euo pipefail

MANIFEST="$HOME/DotCortex/termux/.termux-pkg/manifest/packages.ssv"

if [ ! -f "$MANIFEST" ]; then
  echo "No Termux package manifest at $MANIFEST"
  exit 0
fi

echo "== termux pkg health =="

awk -v FS='[ \t]+' '
  function dequote(s){ return (s=="\"\"" ? "" : s) }
  $0 ~ /^[[:space:]]*#/ || NF==0 { next }
  {
    n1=$1;
    printf "%s\n", dequote(n1)
  }
' "$MANIFEST" | while read -r pkg; do
  [ -n "$pkg" ] || continue
  if command -v "$pkg" >/dev/null 2>&1; then
    echo "[$pkg] in PATH at $(command -v "$pkg")"
  else
    echo "[$pkg] NOT in PATH"
  fi
done
#+END_SRC
