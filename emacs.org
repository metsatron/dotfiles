   '(mixed-pitch org-appear org-modern valign vterm doom-themes centaur-tabs kurecolor ellsp

            shell-default-height 10 shell-default-position 'bottom))

  (when (and gp (file-directory-p (expand-file-name "bin" gp)))
    (add-to-list 'exec-path (expand-file-name "bin" gp))

(defun dotspacemacs/user-init ()
  ;; Every dotfile in $HOME is a stow symlink into the DotCortex git repo, so the
  ;; default `vc-follow-symlinks' value of 'ask fires "Symbolic link to
  ;; Git-controlled source file; follow link?" on essentially every file visit.
  ;; Follow silently: we always want the real file under version control, never
  ;; the link. NOTE this means visiting e.g. ~/.spacemacs.d.claude/init.el lands
  ;; you in TANGLED OUTPUT — read it, never edit it; edit emacs.org instead.
  (setq vc-follow-symlinks t)
  (let* ((cache (expand-file-name "~/.spacemacs.d.claude/.cache/"))
         (elpa  (expand-file-name "elpa" cache))
         (cust  (expand-file-name "custom.el" (expand-file-name "~/.spacemacs.d.claude/"))))
    (unless (file-exists-p cache) (make-directory cache t))
    (setq spacemacs-cache-directory cache
          package-user-dir          elpa
          custom-file               cust)
    (setq recentf-save-file (expand-file-name "recentf" cache)
          recentf-max-saved-items 200
          recentf-auto-cleanup 'never)
    (require 'recentf nil t)
    (when (fboundp 'recentf-mode) (recentf-mode 1)))
  (unless (boundp 'treemacs--buffer-name-prefix)
    (defvar treemacs--buffer-name-prefix " *Treemacs-"))
  ;; Prevent evil-collection from touching NEIO-displaced keys
  (setq evil-collection-key-blacklist '("h" "o" "H" "O")))
#+END_SRC

;; those aliases as startup errors even when user config uses no obsolete names.

  (defconst metsatron/spacemacs-hybrid-alias-warnings
    '("hybrid-mode-default-state"
      "hybrid-mode-enable-hjkl-bindings"
      "hybrid-mode-enable-evilified-state"
      "hybrid-mode-use-evil-search-module"))

  (defun metsatron/filter-spacemacs-hybrid-alias-warnings (warnings)
    (seq-remove
     (lambda (warning)
       (seq-some
        (lambda (name) (string-match-p (regexp-quote name) warning))
        metsatron/spacemacs-hybrid-alias-warnings))
     warnings))

  (advice-add 'spacemacs//check-obsolete-variables
              :filter-return #'metsatron/filter-spacemacs-hybrid-alias-warnings))

(with-eval-after-load 'org
  (setq org-hide-emphasis-markers t

  (defvar metsatron/org-stars-hidden t)
  (defvar metsatron/org--hide-stars-rule '(("^\\(\\*+\\) " (1 'org-hide))))
  (defun metsatron/org--apply-stars-visibility ()
    (if metsatron/org-stars-hidden
        (progn
          (setq org-hide-leading-stars t)
          (set-face-attribute 'org-hide nil :foreground (face-background 'default nil t) :inherit nil)
          (font-lock-add-keywords nil metsatron/org--hide-stars-rule 'append))
      (setq org-hide-leading-stars nil)
      (font-lock-remove-keywords nil metsatron/org--hide-stars-rule)
      (set-face-attribute 'org-hide nil :foreground nil :inherit 'shadow))
    (font-lock-flush) (font-lock-ensure))
  (defun metsatron/org-apply-heading-faces ()
    (set-face-attribute 'org-document-title nil :inherit 'variable-pitch :weight 'bold :foreground "#ffffff" :height 173)
    (set-face-attribute 'org-document-info  nil :inherit 'variable-pitch :weight 'bold :foreground "#ffffff")
    (set-face-attribute 'org-level-1 nil :inherit 'variable-pitch :weight 'bold :height 158)
    (set-face-attribute 'org-level-2 nil :inherit 'variable-pitch :weight 'bold :height 143)
    (set-face-attribute 'org-level-3 nil :inherit 'variable-pitch :weight 'bold :height 120)
    (set-face-attribute 'org-level-4 nil :inherit 'variable-pitch :weight 'bold :height 110)
    (dolist (lv '(org-level-5 org-level-6 org-level-7 org-level-8))
      (set-face-attribute lv nil :inherit 'variable-pitch :weight 'bold :height 110)))
  (defun metsatron/org-looks ()
    (variable-pitch-mode 1)
    (dolist (f '(org-code org-verbatim org-table
                          org-block org-block-begin-line org-block-end-line
                          org-meta-line org-special-keyword org-checkbox))
      (set-face-attribute f nil :inherit 'fixed-pitch))
    (metsatron/org-apply-heading-faces)
    (metsatron/org--apply-stars-visibility))
  (add-hook 'org-mode-hook #'metsatron/org-looks)

  ;; Tables that read like documents, not ASCII art.
  ;;
  ;; valign pixel-aligns cells so columns stay true in a variable-pitch buffer;
  ;; valign-fancy-bar draws a real vertical rule instead of the `|' glyph. Both
  ;; are display-only — the buffer text stays a plain Org table, so TAB, column
  ;; motion, sorting and formulas all behave normally.
  ;;
  ;; org-modern is loaded for its TABLE rendering (continuous horizontal rules in
  ;; place of |---+---|). Every org-modern feature that would paint headings,
  ;; stars, emphasis or blocks is switched OFF: metsatron/org-apply-heading-faces
  ;; and metsatron/org--apply-stars-visibility already own those surfaces, and two
  ;; painters on one glyph is how you get a heading that flickers between styles.
  (with-eval-after-load 'org
    (setq valign-fancy-bar t
          org-modern-table t
          org-modern-star nil
          org-modern-hide-stars nil
          org-modern-list nil
          org-modern-checkbox nil
          org-modern-todo nil
          org-modern-tag nil
          org-modern-priority nil
          org-modern-timestamp nil
          org-modern-block-name nil
          org-modern-keyword nil)
    (add-hook 'org-mode-hook #'valign-mode)
    (add-hook 'org-mode-hook #'org-modern-mode))
  (advice-add 'load-theme :after
              (lambda (&rest _)
                (dolist (b (buffer-list))
                  (with-current-buffer b
                    (when (derived-mode-p 'org-mode)
                      (metsatron/org-looks))))))
  (defun metsatron/org-toggle-pretty () (interactive)
    (setq org-hide-emphasis-markers (not org-hide-emphasis-markers))
    (setq metsatron/org-stars-hidden (not metsatron/org-stars-hidden))

  (define-key org-mode-map (kbd "C-c e") #'metsatron/org-toggle-pretty)
  (spacemacs/set-leader-keys "oe" #'metsatron/org-toggle-pretty))

;; --- VTERM: clean monospace, no extra spacing, no prettify-symbols ---

;; Terminal font preference, in the order fonts.org declares under "Preferred
;; SF Mono fallback candidates". The first family actually installed wins, so
;; the declared order is load-bearing rather than decorative. IosevkaTerm is
;; the tail guarantee, not the choice.
;;
;; TUI column alignment does NOT depend on this family: blk-glyphs pins the
;; box-drawing range (U+2500–U+257F) to IosevkaTerm globally, so a proportional
;; slip in the text font cannot break Claude's / any TUI's box rendering.
;;
;; The second entry read "FiraPro Nerd Font" until 2026-07-13. Nerd Fonts ships
;; no such family, so it had always been a silent no-op. Resolved to FiraCode,
;; which is what was meant and what is installed.
(defvar metsatron/terminal-font-preference
  '("MesloLGS Nerd Font Mono"
    "FiraCode Nerd Font Mono"
    "Mononoki Nerd Font Mono"
    "IosevkaTerm Nerd Font Mono")
  "Monospace families for terminal buffers, most-preferred first.
Order is declared in `fonts.org'. `metsatron/terminal-font' picks the first
family that is actually installed.")

(defvar metsatron/terminal-font-height 105
  "Face height for terminal buffers (1/10 pt).
Chosen by measurement, not taste: at 105 MesloLGS renders an 11px glyph, which
matches the rest of Emacs and the standalone terminal. (org-mode sits at 12px
by its own choice.) The previous 100 rendered 10px — a pixel short of everything
around it, which is what made terminal buffers read as subtly undersized.

Beware the two different numbers here: Emacs reports this font as pixel size 14
in its XLFD (the em box), while the glyph you actually see is 11px. Measure the
glyph, not the XLFD.

Change the size HERE. `text-scale-mode' (C-x C-+) is buffer-local, per-session,
and dies with the Emacs process — it cannot express a fleet-wide default.")

(defun metsatron/terminal-font ()
  "First installed family from `metsatron/terminal-font-preference'.
Returns nil if none are installed, in which case the frame default stands."
  (seq-find (lambda (family) (member family (font-family-list)))
            metsatron/terminal-font-preference))

(with-eval-after-load 'vterm
  (defun metsatron/vterm-fonts ()
    (variable-pitch-mode 0)
    ;; turn off prettify-symbols (can distort TUIs)
    (setq-local prettify-symbols-alist nil)
    (when (boundp 'prettify-symbols-mode)
      (prettify-symbols-mode -1))
    (when-let ((family (metsatron/terminal-font)))
      (let ((spec `(:family ,family :height ,metsatron/terminal-font-height)))
        (face-remap-add-relative 'default     spec)
        (face-remap-add-relative 'fixed-pitch spec)))
    ;; no extra line padding
    (setq-local line-spacing 0))
  (add-hook 'vterm-mode-hook #'metsatron/vterm-fonts))

;; --- Scrolling a full-screen TUI (the Claude CLI) ---
;;
;; The Claude CLI paints on the terminal's ALTERNATE SCREEN, which by terminal
;; convention writes nothing to the scrollback ring. Its buffer therefore holds
;; exactly one screenful, forever. Scrollback exists — but it is VIRTUAL, owned
;; by the CLI itself (enabled by CLAUDE_CODE_NO_FLICKER=1, see blk-claude-agents)
;; and reachable only by sending scroll keys THROUGH to the application.
;;
;; The trap: Emacs binds the wheel to `mwheel-scroll' and S-PageUp to
;; `scroll-down-command'. Both scroll the *Emacs buffer* — which is one screen
;; tall — so they appear completely dead, while plain PageUp (which vterm
;; forwards to the app as `vterm--self-insert') scrolls perfectly. The keys were
;; never broken; Emacs was intercepting them before the CLI ever saw them.
;;
;; So: in Claude buffers ONLY, forward the wheel and S-PageUp/S-PageDown to the
;; application. A plain shell vterm keeps Emacs-side scrolling, which is correct
;; there — it has a real scrollback ring (vterm-max-scrollback) to scroll through.
(defun metsatron/claude-tui-buffer-p ()
  "Non-nil if the current buffer hosts a Claude CLI TUI on the alternate screen.
Matches both clients: `*claude-code[PROJECT]*' (claude-code-ide) and
`*claude:DIR:NAME*' (claude-code.el). A plain `*vterm*' shell does not match."
  (and (derived-mode-p 'vterm-mode)
       (string-match-p "claude" (buffer-name))))

(defun metsatron/claude-scroll-back (&optional _event)
  "Scroll the Claude TUI back a page, by sending PageUp to the application."
  (interactive)
  (vterm-send-prior))

(defun metsatron/claude-scroll-forward (&optional _event)
  "Scroll the Claude TUI forward a page, by sending PageDown to the application."
  (interactive)
  (vterm-send-next))

(defun metsatron/claude-scroll-keys ()
  "Route scroll gestures to the CLI in Claude buffers instead of to Emacs."
  (when (metsatron/claude-tui-buffer-p)
    (let ((map (make-sparse-keymap)))
      (set-keymap-parent map (current-local-map))
      ;; wheel: X11 sends mouse-4/5, GUI Emacs also sends wheel-up/down
      (define-key map [wheel-up]      #'metsatron/claude-scroll-back)
      (define-key map [wheel-down]    #'metsatron/claude-scroll-forward)
      (define-key map [mouse-4]       #'metsatron/claude-scroll-back)
      (define-key map [mouse-5]       #'metsatron/claude-scroll-forward)
      (define-key map (kbd "S-<prior>") #'metsatron/claude-scroll-back)
      (define-key map (kbd "S-<next>")  #'metsatron/claude-scroll-forward)
      (use-local-map map))))

(with-eval-after-load 'vterm
  (add-hook 'vterm-mode-hook #'metsatron/claude-scroll-keys))

(with-eval-after-load 'vterm
  ;; SINGLE OWNER of vterm-max-scrollback. It used to be set in three separate
  ;; blocks — 100000 here, 100000 again below, then 10000 in `blk-vterm', which
  ;; is assembled LAST and therefore silently won. The declared 100000 had never
  ;; once been in effect. Set it here and nowhere else.
  ;;
  ;; Note this governs the PRIMARY screen only. A full-screen TUI on the
  ;; alternate screen (the Claude CLI) writes no scrollback ring at all, by
  ;; terminal convention — see `CLAUDE_CODE_NO_FLICKER' in blk-claude-agents.
  ;; Raising this number will never give the Claude buffer scrollback.
  (setq vterm-max-scrollback 100000)

  (defun la/vterm-freeze-when-scrolled ()
    "Freeze vterm output when not at EOB by enabling vterm-copy-mode."
    (when (derived-mode-p 'vterm-mode)

** Claude Code inside Emacs — three clients, one leader prefix
Three clients live side by side under =SPC o= (the existing "open" prefix). They are not redundant: each is strong somewhere the others are weak, and the differences are load-bearing.

=claude-code-ide= is the default for real work. It runs the actual =claude= binary in a terminal buffer, so the DotCortex harness — every hook in =hooks.org=, every skill in =skills.org=, every plugin — fires exactly as it does in a bare terminal. On top of that it bridges Emacs to Claude over MCP, exposing xref, imenu, project info and flycheck diagnostics as tools, so Claude can see which buffer you are in.

=agent-shell= is the opposite trade. It does not run the CLI at all: it is a comint buffer speaking ACP (Agent Client Protocol) to Zed's =claude-code-acp= adapter. Because it is a native Emacs buffer rather than a TUI in a terminal emulator, Evil behaves properly in it — which is the single most-complained-about friction in the terminal-backed clients. **But hooks are not supported over ACP and skills surface only partially: the DotCortex harness is INERT inside agent-shell, and it fails silently — nothing tells you the guards did not run.** Use it for exploration and prose, never for governed DotCortex work.

=claude-code= is the thin one: real CLI, terminal buffer, no MCP bridge. It is here as the minimal comparison point, and because it carries a =ghostel= backend (libghostty) alongside =eat= and =vterm=.

Bindings are mnemonic letters per the Colemak-NEIO law (rule 18): =c= claude(-ide), =a= agent-shell, =s= simple. No movement keys are consumed.

** Claude workspace layout

A persp-mode layout on =SPC l c= (=@Claude=): Treemacs | editor | Claude, with
Claude at 80 columns.

It is a *layout*, not a window command, and the distinction is the whole design.
The body runs inside a new perspective, so invoking it leaves the layout you are
currently in — the workbench, say — completely intact.

The first version was an =interactive= command that ran =delete-other-windows= on
the current frame. It was picked up by =M-x= autocompletion and run by accident,
and it did exactly what it was written to do: it demolished the live workbench.
Two lessons, both paid for in 2026-07-13:

1. A "new workspace" belongs in the workspace system
   (=spacemacs|define-custom-layout=), not in a function that rearranges the
   frame you are standing in. "New" must mean *additional*, not *in place of*.
2. =(interactive)= is a *capability grant*, not a convenience. It publishes a
   command to =M-x= and to every completion frontend, where a destructive one
   will eventually be selected by accident. =metsatron/claude-workspace-build=
   is therefore deliberately NOT interactive: the layout is the only door in.
   Do not add an =(interactive)= form to it, and do not give it a global key.

The trap this exists to route around: =claude-code-ide-window-width= is only
consulted by =claude-code-ide--display-buffer-in-side-window=, which binds a
=display-buffer-alist= entry with a =window-width= resizer. If the Claude buffer
reaches the screen by any *other* path — an ordinary =display-buffer=, or a
window layout restored by a session/desktop restore — it lands in a plain window
with =window-side= nil, and the width variable is never read at all. Setting
=claude-code-ide-window-width= then appears to do nothing, at any value,
because the knob is not wired to that window. (Observed 2026-07-13: a 73-column
window on a 211-column frame, with the variable set to 100.)

So this layout does not *ask* for a width — it rebuilds the frame and displays
Claude through =claude-code-ide='s own side-window path, which is the only path
that applies it.

(defun metsatron/layout--reset ()
  "Collapse the current perspective to ONE ordinary window.
This is the foundation both layouts stand on, and it is deliberately paranoid.

`delete-other-windows' is a FALSE RESET. Side windows (Treemacs) are exempt from
it by design, and claude-code-ide stamps its window with `no-delete-other-windows'
which vetoes it outright. A layout that \"resets\" with a plain call therefore does
not start from a blank frame at all — it builds on top of whatever survived. That
is how the old workbench produced a duplicate Treemacs, a Claude buffer displaced
into Treemacs' left slot, and finally a window tree containing no ordinary window
at all (2026-07-14; recovering it needed a full-tree parameter walk).

Two things make this reliable where the naive version was not:

1. `window-side' also lives on the INTERNAL parent windows, which `window-list'
   never returns. The tree is walked in full, internal nodes included — clearing
   only the leaves leaves the tree corrupt and Emacs signals \"Window X has not
   same side left as its parent\".
2. `ignore-window-parameters' is bound so `no-delete-other-windows' cannot veto
   the collapse.

NOTE this deletes WINDOWS, never buffers or processes. No Claude session dies
here — see `metsatron/claude--live-buffer' for why that matters."
  (letrec ((clear (lambda (w)
                    (set-window-parameter w 'window-side nil)
                    (set-window-parameter w 'window-slot nil)
                    (set-window-parameter w 'no-delete-other-windows nil)
                    (let ((c (window-child w)))
                      (while c
                        (funcall clear c)
                        (setq c (window-next-sibling c)))))))
    (funcall clear (frame-root-window)))
  (let ((ignore-window-parameters t))
    (delete-other-windows))
  ;; Transient popups can otherwise reclaim a side slot the moment we rebuild.
  (dolist (name '("*helm mini*" "*helm buffers*" "*which-key*"))
    (when (get-buffer name)
      (let ((kill-buffer-query-functions nil))
        (ignore-errors (kill-buffer name))))))

  (or (seq-find (lambda (b)

                       (not (string-prefix-p " " (buffer-name b)))))

  (when-let ((editor (seq-find (lambda (w) (not (window-parameter w 'window-side)))
                               (window-list))))
    (select-window editor)))

(defun metsatron/layout--vterm-in (window)
  "Put a shell in WINDOW — reusing the existing *vterm* rather than spawning."
  (select-window window)
  (if-let ((existing (get-buffer "*vterm*")))
      (set-window-buffer window existing)
    (if (fboundp 'vterm) (vterm) (ansi-term (or (getenv "SHELL") "/bin/bash")))))

(defun metsatron/claude--live-buffer ()
  "A Claude TUI buffer with a LIVE process, from ANY client, or nil.
Prefers claude-code-ide's buffer for this project, then any other live Claude
buffer — `claude-code.el' names its buffer `*claude:DIR:NAME*', which the old
builder did not recognise.

This exists because ADOPTING a running session is the only safe move. The old
builder looked only for claude-code-ide's buffer, failed to find it while a
claude-code.el session was running, and started a fresh CLI with `-r' — which
raced the live session and killed it, taking the user's window with it. Never
spawn a Claude process when one is already running. Never do window surgery
around a live Claude buffer beyond moving it."
  (let ((ide (and (fboundp 'claude-code-ide--get-buffer-name)
                  (get-buffer (claude-code-ide--get-buffer-name)))))
    (or (and ide (get-buffer-process ide) ide)
        (seq-find (lambda (b)
                    (and (buffer-live-p b)
                         (string-match-p "claude" (buffer-name b))
                         (with-current-buffer b (derived-mode-p 'vterm-mode))
                         (get-buffer-process b)))
                  (buffer-list)))))

;;; --- @Claude -------------------------------------------------------------

(defun metsatron/claude-workspace-build ()
  "Treemacs | editor | [ Claude / vterm ].
Not interactive on purpose — reach it through the @Claude layout. See below."
  (metsatron/layout--reset)
  (switch-to-buffer (metsatron/layout--editor-buffer))
  (metsatron/layout--treemacs)
  (let* ((editor (selected-window))
         (right  (split-window-right
                  (max 40 (- (window-total-width editor)
                             metsatron/claude-workspace-width))))
         (claude (metsatron/claude--live-buffer)))
    (if claude
        ;; ADOPT the running session. `set-window-buffer' just moves the buffer
        ;; into a window we made — it never touches the process, so the session
        ;; cannot die here. This is the path that must always be taken when a
        ;; Claude is already running, whichever client started it.
        (set-window-buffer right claude)
      ;; Nothing running: start one. claude-code-ide displays the buffer through
      ;; its own path, and its width defcustom is set globally (NOT let-bound)
      ;; because that display happens ASYNCHRONOUSLY, after any `let' here has
      ;; already exited — which is why a let-bound 80 silently came out as 98.
      (delete-window right)
      (setq claude-code-ide-use-side-window t
            claude-code-ide-window-side 'right
            claude-code-ide-window-width metsatron/claude-workspace-width)
      (claude-code-ide)
      (setq right (and (fboundp 'claude-code-ide--get-buffer-name)
                       (get-buffer-window (claude-code-ide--get-buffer-name)))))
    ;; The vterm strip lives UNDER the Claude panel — the loom console, where
    ;; shell work happens without leaving the workspace.
    ;; A NEGATIVE size means "give the NEW window this many lines". Sizing the
    ;; split directly is deterministic; measuring afterwards and resizing is not.
    ;; +1 for the mode line, so the BODY is exactly term-height.
    ;;
    ;; A post-hoc `window-resize' against `window-body-height' was tried and is
    ;; WRONG: the window has not settled when it is measured, so the correction
    ;; compounds and the layout DRIFTS a line on every rebuild (observed: vterm
    ;; 11 -> 12, Claude 37 -> 38 across two builds). A reset that does not
    ;; converge is not a reset. Do not reintroduce it.
    (when (window-live-p right)
      (let ((term (split-window right
                                (- (1+ metsatron/claude-workspace-term-height))
                                'below)))
        (metsatron/layout--vterm-in term)))
    ;; Width is safe to normalise: it is measured against the editor split, which
    ;; has settled by now. `split-window' sizes the TOTAL window, so fringes eat
    ;; into it — an 80-column split yields a 78-column body.
    (when (window-live-p right)
      (let ((dw (- metsatron/claude-workspace-width (window-body-width right))))
        (unless (zerop dw) (ignore-errors (window-resize right dw t)))))
    (when (window-live-p editor) (select-window editor)))
  ;; vterm learns its size from the window; after the splits above the TUI can be
  ;; left painted at the old width. Make it resync rather than wait for a keypress.
  (when (fboundp 'claude-code-ide--sync-terminal-dimensions)
    (ignore-errors (claude-code-ide--sync-terminal-dimensions))))

** Loom verb runner — annotated completion (after Josep Bigorra's heks-emacs)
Run any loom verb from Emacs with =SPC o l= (or =C-c m l=): pick a verb via =completing-read= with its description as an aligned annotation, then watch it run in a =compilation= buffer from =~/DotCortex=.

Gratefully adapted from Josep Bigorra's maak task runner in [[https://codeberg.org/jjba23/heks-emacs/src/branch/trunk/src/modules/heks-emacs.el][heks-emacs]] — the project whose =maak= sparked loom in the first place. His version statically parses project =maak.scm= files for =(define (task …))= forms; loom registers verbs as =(task 'name …)= data in =~/.config/maak/maak.scm=, so this adaptation asks the live =loom= binary for its listing instead. Same UX, and the picker can never drift from what =loom.org= actually registers. Kept from his design: the =defgroup=/=defcustom=/=defface= customization surface, =--= private helper naming, =user-error= for expected failures, the =:annotation-function= alignment technique, =shell-quote-argument= command building, and =compile= as the runner.

#+NAME: blk-loom
#+BEGIN_SRC emacs-lisp
;; Loom verb runner — adapted with gratitude from Josep Bigorra's heks-emacs
;; maak runner: https://codeberg.org/jjba23/heks-emacs — jjba23
(defgroup loom nil
  "Customizations for the DotCortex loom task runner."
  :group 'tools)

(defface loom-task-documentation-face
  '((t (:inherit font-lock-comment-face)))
  "Face for loom verb descriptions in completion annotations."
  :group 'loom)

(defcustom loom-annotation-padding 4
  "Columns between the longest verb name and its description."
  :type 'integer
  :group 'loom)

(defun loom--binary ()
  "Return the loom executable, or signal a `user-error' when absent."
  (or (executable-find "loom")
      (let ((fallback (expand-file-name "~/.local/bin/loom")))
        (and (file-executable-p fallback) fallback))
      (user-error "No `loom' executable found; is this a stowed DotCortex machine?")))

(defun loom--get-tasks ()
  "Return an alist of (VERB . DESCRIPTION) from the live `loom' listing.
Verb lines are two-space-indented lowercase names; option lines like
`-h' and section headers do not match."
  (let (tasks)
    (with-temp-buffer
      (call-process (loom--binary) nil t nil)
      (goto-char (point-min))
      (while (re-search-forward
              "^  \\([a-z][a-zA-Z0-9:!_-]*\\)[ \t]+\\(.+\\)$" nil t)
        (push (cons (match-string 1) (match-string 2)) tasks)))
    (nreverse tasks)))

(defun loom--make-annotation-function (tasks)
  "Return an annotation function aligning TASKS descriptions in one column."
  (let* ((max-len (seq-max (mapcar (lambda (cell) (length (car cell))) tasks)))
         (doc-align (+ max-len loom-annotation-padding)))
    (lambda (cand)
      (let ((desc (cdr (assoc cand tasks))))
        (if desc
            (concat (propertize " " 'display `(space :align-to ,doc-align))
                    (propertize desc 'face 'loom-task-documentation-face))
          "")))))

(defun loom-run-task (verb)
  "Select a loom VERB with completion and run it in a compilation buffer.
Verbs ending in `!' are destructive by DotCortex convention and ask
for confirmation first. Runs from ~/DotCortex so relative paths in
loom recipes resolve as they do in a shell."
  (interactive
   (let ((tasks (loom--get-tasks)))
     (unless tasks
       (user-error "No loom verbs found; check that `loom' runs"))
     (let ((completion-extra-properties
            `(:annotation-function ,(loom--make-annotation-function tasks))))
       (list (completing-read "Run loom verb: " tasks nil t)))))
  (when (and (string-suffix-p "!" verb)
             (not (yes-or-no-p (format "`%s' is destructive — run it? " verb))))
    (user-error "Aborted"))
  (let ((default-directory (expand-file-name "~/DotCortex/")))
    (compile (concat (shell-quote-argument (loom--binary)) " "
                     (shell-quote-argument verb)))))

(global-set-key (kbd "C-c m l") #'loom-run-task)
(when (fboundp 'spacemacs/set-leader-keys)
  (spacemacs/set-leader-keys "ol" #'loom-run-task))
#+END_SRC
