* Shared spoken-text normaliser

=voice-text-normalize= is the single fleet speech boundary.  It lives in the
universal =all/.local/bin= overlay because Claude, OpenCode/DeepSeek's operator
relay, Honey's Claude bots and MailCortex all need the same transformation but
do not share a TTS engine.  Callers pass reply text on stdin, or a complete hook
payload with =--hook-json=.  MarkdownV2 unescaping is format-aware; the remaining
cleanup applies to every reply format.  Fenced code stays intelligible when it
is short, while a block over 600 characters or 20 lines becomes the bounded
phrase =code block omitted=.

#+BEGIN_SRC python :tangle all/.local/bin/voice-text-normalize :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
"""Turn Telegram Markdown/MarkdownV2 reply text into bounded plain speech."""

import argparse
import json
import re
import sys


FENCED_CODE_CHAR_LIMIT = 600
FENCED_CODE_LINE_LIMIT = 20
MARKDOWN_V2_RESERVED = set("_*[]()~`>#+-=|{}.!" + chr(92))
FENCE_START = re.compile(r"^[ \t]{0,3}(`{3,}|~{3,})[^\r\n]*$")
LINK = re.compile(r"!?\[([^\]\n]+)\]\((?:[^()\n]|\([^()\n]*\))*\)")
REFERENCE_LINK = re.compile(r"!?\[([^\]\n]+)\]\[[^\]\n]*\]")
INLINE_CODE = re.compile(r"(`+)(.*?)\1", re.DOTALL)
LIST_MARKER = re.compile(r"^[ \t]*(?:[•*-]|[0-9]+[.)])[ \t]+")
HEADING = re.compile(r"^[ \t]{0,3}#{1,6}[ \t]+")
QUOTE = re.compile(r"^[ \t]{0,3}>[ \t]?")
HTML_TAG = re.compile(r"<[^>\n]+>")
BARE_URL = re.compile(r"(?:https?|file)://\S+")
UUID = re.compile(r"(?i)(?<![0-9a-z])[0-9a-f]{8}(?:-[0-9a-f]{4}){3}-[0-9a-f]{12}(?![0-9a-z])")
HEX_HASH = re.compile(r"(?i)(?<![0-9a-z])(?:[0-9a-f]{40}|(?=[0-9a-f]*[0-9])[0-9a-f]{7,39})(?![0-9a-z])")
MAIL_MSG_ID = re.compile(r"(?<!\w)\d{10,}\.\d+\.\d+(?:@[\w.-]+)?(?!\w)")
MAIL_KEY = re.compile(r"(?<!\w)(?:\.[0-9]+_)?[0-9a-fA-F]{16,}(?!\w)")
LARGE_NUMBER = re.compile(r"(?<!\d)(?:\d{5,}|\d{2,3}(?:[,_]\d{3})+|\d{1,3}(?:[,_]\d{3}){2,})(?!\d)")


def markdown_v2_unescape(text: str) -> str:
    """Unescape only Telegram MarkdownV2's reserved character set."""
    result = []
    index = 0
    while index < len(text):
        if (text[index] == "\\" and index + 1 < len(text)
                and text[index + 1] in MARKDOWN_V2_RESERVED):
            result.append(text[index + 1])
            index += 2
        else:
            result.append(text[index])
            index += 1
    return "".join(result)


def strip_fenced_code(text: str) -> str:
    """Keep short fence contents; replace long or unterminated blocks safely."""
    lines = text.splitlines()
    result = []
    index = 0
    while index < len(lines):
        match = FENCE_START.match(lines[index])
        if match is None:
            result.append(lines[index])
            index += 1
            continue
        marker = match[1]
        character, width = marker[0], len(marker)
        closing = re.compile(
            r"^[ \t]{0,3}" + re.escape(character) + "{" + str(width) + r",}[ \t]*$")
        block = []
        index += 1
        while index < len(lines) and closing.match(lines[index]) is None:
            block.append(lines[index])
            index += 1
        if index < len(lines):
            index += 1
        content = "\n".join(block).strip()
        if (len(content) > FENCED_CODE_CHAR_LIMIT
                or len(block) > FENCED_CODE_LINE_LIMIT):
            result.append("code block omitted")
        elif content:
            result.append(content)
    return "\n".join(result)


def strip_links(text: str) -> str:
    previous = None
    while text != previous:
        previous = text
        text = LINK.sub(r"\1", text)
        text = REFERENCE_LINK.sub(r"\1", text)
    return text


def strip_block_markers(text: str) -> str:
    result = []
    for raw_line in text.splitlines():
        line = raw_line
        if re.fullmatch(r"[ \t]*[-*_~=+#|]{3,}[ \t]*", line):
            result.append("")
            continue
        while QUOTE.match(line):
            line = QUOTE.sub("", line, count=1)
        line = HEADING.sub("", line, count=1)
        line = re.sub(r"[ \t]+#+[ \t]*$", "", line)
        listed = LIST_MARKER.match(line) is not None
        if listed:
            line = LIST_MARKER.sub("", line, count=1).strip()
            if line and line[-1] not in ".!?:;":
                line += "."
        result.append(line)
    return "\n".join(result)


def strip_emphasis(text: str) -> str:
    for pattern in (
            re.compile(r"\|\|(?=\S)(.*?\S)\|\|", re.DOTALL),
            re.compile(r"~~(?=\S)(.*?\S)~~", re.DOTALL),
            re.compile(r"(?<!~)~(?=\S)(.*?\S)~(?!~)", re.DOTALL),
    ):
        previous = None
        while text != previous:
            previous = text
            text = pattern.sub(r"\1", text)
    # Underscores are markup only at word boundaries.  Internal identifier
    # underscores therefore survive unchanged.
    text = re.sub(r"(?<![\w_])_{1,2}(?=\S)", "", text)
    text = re.sub(r"(?<=\S)_{1,2}(?![\w_])", "", text)
    # Asterisks and backslashes are forbidden in the final speech stream.
    return text.replace("*", "").replace("\\", "").replace("~", "").replace("|", " ")


def normalize(text: str, format_name: str = "markdown") -> str:
    text = text.replace("\r\n", "\n").replace("\r", "\n")
    text = "".join(char if char in "\n\t" or ord(char) >= 32 else " " for char in text)
    if format_name.casefold().replace("_", "") == "markdownv2":
        text = markdown_v2_unescape(text)
    text = strip_fenced_code(text)
    text = INLINE_CODE.sub(r"\2", text).replace("`", "")
    text = strip_links(text)
    text = HTML_TAG.sub("", text)
    text = BARE_URL.sub("", text)
    text = MAIL_MSG_ID.sub("", text)
    text = MAIL_KEY.sub("", text)
    text = UUID.sub("", text)
    text = HEX_HASH.sub("", text)
    text = LARGE_NUMBER.sub("", text)
    text = strip_block_markers(text)
    text = strip_emphasis(text)
    text = text.replace("[", "").replace("]", "").replace("{", "").replace("}", "")
    # Replace slashes from paths and options with spaces so 'slash' is not spoken
    text = text.replace("/", " ")
    text = re.sub(r"(?<!\w)[#=+<>^|~-]{2,}(?!\w)", " ", text)
    text = re.sub(r"([#=+<>^])(?:[ \t]*\1)+", r"\1", text)
    text = re.sub(r"([.!?,;:])\1+", r"\1", text)
    text = re.sub(r"[ \t]+", " ", text)
    text = re.sub(r" *\n *", "\n", text)
    text = re.sub(r"\n{3,}", "\n\n", text)
    # Defence in depth for future transforms: these may never reach speech.
    text = text.replace("\\", "").replace("*", "")
    cleaned_lines = []
    for line in text.splitlines():
        trimmed = line.strip()
        if not trimmed or re.fullmatch(r"(?:Message-ID\s*[:.]?|[\s.!?,;:-])*", trimmed, re.I):
            continue
        cleaned_line = re.sub(r"\s*(?:on\s+)?Message-ID\s*[:.]?", "", line, flags=re.I).rstrip()
        cleaned_lines.append(cleaned_line)
    text = "\n".join(cleaned_lines)
    return text.strip()


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--format", default="markdown")
    parser.add_argument("--hook-json", action="store_true",
                        help="extract tool_input.text and tool_input.format from stdin JSON")
    args = parser.parse_args(argv)
    try:
        if args.hook_json:
            payload = json.load(sys.stdin)
            tool_input = payload.get("tool_input", {}) if isinstance(payload, dict) else {}
            text = tool_input.get("text", "") if isinstance(tool_input, dict) else ""
            format_name = tool_input.get("format", args.format) if isinstance(tool_input, dict) else args.format
            if not isinstance(text, str) or not isinstance(format_name, str):
                raise ValueError("hook text and format must be strings")
        else:
            text = sys.stdin.read()
            format_name = args.format
        sys.stdout.write(normalize(text, format_name))
        return 0
    except (OSError, UnicodeError, ValueError, json.JSONDecodeError) as error:
        print(f"voice-text-normalize: {error}", file=sys.stderr)
        return 2


if __name__ == "__main__":
    raise SystemExit(main())

* Provider-aware rollout watcher

The Ductor Codex path cannot use the Claude =Stop= hook.  This watcher follows only the provider session IDs named by Ductor's Telegram session registry, starts new rollout files at EOF, and persists delivery keys before queueing speech.  Its owner-chat allow-list is read from the private Ductor config at runtime; no chat ID, token, host address, or private identifier belongs in this public repository.  The generated file is the canonical implementation staged into Honey's existing =honey-voice= engine by the host-layer installer.

#+BEGIN_SRC python :tangle all/.local/share/dotcortex/honey-voice/bin/provider-rollout-watcher.py :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
"""Provider-aware rollout watcher for Honey voice instances.

Each OpenRC instance supplies its own Ductor home, provider, rollout home,
state file, bot name, and optional chat allow-list.  A newly discovered
rollout is positioned at EOF, so startup never replays history.  Delivery
keys are persisted before spawning TTS, preventing duplicate voice notes after
an unclean watcher restart.
"""
import glob
import hashlib
import json
import os
import subprocess
import sys
import time
import uuid


HOME = os.path.expanduser("~")
BASE = os.path.join(HOME, ".local", "share", "honey-voice")
BOT = os.environ.get("HONEY_VOICE_BOT", "badger")
DUCTOR = os.environ.get("HONEY_VOICE_DUCTOR_HOME", os.path.join(HOME, ".ductor"))
PROVIDER = os.environ.get("HONEY_VOICE_PROVIDER", "codex")
ROLLOUT_HOME = os.environ.get("HONEY_VOICE_ROLLOUT_HOME", HOME)
READ_AS = os.environ.get("HONEY_VOICE_READ_AS", "")
STATE = os.environ.get("HONEY_VOICE_STATE", os.path.join(HOME, ".local", "state", "honey-voice"))
OFFSETS = os.path.join(STATE, "%s-offsets.json" % BOT)
PENDING = os.path.join(HOME, ".cache", "honey-voice", "pending")
DELIVERIES = os.path.join(STATE, "%s-deliveries.json" % BOT)
POLL_S = float(os.environ.get("HONEY_VOICE_POLL_SECONDS", "2"))
SESSIONS_GLOB = os.path.join(ROLLOUT_HOME, ".codex", "sessions", "**", "rollout-*%s.jsonl")


def say(msg):
    print("%s %s" % (time.strftime("%Y-%m-%d %H:%M:%S"), msg), flush=True)


_threads_cache = {"mtime": None, "out": {}}


def read_text(path):
    try:
        with open(path) as f:
            return f.read()
    except PermissionError:
        if not READ_AS:
            raise
        done = subprocess.run(["sudo", "-n", "-u", READ_AS, "cat", "--", path],
                              capture_output=True, text=True, timeout=10)
        if done.returncode != 0:
            raise OSError("sudo cat failed")
        return done.stdout


def allowed_chat_ids():
    """Read the private Ductor allow-list; never put owner IDs in service source."""
    configured = os.environ.get("HONEY_VOICE_ALLOWED_CHAT_IDS", "")
    if configured:
        return {int(x) for x in configured.split(",") if x.strip().lstrip("-").isdigit()}
    try:
        data = json.loads(read_text(os.path.join(DUCTOR, "config", "config.json")))
        return {int(x) for x in data.get("allowed_user_ids", [])}
    except (OSError, ValueError, TypeError, KeyError):
        return set()


def threads():
    path = os.path.join(DUCTOR, "sessions.json")
    try:
        mtime = os.stat(path).st_mtime_ns
        if mtime == _threads_cache["mtime"]:
            return _threads_cache["out"]
        data = json.loads(read_text(path))
    except (OSError, ValueError, subprocess.SubprocessError):
        return _threads_cache.get("out", {})
    out = {}
    for entry in data.values():
        if entry.get("transport") != "tg":
            continue
        try:
            chat = int(entry["chat_id"])
        except (KeyError, TypeError, ValueError):
            continue
        allow = allowed_chat_ids()
        if allow and chat not in allow:
            continue
        provider_entry = (entry.get("provider_sessions") or {}).get(PROVIDER) or {}
        sid = provider_entry.get("session_id")
        if sid:
            out[str(sid)] = (chat, entry.get("topic_id"))
    _threads_cache.update(mtime=mtime, out=out)
    return out


def load_json(path):
    try:
        with open(path) as f:
            value = json.load(f)
        return value if isinstance(value, dict) else {}
    except (OSError, ValueError):
        return {}


def save_json(path, value):
    os.makedirs(os.path.dirname(path), mode=0o700, exist_ok=True)
    tmp = path + ".tmp.%s" % uuid.uuid4().hex
    fd = os.open(tmp, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(fd, "w") as f:
        json.dump(value, f, separators=(",", ":"))
    os.replace(tmp, path)


def queue_voice(text, chat):
    os.makedirs(PENDING, mode=0o700, exist_ok=True)
    path = os.path.join(PENDING, "%s-%s.txt" % (BOT, uuid.uuid4().hex))
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(fd, "w") as f:
        f.write(text)
    subprocess.Popen([sys.executable, os.path.join(BASE, "bin", "honey_speak.py"),
                      "--bot", BOT, "--chat", str(chat), "--text-file", path],
                     stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL,
                     stderr=subprocess.DEVNULL, start_new_session=True, close_fds=True)


def main():
    once = "--once" in sys.argv
    offsets = load_json(OFFSETS)
    deliveries = load_json(DELIVERIES)
    paths = {}
    say("%s watcher started provider=%s" % (BOT, PROVIDER))
    while True:
        changed = False
        for sid, (chat, _topic) in threads().items():
            path = paths.get(sid)
            if not path or not os.path.exists(path):
                found = sorted(glob.glob(SESSIONS_GLOB % sid, recursive=True), key=os.path.getmtime)
                if not found:
                    continue
                path = paths[sid] = found[-1]
            try:
                size = os.path.getsize(path)
            except OSError:
                continue
            if path not in offsets or offsets[path] > size:
                offsets[path] = size
                changed = True
                say("following %s" % os.path.basename(path)[:60])
                continue
            if size == offsets[path]:
                continue
            start = offsets[path]
            with open(path, "rb") as f:
                f.seek(start)
                chunk = f.read(size - start)
            end = chunk.rfind(b"\n")
            if end < 0:
                continue
            for line in chunk[:end + 1].splitlines(keepends=True):
                line_start = start
                start += len(line)
                try:
                    event = json.loads(line)
                except ValueError:
                    continue
                payload = event.get("payload") or {}
                if event.get("type") != "event_msg" or not isinstance(payload, dict):
                    continue
                if payload.get("type") != "task_complete":
                    continue
                text = payload.get("last_agent_message") or ""
                if not text.strip():
                    continue
                key = hashlib.sha256((path + ":" + str(line_start)).encode()).hexdigest()
                if key in deliveries:
                    continue
                # Record before spawning: a restart cannot deliver this line twice.
                deliveries[key] = int(time.time())
                if len(deliveries) > 10000:
                    for old in sorted(deliveries, key=deliveries.get)[:2000]:
                        deliveries.pop(old, None)
                save_json(DELIVERIES, deliveries)
                queue_voice(text, chat)
                say("turn complete in %s -> voice queued (%d chars)" % (sid[:8], len(text)))
            offsets[path] = start
            changed = True
        if changed:
            save_json(OFFSETS, offsets)
        if once:
            return 0
        time.sleep(POLL_S)


if __name__ == "__main__":
    try:
        sys.exit(main())
    except KeyboardInterrupt:
        sys.exit(0)

* Honey voice engine (Badger and Flying Fox)

Honey's two Ductor bots speak their Telegram replies through a private engine
in =~/.local/share/honey-voice= of the voice account: Honey Badger (Claude)
through a =Stop= hook, Flying Fox (Codex) through the rollout watcher above.
Both hand the reply to =honey_speak.py=, which renders it with the selected
pvox clone profile and posts it with =sendVoice=.  The Piper venv, voice models
and =voices.json= stay machine-local; this section owns the code, the hook
settings template and a user-level installer.  The OpenRC bodies, their
=/etc/conf.d= templates and the root installer are the host module's
([[file:agents-bots-honey.org][agents-bots-honey.org]], "Ductor voice service lane").

*Why this lives here (2026-11-26).*  On 2026-11-19 16:07 the pvox clone
installer (=clone_backend.py install=, HelmCortex =FORGE/VoxForge=) replaced
=~/.local/share/pvox-clone/client/bin/pvox= with a copy at mode 0664: it copies
code with =shutil.copy2=, which carries the mode of the package it was run
from, and only the Honey adapter got an explicit =chmod=.  =honey_speak.py=
executed the client directly, so every Badger and Fox voice note failed with
=PermissionError= until a live hot-fix on 2026-11-26 (81 Badger and 12 Fox
notes were lost; the text replies were not affected).  Aura's hook had the same
fault and the same cure (=sys.executable=, 2026-11-25).  The rule since: *every
entry point in this engine is started through an interpreter*, never through
its execute bit, and the installer reports a client that lost the bit.

** Renderer: honey_speak.py

Byte-for-byte the HelmCortex =FORGE/VoxForge/scripts/honey_speak.py= adapter
with the one fix: the pvox client runs as =[sys.executable, pvox, "render", ...]=.
The bot token is read at run time from the bot's own Ductor config (through
=read_as= for the Fox home) and is never logged; the log carries no reply text.

#+BEGIN_SRC python :tangle all/.local/share/dotcortex/honey-voice/bin/honey_speak.py :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
"""honey_speak.py — render a Ductor bot reply as a Telegram voice note (Honey, 2026-09-16).

Modeled on Aura's aura_voice.py (Hermes) but fully separate: own venv/voices under
~/.local/share/honey-voice, delivery straight to the Telegram Bot API (Ductor has no
post-response hook and uploads .ogg as a document, not a voice bubble).

  honey_speak.py --bot badger --session <id> --text-file <path>   (chat resolved from sessions.json, retries)
  honey_speak.py --bot fox --chat <id> --text-file <path>
  honey_speak.py --bot badger --text "..." --no-send               (render only; prints ogg path)
  honey_speak.py --sanitize < file                                   (debug the ear-text)

Config: ~/.local/share/honey-voice/voices.json  ("voice": null = wired but silent).
Token: read at runtime from <ductor_home>/config/config.json — never logged or copied.
Log: ~/.local/state/honey-voice/voice.log (0600, no reply text). Cache: ~/.cache/honey-voice (pruned >3d).
"""
import argparse, fcntl, json, os, re, subprocess, sys, time, uuid, urllib.request
import tempfile
import wave

HOME = os.path.expanduser("~")
BASE = os.path.join(HOME, ".local", "share", "honey-voice")
PY = os.path.join(BASE, "venv", "bin", "python")
VOICES = os.path.join(BASE, "voices")
CONF = os.path.join(BASE, "voices.json")

def log(msg):
    os.makedirs(STATE, mode=0o700, exist_ok=True)
    fd = os.open(LOG, os.O_WRONLY | os.O_CREAT | os.O_APPEND, 0o600)
    with os.fdopen(fd, "a") as f:
        f.write("%s %s\n" % (time.strftime("%Y-%m-%d %H:%M:%S"), msg))


def sanitize(text, max_chars=None):
    t = text
    t = re.sub(r"```.*?```", " (code omitted) ", t, flags=re.S)
    t = re.sub(r"\[button:[^\]]*\]", " ", t)
    t = re.sub(r"<file:[^>]*>", " (file attached) ", t)
    t = re.sub(r"\[([^\]]+)\]\((?:https?://)[^)]+\)", r"\1", t)
    t = re.sub(r"https?://\S+", " link ", t)
    lines = []
    for line in t.splitlines():
        s = line.strip()
        if s.startswith("|") or re.fullmatch(r"[-|: ]{3,}", s or "x"):
            continue                                  # tables
        if s.startswith("[Aura context]"):
            continue                                  # internal relay line
        s = re.sub(r"^#{1,6}\s*", "", s)              # headings
        s = re.sub(r"^\s*(?:[-*•]|\d+[.)])\s+", "", s)  # bullets / numbering
        s = re.sub(r"^>\s?", "", s)                   # quotes
        lines.append(s)
    t = "\n".join(lines)
    t = re.sub(r"`([^`]*)`", r"\1", t)
    t = re.sub(r"(\*\*|__|\*|_|~~)(?=\S)(.+?)(?<=\S)\1", r"\2", t)
    t = re.sub(r"\s*→\s*", ", then ", t)
    t = t.replace(" & ", " and ")
    t = re.sub(r"[ \t]+", " ", t)
    t = re.sub(r"\n{2,}", "\n", t).strip()
    if max_chars and len(t) > max_chars:            # no cap unless voices.json sets one (2026-10-07)
        cut = t[:max_chars]
        m = max(cut.rfind(". "), cut.rfind("! "), cut.rfind("? "), cut.rfind("\n"))
        t = (cut[: m + 1] if m > max_chars * 0.5 else cut).rstrip() + " … The rest is in the chat."
    return t


def load_conf(bot):
    with open(CONF) as f:
        c = json.load(f)[bot]
    return c


def resolve_chat(conf, session_id, wait_s=25):
    """Map a provider session id to a Telegram chat via Ductor's sessions.json.
    Ductor records a NEW session id only after the CLI turn ends, so retry briefly."""
    path = os.path.join(conf["ductor_home"], "sessions.json")
    deadline = time.time() + wait_s
    while True:
        try:
            with open(path) as f:
                data = json.load(f)
            for entry in data.values():
                if entry.get("transport") != "tg":
                    continue
                ps = (entry.get("provider_sessions") or {}).get(conf["provider"]) or {}
                if ps.get("session_id") == session_id:
                    return int(entry["chat_id"]), entry.get("topic_id")
        except (OSError, ValueError, KeyError):
            pass
        if time.time() >= deadline:
            return None, None
        time.sleep(2)


def render_clone(text, agent):
    """Use the selected pvox profile; preserve the existing Telegram transport."""
    os.makedirs(CACHE, mode=0o700, exist_ok=True)
    env = dict(os.environ)
    for key in ("PYTHONPATH", "PYTHONHOME", "PYTHONSTARTUP", "PYTHONEXECUTABLE"):
        env.pop(key, None)
    pvox = os.path.join(HOME, ".local", "share", "pvox-clone", "client", "bin", "pvox")

def render(text, voice, pvox_agent=None):
    if pvox_agent:
        return render_clone(text, pvox_agent)
    onnx = os.path.join(VOICES, voice + ".onnx")
    with open(onnx + ".json") as f:
        rate = int(json.load(f)["audio"]["sample_rate"])
    os.makedirs(CACHE, mode=0o700, exist_ok=True)
    stamp = time.strftime("%Y%m%d-%H%M%S") + "-" + uuid.uuid4().hex[:6]
    ogg = os.path.join(CACHE, "%s.ogg" % stamp)
    piper = subprocess.run([PY, "-m", "piper", "-m", onnx, "--output-raw"], input=text.encode(),
                           stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=180 + len(text) // 20)
    if piper.returncode != 0 or not piper.stdout:
        raise RuntimeError("piper failed rc=%s %s" % (piper.returncode, piper.stderr[-200:]))
    ff = subprocess.run(["ffmpeg", "-y", "-loglevel", "error", "-f", "s16le", "-ar", str(rate), "-ac", "1",
                         "-i", "-", "-c:a", "libopus", "-b:a", "32k", ogg],
                        input=piper.stdout, stdout=subprocess.DEVNULL, stderr=subprocess.PIPE, timeout=120 + len(text) // 40)
    if ff.returncode != 0 or not os.path.exists(ogg):
        raise RuntimeError("ffmpeg failed rc=%s %s" % (ff.returncode, ff.stderr[-200:]))
    os.chmod(ogg, 0o600)
    return ogg, len(piper.stdout) / 2 / rate


def send_voice(token, chat_id, ogg, duration, topic_id=None, caption=None):
    boundary = "----honeyvoice" + uuid.uuid4().hex
    parts = []
    def field(name, value):
        parts.append(("--%s\r\nContent-Disposition: form-data; name=\"%s\"\r\n\r\n%s\r\n" % (boundary, name, value)).encode())
    field("chat_id", str(chat_id))
    field("duration", str(int(round(duration))))
    if topic_id:
        field("message_thread_id", str(topic_id))
    if caption:
        field("caption", caption[:1000])
    with open(ogg, "rb") as f:
        audio = f.read()
    parts.append(("--%s\r\nContent-Disposition: form-data; name=\"voice\"; filename=\"voice.ogg\"\r\nContent-Type: audio/ogg\r\n\r\n" % boundary).encode())
    parts.append(audio + b"\r\n")
    parts.append(("--%s--\r\n" % boundary).encode())
    req = urllib.request.Request("https://api.telegram.org/bot%s/sendVoice" % token, data=b"".join(parts),
                                 headers={"Content-Type": "multipart/form-data; boundary=%s" % boundary})
    with urllib.request.urlopen(req, timeout=60 + len(audio) // 20000) as r:
        body = json.load(r)
    if not body.get("ok"):
        raise RuntimeError("telegram not ok: %s" % str(body)[:200])
    return body


def prune():
    now = time.time()
    for root in (CACHE, os.path.join(CACHE, "pending")):
        try:
            for name in os.listdir(root):
                p = os.path.join(root, name)
                if os.path.isfile(p) and now - os.path.getmtime(p) > PRUNE_AGE_S:
                    os.unlink(p)
        except OSError:
            pass

"""claude_stop_hook.py — Honey Badger voice hook (Claude Code Stop event, Honey 2026-09-16).
Loaded ONLY for Honey Badger's own CLI via `--settings ~/.local/share/honey-voice/claude-settings.json`

** Flying Fox entrypoint

The Fox unit exports =HONEY_FOX_HOME= and =HONEY_FOX_READ_AS= from

#+BEGIN_SRC python :tangle all/.local/share/dotcortex/honey-voice/bin/codex_rollout_watcher.py :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
"""Flying Fox entrypoint for the provider-aware rollout watcher.

The honey-voice-fox unit exports HONEY_FOX_HOME (the Flying Fox agent
account's home) and HONEY_FOX_READ_AS (the account its private files are read
as).  This maps them onto the generic watcher's settings and runs
provider-rollout-watcher.py from this directory by path, so the engine needs
no second, importable copy of the watcher.  Explicit HONEY_VOICE_* values win.
"""
import importlib.util
import os
import sys

HERE = os.path.dirname(os.path.abspath(__file__))
WATCHER = os.path.join(HERE, "provider-rollout-watcher.py")


def configure(env=os.environ):
    fox_home = env.get("HONEY_FOX_HOME", "")
    if not fox_home:
        raise SystemExit("codex_rollout_watcher: HONEY_FOX_HOME is not set (see /etc/conf.d/honey-voice-fox)")
    env.setdefault("HONEY_VOICE_BOT", "fox")
    env.setdefault("HONEY_VOICE_PROVIDER", "codex")
    env.setdefault("HONEY_VOICE_DUCTOR_HOME", os.path.join(fox_home, ".ductor-codex"))
    env.setdefault("HONEY_VOICE_ROLLOUT_HOME", fox_home)
    read_as = env.get("HONEY_FOX_READ_AS", "")
    if read_as:
        env.setdefault("HONEY_VOICE_READ_AS", read_as)
    return env


def load_watcher(path=WATCHER):
    # The watcher reads its settings at import time, so configure() runs first.
    spec = importlib.util.spec_from_file_location("provider_rollout_watcher", path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


if __name__ == "__main__":
    configure()
    try:
        raise SystemExit(load_watcher().main())
    except KeyboardInterrupt:
        sys.exit(0)

** Engine installer: honey-voice-engine-install

Run as the voice account.  =check= (the default) compares the four engine
scripts and the rendered hook settings with the installed copies, content and
mode, checks =voices.json=, and warns when the pvox client is missing its
execute bit.  =install= replaces only what differs, keeps a
=<file>.bak-<timestamp>= of each replaced file, then verifies.  =--dry-run=

    case "$arg" in

template="$(<"$LANE/claude-settings.json.template")"

# the modes are belt and braces, but a drifted mode is still reported.
targets=(
    "$LANE/bin/honey_speak.py|$ENGINE_ROOT/bin/honey_speak.py|755"
    "$LANE/bin/claude_stop_hook.py|$ENGINE_ROOT/bin/claude_stop_hook.py|755"

    "$tmp/claude-settings.json|$ENGINE_ROOT/claude-settings.json|644"
)

warn() { echo "?? $*"; }

place() {
    local pair src dst want have
    for pair in "${targets[@]}"; do
        IFS='|' read -r src dst want <<<"$pair"
        [ -r "$src" ] || { echo "missing source $src" >&2; exit 1; }

            have="$(stat -c %a -- "$dst")"

            if [ "$VERB" = check ]; then problem "mode $have (want $want) $dst"; continue; fi
            if [ "$DRY" -eq 1 ]; then echo "would: chmod $want $dst"; continue; fi
            chmod -- "$want" "$dst"
            echo "chmod $want $dst"
            continue
        fi
        if [ "$VERB" = check ]; then
            if [ -e "$dst" ]; then problem "drift $dst"; else problem "missing $dst"; fi
            continue
        fi
        if [ "$DRY" -eq 1 ]; then echo "would: backup/install $dst"; continue; fi
        mkdir -p -- "$(dirname -- "$dst")"
        [ -e "$dst" ] && cp -p -- "$dst" "$dst.bak-$ts"
        install -m "$want" -- "$src" "$dst.pending"
        mv -f -- "$dst.pending" "$dst"
        echo "installed $dst"
    done
}

        echo "ok $conf"

    fi
    if [ ! -f "$PVOX_CLIENT" ]; then

        '#!'*python*) ;;

    mode="$(stat -c %a -- "$PVOX_CLIENT")"
    if [ -x "$PVOX_CLIENT" ]; then
        echo "ok pvox client $mode $PVOX_CLIENT"
    else
        warn "pvox client not executable (mode $mode): voice notes still render through the interpreter, but direct callers fail; fix: chmod 0755 $PVOX_CLIENT"
    fi
    [ ! -e "$ENGINE_ROOT/bin/provider_rollout_watcher.py" ] ||
        echo "-- legacy $ENGINE_ROOT/bin/provider_rollout_watcher.py: unused by the canonical Fox entrypoint; remove after honey-voice-fox restarts on it"
}

place
if [ "$VERB" = install ] && [ "$DRY" -eq 0 ]; then
    VERB=check
    echo '== verify'
    place
fi
machine_local
echo 'No service was started, restarted, or enabled.'
exit "$bad"
