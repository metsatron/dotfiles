* Shared spoken-text normaliser

=voice-text-normalize= is the single fleet speech boundary.  It lives in the
universal =all/.local/bin= overlay because Claude, OpenCode/DeepSeek's operator
relay, Honey's Claude bots and MailCortex all need the same transformation but
do not share a TTS engine.  Callers pass reply text on stdin, or a complete hook
payload with =--hook-json=.  MarkdownV2 unescaping is format-aware; the remaining
cleanup applies to every reply format.  Fenced code stays intelligible when it
is short, while a block over 600 characters or 20 lines becomes the bounded
phrase =code block omitted=.

#+BEGIN_SRC python :tangle all/.local/bin/voice-text-normalize :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
"""Turn Telegram Markdown/MarkdownV2 reply text into bounded plain speech."""

import argparse
import json
import re
import sys


FENCED_CODE_CHAR_LIMIT = 600
FENCED_CODE_LINE_LIMIT = 20
MARKDOWN_V2_RESERVED = set("_*[]()~`>#+-=|{}.!" + chr(92))
FENCE_START = re.compile(r"^[ \t]{0,3}(`{3,}|~{3,})[^\r\n]*$")
LINK = re.compile(r"!?\[([^\]\n]+)\]\((?:[^()\n]|\([^()\n]*\))*\)")
REFERENCE_LINK = re.compile(r"!?\[([^\]\n]+)\]\[[^\]\n]*\]")
INLINE_CODE = re.compile(r"(`+)(.*?)\1", re.DOTALL)
LIST_MARKER = re.compile(r"^[ \t]*(?:[•*-]|[0-9]+[.)])[ \t]+")
HEADING = re.compile(r"^[ \t]{0,3}#{1,6}[ \t]+")
QUOTE = re.compile(r"^[ \t]{0,3}>[ \t]?")
HTML_TAG = re.compile(r"<[^>\n]+>")
BARE_URL = re.compile(r"(?:https?|file)://\S+")
LARGE_NUMBER = re.compile(r"\b(?:\d{5,}|\d{2,3}(?:[,_]\d{3})+|\d{1,3}(?:[,_]\d{3}){2,})\b")
UUID = re.compile(r"\b[0-9a-fA-F]{8}(?:-[0-9a-fA-F]{4}){3}-[0-9a-fA-F]{12}\b")


def markdown_v2_unescape(text: str) -> str:
    """Unescape only Telegram MarkdownV2's reserved character set."""
    result = []
    index = 0
    while index < len(text):
        if (text[index] == "\\" and index + 1 < len(text)
                and text[index + 1] in MARKDOWN_V2_RESERVED):
            result.append(text[index + 1])
            index += 2
        else:
            result.append(text[index])
            index += 1
    return "".join(result)


def strip_fenced_code(text: str) -> str:
    """Keep short fence contents; replace long or unterminated blocks safely."""
    lines = text.splitlines()
    result = []
    index = 0
    while index < len(lines):
        match = FENCE_START.match(lines[index])
        if match is None:
            result.append(lines[index])
            index += 1
            continue
        marker = match[1]
        character, width = marker[0], len(marker)
        closing = re.compile(
            r"^[ \t]{0,3}" + re.escape(character) + "{" + str(width) + r",}[ \t]*$")
        block = []
        index += 1
        while index < len(lines) and closing.match(lines[index]) is None:
            block.append(lines[index])
            index += 1
        if index < len(lines):
            index += 1
        content = "\n".join(block).strip()
        if (len(content) > FENCED_CODE_CHAR_LIMIT
                or len(block) > FENCED_CODE_LINE_LIMIT):
            result.append("code block omitted")
        elif content:
            result.append(content)
    return "\n".join(result)


def strip_links(text: str) -> str:
    previous = None
    while text != previous:
        previous = text
        text = LINK.sub(r"\1", text)
        text = REFERENCE_LINK.sub(r"\1", text)
    return text


def strip_block_markers(text: str) -> str:
    result = []
    for raw_line in text.splitlines():
        line = raw_line
        if re.fullmatch(r"[ \t]*[-*_~=+#|]{3,}[ \t]*", line):
            result.append("")
            continue
        while QUOTE.match(line):
            line = QUOTE.sub("", line, count=1)
        line = HEADING.sub("", line, count=1)
        line = re.sub(r"[ \t]+#+[ \t]*$", "", line)
        listed = LIST_MARKER.match(line) is not None
        if listed:
            line = LIST_MARKER.sub("", line, count=1).strip()
            if line and line[-1] not in ".!?:;":
                line += "."
        result.append(line)
    return "\n".join(result)


def strip_emphasis(text: str) -> str:
    for pattern in (
            re.compile(r"\|\|(?=\S)(.*?\S)\|\|", re.DOTALL),
            re.compile(r"~~(?=\S)(.*?\S)~~", re.DOTALL),
            re.compile(r"(?<!~)~(?=\S)(.*?\S)~(?!~)", re.DOTALL),
    ):
        previous = None
        while text != previous:
            previous = text
            text = pattern.sub(r"\1", text)
    # Underscores are markup only at word boundaries.  Internal identifier
    # underscores therefore survive unchanged.
    text = re.sub(r"(?<![\w_])_{1,2}(?=\S)", "", text)
    text = re.sub(r"(?<=\S)_{1,2}(?![\w_])", "", text)
    # Asterisks and backslashes are forbidden in the final speech stream.
    return text.replace("*", "").replace("\\", "").replace("~", "").replace("|", " ")


def normalize(text: str, format_name: str = "markdown") -> str:
    text = text.replace("\r\n", "\n").replace("\r", "\n")
    text = "".join(char if char in "\n\t" or ord(char) >= 32 else " " for char in text)
    if format_name.casefold().replace("_", "") == "markdownv2":
        text = markdown_v2_unescape(text)
    text = strip_fenced_code(text)
    text = INLINE_CODE.sub(r"\2", text).replace("`", "")
    text = strip_links(text)
    text = HTML_TAG.sub("", text)
    text = BARE_URL.sub("", text)
    text = UUID.sub("", text)
    text = LARGE_NUMBER.sub("", text)
    text = strip_block_markers(text)
    text = strip_emphasis(text)
    text = text.replace("[", "").replace("]", "").replace("{", "").replace("}", "")
    # Replace slashes from paths and options with spaces so 'slash' is not spoken
    text = text.replace("/", " ")
    text = re.sub(r"(?<!\w)[#=+<>^|~-]{2,}(?!\w)", " ", text)
    text = re.sub(r"([#=+<>^])(?:[ \t]*\1)+", r"\1", text)
    text = re.sub(r"([.!?,;:])\1+", r"\1", text)
    text = re.sub(r"[ \t]+", " ", text)
    text = re.sub(r" *\n *", "\n", text)
    text = re.sub(r"\n{3,}", "\n\n", text)
    # Defence in depth for future transforms: these may never reach speech.
    text = text.replace("\\", "").replace("*", "")
    cleaned_lines = []
    for line in text.splitlines():
        trimmed = line.strip()
        if not trimmed or re.fullmatch(r"(?:Message-ID\s*[:.]?|[\s.!?,;:-])*", trimmed, re.I):
            continue
        cleaned_line = re.sub(r"\s*(?:on\s+)?Message-ID\s*[:.]?", "", line, flags=re.I).rstrip()
        cleaned_lines.append(cleaned_line)
    text = "\n".join(cleaned_lines)
    return text.strip()


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--format", default="markdown")
    parser.add_argument("--hook-json", action="store_true",
                        help="extract tool_input.text and tool_input.format from stdin JSON")
    args = parser.parse_args(argv)
    try:
        if args.hook_json:
            payload = json.load(sys.stdin)
            tool_input = payload.get("tool_input", {}) if isinstance(payload, dict) else {}
            text = tool_input.get("text", "") if isinstance(tool_input, dict) else ""
            format_name = tool_input.get("format", args.format) if isinstance(tool_input, dict) else args.format
            if not isinstance(text, str) or not isinstance(format_name, str):
                raise ValueError("hook text and format must be strings")
        else:
            text = sys.stdin.read()
            format_name = args.format
        sys.stdout.write(normalize(text, format_name))
        return 0
    except (OSError, UnicodeError, ValueError, json.JSONDecodeError) as error:
        print(f"voice-text-normalize: {error}", file=sys.stderr)
        return 2


if __name__ == "__main__":
    raise SystemExit(main())

* Provider-aware rollout watcher

The Ductor Codex path cannot use the Claude =Stop= hook.  This watcher follows only the provider session IDs named by Ductor's Telegram session registry, starts new rollout files at EOF, and persists delivery keys before queueing speech.  Its owner-chat allow-list is read from the private Ductor config at runtime; no chat ID, token, host address, or private identifier belongs in this public repository.  The generated file is the canonical implementation staged into Honey's existing =honey-voice= engine by the host-layer installer.

#+BEGIN_SRC python :tangle all/.local/share/dotcortex/honey-voice/bin/provider-rollout-watcher.py :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
"""Provider-aware rollout watcher for Honey voice instances.

Each OpenRC instance supplies its own Ductor home, provider, rollout home,
state file, bot name, and optional chat allow-list.  A newly discovered
rollout is positioned at EOF, so startup never replays history.  Delivery
keys are persisted before spawning TTS, preventing duplicate voice notes after
an unclean watcher restart.
"""
import glob
import hashlib
import json
import os
import subprocess
import sys
import time
import uuid


HOME = os.path.expanduser("~")
BASE = os.path.join(HOME, ".local", "share", "honey-voice")
BOT = os.environ.get("HONEY_VOICE_BOT", "badger")
DUCTOR = os.environ.get("HONEY_VOICE_DUCTOR_HOME", os.path.join(HOME, ".ductor"))
PROVIDER = os.environ.get("HONEY_VOICE_PROVIDER", "codex")
ROLLOUT_HOME = os.environ.get("HONEY_VOICE_ROLLOUT_HOME", HOME)
READ_AS = os.environ.get("HONEY_VOICE_READ_AS", "")
STATE = os.environ.get("HONEY_VOICE_STATE", os.path.join(HOME, ".local", "state", "honey-voice"))
OFFSETS = os.path.join(STATE, "%s-offsets.json" % BOT)
PENDING = os.path.join(HOME, ".cache", "honey-voice", "pending")
DELIVERIES = os.path.join(STATE, "%s-deliveries.json" % BOT)
POLL_S = float(os.environ.get("HONEY_VOICE_POLL_SECONDS", "2"))
SESSIONS_GLOB = os.path.join(ROLLOUT_HOME, ".codex", "sessions", "**", "rollout-*%s.jsonl")


def say(msg):
    print("%s %s" % (time.strftime("%Y-%m-%d %H:%M:%S"), msg), flush=True)


_threads_cache = {"mtime": None, "out": {}}


def read_text(path):
    try:
        with open(path) as f:
            return f.read()
    except PermissionError:
        if not READ_AS:
            raise
        done = subprocess.run(["sudo", "-n", "-u", READ_AS, "cat", "--", path],
                              capture_output=True, text=True, timeout=10)
        if done.returncode != 0:
            raise OSError("sudo cat failed")
        return done.stdout


def allowed_chat_ids():
    """Read the private Ductor allow-list; never put owner IDs in service source."""
    configured = os.environ.get("HONEY_VOICE_ALLOWED_CHAT_IDS", "")
    if configured:
        return {int(x) for x in configured.split(",") if x.strip().lstrip("-").isdigit()}
    try:
        data = json.loads(read_text(os.path.join(DUCTOR, "config", "config.json")))
        return {int(x) for x in data.get("allowed_user_ids", [])}
    except (OSError, ValueError, TypeError, KeyError):
        return set()


def threads():
    path = os.path.join(DUCTOR, "sessions.json")
    try:
        mtime = os.stat(path).st_mtime_ns
        if mtime == _threads_cache["mtime"]:
            return _threads_cache["out"]
        data = json.loads(read_text(path))
    except (OSError, ValueError, subprocess.SubprocessError):
        return _threads_cache.get("out", {})
    out = {}
    for entry in data.values():
        if entry.get("transport") != "tg":
            continue
        try:
            chat = int(entry["chat_id"])
        except (KeyError, TypeError, ValueError):
            continue
        allow = allowed_chat_ids()
        if allow and chat not in allow:
            continue
        provider_entry = (entry.get("provider_sessions") or {}).get(PROVIDER) or {}
        sid = provider_entry.get("session_id")
        if sid:
            out[str(sid)] = (chat, entry.get("topic_id"))
    _threads_cache.update(mtime=mtime, out=out)
    return out


def load_json(path):
    try:
        with open(path) as f:
            value = json.load(f)
        return value if isinstance(value, dict) else {}
    except (OSError, ValueError):
        return {}


def save_json(path, value):
    os.makedirs(os.path.dirname(path), mode=0o700, exist_ok=True)
    tmp = path + ".tmp.%s" % uuid.uuid4().hex
    fd = os.open(tmp, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(fd, "w") as f:
        json.dump(value, f, separators=(",", ":"))
    os.replace(tmp, path)


def queue_voice(text, chat):
    os.makedirs(PENDING, mode=0o700, exist_ok=True)
    path = os.path.join(PENDING, "%s-%s.txt" % (BOT, uuid.uuid4().hex))
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(fd, "w") as f:
        f.write(text)
    subprocess.Popen([sys.executable, os.path.join(BASE, "bin", "honey_speak.py"),
                      "--bot", BOT, "--chat", str(chat), "--text-file", path],
                     stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL,
                     stderr=subprocess.DEVNULL, start_new_session=True, close_fds=True)


def main():
    once = "--once" in sys.argv
    offsets = load_json(OFFSETS)
    deliveries = load_json(DELIVERIES)
    paths = {}
    say("%s watcher started provider=%s" % (BOT, PROVIDER))
    while True:
        changed = False
        for sid, (chat, _topic) in threads().items():
            path = paths.get(sid)
            if not path or not os.path.exists(path):
                found = sorted(glob.glob(SESSIONS_GLOB % sid, recursive=True), key=os.path.getmtime)
                if not found:
                    continue
                path = paths[sid] = found[-1]
            try:
                size = os.path.getsize(path)
            except OSError:
                continue
            if path not in offsets or offsets[path] > size:
                offsets[path] = size
                changed = True
                say("following %s" % os.path.basename(path)[:60])
                continue
            if size == offsets[path]:
                continue
            start = offsets[path]
            with open(path, "rb") as f:
                f.seek(start)
                chunk = f.read(size - start)
            end = chunk.rfind(b"\n")
            if end < 0:
                continue
            for line in chunk[:end + 1].splitlines(keepends=True):
                line_start = start
                start += len(line)
                try:
                    event = json.loads(line)
                except ValueError:
                    continue
                payload = event.get("payload") or {}
                if event.get("type") != "event_msg" or not isinstance(payload, dict):
                    continue
                if payload.get("type") != "task_complete":
                    continue
                text = payload.get("last_agent_message") or ""
                if not text.strip():
                    continue
                key = hashlib.sha256((path + ":" + str(line_start)).encode()).hexdigest()
                if key in deliveries:
                    continue
                # Record before spawning: a restart cannot deliver this line twice.
                deliveries[key] = int(time.time())
                if len(deliveries) > 10000:
                    for old in sorted(deliveries, key=deliveries.get)[:2000]:
                        deliveries.pop(old, None)
                save_json(DELIVERIES, deliveries)
                queue_voice(text, chat)
                say("turn complete in %s -> voice queued (%d chars)" % (sid[:8], len(text)))
            offsets[path] = start
            changed = True
        if changed:
            save_json(OFFSETS, offsets)
        if once:
            return 0
        time.sleep(POLL_S)


if __name__ == "__main__":
    try:
        sys.exit(main())
    except KeyboardInterrupt:
        sys.exit(0)

class NormalizerTests(unittest.TestCase):
    def normalize(self, text, format_name="markdown"):
        result = subprocess.run(
            [str(NORMALIZER), "--format", format_name], input=text,
            text=True, capture_output=True, check=False,
        )
        self.assertEqual(result.returncode, 0, result.stderr)
        return result.stdout

    def test_real_markdown_v2_fixture_is_plain_speech(self):
        raw = (
            "\\# \\*Fleet reply\\*\n"
            "\\- Use `codex_telegram` with \\(snake_case\\)\\.\n"
            "\\- Read [the guide](https://example.invalid/guide)\\!\n"
            "1\\. Keep *bold*, __underlined__, ~struck~, and ||hidden|| text\\."
        )
        spoken = self.normalize(raw, "markdownv2")
        self.assertIn("Fleet reply", spoken)
        self.assertIn("codex_telegram", spoken)
        self.assertIn("snake_case", spoken)
        self.assertIn("Read the guide", spoken)
        self.assertIn("bold", spoken)
        self.assertNotIn("https", spoken)
        self.assertNotIn("\\", spoken)
        self.assertNotIn("*", spoken)

    def test_plain_markdown_structure_becomes_speech(self):
        raw = "## Heading\n> **Quoted** words\n\n* first\n* second with `inline_code`"
        spoken = self.normalize(raw)
        self.assertIn("Heading", spoken)
        self.assertIn("Quoted words", spoken)
        self.assertIn("first", spoken)
        self.assertIn("second with inline_code", spoken)
        self.assertNotIn("#", spoken)
        self.assertNotIn(">", spoken)
        self.assertNotIn("`", spoken)
        self.assertNotIn("*", spoken)

    def test_long_fence_is_omitted_and_short_fence_is_kept(self):
        long_block = "before\n```python\n" + ("print('too long')\n" * 80) + "```\nafter"
        spoken = self.normalize(long_block)
        self.assertIn("before", spoken)
        self.assertIn("code block omitted", spoken)
        self.assertIn("after", spoken)
        self.assertNotIn("too long", spoken)
        short = self.normalize("```python\nprint('kept')\n```")
        self.assertIn("print('kept')", short)
        self.assertNotIn("`", short)

    def test_identifiers_keep_internal_underscores_and_forbidden_symbols_never_survive(self):
        spoken = self.normalize(r"_soft_ snake_case codex_telegram C:\\temp **bold**")
        self.assertIn("snake_case", spoken)
        self.assertIn("codex_telegram", spoken)
        self.assertNotIn("\\", spoken)
        self.assertNotIn("*", spoken)
        self.assertNotIn("/", spoken)

    def test_paths_filter_slashes_to_spaces_and_large_numbers_are_omitted(self):
        raw = (
            "Read MailCortex delivery <179388615243.9498.7478661436905674088@kikin.helm>.\n"
            "Checked /home/metsatron/foo/bar.py and ~/.ductor/config/config.json with NEXUS/stow/ and and/or.\n"
            "Found 45 skills on port 8080 in year 2026, with 100,000 items and UUID 01a1a5f3-530f-7920-a7d1-3766bfd6a64b."
        )
        spoken = self.normalize(raw)
        self.assertNotIn("/", spoken)
        self.assertNotIn("179388615243", spoken)
        self.assertNotIn("7478661436905674088", spoken)
        self.assertNotIn("100,000", spoken)
        self.assertNotIn("01a1a5f3", spoken)
        self.assertIn("home metsatron foo bar.py", spoken)
        self.assertIn(".ductor config config.json", spoken)
        self.assertIn("NEXUS stow", spoken)
        self.assertIn("and or", spoken)
        self.assertIn("45", spoken)
        self.assertIn("8080", spoken)
        self.assertIn("2026", spoken)
