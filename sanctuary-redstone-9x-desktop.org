#------------------------------------------------------------------------------
# This one moves and then raises the window if you drag the mouse,
# only raises the window if you click, or does a vertical maximize if
# you double click (full screen maximize would be 'Maximize 100 100')
#------------------------------------------------------------------------------

DestroyFunc "Move-or-Raise"
AddToFunc "Move-or-Raise" "M" Move
+                         "M" Raise
+                         "C" Raise
+                         "D" Maximize 100 96



#------------------------------------------------------------------------------
# This one moves and then lowers the window if you drag the mouse,
# only lowers the window if you click, or does a RaiseLower if you double
# click
#------------------------------------------------------------------------------

DestroyFunc "Move-or-Lower"
AddToFunc "Move-or-Lower" "M" Move
+                         "M" Lower
+                         "C" Lower
+                         "D" RaiseLower


#------------------------------------------------------------------------------
# This one moves or (de)iconifies:
#------------------------------------------------------------------------------

DestroyFunc "Move-or-Iconify"
AddToFunc "Move-or-Iconify" "M" Move
+                           "D" Iconify

#------------------------------------------------------------------------------
# This one maximizes vertically if you click (leaving room for the taskbar
# at the bottom, or does a full maximization if you double click, or a
# true full vertical maximization if you just hold the mouse button down.
#------------------------------------------------------------------------------


DestroyFunc "Maximize-Func"
AddToFunc "Maximize-Func" "M"	Maximize 0 96
+ 			  "C"	Maximize 0 100
+ 			  "D"	Maximize 100 96

#------------------------------------------------------------------------------
# This one resizes and then raises the window if you drag the mouse,
# only raises the window if you click,  or does a RaiseLower if you double
# click
#------------------------------------------------------------------------------

DestroyFunc "Resize-or-Raise"
AddToFunc "Resize-or-Raise" "M" Resize
+                           "M" Raise
+                           "C" Raise
+                           "D" RaiseLower


#------------------------------------------------------------------------------
# This one focuses and then 'warps' to the center of the window.
#------------------------------------------------------------------------------


DestroyFunc "FocusAndWarp"
AddToFunc FocusAndWarp "I" Focus

#------------------------------------------------------------------------------
# Like the precedent, but de-iconifies first.

#------------------------------------------------------------------------------
# First, for the mouse in the root window (Context=R)
# Button 1 gives the Start menu
# Button 2 gives the Window Ops menu
# Button 3 gives the WindowList
#------------------------------------------------------------------------------


#     Button	Context Modifi 	Function

Mouse 1		R   	A       Menu "StartMenu" Nop
Mouse 2		R    	A       Menu "Window" Nop
# Root button3 is intentionally NOT bound: the xclass desktop manager
# installs its own XGrabButton(AnyButton) on the root to show its native
# context menu (New / Arrange Icons By / Line Up Icons / Auto Arrange /
# Properties). A fvwm95 root button3 binding would conflict (BadAccess) and
# prevent the xclass menu — and thus Arrange Icons By / Auto Arrange —
# from ever working.
#Mouse 3         R       A       Menu "DeskCtx" Nop
#Mouse 3	R    	A      	Module "winlist" FvwmWinList transient


#------------------------------------------------------------------------------
# Now the title bar buttons
# Any button in the left title-bar button gives the window ops menu
# Any button in the first right title-bar button Iconifies the window
# Any button in the second title-bar button full-maximizes
# Any button in the third (rightmost) title-bar button closes
# Note the use of "Mouse 0" for AnyButton.
#------------------------------------------------------------------------------


#     Button	Context Modif 	Function
Mouse 0		1    	A      	Function "window_ops_func"
Mouse 1         2       A       Delete
Mouse 0		4    	A     	Maximize 100 96
Mouse 0		6    	A     	Iconify


#------------------------------------------------------------------------------
# Now the rest of the frame
# Here I invoke my complex functions for Move-or-lower, Move-or-raise,
# and Resize-or-Raise.
#------------------------------------------------------------------------------

# Button 1 in the corners or sides, with any modifiers, gives resize or raise
Mouse 1		FS	A	Function "Resize-or-Raise"

# Button 1 in the title, sides, or icon, w/ any modifiers, gives move or raise
Mouse 1		T	A	Function "Move-or-Raise"

# Button 1 in an icons gives move for a drag, de-iconify for a double-click,
# nothing for a single click
Mouse 1		I	A	Function "Move-or-Iconify"

# Button 2 in an icon, w/ any modifiers, gives de-iconify
Mouse 2		I	A	Iconify

# Button 2 in the corners, sides, or title-bar gives the window ops menu
Mouse 2		FST	A	Function "window_ops_func"

# Button 3 anywhere in the decoration (except the title-bar buttons)
# does a raise-lower
Mouse 3		TSIF	A	RaiseLower

# Button 3 in the window, with the Modifier-1 key (usually alt or diamond)
# gives Raise-Lower. Used to use control here, but that interferes with xterm
Mouse 3         W       M       RaiseLower

PCManFM desktop icon grid helper.

PCManFM exposes =Snap to Grid= as an internal desktop-context action, not as
a command-line option.  This helper gives the IceWM menu an equivalent
all-icons action: stop the desktop layer, preserve a timestamped copy of its
position file, snap only desktop-icon coordinates to Redstone's 80-pixel grid,
then start the desktop layer again.  It does not enable auto-arrange, so later
manual movement remains possible.

#+BEGIN_SRC bash :tangle linux/.local/bin/redstone-9x-icons-grid :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

usage() {
  cat <<'EOF'
redstone-9x-icons-grid - snap PCManFM desktop icons to Redstone's grid

Usage: redstone-9x-icons-grid
       redstone-9x-icons-grid -h | --help

Stops the PCManFM desktop layer, backs up desktop-items-0.conf, rounds desktop
icon x/y coordinates to the nearest 80 pixels, and starts the desktop again.
EOF
}

case "${1:-}" in
  -h|--help)
    usage
    exit 0
    ;;
  "")
    ;;
  *)
    usage >&2
    exit 2
    ;;
esac

if (( $# > 1 )); then
  usage >&2
  exit 2
fi

config_home="${XDG_CONFIG_HOME:-${HOME}/.config}"
config="${config_home}/pcmanfm/default/desktop-items-0.conf"
cache_dir="${XDG_CACHE_HOME:-${HOME}/.cache}/redstone-9x"
grid=80

if [[ ! -f "${config}" ]]; then
  printf 'redstone-9x-icons-grid: PCManFM desktop config not found: %s\n' "${config}" >&2
  exit 1
fi

mkdir -p "${cache_dir}"
backup="${config}.backup-$(date +%Y%m%d-%H%M%S)-$$"
temp="$(mktemp "${config}.tmp.XXXXXX")"
restart_needed=0

cleanup() {
  [[ -z "${temp:-}" ]] || rm -f -- "${temp}"
  if (( restart_needed )); then
    pcmanfm --desktop >"${cache_dir}/pcmanfm-desktop.log" 2>&1 &
  fi
}
trap cleanup EXIT

pcmanfm --desktop-off >/dev/null 2>&1
restart_needed=1
sleep 0.5

cp -p -- "${config}" "${backup}"

awk -v grid="${grid}" '
function snap(value) {
  if (value >= 0) return int((value + grid / 2) / grid) * grid
  return -int((-value + grid / 2) / grid) * grid
}

/^\[/ {
  in_desktop_icon = ($0 ~ /^\[[^]]+\.desktop\]$/)
}

in_desktop_icon && /^x=-?[0-9]+$/ {
  sub(/^x=/, "")
  print "x=" snap($0 + 0)
  next
}

in_desktop_icon && /^y=-?[0-9]+$/ {
  sub(/^y=/, "")
  print "y=" snap($0 + 0)
  next
}

{ print }
' "${config}" >"${temp}"

chmod --reference="${config}" "${temp}"
mv -f -- "${temp}" "${config}"
temp=""

pcmanfm --desktop >"${cache_dir}/pcmanfm-desktop.log" 2>&1 &
restart_needed=0
trap - EXIT

printf 'Desktop icons snapped to a %s-pixel grid. Backup: %s\n' "${grid}" "${backup}"
#+END_SRC

# --- Why -dmenu, not run mode (rofi 2.0.0, verified 2026-07-18) ---------------
# rofi fires a themed button's `action:` on MOUSE CLICK only in -dmenu mode; in
# -modes run / -show run the click is silently swallowed, so the Browse and
# dropdown buttons were dead however they were styled. So the box is driven as
# -dmenu and this wrapper runs the accepted command itself — which also matches
# the real Win95 Run box, that does not autocomplete. The Run history IS the
# -dmenu row list, fed on stdin; it stays hidden (listview enabled:false) until
# the ▼ dropdown reveals it. A trailing blank line is always fed as a guard:
# clicking any button while the dmenu list is EMPTY segfaults rofi, so >=1 row
# must always be present.
#
# Button -> exit code -> handled below (no rofi binary patch):
#   Return / OK  (kb-accept-custom) -> exit 0  -> run the typed/selected command
#   Browse...    (button-browse -> kb-custom-1) -> exit 10 -> drun app launcher
#   ▼ dropdown   (combo-button  -> kb-custom-2) -> exit 11 -> reveal Run history
#   Cancel / Esc (kb-cancel)                    -> exit 1  -> do nothing
#
# Once IceWM maps the Rofi window, retitle it ("Run", not "rofi - Run") and
# centre it — rofi ignores its own theme location under -normal-window, so IceWM
# would otherwise pin it top-left and clip the titlebar. The map + WM handshake
# outruns a fixed delay, so poll (icesh ships with IceWM).
_frame_rofi() {
  local title="$1"
  ( for _i in $(seq 1 50); do
      if icesh -c Rofi list 2>/dev/null | grep -q Rofi; then
        icesh -c Rofi setWindowTitle "${title}" 2>/dev/null
        icesh -c Rofi center 2>/dev/null
        break
      fi
      sleep 0.08
    done ) &
}

# Most-recent-first, de-duplicated, non-blank history rows.
_history_rows() { tac "$histfile" 2>/dev/null | awk 'NF && !seen[$0]++'; }

# Prepend a command to the history (dedup, cap 50). Never fails the script.
_record() {
  local c="$1"
  [ -z "$c" ] && return 0
  { printf '%s\n' "$c"; grep -vxF "$c" "$histfile" 2>/dev/null || true; } \
    | awk 'NF' | head -n 50 >"$histfile.tmp" 2>/dev/null \
    && mv "$histfile.tmp" "$histfile" 2>/dev/null || true
  return 0
}

view="run"
while : ; do
  args=( -dmenu -normal-window -config "$config" -theme "$theme"
         -kb-accept-entry "Control+j,Control+m"
         -kb-accept-custom "Return,KP_Enter,Control+Return"
         -pid "$pidfile" )
  if [ "$view" = history ]; then
    # ▼ dropdown view: reveal the history list, hide the chevron (list is open).
    # Down navigates the list here (rofi default), so no key override.
    args+=( -theme-str 'listview { enabled: true; }'
            -theme-str 'combo-frame { enabled: false; }' )
    _frame_rofi "Run history"
  else
    # Run view: Down / Alt+Down / F4 open the dropdown, exactly like a real Win95
    # combobox. Move row-down off Down first (the list is hidden here, so row
    # navigation is unused) — rofi rejects a key bound to two actions.
    args+=( -kb-row-down "Control+n"
            -kb-custom-2 "Down,Alt+Down,F4" )
    _frame_rofi "Run"
  fi

  rc=0
  cmd="$( { _history_rows; printf '\n'; } | rofi "${args[@]}" )" || rc=$?

  case "$rc" in
    10)
      # Browse... -> the drun application launcher (a normal rofi window so its
      # listview grows without the -normal-window fixed-height oscillation).
      exec rofi -config "$config" -show drun -show-icons -display-drun "Programs"
      ;;
    11)
      view="history"
      continue
      ;;
    0)
      if printf '%s' "$cmd" | grep -q '[^[:space:]]'; then
        _record "$cmd"
        setsid bash -c "$cmd" >/dev/null 2>&1 &
      fi
      break
      ;;
    *)
      break
      ;;
  esac
done

Recycle Bin icon-state watcher.  PCManFM's own native trash desktop object
would track full/empty state for free, but room-gaming deliberately has no
=gvfs= (it crashes =pcmanfm --desktop= — see the running registry). This
polls the plain on-disk Trash folder instead — no GVFS, no new package,
just =ls=/=sed= against a path that already exists — and rewrites the
static launcher's =Icon== key directly. =pcmanfm --desktop= already
monitors =~/Desktop/= for content changes (that's how manual icon
repositioning/renaming persists), so an in-place edit to the =.desktop=
file's icon is picked up the same way.

#+BEGIN_SRC bash :tangle linux/.local/bin/redstone-9x-recycle-bin-watch :shebang "#!/bin/bash" :tangle-mode (identity #o755)
# Redstone 9X — Recycle Bin icon state watcher.
# Toggles ~/Desktop/recycle-bin.desktop's Icon= between user-trash and
# user-trash-full to match the real on-disk Trash folder, since the static
# launcher has no dynamic state of its own (no gvfs in this room — see
# redstone-9x-state-digest.md). Plain poll loop; no inotify dependency.
set -euo pipefail

TRASH_FILES="${XDG_DATA_HOME:-${HOME}/.local/share}/Trash/files"
DESKTOP_ICON="${HOME}/Desktop/recycle-bin.desktop"

while true; do
  if [[ -d "${TRASH_FILES}" ]] && [[ -n "$(ls -A "${TRASH_FILES}" 2>/dev/null)" ]]; then
    icon="user-trash-full"
  else
    icon="user-trash"
  fi
  # Compare against the file's ACTUAL current Icon= line, not a remembered
  # in-process variable — a relaunch reprojects a fresh copy of this file
  # over top of a still-running watcher, and an in-memory "last state" would
  # never notice the reset, silently going stale (found 2026-07-07).
  if [[ -f "${DESKTOP_ICON}" ]] && ! grep -qx "Icon=${icon}" "${DESKTOP_ICON}"; then
    sed -i "s/^Icon=.*/Icon=${icon}/" "${DESKTOP_ICON}"
  fi
  sleep 3
done
#+END_SRC

** =Exec=" must be QUOTED when the path contains spaces

A =.desktop= =Exec== line is not a shell command; the launcher runs it through
=g_shell_parse_argv=, which splits on whitespace exactly like a POSIX shell.  Every
launcher in the ports tree is named like =Ultimate Doom (MP3) (crispy).sh=, so an
unquoted =Exec== resolves =argv[0]= to =.../ports/doom/Ultimate= and the icon does
nothing at all — no error, no window, no log line.  The Ultimate Doom entry shipped
broken this way and nobody noticed, because a desktop icon that silently does nothing
looks exactly like a desktop icon nobody clicked.  Quote the path.  Spaces AND
parentheses are reserved characters in the spec.

Stages Microsoft and period-appropriate fonts into the room's private font directory
at =~/.local/share/dotcortex/guests/sanctuary-redstone-9x/fonts/=.  This path is

#+BEGIN_SRC bash :tangle linux/.local/bin/redstone-9x-font-provision :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

SANCTUARY="sanctuary-redstone-9x"
GUESTS_DIR="${XDG_DATA_HOME:-$HOME/.local/share}/dotcortex/guests"
FONT_STAGE="${GUESTS_DIR}/${SANCTUARY}/fonts"
FONT_WINEPREFIX="${GUESTS_DIR}/${SANCTUARY}/font-provision-prefix"

WINE_PROFILE="${HOME}/.guix-extra-profiles/room-windows-compat/room-windows-compat"

  if [[ -d "${PREFIX_FONTS}" ]]; then
    for font in \
      andale32.exe arial.ttf arialbd.ttf arialbi.ttf ariali.ttf ariblk.ttf \
      comic.ttf comicbd.ttf cour.ttf courbd.ttf courbi.ttf couri.ttf \
      georgia.ttf georgiab.ttf georgiai.ttf georgiaz.ttf impact.ttf \

GEPROTON_FONTS="/home/metsatron/mnt/x230/.steam/compatibilitytools.d/GE-Proton7-45/files/share/wine/fonts"

GUIX_WINE_FONTS="${WINE_PROFILE}/share/wine/fonts"

CASCADIA_TTF="${FONT_STAGE}/CascadiaMono.ttf"

* Redstone PowerShell launcher

PowerShell interactive redraw is sensitive to the terminal line editor and
terminal identity.  Keep this isolated from Bash/CMD launchers.

#+BEGIN_SRC bash :tangle linux/.local/bin/redstone-9x-pwsh :shebang "#!/bin/bash" :tangle-mode (identity #o755)
set -euo pipefail

export TERM="${TERM:-xterm-256color}"
export POWERSHELL_UPDATECHECK=Off
export DOTNET_CLI_TELEMETRY_OPTOUT=1

# .NET 8 System.Console locates the terminfo database via $TERMINFO, ~/.terminfo,
# then fixed roots (/usr/share/terminfo, ...) -- it NEVER consults $TERMINFO_DIRS.
# Inside this rootless Distrobox container /usr/share/terminfo has no 'xterm'
# entry (only /run/host/usr/share/terminfo/x/xterm does), so .NET loads no
# terminfo, CursorAddress (cup) is null, Console.SetCursorPosition silently
# emits nothing while still caching the move, and PSReadLine's full-buffer
# redraw never repositions to the prompt origin -> the interactive "smear".
# CPR (ESC[6n) is unaffected because .NET hard-codes it.  Point .NET at the
# host terminfo tree so cup is emitted and the redraw is correct.  (Root cause
# confirmed 2026-07-06 via strace: openat of the host xterm entry + 26 CUP
# writes appear, smear gone.)
if [ -d /run/host/usr/share/terminfo ]; then
  export TERMINFO=/run/host/usr/share/terminfo
fi

# Let pwsh handle the TTY directly.  PSReadLine (PowerShell 7's readline
# engine) sets raw terminal mode itself and handles all echoing; running
# stty -echo before launch creates a double-echo conflict with PSReadLine.
pwsh -NoLogo -NoProfile
#+END_SRC

#+BEGIN_SRC powershell :tangle linux/.config/redstone-9x/powershell/basic-repl.ps1 :mkdirp yes
$host.UI.RawUI.WindowTitle = "Windows PowerShell"
Remove-Module PSReadLine -Force -ErrorAction SilentlyContinue

while ($true) {
    [Console]::Write("PS $($PWD.Path)> ")
    $line = [Console]::ReadLine()
    if ($null -eq $line) { break }
    if ($line.Trim() -eq "exit") { break }
    if ($line.Trim().Length -eq 0) { continue }

    try {
        Invoke-Expression $line
    }
    catch {
        Write-Error $_
    }
}
#+END_SRC

** Wine tool dispatcher — redstone-9x-wine

One env-wrapped entry point for every Wine tool.  It sources the same three Guix
profiles and Wine environment as =redstone-9x-run=, then dispatches to a tool
chosen either by an explicit first argument (how the IceWM menu calls it, e.g.
=redstone-9x-wine winecfg=) or by its own invoked name (how the Run-box PATH
shims call it — =~/.local/bin/explorer= is a symlink to this script, so typing
=explorer= in the Run box lands here with =$0= = =explorer=).  Native profile
binaries (winetricks, wineboot, msiexec, regsvr32) are exec'd by ABSOLUTE path so
a same-named PATH shim can never recurse into itself; the =.exe= tools go through
=wine=, which is never shadowed.

#+BEGIN_SRC bash :tangle linux/.local/bin/redstone-9x-wine :shebang "#!/bin/bash" :tangle-mode (identity #o755)
set -euo pipefail

GUIX_PROFILES="${HOME}/.guix-extra-profiles"
WINCOMPAT="${GUIX_PROFILES}/room-windows-compat/room-windows-compat"

source_guix_profile() {
  local profile="$1"
  local status=0
  local had_nounset=0

  case $- in
    *u*) had_nounset=1; set +u ;;
  esac

  GUIX_PROFILE="$profile"
  export GUIX_PROFILE

  if . "$profile/etc/profile"; then
    status=0
  else
    status=$?
  fi

  (( had_nounset )) && set -u
  return "$status"
}

for _prof in desktop-common room-gaming room-windows-compat; do
  source_guix_profile "${GUIX_PROFILES}/${_prof}/${_prof}"
done

export DISPLAY="${DISPLAY:-:93}"
export XDG_RUNTIME_DIR="${XDG_RUNTIME_DIR:-/run/user/$(id -u)}"
export WINEPREFIX="${WINEPREFIX:-${HOME}/.wine-redstone}"
export WINEDEBUG="${WINEDEBUG:--all}"
export DOTCORTEX_DIR="${DOTCORTEX_DIR:-${HOME}/DotCortex}"

# Pick the tool: explicit first arg when invoked as redstone-9x-wine, otherwise
# the name we were called by (a PATH shim symlink).
self="$(basename "$0")"
if [ "${self}" = "redstone-9x-wine" ]; then
  tool="${1:-explorer}"
  shift || true
else
  tool="${self}"
fi

case "${tool}" in
  # Native profile binaries — absolute path so a same-named shim can't recurse.
  winetricks)
    if [ "$#" -eq 0 ] && command -v zenity >/dev/null 2>&1; then
      exec "${WINCOMPAT}/bin/winetricks" --gui
    fi
    exec "${WINCOMPAT}/bin/winetricks" "$@" ;;
  wineboot|msiexec|regsvr32|winedbg|winepath)
    exec "${WINCOMPAT}/bin/${tool}" "$@" ;;
  # Friendly aliases.
  ie|iexplore) exec wine iexplore.exe "$@" ;;
  uninstall|addremove) exec wine uninstaller.exe "$@" ;;
  cmd|command) exec wine cmd.exe "$@" ;;
  # Everything else is a Wine built-in .exe reached through the wine loader.
  *) exec wine "${tool}" "$@" ;;
esac
#+END_SRC

# Redstone 9X sanctuary bash rc — Win9x/MS-DOS shell identity.
# Shared sanctuary grammar: symlinked to ~/.bashrc by sanctuary-redstone-9x-launch.
# Interactive shells only.
case $- in *i*) ;; *) return ;; esac

# Map the Unix path to a DOS drive path: $HOME collapses to Z:\ , anything above
# $HOME shows as a C:\ path, forward slashes become backslashes. The final result
# is doubled ('\' -> '\\') because it lands in PS1, which bash re-expands as prompt
# escapes — an un-doubled '\h' '\s' '\d' '\w' '\u' would render as the
# hostname/shell/date/cwd/user. (Verified via ${PS1@P} rendering, 2026-07-15.)
_redstone_dospath() {
  local p="${PWD}" rel dos
  if [ "${p}" = "${HOME}" ]; then
    dos='Z:\'
  elif [ "${p#${HOME}/}" != "${p}" ]; then
    rel="${p#${HOME}/}"; dos="Z:\\${rel//\//\\}"
  else
    rel="${p#/}"; dos="C:\\${rel//\//\\}"
  fi
  printf '%s' "${dos//\\/\\\\}"
}
PROMPT_COMMAND='PS1="BASH $(_redstone_dospath)> "'

# First interactive shell in each terminal: open at HOME, print the boot banner.
# REDSTONE_SHELL_INIT is exported so nested subshells inherit it and stay quiet.
if [ -z "${REDSTONE_SHELL_INIT:-}" ]; then
  export REDSTONE_SHELL_INIT=1
  cd "${HOME}" 2>/dev/null || true
  printf '\033[97mMineshaft Redstone 9X.0.26715\033[0m\n'
fi
