* Host inotify capacity

GTK accessibility bridges and game launchers can consume one inotify instance
per process. The default Devuan ceiling is too small for a long-running gaming
sanctuary, so the host policy is declared here and applied before a
RetroPie-backed sanctuary starts. The helper writes a persistent
=/etc/sysctl.d/= drop-in and applies the same values immediately. It requires
the existing non-interactive sudo policy when launched as the normal user.

#+BEGIN_SRC bash :tangle linux/.local/bin/sanctuary-gaming-inotify-apply :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

CONFIG_PATH="${SANCTUARY_GAMING_INOTIFY_CONFIG:-/etc/sysctl.d/90-dotcortex-gaming-sanctuaries.conf}"
WATCHES="${SANCTUARY_GAMING_INOTIFY_WATCHES:-1048576}"
INSTANCES="${SANCTUARY_GAMING_INOTIFY_INSTANCES:-1024}"

usage() {
  cat <<'EOF'
Usage: sanctuary-gaming-inotify-apply [--check]

Install and immediately apply the host inotify ceilings used by gaming
sanctuaries. --check only reports the live kernel values and config path.
EOF
}

[[ "${WATCHES}" =~ ^[0-9]+$ && "${INSTANCES}" =~ ^[0-9]+$ ]] || {
  echo "sanctuary-gaming-inotify-apply: limits must be numeric" >&2
  exit 2
}

if [[ "${1:-}" == "--check" ]]; then
  printf 'config   : %s\n' "${CONFIG_PATH}"
  printf 'watches  : %s\n' "$(sysctl -n fs.inotify.max_user_watches 2>/dev/null || echo unavailable)"
  printf 'instances: %s\n' "$(sysctl -n fs.inotify.max_user_instances 2>/dev/null || echo unavailable)"
  exit 0
fi
[[ "${1:-}" == "" ]] || { usage >&2; exit 2; }

tmp="$(mktemp)"
trap 'rm -f -- "$tmp"' EXIT
printf '%s\n' \
  '# Managed by DotCortex: shared gaming sanctuary capacity.' \
  "fs.inotify.max_user_watches = ${WATCHES}" \
  "fs.inotify.max_user_instances = ${INSTANCES}" >"${tmp}"

if [[ "$(id -u)" -eq 0 ]]; then
  install -D -m 0644 "${tmp}" "${CONFIG_PATH}"
  sysctl -w "fs.inotify.max_user_watches=${WATCHES}" \
          "fs.inotify.max_user_instances=${INSTANCES}"
else
  command -v sudo >/dev/null 2>&1 || {
    echo "sanctuary-gaming-inotify-apply: sudo is required to provision ${CONFIG_PATH}" >&2
    exit 127
  }
  sudo -n install -D -m 0644 "${tmp}" "${CONFIG_PATH}"
  sudo -n sysctl -w "fs.inotify.max_user_watches=${WATCHES}" \
                   "fs.inotify.max_user_instances=${INSTANCES}"
fi

printf 'sanctuary-gaming-inotify-apply: watches=%s instances=%s\n' \
  "${WATCHES}" "${INSTANCES}"
#+END_SRC

* Guarded RetroPie hub projection

=~/RetroPie/stow= is a shared game-data hub. It is deliberately not
added to =loom stow:t480s= or any normal DotCortex package set: the hub is
outside this checkout and belongs to the fileserver migration boundary.

The helper below is the sanctuary-safe equivalent of stowing a hub entry. A
normal GNU Stow run would calculate links from the host path, which is not the
path visible inside Distrobox. These mappings therefore create links to the
guest-visible bind path, refuse arbitrary package names, and preserve local
state by timestamped quarantine. It is dry-run by default.

#+BEGIN_SRC bash :tangle linux/.local/bin/sanctuary-retropie-stow :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

RETROPIE_HUB_ROOT="${RETROPIE_HUB_ROOT:-${HOME}/RetroPie}"
RETROPIE_CONTAINER_ROOT="${RETROPIE_CONTAINER_ROOT:-/run/dotcortex/retropie}"
GAMES_HOST_ROOT="${GAMES_HOST_ROOT:-${HOME}/Games}"
GAMES_CONTAINER_ROOT="${GAMES_CONTAINER_ROOT:-/run/dotcortex/games}"
GUEST_HOME="${SANCTUARY_GUEST_HOME:-}"
APPLY=0
MAPPING=""

usage() {
  cat <<'EOF'
Usage: sanctuary-retropie-stow --guest-home PATH [--mapping NAME] [--apply]

Mappings:
  openra-content   RetroPie/stow/.openra/Content -> OpenRA Flatpak Content
                   (also composes Dune 2000 v3 from the shared Games tree)
  openra-settings   RetroPie/stow/.openra/settings.yaml -> OpenRA Flatpak settings

The default is a dry-run. --apply is required for filesystem changes.
EOF
}

while (( $# > 0 )); do
  case "$1" in
    --guest-home) [[ $# -ge 2 ]] || { usage >&2; exit 2; }; GUEST_HOME="$2"; shift 2 ;;
    --mapping) [[ $# -ge 2 ]] || { usage >&2; exit 2; }; MAPPING="$2"; shift 2 ;;
    --apply) APPLY=1; shift ;;
    -h|--help) usage; exit 0 ;;
    *) usage >&2; exit 2 ;;
  esac
done

[[ -n "${GUEST_HOME}" && -n "${MAPPING}" ]] || { usage >&2; exit 2; }

hub_root="$(readlink -f -- "${RETROPIE_HUB_ROOT}" 2>/dev/null || true)"
[[ -d "${hub_root}/roms/ports" && -d "${hub_root}/stow" ]] || {
  echo "sanctuary-retropie-stow: RetroPie hub is unavailable: ${RETROPIE_HUB_ROOT}" >&2
  exit 1
}

guest_root="$(readlink -f -- "${GUEST_HOME}" 2>/dev/null || true)"
guests_root="$(readlink -f -- "${XDG_DATA_HOME:-${HOME}/.local/share}/dotcortex/guests" 2>/dev/null || true)"
[[ -n "${guest_root}" && -d "${guest_root}" ]] || {
  echo "sanctuary-retropie-stow: guest home is unavailable: ${GUEST_HOME}" >&2
  exit 1
}
case "${guest_root}" in
  "${guests_root}"/sanctuary-*/home) ;;
  *) echo "sanctuary-retropie-stow: refusing target outside a sanctuary guest home: ${guest_root}" >&2; exit 1 ;;
esac

COMPOSE_DUNE=0
games_host_root="$(readlink -f -- "${GAMES_HOST_ROOT}" 2>/dev/null || true)"
DUNE_DATA_HOST_ROOT="${games_host_root}/dune-2000/dosdevices/c:/Program Files/Gruntmods Studios/Dune 2000/data"
DUNE_DATA_VISIBLE_ROOT="${GAMES_CONTAINER_ROOT}/dune-2000/dosdevices/c:/Program Files/Gruntmods Studios/Dune 2000/data"

dune_source_ready() {
  local path
  for path in \
    Movies Music GAMESFX BLOXBAT.R16 BLOXBASE.R16 BLOXBGBS.R16 \
    BLOXICE.R16 BLOXTREE.R16 BLOXWAST.R16 MOUSE.R16 DATA.R16; do
    [[ -e "${DUNE_DATA_HOST_ROOT}/${path}" ]] || return 1
  done
  [[ -e "${DUNE_DATA_HOST_ROOT}/bin/PALETTE.BIN" \
    && -e "${DUNE_DATA_HOST_ROOT}/GAMESFX/SOUND.RS" ]]
}

case "${MAPPING}" in
  openra-content)
    source_rel='.openra/Content'
    target_rel='.var/app/net.openra.OpenRA/.openra/Content'
    visible_rel="${RETROPIE_CONTAINER_ROOT}/stow/.openra/Content"
    if [[ -d "${DUNE_DATA_HOST_ROOT}" ]] && dune_source_ready; then
      COMPOSE_DUNE=1
    fi
    ;;
  openra-settings)
    source_rel='.openra/settings.yaml'
    target_rel='.var/app/net.openra.OpenRA/.openra/settings.yaml'
    visible_rel="${RETROPIE_CONTAINER_ROOT}/stow/.openra/settings.yaml"
    ;;
  *) echo "sanctuary-retropie-stow: mapping is not allowlisted: ${MAPPING}" >&2; exit 2 ;;
esac

source_path="${hub_root}/stow/${source_rel}"
target_path="${guest_root}/${target_rel}"
[[ -e "${source_path}" ]] || {
  echo "sanctuary-retropie-stow: source is unavailable: ${source_path}" >&2
  exit 1
}

if [[ "${MAPPING}" = openra-content && "${COMPOSE_DUNE}" -eq 1 \
  && -d "${target_path}" && ! -L "${target_path}" \
  && "$(readlink -- "${target_path}/cnc" 2>/dev/null || true)" = "${visible_rel}/cnc" \
  && "$(readlink -- "${target_path}/d2k/v2" 2>/dev/null || true)" = "${visible_rel}/d2k/v2" \
  && "$(readlink -- "${target_path}/d2k/v3/Movies" 2>/dev/null || true)" = "${DUNE_DATA_VISIBLE_ROOT}/Movies" \
  && "$(readlink -- "${target_path}/d2k/v3/Music" 2>/dev/null || true)" = "${DUNE_DATA_VISIBLE_ROOT}/Music" ]]; then
  echo "sanctuary-retropie-stow: already projected ${MAPPING} with Dune v3 data"
  exit 0
fi

if [[ "${MAPPING}" != openra-content || "${COMPOSE_DUNE}" -eq 0 ]] \
  && [[ -L "${target_path}" && "$(readlink -- "${target_path}")" == "${visible_rel}" ]]; then
  echo "sanctuary-retropie-stow: already projected ${MAPPING} -> ${visible_rel}"
  exit 0
fi

if [[ "${APPLY}" -eq 0 ]]; then
  if [[ -e "${target_path}" || -L "${target_path}" ]]; then
    echo "would quarantine: ${target_path}"
  fi
  if [[ "${MAPPING}" = openra-content && "${COMPOSE_DUNE}" -eq 1 ]]; then
    echo "would compose: ${target_path} (shared C&C/RA/Dune v2 plus Games Dune v3)"
  else
    echo "would link: ${target_path} -> ${visible_rel}"
  fi
  exit 0
fi

mkdir -p -- "$(dirname -- "${target_path}")"
if [[ -e "${target_path}" || -L "${target_path}" ]]; then
  quarantine="${target_path}.local-$(date +%Y%m%d-%H%M%S)"
  mv -- "${target_path}" "${quarantine}"
  echo "sanctuary-retropie-stow: quarantined local target: ${quarantine}" >&2
fi
if [[ "${MAPPING}" = openra-content && "${COMPOSE_DUNE}" -eq 1 ]]; then
  # The RetroPie export is read-only on some hosts. Keep the content root
  # local, but link every game payload back to the guest-visible shared trees;
  # this adds Dune's current v3 layout without copying or renaming its files.
  mkdir -p -- "${target_path}/d2k/v3"
  for mod in cnc ra; do
    [[ -d "${source_path}/${mod}" ]] \
      && ln -s -- "${visible_rel}/${mod}" "${target_path}/${mod}"
  done
  [[ -d "${source_path}/d2k/v2" ]] \
    && ln -s -- "${visible_rel}/d2k/v2" "${target_path}/d2k/v2"
  ln -s -- "${DUNE_DATA_VISIBLE_ROOT}/Movies" "${target_path}/d2k/v3/Movies"
  ln -s -- "${DUNE_DATA_VISIBLE_ROOT}/Music" "${target_path}/d2k/v3/Music"
  ln -s -- "${DUNE_DATA_VISIBLE_ROOT}/GAMESFX" "${target_path}/d2k/v3/GAMESFX"
  for file in \
    BLOXBAT.R16 BLOXBASE.R16 BLOXBGBS.R16 BLOXICE.R16 BLOXTREE.R16 \
    BLOXWAST.R16 MOUSE.R16 DATA.R16; do
    ln -s -- "${DUNE_DATA_VISIBLE_ROOT}/${file}" "${target_path}/d2k/v3/${file}"
  done
  ln -s -- "${DUNE_DATA_VISIBLE_ROOT}/bin/PALETTE.BIN" "${target_path}/d2k/v3/PALETTE.BIN"
  ln -s -- "${DUNE_DATA_VISIBLE_ROOT}/GAMESFX/SOUND.RS" "${target_path}/d2k/v3/SOUND.RS"
  echo "sanctuary-retropie-stow: composed ${MAPPING} with Dune v3 data from Games"
else
  ln -s -- "${visible_rel}" "${target_path}"
  echo "sanctuary-retropie-stow: projected ${MAPPING} -> ${visible_rel}"
fi
#+END_SRC

The launcher for each RetroPie-backed gaming sanctuary invokes the inotify
helper before a desktop starts. Redstone 9X additionally invokes the
=openra-content= mapping, while the existing shared projection keeps the
guest's private settings and logs separate from the hub.
