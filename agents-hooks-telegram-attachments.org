* Adoption contract

Linux native Claude only: exact own =~/.npm-global/bin/claude= executable inode
and kernel PID/start identity shared by plugin and PermissionRequest child.
The first native download hook binds that owner generation to session_id;
no tool JSON, transcript, filename or size grants authority. Sessions never
transfer. A plugin restart invalidates every old receipt. Fresh resend required
for historical attachments. No historical attestation/replay shortcut exists.

Private 0700 inbox and custody/registry directories are required. Shared
channel ancestors may remain 0770/0775: descriptors prevent path escape and
current-inode checks refuse raced paths. Existing permissive inboxes are refused,
never automatically chmodded. Parent owns any reviewed private-inbox migration.
Only exact produced bytes gain read_attachment MCP text access; native Read,
Write and Edit have no attachment carveout. download_attachment returns a
confirmation, never a path for native Read. read_attachment accepts only file_id
and rechecks native owner/plugin generation, bound session, receipt, inode and
content hash. UTF-8 text (including Markdown) is supported up to 256 KiB.
PNG/JPEG/PDF and all other binary formats fail closed with an explicit limit;
no image/PDF rendering or broad all-format access is claimed. Content dispatch
never trusts an uploader filename, extension or MIME. MCP text is untrusted
attachment data, not instructions or authority. Linux proc-fd floors remain.
All applier/helper security checks are unconditional, including under -O.
Same-UID arbitrary code execution is outside this filesystem trust boundary.

Preflight defaults to no mutation. Explicit --apply atomically promotes one
managed server.ts only after BOTH pin checks and complete-series compile check.
--rollback --apply restores only exact known managed bytes. No package, bot,
credential, service, queue or process lifecycle operation is performed.
The coordinator deploys helper/applier/assets first, reviews both patch engines,
then applies the source-owned patch and refreshes the main native MCP session.

#!/usr/bin/env python3
"""Preflight the FULL ordered pinned series before an explicit --apply.
Only server.ts is managed. No bot startup, credential read, install or restart.
"""
import argparse
import hashlib
import json
import os
from pathlib import Path
import shutil
import stat
import subprocess
import tempfile

def require(ok, message='attachment patch refused'):
    if not ok:
        raise ValueError(message)

def sha(data):
    return hashlib.sha256(data).hexdigest()

def main():
    p = argparse.ArgumentParser()
    p.add_argument('runtime', type=Path)
    p.add_argument('--apply', action='store_true')
    p.add_argument('--rollback', action='store_true')
    p.add_argument('--engine', choices=('git', 'patch'), default='git')
    a = p.parse_args()
    assets = Path(__file__).resolve().parents[1] / 'share/dotcortex/telegram-attachments'
    m = json.loads((assets / 'manifest.json').read_text())
    pristine = (assets / 'server.ts.pristine').read_bytes()
    require(sha(pristine) == m['pristine_sha256'], 'pristine drift')
    with tempfile.TemporaryDirectory(prefix='telegram-series-', dir=os.environ.get('TMPDIR')) as scratch:
        stage = Path(scratch)
        (stage / 'server.ts').write_bytes(pristine)
        for name in m['series']:
            patch = assets / name
            argv = (['git', 'apply', '--verbose', str(patch)] if a.engine == 'git'
                    else ['patch', '--batch', '--fuzz=0', '-p1', '-i', str(patch)])
            r = subprocess.run(argv, cwd=stage, capture_output=True, text=True, check=True)
            require(not any((x in (r.stdout + r.stderr).lower() for x in ('offset', 'fuzz', 'reversed'))), 'nonexact patch')
        expected = (stage / 'server.ts').read_bytes()
        require(sha(expected) == m['expected_sha256'], 'series drift')
        # Compile only; external packages stay external and no code is executed.
        subprocess.run(['bun', 'build', str(stage / 'server.ts'), '--target=bun', '--packages=external',
                        '--outfile', str(stage / 'check.js')], capture_output=True, check=True)
        require(expected.count(b'await safeAttachmentDownload(args)') == 1)
        require(expected.count(b"case 'read_attachment':") == 1)
        require(expected.count(b"attachmentCustody('receive'") == 1)
        require(sha((assets / 'custody.py').read_bytes()) == m['custody_sha256'], 'custody helper drift')
        for name, expected_sha in m['upstream_metadata'].items():
            require(sha((a.runtime / name).read_bytes()) == expected_sha, 'runtime metadata incompatible')
        compile((assets / 'custody.py').read_text(), 'custody.py', 'exec')
        # Walk fixed descriptors: no leaf/parent symlink or shared runtime writer.
        fd = os.open('/', os.O_RDONLY | os.O_DIRECTORY)
        try:
            for part in a.runtime.absolute().parts[1:]:
                require(part not in ('', '.', '..'))
                child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
                os.close(fd)
                fd = child
            root = os.fstat(fd)
            require(root.st_uid == os.getuid() and (not root.st_mode & 0o022), 'unsafe runtime directory')
            f = os.open('server.ts', os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=fd)
            with os.fdopen(f, 'rb') as src:
                st = os.fstat(src.fileno())
                require(stat.S_ISREG(st.st_mode) and st.st_nlink == 1 and (st.st_uid == os.getuid()) and (not st.st_mode & 0o022))
                current = src.read()
            require(sha(current) in (m['pristine_sha256'], m['expected_sha256']), 'live drift; unchanged')
            desired = pristine if a.rollback else expected
            if a.apply and current != desired:
                # One-file atomic promotion only AFTER whole-series preflight.
                temp = '.server.ts.custody-' + os.urandom(16).hex()
                out = os.open(temp, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, stat.S_IMODE(st.st_mode), dir_fd=fd)
                with os.fdopen(out, 'wb') as dest:
                    os.fchmod(dest.fileno(), stat.S_IMODE(st.st_mode))
                    os.fchown(dest.fileno(), -1, st.st_gid)
                    dest.write(desired); dest.flush(); os.fsync(dest.fileno())
                # Recheck exact source inode/bytes before atomic replacement.
                check = os.open('server.ts', os.O_RDONLY | os.O_NOFOLLOW, dir_fd=fd)
                with os.fdopen(check, 'rb') as src:
                    now = os.fstat(src.fileno())
                    require((now.st_dev, now.st_ino) == (st.st_dev, st.st_ino) and src.read() == current, 'concurrent drift')
                os.rename(temp, 'server.ts', src_dir_fd=fd, dst_dir_fd=fd)
                os.fsync(fd)
            print(json.dumps({'preflight':'complete-series', 'changed':bool(a.apply and current != desired),
                              'target_sha256':sha(desired), 'mode':oct(stat.S_IMODE(st.st_mode)),
                              'action':'rollback' if a.rollback else 'adopt'}))
        finally:
            os.close(fd)

if __name__ == '__main__':
    main()

#!/usr/bin/env python3
"""Linux descriptor-confined Telegram custody. No credentials or document parsing.
Native Claude executable ancestry is the owner; the hook alone binds session_id.
Same-UID code execution is not an OS security boundary. Metadata is never input
from tool arguments. Restart/expiry/identity uncertainty fails closed.
"""
import base64
import fcntl
import hashlib
import json
import os
from pathlib import Path
import re
import secrets
import stat
import sys
import time

LIMIT = 20 * 1024 * 1024
TTL = 900
ID = re.compile(r'[A-Za-z0-9_-]{1,256}\Z')
DOWNLOAD = 'mcp__plugin_telegram_telegram__download_attachment'
READ = 'mcp__plugin_telegram_telegram__read_attachment'
TEXT_LIMIT = 256 * 1024

def require(ok):
    if not ok:
        raise ValueError('attachment custody refused')

def identity(pid):
    p = Path('/proc') / str(pid)
    fields = (p / 'stat').read_text().rsplit(')', 1)[1].split()
    require(p.stat().st_uid == os.getuid())
    boot = Path('/proc/sys/kernel/random/boot_id').read_text().strip()
    return {'pid': pid, 'start': fields[19], 'boot': boot}, int(fields[1])

def owner():
    expected = Path.home() / '.npm-global/bin/claude'
    expected_stat = expected.stat()
    # A Claude Code auto-update swaps the package dir under live seats: their
    # exe becomes '<root>/.claude-code-XXXX/bin/claude.exe (deleted)' and no
    # longer shares the new binary's inode. Match that exact npm layout too, or
    # every seat older than the update silently loses its attachments.
    package_root = os.path.dirname(os.path.dirname(os.path.dirname(os.path.realpath(expected))))
    replaced = re.compile(re.escape(package_root) + r'/\.?claude-code(?:-[A-Za-z0-9]+)?/bin/claude\.exe(?: \(deleted\))?\Z')
    pid = os.getpid()
    for _ in range(64):
        item, parent = identity(pid)
        exe = os.stat('/proc/%s/exe' % pid)
        if (exe.st_dev, exe.st_ino) == (expected_stat.st_dev, expected_stat.st_ino):
            return item
        if replaced.fullmatch(os.readlink('/proc/%s/exe' % pid)):
            return item
        require(parent > 1 and parent != pid)
        pid = parent
    raise ValueError('no native Claude owner')

def alive(item):
    try:
        return identity(item['pid'])[0] == item
    except (OSError, KeyError, ValueError):
        return False

class Directory:
    def __init__(self, path, create=False, private=True):
        require(isinstance(path, str) and path.startswith('/') and os.path.normpath(path) == path)
        parts = path.split('/')[1:]
        require(parts and all(x not in ('', '.', '..') for x in parts))
        fd = os.open('/', os.O_RDONLY | os.O_DIRECTORY)
        try:
            for part in parts:
                if create:
                    try:
                        os.mkdir(part, 0o700, dir_fd=fd)
                    except FileExistsError:
                        pass
                child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
                os.close(fd)
                fd = child
            info = os.fstat(fd)
            require(info.st_uid == os.getuid() and (not private or info.st_mode & 0o077 == 0))
            self.fd, self.path, self.info = fd, path, info
        except BaseException:
            os.close(fd)
            raise
    def close(self):
        os.close(self.fd)
    def current(self):
        other = Directory(self.path)
        try:
            require((other.info.st_dev, other.info.st_ino) == (self.info.st_dev, self.info.st_ino))
        finally:
            other.close()
    def regular(self, name, flags=os.O_RDONLY):
        require('/' not in name and name not in ('', '.', '..'))
        fd = os.open(name, flags | os.O_NOFOLLOW | os.O_NONBLOCK, 0o600, dir_fd=self.fd)
        try:
            st = os.fstat(fd)
            require(stat.S_ISREG(st.st_mode) and st.st_uid == os.getuid()
                    and st.st_nlink == 1 and st.st_mode & 0o077 == 0)
            return fd
        except BaseException:
            os.close(fd)
            raise
    def read(self, name):
        fd = self.regular(name)
        with os.fdopen(fd, 'rb') as f:
            raw = f.read(256 * 1024 + 1)
        require(len(raw) <= 256 * 1024)
        return json.loads(raw)
    def write(self, name, value, replace=False):
        raw = json.dumps(value, separators=(',', ':')).encode()
        require(len(raw) <= 256 * 1024)
        temp = '.txn-' + secrets.token_hex(16) if replace else name
        fd = self.regular(temp, os.O_WRONLY | os.O_CREAT | os.O_EXCL)
        with os.fdopen(fd, 'wb') as f:
            f.write(raw)
            f.flush()
            os.fsync(f.fileno())
        if replace:
            os.rename(temp, name, src_dir_fd=self.fd, dst_dir_fd=self.fd)
        self.current()

def registry(create=False):
    return Directory(str(Path.home() / '.local/state/dotcortex/telegram-attachments'), create)

def key(o):
    return str(o['pid']) + '-' + o['start'] + '.json'

def generation(state, nonce):
    require(isinstance(nonce, str) and re.fullmatch('[a-f0-9]{32}', nonce))
    # Only inbox owns bytes/proofs; channel ancestors may be shared 0770/0775.
    inbox = Directory(state + '/inbox', True)
    try:
        proof = Directory(state + '/inbox/.custody', True)
    finally:
        inbox.close()
    return proof

def locked(reg):
    fd = reg.regular('.lock', os.O_RDWR | os.O_CREAT)
    fcntl.flock(fd, fcntl.LOCK_EX)
    return fd

def begin(state, nonce):
    o = owner()
    plugin = identity(os.getppid())[0]
    reg = registry(True)
    lock = locked(reg)
    try:
        try:
            old = reg.read(key(o))
            require(not alive(old['plugin']) or old['plugin'] == plugin and old['nonce'] == nonce)
        except FileNotFoundError:
            pass
        p = generation(state, nonce)
        inbox = Directory(state + '/inbox')
        root = {'dev': inbox.info.st_dev, 'ino': inbox.info.st_ino}
        inbox.close()
        try:
            p.write(nonce + '.json', {'owner': o, 'plugin': plugin, 'nonce': nonce,
                    'session': None, 'receipts': {}, 'files': {}, 'root': root, 'anchor': None })
        finally:
            p.close()
        reg.write(key(o), {'owner': o, 'plugin': plugin, 'nonce': nonce, 'state': state}, True)
        return root
    finally:
        os.close(lock)
        reg.close()

def with_record(action):
    reg = registry()
    lock = locked(reg)
    proof = None
    try:
        o = owner()
        index = reg.read(key(o))
        require(index['owner'] == o and alive(index['plugin']))
        proof = generation(index['state'], index['nonce'])
        record = proof.read(index['nonce'] + '.json')
        require(record['owner'] == o and record['plugin'] == index['plugin'] and record['nonce'] == index['nonce'])
        result, dirty = action(index, record)
        if dirty:
            proof.write(index['nonce'] + '.json', record, True)
        return result
    finally:
        if proof:
            proof.close()
        os.close(lock)
        reg.close()

def verified_bytes(index, r, fid):
    now = time.time()
    require(fid in r['receipts'] and 0 <= now-r['receipts'][fid]['at'] < TTL)
    matches = [(name, rec) for name, rec in r['files'].items() if rec['id'] == fid]
    require(bool(matches))
    name, rec = matches[-1]
    require(re.fullmatch(r'attachment-[a-f0-9]{48}\.bin', name) and 0 <= now-rec['at'] < TTL)
    inbox = Directory(index['state'] + '/inbox')
    try:
        anchor = os.stat(r['anchor'])
        require((anchor.st_dev, anchor.st_ino) == (inbox.info.st_dev, inbox.info.st_ino)
                == (r['root']['dev'], r['root']['ino']))
        fd = inbox.regular(name)
        with os.fdopen(fd, 'rb') as f:
            st = os.fstat(f.fileno())
            require((st.st_dev, st.st_ino, st.st_size) == (rec['dev'], rec['ino'], rec['size']) and st.st_size <= LIMIT)
            raw = f.read(LIMIT+1)
        require(len(raw) == rec['size'] and hashlib.sha256(raw).hexdigest() == rec['sha'])
        inbox.current()
        return raw
    finally:
        inbox.close()

def text_content(raw):
    # Content, never uploader filename/MIME, determines this bounded capability.
    # Image/PDF decoding is deliberately NOT implemented by this text-only tool.
    unsupported = {'content': [{'type': 'text', 'text':
        'Unsupported attachment format or size: read_attachment supports only UTF-8 text up to 256 KiB; PNG, JPEG, PDF and other binary formats are not supported.'}], 'isError': True}
    if len(raw) > TEXT_LIMIT or raw.startswith((b'\x89PNG\r\n\x1a\n', b'\xff\xd8\xff', b'%PDF-')):
        return unsupported
    try:
        text = raw.decode('utf-8', errors='strict')
    except UnicodeDecodeError:
        return unsupported
    if any(ord(ch) < 32 and ch not in '\t\r\n' or 127 <= ord(ch) <= 159 for ch in text):
        return unsupported
    return {'content': [{'type': 'text', 'text': text}]}

def producer(op, data):
    def act(index, r):
        require(identity(os.getppid())[0] == r['plugin'] and data['nonce'] == r['nonce'])
        now = time.time()
        r['receipts'] = {k:v for k,v in r['receipts'].items() if 0 <= now - v['at'] < TTL}
        if op == 'anchor':
            require(set(data) == {'nonce', 'fd'} and type(data['fd']) is int and 0 <= data['fd'] <= 100000)
            anchor = '/proc/%s/fd/%s' % (r['plugin']['pid'], data['fd'])
            st = os.stat(anchor)
            require(stat.S_ISDIR(st.st_mode) and (st.st_dev, st.st_ino) == (r['root']['dev'], r['root']['ino']))
            r['anchor'] = anchor
            return True, True
        fid = data['file_id']
        require(isinstance(fid, str) and ID.fullmatch(fid))
        if op == 'receive':
            require(set(data) == {'nonce', 'file_id', 'sender', 'chat', 'message', 'size'})
            require(all(isinstance(data[k], str) and re.fullmatch(r'-?[0-9]{1,20}', data[k]) for k in ('sender', 'chat', 'message')))
            require(data['size'] is None or type(data['size']) is int and 0 <= data['size'] <= LIMIT)
            require(len(r['receipts']) < 128)
            r['receipts'][fid] = {'at':now, 'sender':data['sender'], 'chat':data['chat'], 'message':data['message']}
            return True, True
        if op == 'revoke':
            r['receipts'].pop(fid, None)
            return True, True
        require(fid in r['receipts'] and isinstance(r['session'], str) and isinstance(r['anchor'], str))
        if op == 'read':
            require(set(data) == {'nonce', 'file_id'})
            return text_content(verified_bytes(index, r, fid)), False
        if op == 'check':
            require(set(data) == {'nonce', 'file_id'})
            return True, False
        require(op == 'store' and set(data) == {'nonce', 'file_id', 'bytes'})
        buf = base64.b64decode(data['bytes'], validate=True)
        require(len(buf) <= LIMIT and len(r['files']) < 128)
        inbox = Directory(index['state'] + '/inbox')
        try:
            anchor_stat = os.stat(r['anchor'])
            require((anchor_stat.st_dev, anchor_stat.st_ino) == (inbox.info.st_dev, inbox.info.st_ino))
            name = 'attachment-' + secrets.token_hex(24) + '.bin'
            fd = inbox.regular(name, os.O_WRONLY | os.O_CREAT | os.O_EXCL)
            with os.fdopen(fd, 'wb') as f:
                f.write(buf)
                f.flush()
                os.fsync(f.fileno())
                st = os.fstat(f.fileno())
            inbox.current()
            r['files'][name] = {'id':fid, 'at':now, 'dev':st.st_dev, 'ino':st.st_ino,
                                'size':st.st_size, 'sha':hashlib.sha256(buf).hexdigest()}
            return r['anchor'] + '/' + name, True
        finally:
            inbox.close()
    return with_record(act)

def approval(event, bind=False):
    tool, args = event.get('tool_name'), event.get('tool_input')
    require(event.get('hook_event_name') == 'PermissionRequest' and isinstance(args, dict))
    sid = event.get('session_id')
    require(isinstance(sid, str) and re.fullmatch(r'[A-Za-z0-9_-]{1,128}', sid))
    require(tool in (DOWNLOAD, READ))
    require(set(args) == {'file_id'} and isinstance(args['file_id'], str) and ID.fullmatch(args['file_id']))
    def act(index, r):
        dirty = False
        # First trusted native PermissionRequest pins this Claude generation to
        # its session. A different session never inherits receipts, even if the
        # same executable survives. No tool-supplied owner/session is consulted.
        if r['session'] is None and bind and tool == DOWNLOAD and args['file_id'] in r['receipts']:
            r['session'], dirty = sid, True
        require(r['session'] == sid)
        now = time.time()
        if tool == DOWNLOAD:
            rec = r['receipts'][args['file_id']]
            require(0 <= now-rec['at'] < TTL)
            return True, dirty
        verified_bytes(index, r, args['file_id'])
        return True, False
    return with_record(act)

def main():
    op = sys.argv[1]
    raw = sys.stdin.buffer.read(LIMIT * 2 + 4097)
    require(len(raw) <= LIMIT * 2 + 4096)
    data = json.loads(raw)
    if op == 'begin':
        result = begin(data['state'], data['nonce'])
    else:
        result = producer(op, data)
    print(json.dumps(result))

if __name__ == '__main__':
    try:
        main()
    except Exception:
        print('attachment custody refused', file=sys.stderr)
        sys.exit(1)

function gate(ctx: Context): GateResult {
  const access = loadAccess()

  if (access.dmPolicy === 'disabled') return { action: 'drop' }

    if (access.allowFrom.includes(senderId)) return { action: 'deliver', access }

    for (const [code, p] of Object.entries(access.pending)) {
      if (p.senderId === senderId) {
        // Reply twice max (initial + one reminder), then go silent.
        if ((p.replies ?? 1) >= 2) return { action: 'drop' }
        p.replies = (p.replies ?? 1) + 1
        saveAccess(access)
        return { action: 'pair', code, isResend: true }
      }
    }
    // Cap pending at 3. Extra attempts are silently dropped.
    if (Object.keys(access.pending).length >= 3) return { action: 'drop' }

    const code = randomBytes(3).toString('hex') // 6 hex chars
    const now = Date.now()
    access.pending[code] = {
      senderId,
      chatId: String(ctx.chat!.id),
      createdAt: now,
      expiresAt: now + 60 * 60 * 1000, // 1h
      replies: 1,
    }
    saveAccess(access)
    return { action: 'pair', code, isResend: false }
  }

  if (chatType === 'group' || chatType === 'supergroup') {
    const groupId = String(ctx.chat!.id)
    const policy = access.groups[groupId]
    if (!policy) return { action: 'drop' }
    const groupAllowFrom = policy.allowFrom ?? []
    const requireMention = policy.requireMention ?? true
    if (groupAllowFrom.length > 0 && !groupAllowFrom.includes(senderId)) {
      return { action: 'drop' }
    }
    if (requireMention && !isMentioned(ctx, access.mentionPatterns)) {
      return { action: 'drop' }
    }
    return { action: 'deliver', access }
  }

// Telegram caps messages at 4096 chars. Split long replies, preferring
// paragraph boundaries when chunkMode is 'newline'.

function chunk(text: string, limit: number, mode: 'length' | 'newline'): string[] {
  if (text.length <= limit) return [text]
  const out: string[] = []
  let rest = text
  while (rest.length > limit) {
    let cut = limit
    if (mode === 'newline') {
      // Prefer the last double-newline (paragraph), then single newline,
      // then space. Fall back to hard cut.
      const para = rest.lastIndexOf('\n\n', limit)
      const line = rest.lastIndexOf('\n', limit)
      const space = rest.lastIndexOf(' ', limit)
      cut = para > limit / 2 ? para : line > limit / 2 ? line : space > 0 ? space : limit
    }
    out.push(rest.slice(0, cut))
    rest = rest.slice(cut).replace(/^\n+/, '')
  }
  if (rest) out.push(rest)
  return out
}

mcp.setRequestHandler(CallToolRequestSchema, async req => {
  const args = (req.params.arguments ?? {}) as Record<string, unknown>
  try {
    switch (req.params.name) {
      case 'reply': {
        const chat_id = args.chat_id as string
        const text = args.text as string
        const reply_to = args.reply_to != null ? Number(args.reply_to) : undefined
        const files = (args.files as string[] | undefined) ?? []
        const format = (args.format as string | undefined) ?? 'text'
        const parseMode = format === 'markdownv2' ? 'MarkdownV2' as const : undefined

        assertAllowedChat(chat_id)

        for (const f of files) {
          assertSendable(f)
          const st = statSync(f)
          if (st.size > MAX_ATTACHMENT_BYTES) {
            throw new Error(`file too large: ${f} (${(st.size / 1024 / 1024).toFixed(1)}MB, max 50MB)`)
          }
        }

        const access = loadAccess()
        const limit = Math.max(1, Math.min(access.textChunkLimit ?? MAX_CHUNK_LIMIT, MAX_CHUNK_LIMIT))
        const mode = access.chunkMode ?? 'length'
        const replyMode = access.replyToMode ?? 'first'
        const chunks = chunk(text, limit, mode)
        const sentIds: number[] = []

        try {
          for (let i = 0; i < chunks.length; i++) {
            const shouldReplyTo =
              reply_to != null &&
              replyMode !== 'off' &&
              (replyMode === 'all' || i === 0)
            const sent = await bot.api.sendMessage(chat_id, chunks[i], {
              ...(shouldReplyTo ? { reply_parameters: { message_id: reply_to } } : {}),
              ...(parseMode ? { parse_mode: parseMode } : {}),
            })
            sentIds.push(sent.message_id)
          }
        } catch (err) {
          const msg = err instanceof Error ? err.message : String(err)
          throw new Error(
            `reply failed after ${sentIds.length} of ${chunks.length} chunk(s) sent: ${msg}`,
          )
        }

        // Files go as separate messages (Telegram doesn't mix text+file in one
        // sendMessage call). Thread under reply_to if present.
        for (const f of files) {
          const ext = extname(f).toLowerCase()
          const input = new InputFile(f)
          const opts = reply_to != null && replyMode !== 'off'
            ? { reply_parameters: { message_id: reply_to } }
            : undefined
          if (PHOTO_EXTS.has(ext)) {
            const sent = await bot.api.sendPhoto(chat_id, input, opts)
            sentIds.push(sent.message_id)
          } else {
            const sent = await bot.api.sendDocument(chat_id, input, opts)
            sentIds.push(sent.message_id)
          }
        }

        const result =
          sentIds.length === 1
            ? `sent (id: ${sentIds[0]})`
            : `sent ${sentIds.length} parts (ids: ${sentIds.join(', ')})`
        return { content: [{ type: 'text', text: result }] }
      }
      case 'react': {
        assertAllowedChat(args.chat_id as string)
        await bot.api.setMessageReaction(args.chat_id as string, Number(args.message_id), [
          { type: 'emoji', emoji: args.emoji as ReactionTypeEmoji['emoji'] },
        ])
        return { content: [{ type: 'text', text: 'reacted' }] }
      }
      case 'download_attachment': {
        const file_id = args.file_id as string
        const file = await bot.api.getFile(file_id)
        if (!file.file_path) throw new Error('Telegram returned no file_path — file may have expired')
        const url = `https://api.telegram.org/file/bot${TOKEN}/${file.file_path}`
        const res = await fetch(url)
        if (!res.ok) throw new Error(`download failed: HTTP ${res.status}`)
        const buf = Buffer.from(await res.arrayBuffer())
        // file_path is from Telegram (trusted), but strip to safe chars anyway
        // so nothing downstream can be tricked by an unexpected extension.
        const rawExt = file.file_path.includes('.') ? file.file_path.split('.').pop()! : 'bin'
        const ext = rawExt.replace(/[^a-zA-Z0-9]/g, '') || 'bin'
        const uniqueId = (file.file_unique_id ?? '').replace(/[^a-zA-Z0-9_-]/g, '') || 'dl'
        const path = join(INBOX_DIR, `${Date.now()}-${uniqueId}.${ext}`)
        mkdirSync(INBOX_DIR, { recursive: true })
        writeFileSync(path, buf)
        return { content: [{ type: 'text', text: path }] }
      }
      case 'edit_message': {
        assertAllowedChat(args.chat_id as string)
        const editFormat = (args.format as string | undefined) ?? 'text'
        const editParseMode = editFormat === 'markdownv2' ? 'MarkdownV2' as const : undefined
        const edited = await bot.api.editMessageText(
          args.chat_id as string,
          Number(args.message_id),
          args.text as string,
          ...(editParseMode ? [{ parse_mode: editParseMode }] : []),
        )
        const id = typeof edited === 'object' ? edited.message_id : args.message_id
        return { content: [{ type: 'text', text: `edited (id: ${id})` }] }
      }
      default:
        return {
          content: [{ type: 'text', text: `unknown tool: ${req.params.name}` }],
          isError: true,
        }
    }
  } catch (err) {
    const msg = err instanceof Error ? err.message : String(err)
    return {
      content: [{ type: 'text', text: `${req.params.name} failed: ${msg}` }],
      isError: true,
    }
  }
})

// When Claude Code closes the MCP connection, stdin gets EOF. Without this
// the bot keeps polling forever as a zombie, holding the token and blocking
// the next session with 409 Conflict.
let shuttingDown = false
function shutdown(): void {
  if (shuttingDown) return
  shuttingDown = true
  process.stderr.write('telegram channel: shutting down\n')
  try {
    if (parseInt(readFileSync(PID_FILE, 'utf8'), 10) === process.pid) rmSync(PID_FILE)
  } catch {}
  // bot.stop() signals the poll loop to end; the current getUpdates request
  // may take up to its long-poll timeout to return. Force-exit after 2s.
  setTimeout(() => process.exit(0), 2000)
  void Promise.resolve(bot.stop()).finally(() => process.exit(0))
}
process.stdin.on('end', shutdown)
process.stdin.on('close', shutdown)
process.on('SIGTERM', shutdown)
process.on('SIGINT', shutdown)
process.on('SIGHUP', shutdown)

// Orphan watchdog: belt-and-suspenders for the stdin 'end'/'close' handlers
// above. Stdin is the MCP transport pipe inherited straight from the CLI; the
// kernel closes it on any CLI death (clean, crash, SIGKILL, OOM) regardless of
// intermediate wrappers. A ppid-change check used to live here but it
// false-fires when the bun-run/shell wrapper exits or execs during normal
// startup and we get reparented to init.
setInterval(() => {
  if (process.stdin.destroyed || process.stdin.readableEnded) shutdown()
}, 5000).unref()

// Inline-button handler for permission requests. Callback data is
// `perm:allow:<id>`, `perm:deny:<id>`, or `perm:more:<id>`.
// Security mirrors the text-reply path: allowFrom must contain the sender.
bot.on('callback_query:data', async ctx => {
  const data = ctx.callbackQuery.data
  const m = /^perm:(allow|deny|more):([a-km-z]{5})$/.exec(data)
  if (!m) {
    await ctx.answerCallbackQuery().catch(() => {})
    return
  }
  const access = loadAccess()
  const senderId = String(ctx.from.id)
  if (!access.allowFrom.includes(senderId)) {
    await ctx.answerCallbackQuery({ text: 'Not authorized.' }).catch(() => {})
    return
  }
  const [, behavior, request_id] = m

  if (behavior === 'more') {
    const details = pendingPermissions.get(request_id)
    if (!details) {
      await ctx.answerCallbackQuery({ text: 'Details no longer available.' }).catch(() => {})
      return
    }
    const { tool_name, description, input_preview } = details
    let prettyInput: string
    try {
      prettyInput = JSON.stringify(JSON.parse(input_preview), null, 2)
    } catch {
      prettyInput = input_preview
    }
    const expanded =
      `🔐 Permission: ${tool_name}\n\n` +
      `tool_name: ${tool_name}\n` +
      `description: ${description}\n` +
      `input_preview:\n${prettyInput}`
    const keyboard = new InlineKeyboard()
      .text('✅ Allow', `perm:allow:${request_id}`)
      .text('❌ Deny', `perm:deny:${request_id}`)
    await ctx.editMessageText(expanded, { reply_markup: keyboard }).catch(() => {})
    await ctx.answerCallbackQuery().catch(() => {})
    return
  }

  void mcp.notification({
    method: 'notifications/claude/channel/permission',
    params: { request_id, behavior },
  })
  pendingPermissions.delete(request_id)
  const label = behavior === 'allow' ? '✅ Allowed' : '❌ Denied'
  await ctx.answerCallbackQuery({ text: label }).catch(() => {})
  // Replace buttons with the outcome so the same request can't be answered
  // twice and the chat history shows what was chosen.
  const msg = ctx.callbackQuery.message
  if (msg && 'text' in msg && msg.text) {
    await ctx.editMessageText(`${msg.text}\n\n${label}`).catch(() => {})
  }
})

type AttachmentMeta = {
  kind: string
  file_id: string
  size?: number
  mime?: string
  name?: string
}

// Filenames and titles are uploader-controlled. They land inside the <channel>
// notification — delimiter chars would let the uploader break out of the tag
// or forge a second meta entry.
function safeName(s: string | undefined): string | undefined {
  return s?.replace(/[<>\[\]\r\n;]/g, '_')
}

async function handleInbound(
  ctx: Context,
  text: string,
  downloadImage: (() => Promise<string | undefined>) | undefined,
  attachment?: AttachmentMeta,
): Promise<void> {
  const result = gate(ctx)

  if (result.action === 'drop') return

  if (result.action === 'pair') {
    const lead = result.isResend ? 'Still pending' : 'Pairing required'
    await ctx.reply(
      `${lead} — run in Claude Code:\n\n/telegram:access pair ${result.code}`,
    )
    return
  }

  const access = result.access
  const from = ctx.from!
  const chat_id = String(ctx.chat!.id)
  const msgId = ctx.message?.message_id

  // Permission-reply intercept: if this looks like "yes xxxxx" for a
  // pending permission request, emit the structured event instead of
  // relaying as chat. The sender is already gate()-approved at this point
  // (non-allowlisted senders were dropped above), so we trust the reply.
  const permMatch = PERMISSION_REPLY_RE.exec(text)
  if (permMatch) {
    void mcp.notification({
      method: 'notifications/claude/channel/permission',
      params: {
        request_id: permMatch[2]!.toLowerCase(),
        behavior: permMatch[1]!.toLowerCase().startsWith('y') ? 'allow' : 'deny',
      },
    })
    if (msgId != null) {
      const emoji = permMatch[1]!.toLowerCase().startsWith('y') ? '✅' : '❌'
      void bot.api.setMessageReaction(chat_id, msgId, [
        { type: 'emoji', emoji: emoji as ReactionTypeEmoji['emoji'] },
      ]).catch(() => {})
    }
    return
  }

  // Typing indicator — signals "processing" until we reply (or ~5s elapses).
  void bot.api.sendChatAction(chat_id, 'typing').catch(() => {})

  // Ack reaction — lets the user know we're processing. Fire-and-forget.
  // Telegram only accepts a fixed emoji whitelist — if the user configures
  // something outside that set the API rejects it and we swallow.
  if (access.ackReaction && msgId != null) {
    void bot.api
      .setMessageReaction(chat_id, msgId, [
        { type: 'emoji', emoji: access.ackReaction as ReactionTypeEmoji['emoji'] },
      ])
      .catch(() => {})
  }

  const imagePath = downloadImage ? await downloadImage() : undefined

  // image_path goes in meta only — an in-content "[image attached — read: PATH]"
  // annotation is forgeable by any allowlisted sender typing that string.
  mcp.notification({
    method: 'notifications/claude/channel',
    params: {
      content: text,
      meta: {
        chat_id,
        ...(msgId != null ? { message_id: String(msgId) } : {}),
        user: from.username ?? String(from.id),
        user_id: String(from.id),
        ts: new Date((ctx.message?.date ?? 0) * 1000).toISOString(),
        ...(imagePath ? { image_path: imagePath } : {}),
        ...(attachment ? {
          attachment_kind: attachment.kind,
          attachment_file_id: attachment.file_id,
          ...(attachment.size != null ? { attachment_size: String(attachment.size) } : {}),
          ...(attachment.mime ? { attachment_mime: attachment.mime } : {}),
          ...(attachment.name ? { attachment_name: attachment.name } : {}),
        } : {}),
      },
    },
  }).catch(err => {
    process.stderr.write(`telegram channel: failed to deliver inbound to Claude: ${err}\n`)
  })
}

// Retry polling with backoff on any error. Previously only 409 was retried —
// a single ETIMEDOUT/ECONNRESET/DNS failure rejected bot.start(), the catch
// returned, and polling stopped permanently while the process stayed alive
// (MCP stdin keeps it running). Outbound tools kept working but the bot was
// deaf to inbound messages until a full restart.
void (async () => {
  for (let attempt = 1; ; attempt++) {
    try {
      await bot.start({
        onStart: info => {
          attempt = 0
          botUsername = info.username
          process.stderr.write(`telegram channel: polling as @${info.username}\n`)
          void bot.api.setMyCommands(
            [
              { command: 'start', description: 'Welcome and setup guide' },
              { command: 'help', description: 'What this bot can do' },
              { command: 'status', description: 'Check your pairing status' },
            ],
            { scope: { type: 'all_private_chats' } },
          ).catch(() => {})
        },
      })
      return // bot.stop() was called — clean exit from the loop
    } catch (err) {
      if (shuttingDown) return
      // bot.stop() mid-setup rejects with grammy's "Aborted delay" — expected, not an error.
      if (err instanceof Error && err.message === 'Aborted delay') return
      const is409 = err instanceof GrammyError && err.error_code === 409
