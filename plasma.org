* Apply Preferences
This file intentionally tangles scripts, not whole mutable KDE config files.
The live config files remain private GUI-managed state; this script only sets
explicit preferences through KDE's own CLI config tools.

need() {
  command -v "$1" >/dev/null 2>&1 || {
    printf 'missing required command: %s\n' "$1" >&2
    exit 1
  }
}

backup_once() {
  local path="$1"
  [ -e "$path" ] || return 0
  cp "$path" "$path.backup-$(date +%Y%m%d-%H%M%S)"
}

kset() {
  "$KWRITE" --file "$KGLOBAL" --group "$1" --key "$2" "$3"
}

* Plasmoid Manifest — pinned panel applets
Pinned upstream Plasma 6 applets installed per-user with =kpackagetool6=.
Each row is locked to a tag *and* its resolved commit; =plasmoid-apply=
refuses to install if the tag has moved. Plugin IDs are recorded from each
package's =metadata.json=, and re-verified against the source at apply time.

Update flow: bump TAG, resolve =git ls-remote --tags <upstream>= for the new
commit, update COMMIT, then =loom plasmoid:apply= and =loom plasmoid:reload=.
COMMIT must be the *peeled* hash: for annotated tags use the =^{}= line from
ls-remote, not the tag object hash.

#+BEGIN_SRC text :tangle linux/.plasmoid/manifest/applets.ssv :padline no
# NAME ID UPSTREAM TAG COMMIT PKGROOT TYPE
plasma6-window-title-applet org.kde.windowtitle https://github.com/dhruv8sh/plasma6-window-title-applet.git v0.9.0 a6eaf5086a473919ed2fffc5d3b8d98237c2dd41 . plasma-applet
application-title-bar com.github.antroids.application-title-bar https://github.com/antroids/application-title-bar.git v0.9.1 7a1f0ef0d1c7ae789f0a30026ca2115e7a0167fe package plasma-applet
#+END_SRC

* Plasmoid Apply
Clones/updates each upstream into the shared DotCortex source cache,
checks out the pinned commit, verifies the lock, then installs or upgrades
via =kpackagetool6=. Never restarts plasmashell — use =plasmoid-reload=.

#+BEGIN_SRC bash :tangle linux/.local/bin/plasmoid-apply :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

MANIFEST="${PLASMOID_SSV:-$HOME/.plasmoid/manifest/applets.ssv}"
SRC_ROOT="${PLASMOID_SRC_ROOT:-$HOME/.local/share/dotcortex/src}"
KPKG="${KPKG:-kpackagetool6}"
USER_PLASMOIDS="$HOME/.local/share/plasma/plasmoids"

die() { printf 'plasmoid-apply: %s\n' "$*" >&2; exit 1; }

command -v "$KPKG" >/dev/null 2>&1 || die "missing $KPKG (not a Plasma 6 machine?)"
command -v git >/dev/null 2>&1 || die "missing git"
[ -f "$MANIFEST" ] || die "missing manifest: $MANIFEST"

meta_field() {
  python3 -c 'import json,sys; print(json.load(open(sys.argv[1]))["KPlugin"][sys.argv[2]])' "$1" "$2"
}

mkdir -p "$SRC_ROOT"

while read -r name id upstream tag commit pkgroot type; do
  [ "$type" = "plasma-applet" ] || die "$name: unsupported install_type '$type'"
  src="$SRC_ROOT/$name"
  echo "== $name $tag ($commit)"

  if [ -d "$src/.git" ]; then
    git -C "$src" fetch --quiet --tags origin
  else
    git clone --quiet "$upstream" "$src"
  fi
  git -C "$src" checkout --quiet --detach "$commit" \
    || die "$name: pinned commit not present in $upstream"
  head="$(git -C "$src" rev-parse HEAD)"
  [ "$head" = "$commit" ] || die "$name: HEAD $head != pinned $commit"
  tag_commit="$(git -C "$src" rev-list -n1 "refs/tags/$tag" 2>/dev/null || true)"
  if [ -n "$tag_commit" ] && [ "$tag_commit" != "$commit" ]; then
    die "$name: tag $tag moved upstream to $tag_commit (pinned $commit) — re-resolve and update the manifest"
  fi

  dir="$src"
  [ "$pkgroot" = "." ] || dir="$src/$pkgroot"
  meta="$dir/metadata.json"
  [ -f "$meta" ] || die "$name: no metadata.json under $dir"
  real_id="$(meta_field "$meta" Id)"
  [ "$real_id" = "$id" ] || die "$name: metadata Id '$real_id' != manifest ID '$id'"
  want_ver="$(meta_field "$meta" Version)"

  installed_meta="$USER_PLASMOIDS/$id/metadata.json"
  if [ -f "$installed_meta" ]; then
    have_ver="$(meta_field "$installed_meta" Version)"
    if [ "$have_ver" = "$want_ver" ]; then
      echo "   up-to-date: $id $have_ver"
    else
      "$KPKG" --type Plasma/Applet --upgrade "$dir"
      echo "   upgraded: $id $have_ver -> $want_ver"
    fi
  else
    "$KPKG" --type Plasma/Applet --install "$dir"
    echo "   installed: $id $want_ver"
  fi
done < <(grep -vE '^[[:space:]]*(#|$)' "$MANIFEST")

echo
echo "plasmoid-apply: done — reload the shell with 'plasmoid-reload' to pick up changes"
#+END_SRC

* Plasmoid Diff
Reports manifest vs installed state without touching the network.

#+BEGIN_SRC bash :tangle linux/.local/bin/plasmoid-diff :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

MANIFEST="${PLASMOID_SSV:-$HOME/.plasmoid/manifest/applets.ssv}"
USER_PLASMOIDS="$HOME/.local/share/plasma/plasmoids"

[ -f "$MANIFEST" ] || { printf 'plasmoid-diff: missing manifest: %s\n' "$MANIFEST" >&2; exit 1; }

ver_of() {
  python3 -c 'import json,sys; print(json.load(open(sys.argv[1]))["KPlugin"]["Version"])' "$1" 2>/dev/null || printf 'unknown'
}

while read -r name id upstream tag commit pkgroot type; do
  installed_meta="$USER_PLASMOIDS/$id/metadata.json"
  if [ -f "$installed_meta" ]; then
    have="$(ver_of "$installed_meta")"
    want="${tag#v}"
    if [ "$have" = "$want" ]; then
      printf 'ok        %-50s %s\n' "$id" "$have"
    else
      printf 'outdated  %-50s installed=%s pinned=%s\n' "$id" "$have" "$want"
    fi
  else
    printf 'missing   %-50s pinned=%s\n' "$id" "$tag"
  fi
done < <(grep -vE '^[[:space:]]*(#|$)' "$MANIFEST")
#+END_SRC

* Plasmoid Health
#+BEGIN_SRC bash :tangle linux/.local/bin/plasmoid-health :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

KPKG="${KPKG:-kpackagetool6}"

echo "== tools"
for t in "$KPKG" plasmashell kreadconfig6 kwriteconfig6 qdbus6 git python3; do
  if command -v "$t" >/dev/null 2>&1; then
    printf '  ok       %s\n' "$t"
  else
    printf '  missing  %s\n' "$t"
  fi
done

if command -v plasmashell >/dev/null 2>&1; then
  echo "== plasmashell"
  printf '  %s\n' "$(plasmashell --version 2>/dev/null || echo 'version unavailable')"
fi

echo "== user-installed applets ($KPKG --list)"
"$KPKG" --type Plasma/Applet --list 2>/dev/null | sed 's/^/  /' || echo "  (none)"

echo "== manifest state"
"$(dirname "$0")/plasmoid-diff" | sed 's/^/  /'

echo "== kwin BorderlessMaximizedWindows"
val="$(kreadconfig6 --file kwinrc --group Windows --key BorderlessMaximizedWindows 2>/dev/null || true)"
printf '  %s\n' "${val:-unset (default: false)}"
#+END_SRC

* Plasmoid Reload
Explicit plasmashell replace — the only verb allowed to restart the shell.

#+BEGIN_SRC bash :tangle linux/.local/bin/plasmoid-reload :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

command -v plasmashell >/dev/null 2>&1 || { echo "plasmoid-reload: plasmashell not found" >&2; exit 1; }
[ -n "${DISPLAY:-}" ] || { echo "plasmoid-reload: no DISPLAY — run from the graphical session" >&2; exit 1; }

setsid -f env -u LD_PRELOAD plasmashell --replace </dev/null >/tmp/dotcortex-plasmashell.log 2>&1
echo "plasmoid-reload: plasmashell replaced (log: /tmp/dotcortex-plasmashell.log)"
#+END_SRC

* KWin Borderless Maximized Toggle
Opt-in: hide titlebars/borders only when windows are maximized, so the top
panel (window buttons + title + global menu) carries the chrome instead.
Backs up =kwinrc= before every write; reloads KWin via D-Bus reconfigure
(non-destructive — no session restart).

#+BEGIN_SRC bash :tangle linux/.local/bin/kwin-borderless-maximized :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

KREAD="${KREAD:-kreadconfig6}"
KWRITE="${KWRITE:-kwriteconfig6}"
QDBUS="${QDBUS:-qdbus6}"
KWINRC="$HOME/.config/kwinrc"

usage() {
  cat <<'EOF'
usage: kwin-borderless-maximized {status|enable|disable}

Controls [Windows] BorderlessMaximizedWindows in kwinrc.
enable  = maximized windows lose titlebar/borders
disable = normal titlebars everywhere
EOF
  exit "${1:-0}"
}

current() {
  "$KREAD" --file kwinrc --group Windows --key BorderlessMaximizedWindows 2>/dev/null || true
}

reconfigure() {
  if command -v "$QDBUS" >/dev/null 2>&1 && "$QDBUS" org.kde.KWin /KWin org.kde.KWin.reconfigure >/dev/null 2>&1; then
    echo "kwin reconfigured live"
  else
    echo "could not reach KWin over D-Bus — log out/in to apply"
  fi
}

setval() {
  command -v "$KWRITE" >/dev/null 2>&1 || { echo "missing $KWRITE" >&2; exit 1; }
  [ ! -f "$KWINRC" ] || cp "$KWINRC" "$KWINRC.backup-$(date +%Y%m%d-%H%M%S)"
  "$KWRITE" --file kwinrc --group Windows --key BorderlessMaximizedWindows "$1"
  echo "BorderlessMaximizedWindows=$1"
  reconfigure
}

case "${1:-status}" in
  status)
    val="$(current)"
    echo "BorderlessMaximizedWindows: ${val:-unset (default: false)}"
    ;;
  enable)  setval true ;;
  disable) setval false ;;
  -h|--help) usage ;;
  *) usage 1 ;;
esac
#+END_SRC
