* Services Loom
This loom tracks small service wrappers and helper scripts that live in
the =.local/bin= overlays and are deployed via stow.

* Early OOM
Scripts to apply a conservative, balanced OOM policy.
The X230 runs systemd; the T480s runs Devuan/sysvinit.
Both expose the same =MODE= knob (=balanced= | =proactive=).

** earlyoom-balanced — X230 (systemd drop-in)
Writes a systemd drop-in at =/etc/systemd/system/earlyoom.service.d/override.conf=
and restarts the service.  Avoids =/etc/default/= quoting issues by setting
=ExecStart= directly.

#+BEGIN_SRC bash :tangle x230/.local/bin/earlyoom-balanced :shebang "#!/usr/bin/env bash"
set -euo pipefail

MODE="${MODE:-balanced}"   # balanced or proactive

# Kill this stuff first if we MUST kill something:
#  - non-critical XFCE daemons
#  - media / background apps
#  - tools that autosave (Obsidian, Telegram)
PREFER='^(xfce4-screensaver|xfce4-clipman|xfdesktop|xfce4-notifyd|kdeconnectd|fluidsynth|jellyfin|[Oo]bsidian|Telegram|WebExtensions|qmmp)$'

# Protect these at all costs:
#  - browsers (no autosave guarantees)
#  - Codium / Code (no autosave guarantees in your current setup)
#  - file managers + GVFS plumbing
#  - file transfer / backup tools (rsync, etc.)
#  - xfce4-session so we don't drop the whole session
AVOID='^(firedragon|floorp|zen|ungoogled-chromium|firefox|codium|code|thunar|gvfs.*|rsync|scp|sftp|rclone|syncthing|borg|restic|xfce4-session|barrier|barriers|signal-desktop|Betterbird|betterbird|thunderbird)$'

if [[ "$MODE" = "balanced" ]]; then
  M_TERM=5;  M_KILL=3;  S_TERM=15; S_KILL=10
elif [[ "$MODE" = "proactive" ]]; then
  M_TERM=7;  M_KILL=5;  S_TERM=20; S_KILL=10
else
  echo "Unknown MODE='$MODE' (use balanced or proactive)" >&2
  exit 1
fi

DROPIN_DIR="/etc/systemd/system/earlyoom.service.d"
DROPIN="${DROPIN_DIR}/override.conf"

sudo install -d -m 0755 "$DROPIN_DIR"

# Write drop-in atomically
TMP="$(mktemp)"
cat >"$TMP" <<EOF
[Service]
EnvironmentFile=
ExecStart=
ExecStart=/usr/bin/earlyoom -r 3600 -m ${M_TERM},${M_KILL} -s ${S_TERM},${S_KILL} \
  --prefer ${PREFER} \
  --avoid  ${AVOID}
OOMScoreAdjust=-100
Nice=-20
EOF
sudo install -m 0644 "$TMP" "$DROPIN"
rm -f "$TMP"

sudo systemctl daemon-reload
sudo systemctl restart earlyoom

echo "Applied Early OOM policy: mem ${M_TERM}%/${M_KILL}%, swap ${S_TERM}%/${S_KILL}%"
echo "Prefer: ${PREFER}"
echo "Avoid:  ${AVOID}"

# Show the running command and thresholds
if pidof earlyoom >/dev/null; then
  tr '\0' ' ' </proc/$(pidof earlyoom)/cmdline; echo
  journalctl -u earlyoom -b --no-pager | tail -n 20
else
  echo "earlyoom is not running" >&2
  exit 2
fi
#+END_SRC

** Jellyfin toggle (ThinkPad X230 server)
This script lives in the =x230= overlay, it is specific to the current
X230 that is running Jellyfin as a systemd service.
#+BEGIN_SRC bash :tangle x230/.local/bin/jellyfin-toggle :shebang "#!/bin/sh"
# jellyfin-toggle
# Toggle Jellyfin Media Server on and off via systemd.
# Requires sudoers entry:
#   metsatron ALL=(root) NOPASSWD: /usr/bin/systemctl

if systemctl is-active --quiet jellyfin; then
  exec sudo -n /usr/bin/systemctl stop jellyfin
else
  exec sudo -n /usr/bin/systemctl start jellyfin
fi
#+END_SRC

After first tangle:

#+BEGIN_SRC bash
chmod +x ~/DotCortex/x230/.local/bin/jellyfin-toggle
#+END_SRC

Panel launcher command:

sh -c 'if systemctl is-active --quiet jellyfin; then sudo -n /usr/bin/systemctl stop jellyfin; else sudo -n /usr/bin/systemctl start jellyfin; fi'
#+END_SRC

* Swapfetch
#+NAME: swapfetch
#+BEGIN_SRC bash :tangle linux/.local/bin/swapfetch :shebang "#!/usr/bin/env bash"
set -euo pipefail

# Ensure swapon/free are reachable (Devuan keeps them in /sbin)
[[ ":$PATH:" != *":/sbin:"* ]] && export PATH="$PATH:/sbin"

# Nerd Font icons
ICON_ZRAM="󰓅"   # ZRAM / compression
ICON_SWAP="󰓾"   # swap/disk
ICON_MEM=""    # RAM

# Colors (always on; tune shades in style.org if you like)
C_HDR=$'\e[1;36m'     # cyan bold (headers)
C_LABEL=$'\e[1;33m'   # yellow bold (labels)
C_VAL=$'\e[1;37m'     # white bold  (values)
C_DIM=$'\e[2;37m'     # dim white   (background / secondary)
C_OK=$'\e[1;32m'      # green
C_WARM=$'\e[1;33m'    # yellow
C_HOT=$'\e[1;31m'     # red
C_RESET=$'\e[0m'

C_BOLD=$'\033[1m'

# Bar background: “12th” color – you already settled on #454545
C_BAR_BG=$'\e[38;5;11m'

# Logo color: *this* is the one that makes the banner colorful.
# Use header cyan by default; change to C_LABEL if you want gold.
: "${C_LOGO:=$C_HDR}"

# --- optional watch mode: swapfetch --watch [seconds] ---
if [[ "${SWAPFETCH_SKIP_WATCH:-0}" != "1" ]]; then
  case "${1-}" in
    -w|--watch)
      shift
      interval="${1-5}"
      # default to 5 if not a simple integer
      [[ "$interval" =~ ^[0-9]+$ ]] || interval=5

      while :; do
        clear
        printf "swapfetch (every %ss) - %s\n\n" "$interval" "$(date '+%H:%M:%S')"
        SWAPFETCH_SKIP_WATCH=1 "$0"
        sleep "$interval"
      done
      exit 0
      ;;
  esac
fi

print_header() {
  printf '%b%s%b\n' "$C_HDR" "$1" "$C_RESET"
}

gauge_color_for() {
  local pct=$1
  if   (( pct >= 90 )); then printf '%s' "$C_HOT"
  elif (( pct >= 70 )); then printf '%s' "$C_WARM"
  else                       printf '%s' "$C_OK"
  fi
}

gauge_bar() {
  local pct=$1
  local width=${2:-24}
  local col="${3:-$C_OK}"      # filled color
  local bg="${4:-$C_BAR_BG}"   # empty + right cap color

  ((pct < 0))   && pct=0
  ((pct > 100)) && pct=100

  local filled=$(( (pct * width + 50) / 100 ))
  ((filled > width)) && filled=$width

  local filled_str="" empty_str=""
  local i
  for ((i=1; i<=filled; i++)); do
    filled_str+="█"
  done
  for ((; i<=width; i++)); do
    empty_str+="█"
  done

  # left cap + filled in severity color; empty + right cap in bar-bg color
  printf '%b' "${col}${filled_str}${bg}${empty_str}${C_RESET}"
}

# ---------- ZRAM (zram0) ----------
if [ -r /sys/block/zram0/mm_stat ]; then
  print_header "$ICON_ZRAM  ZRAM (zram0)"

  eval "$(
    awk '{
      og = $1/1024/1024/1024;   # orig_data
      cg = $2/1024/1024/1024;   # compr_data
      mg = $3/1024/1024/1024;   # mem_used_total
      lg = $4/1024/1024/1024;   # mem_limit
      xg = $5/1024/1024/1024;   # max_used
      ratio = ($2 > 0 ? $1 / $2 : 0);
      printf "Z_ORIG=%.2f\n",  og;
      printf "Z_COMPR=%.2f\n", cg;
      printf "Z_MEM=%.2f\n",   mg;
      printf "Z_LIMIT=%.2f\n", lg;
      printf "Z_MAX=%.2f\n",   xg;
      printf "Z_RATIO=%.2f\n", ratio;
    }' /sys/block/zram0/mm_stat
  )"

  # Big ASCII logo (no awk, no escaping hell)
  mapfile -t ZLOGO <<'EOF'
 ____ _ __ __ _ _ __ _ _     ___ __      __ _   _ __
|_  /| '__/ _` | '_ ` _ \ _ / __|\ \ /\ / / _` | '_ \
 / / | | | (_| | | | | | |_|\__ \ \ V  V / (_| | |_) |
/___||_|  \__,_|_| |_| |_|  |___/  \_/\_/ \__,_| .__/
                                                  |_|
EOF

  printf "  %sorig_data%s   : %s%.2f GiB%s  %s%s%s\n" \
    "$C_LABEL" "$C_RESET" "$C_VAL" "$Z_ORIG"  "$C_RESET" "$C_LOGO" "${ZLOGO[0]}" "$C_RESET"

  printf "  %scompr_data%s  : %s%.2f GiB%s  %s%s%s\n" \
    "$C_LABEL" "$C_RESET" "$C_VAL" "$Z_COMPR" "$C_RESET" "$C_LOGO" "${ZLOGO[1]}" "$C_RESET"

  printf "  %smem_used%s    : %s%.2f GiB%s  %s%s%s\n" \
    "$C_LABEL" "$C_RESET" "$C_VAL" "$Z_MEM"   "$C_RESET" "$C_LOGO" "${ZLOGO[2]}" "$C_RESET"

  printf "  %smax_used%s    : %s%.2f GiB%s  %s%s%s\n" \
    "$C_LABEL" "$C_RESET" "$C_VAL" "$Z_MAX"   "$C_RESET" "$C_LOGO" "${ZLOGO[3]}" "$C_RESET"

  if [[ "${Z_LIMIT%.*}" -gt 0 ]] 2>/dev/null; then
    printf "  %smem_limit%s   : %s%.2f GiB%s\n" \
      "$C_LABEL" "$C_RESET" "$C_VAL" "$Z_LIMIT" "$C_RESET"
  fi

  printf "  %scompression%s : %s%.2fx%s  %s%s%s\n\n" \
    "$C_LABEL" "$C_RESET" "$C_VAL" "$Z_RATIO" "$C_RESET" \
    "$C_LOGO" "${ZLOGO[4]}" "$C_RESET"
else
  # No ZRAM — show swap-only banner
  mapfile -t SLOGO <<'EOF'
                            __     _       _
 _____      ____ _ _ __   / _| ___| |_ ___| |__
/ __\ \ /\ / / _` | '_ \ | |_ / _ \ __/ __| '_ \
\__ \\ V  V / (_| | |_) ||  _|  __/ || (__| | | |
|___/ \_/\_/ \__,_| .__/ |_|  \___|\__\___|_| |_|
                  |_|
EOF
  for line in "${SLOGO[@]}"; do
    printf "  %s%s%s\n" "$C_LOGO" "$line" "$C_RESET"
  done
  echo
fi

# --- Swap devices ---------------------------------------------------------
print_header "$ICON_SWAP  Swap devices"

swapon --show=NAME,SIZE,USED,PRIO --bytes 2>/dev/null | \
  awk 'NR==1 {next} {print $1, $2, $3, $4}' | \
  while read -r name size used prio; do
    [ -z "${name:-}" ] && continue
    [ "${size:-0}" -eq 0 ] && continue

    # percentage (rounded)
    pct=$(( (used * 100 + size/2) / size ))
    ((pct < 0))   && pct=0
    ((pct > 100)) && pct=100

    col="$(gauge_color_for "$pct")"
    bar="$(gauge_bar "$pct" 24 "$col")"   # bg defaults to C_BAR_BG now

    # human-readable GiB
    read -r used_gib size_gib <<EOF
$(awk -v u="$used" -v s="$size" 'BEGIN {printf "%.1f %.1f\n", u/1024/1024/1024, s/1024/1024/1024}')
EOF

    printf "  %s%-16s%s %s %s%5.1f/%-5.1f GiB%s (%3d%%) prio %d\n" \
      "$C_LABEL" "$name" "$C_RESET" \
      "$bar" \
      "$C_VAL" "$used_gib" "$size_gib" "$C_RESET" \
      "$pct" "$prio"
  done

echo

# --- Memory summary -------------------------------------------------------
print_header "$ICON_MEM  Memory"

# Raw bytes for math
eval "$(free -b | awk '
/^Mem:/  { printf "MEM_TOTAL=%s MEM_USED=%s MEM_FREE=%s MEM_AVAIL=%s\n", $2,$3,$4,$7 }
/^Swap:/ { printf "SWAP_TOTAL=%s SWAP_USED=%s SWAP_FREE=%s\n", $2,$3,$4 }
')"

mem_pct=0
swap_pct=0
if [[ ${MEM_TOTAL:-0} -gt 0 ]]; then
  mem_pct=$(( (MEM_USED * 100 + MEM_TOTAL/2) / MEM_TOTAL ))
fi
if [[ ${SWAP_TOTAL:-0} -gt 0 ]]; then
  swap_pct=$(( (SWAP_USED * 100 + SWAP_TOTAL/2) / SWAP_TOTAL ))
fi

mem_col=$(gauge_color_for "$mem_pct")
swap_col=$(gauge_color_for "$swap_pct")
mem_bar=$(gauge_bar "$mem_pct" 30 "$mem_col")
swap_bar=$(gauge_bar "$swap_pct" 30 "$swap_col")

# Pretty numbers from free -h
free -h | awk -v label="$C_LABEL" -v val="$C_VAL" -v reset="$C_RESET" '
NR==2 {
  printf "  %sMem %s total=%s%s%s used=%s%s%s free=%s%s%s avail=%s%s%s\n",
         label, reset,
         val, $2, reset,
         val, $3, reset,
         val, $4, reset,
         val, $7, reset;
}
NR==3 {
  printf "  %sSwap%s total=%s%s%s used=%s%s%s free=%s%s%s\n",
         label, reset,
         val, $2, reset,
         val, $3, reset,
         val, $4, reset;
}
'

printf "  %sMem use%s   %s %s%3d%%%s\n" \
  "$C_LABEL" "$C_RESET" "$mem_bar" "$C_VAL" "$mem_pct" "$C_RESET"
printf "  %sSwap use%s  %s %s%3d%%%s\n" \
  "$C_LABEL" "$C_RESET" "$swap_bar" "$C_VAL" "$swap_pct" "$C_RESET"

#+END_SRC

* Swap tools
** swaptop – top by VmSwap (btop-ish)
#+BEGIN_SRC bash :tangle linux/.local/bin/swaptop :shebang "#!/usr/bin/env bash"
set -euo pipefail

interval="${1:-5}"

# === Colors – keep in sync with swapfetch ===

# Nerd Font style: same palette as swapfetch
C_HDR=$'\e[1;36m'     # cyan bold (headers)
C_LABEL=$'\e[1;33m'   # yellow bold (labels)
C_VAL=$'\e[1;37m'     # white bold  (values)
C_DIM=$'\e[2;37m'     # dim white   (background / secondary)
C_OK=$'\e[1;32m'      # green
C_WARM=$'\e[1;33m'    # yellow
C_HOT=$'\e[1;31m'     # red
C_RESET=$'\e[0m'
C_BOLD=$'\033[1m'

# Bar background: “12th” color – aligned with swapfetch’s style
C_BAR_BG=$'\e[38;5;11m'

# Logo/header color
: "${C_LOGO:=$C_HDR}"

gauge_color_for() {
  local pct=$1
  if   (( pct >= 90 )); then printf '%s' "$C_HOT"
  elif (( pct >= 70 )); then printf '%s' "$C_WARM"
  else                       printf '%s' "$C_OK"
  fi
}

gauge_bar() {
  local pct=$1
  local width=${2:-24}
  local col="${3:-$C_OK}"      # filled color
  local bg="${4:-$C_BAR_BG}"   # unfilled “background”

  ((pct < 0))   && pct=0
  ((pct > 100)) && pct=100

  local filled=$(( (pct * width + 50) / 100 ))
  ((filled > width)) && filled=$width

  local filled_str="" empty_str=""
  local i
  for ((i=1; i<=filled; i++)); do
    filled_str+="█"
  done
  for ((; i<=width; i++)); do
    empty_str+="█"
  done

  printf '%b' "${col}${filled_str}${bg}${empty_str}${C_RESET}"
}

# Use tput if available so we can move the cursor without nuking scrollback
HAS_TPUT=0
if command -v tput >/dev/null 2>&1; then
  HAS_TPUT=1
  tput civis 2>/dev/null || true
  trap 'tput cnorm 2>/dev/null || true; exit' INT TERM
fi

# How many process rows we printed last time
LAST_ROWS=0

print_row() {
  local idx="$1"
  local line pid kb pct col bar mib cmdline

  line=${_rows[idx]}
  set -- $line
  pid=$1
  kb=$2

  [[ -z "$pid" || -z "$kb" ]] && return 0

  # percent of total swap
  pct=$(( (kb * 100 + local_total_swap_kb/2) / local_total_swap_kb ))
  ((pct < 0))   && pct=0
  ((pct > 100)) && pct=100

  col="$(gauge_color_for "$pct")"
  bar="$(gauge_bar "$pct" 24 "$col" "$C_BAR_BG")"

  # human readable MiB
  mib=$(awk -v k="$kb" 'BEGIN {printf "%.1f", k/1024}')

  # command line (shortened single-line)
  cmdline=$(ps -p "$pid" -o comm=,args= 2>/dev/null || true)
  cmdline=${cmdline# }
  cmdline=${cmdline//$'\n'/ }
  cmdline=${cmdline:0:60}

  printf "%7s %7s MiB %3d%% %s %s\n" \
    "$pid" "$mib" "$pct" "$bar" "$cmdline"
}

while :; do
  # Terminal height (only if tput exists)
  if (( HAS_TPUT )); then
    term_lines=$(tput lines 2>/dev/null || echo 24)
  else
    term_lines=24
  fi

  # Total swap in KiB
  local_total_swap_kb=$(awk '/^SwapTotal:/ {print $2}' /proc/meminfo)
  (( local_total_swap_kb == 0 )) && local_total_swap_kb=1

  # Collect processes sorted by VmSwap DESC
  mapfile -t _rows < <(
    awk '/^VmSwap:/ {print FILENAME, $2}' /proc/[0-9]*/status 2>/dev/null \
      | sed 's#/proc/\([0-9]*\)/status#\1#' \
      | sort -k2,2nr
  )

  rows_len=${#_rows[@]}
  if (( rows_len == 0 )); then
    current_rows=0
  else
    # 1 title + 1 header + 1 separator + 1 breathing line = 4
    max_rows=$((term_lines - 4))
    ((max_rows < 1)) && max_rows=1
    current_rows=$rows_len
    ((current_rows > max_rows)) && current_rows=$max_rows
  fi

  if (( HAS_TPUT )); then
    # --- Header lines (each one: move, clear line, print) ---

    tput cup 0 0 2>/dev/null
    printf '\033[K'
    printf '%b%s%b  (interval %ss)\n' \
      "$C_HDR" "swaptop – top VmSwap users" "$C_RESET" "$interval"

    tput cup 1 0 2>/dev/null
    printf '\033[K'
    printf '%b%-7s %-8s %-8s %-5s %-s%b\n' \
      "$C_LABEL" "PID" "SWAP" "PCT" "BAR" "COMMAND" "$C_RESET"

    tput cup 2 0 2>/dev/null
    printf '\033[K'
    printf '%s\n' "--------------------------------------------------------------------------"

    if (( current_rows == 0 )); then
      # No rows: print a single status line
      tput cup 3 0 2>/dev/null
      printf '\033[K'
      printf "   (no processes with VmSwap)\n"
    else
      # Print each row at a fixed line, clearing only that line
      for ((idx=0; idx<current_rows; idx++)); do
        line_no=$((3 + idx))
        tput cup "$line_no" 0 2>/dev/null
        printf '\033[K'
        print_row "$idx"
      done
    fi

    # If we had more rows last time, blank the leftovers
    if (( LAST_ROWS > current_rows )); then
      for ((idx=current_rows; idx<LAST_ROWS; idx++)); do
        line_no=$((3 + idx))
        tput cup "$line_no" 0 2>/dev/null
        printf '\033[K'
      done
    fi

  else
    # Fallback: dumb terminal – just print a fresh block each tick
    printf '%b%s%b  (interval %ss)\n' \
      "$C_HDR" "swaptop – top VmSwap users" "$C_RESET" "$interval"
    printf '%b%-7s %-8s %-8s %-5s %-s%b\n' \
      "$C_LABEL" "PID" "SWAP" "PCT" "BAR" "COMMAND" "$C_RESET"
    printf '%s\n' "--------------------------------------------------------------------------"

    if (( current_rows == 0 )); then
      printf "   (no processes with VmSwap)\n"
    else
      for ((idx=0; idx<current_rows; idx++)); do
        print_row "$idx"
      done
    fi
  fi

  LAST_ROWS=$current_rows
  sleep "$interval"
done
#+END_SRC

** xfce4-refresh – recycle XFCE desktop daemons via host-wrap
#+BEGIN_SRC bash :tangle x230/.local/bin/xfce4-refresh :shebang "#!/usr/bin/env bash"
set -euo pipefail

log() {
  printf '[xfce4-refresh] %s\n' "$*"
}

kill_if_running() {
  local name="$1"
  if pgrep -x "$name" >/dev/null 2>&1; then
    log "Killing $name…"
    pkill -x "$name" || true
  fi
}

start_with_host_wrap_if_missing() {
  local name="$1"
  if ! pgrep -x "$name" >/dev/null 2>&1; then
    if command -v host-wrap >/dev/null 2>&1; then
      log "Starting $name via host-wrap…"
      nohup host-wrap "$name" >/dev/null 2>&1 &
    else
      log "host-wrap not found; starting $name directly…"
      nohup "$name" >/dev/null 2>&1 &
    fi
  fi
}

log "Refreshing XFCE desktop daemons"

# 1) Kill screensaver and leave it dead - handle truncated comm name
if pgrep -x xfce4-screensav >/dev/null 2>&1 || pgrep -f /usr/bin/xfce4-screensaver >/dev/null 2>&1; then
  log "Killing xfce4-screensaver..."
  pkill -x xfce4-screensav || pkill -f /usr/bin/xfce4-screensaver || true
fi

# 2) Clipman: kill + restart (via host-wrap)
kill_if_running xfce4-clipman
start_with_host_wrap_if_missing xfce4-clipman

# 3) Desktop: kill + restart (via host-wrap)
kill_if_running xfdesktop
start_with_host_wrap_if_missing xfdesktop

# 4) Notifications: kill + restart (via host-wrap)
if command -v xfce4-notifyd >/dev/null 2>&1; then
  kill_if_running xfce4-notifyd
  start_with_host_wrap_if_missing xfce4-notifyd
fi

# 5) xfrun4 - kill if it is hanging around - do not restart
kill_if_running xfrun4

# 6) Panel + settings: delegate to canonical healer
if command -v xfce4-heal >/dev/null 2>&1; then
  log "Delegating panel + settings restart to xfce4-heal…"
  xfce4-heal
elif command -v xfce4-panel >/dev/null 2>&1; then
  # Fallback path if xfce4-heal is missing for some reason
  if pgrep -x xfce4-panel >/dev/null 2>&1; then
    log "xfce4-heal missing; trying xfce4-panel -r fallback…"
    if ! xfce4-panel -r >/dev/null 2>&1; then
      log "xfce4-panel -r failed, trying kill + restart via host-wrap/direct…"
      kill_if_running xfce4-panel
      start_with_host_wrap_if_missing xfce4-panel
    fi
  else
    log "xfce4-heal missing; xfce4-panel not running, starting fresh via host-wrap…"
    start_with_host_wrap_if_missing xfce4-panel
  fi
fi

log "Done."

** xfce4-refresh-guardian – early panel refresh on high swap + panel revive
#+BEGIN_SRC bash :tangle x230/.local/bin/xfce4-refresh-guardian :shebang "#!/usr/bin/env bash"
#!/usr/bin/env bash
# xfce4-refresh-guardian – early refresh trigger based on swap
# and “is the panel actually visible?”

set -euo pipefail

SWAP_LIMIT="${SWAP_LIMIT:-80}"

DO_RESTART=0
if [[ "${1:-}" == "--restart" || "${1:-}" == "-r" ]]; then
  DO_RESTART=1
fi

# --- helper: is the xfce4-panel actually alive (proc + window)? ---
panel_is_alive=0

if pgrep -x xfce4-panel >/dev/null 2>&1; then
  # We have a process – now check for a real panel window if wmctrl exists.
  if command -v wmctrl >/dev/null 2>&1; then
    if wmctrl -lx 2>/dev/null | grep -qi "xfce4-panel.Xfce4-panel"; then
      panel_is_alive=1
    else
      printf '[xfce4-refresh-guardian] xfce4-panel process found, but no panel window.\n'
    fi
  else
    # No wmctrl? fall back to trusting the process
    panel_is_alive=1
  fi
else
  printf '[xfce4-refresh-guardian] No xfce4-panel process.\n'
fi

# If the panel is missing (no process OR no window), healing is top priority.
if (( panel_is_alive == 0 )); then
  printf '[xfce4-refresh-guardian] xfce4-panel is NOT alive, attempting heal via xfce4-heal…\n'
  if command -v xfce4-heal >/dev/null 2>&1; then
    xfce4-heal
    # Give it a moment, then re-check both process + window.
    sleep 1

    panel_is_alive_after=0
    if pgrep -x xfce4-panel >/dev/null 2>&1; then
      if command -v wmctrl >/dev/null 2>&1; then
        if wmctrl -lx 2>/dev/null | grep -qi "xfce4-panel.Xfce4-panel"; then
          panel_is_alive_after=1
        fi
      else
        panel_is_alive_after=1
      fi
    fi

    if (( panel_is_alive_after == 1 )); then
      printf '[xfce4-refresh-guardian] Panel is now running (process + window detected).\n'
    else
      printf '[xfce4-refresh-guardian] Heal attempted, but panel still not detected.\n'
    fi
  else
    printf '[xfce4-refresh-guardian] xfce4-heal not found; cannot restore panel.\n'
  fi

  # Panel health is a higher priority than swap tuning – stop here.
  exit 0
fi

# --- at this point: panel_is_alive == 1, proceed with swap threshold logic ---

# Read swap from /proc/meminfo (KiB)
swap_total_kb=$(awk '/^SwapTotal:/ {print $2}' /proc/meminfo)
swap_free_kb=$(awk '/^SwapFree:/  {print $2}' /proc/meminfo)

if (( swap_total_kb == 0 )); then
  SWAP_PCT=0
else
  swap_used_kb=$((swap_total_kb - swap_free_kb))
  SWAP_PCT=$(( (swap_used_kb * 100 + swap_total_kb/2) / swap_total_kb ))
fi

printf '[xfce4-refresh-guardian] THRESHOLD: SWAP=%d%% (limit %d%%)\n' \
  "$SWAP_PCT" "$SWAP_LIMIT"

if (( SWAP_PCT >= SWAP_LIMIT )); then
  if (( DO_RESTART )); then
    if command -v xfce4-refresh >/dev/null 2>&1; then
      printf '[xfce4-refresh-guardian] Triggering xfce4-refresh (early recycle)…\n'
      xfce4-refresh
    else
      printf '[xfce4-refresh-guardian] xfce4-refresh not found, nothing to do.\n'
    fi
  else
    printf '[xfce4-refresh-guardian] Would run xfce4-refresh (dry run).\n'
  fi
else
  printf '[xfce4-refresh-guardian] Below threshold, nothing to do.\n'
fi
#+END_SRC

    log "xfdesktop not running, starting…"
    if command -v host-wrap >/dev/null 2>&1; then
      nohup host-wrap xfdesktop >/dev/null 2>&1 &

      nohup xfdesktop >/dev/null 2>&1 &
    fi

  if ! command -v wmctrl >/dev/null 2>&1; then
    log "wmctrl not found, cannot toggle show desktop"

  local state
  state=$(xprop -root _NET_SHOWING_DESKTOP 2>/dev/null | awk '{print $3}' || echo 0)

  if [[ "$state" = "0" ]]; then
    log "Enabling show desktop"
    wmctrl -k on
  else
    log "Disabling show desktop"
    wmctrl -k off
  fi
}

def proc_metrics(pid_str):
    m = {"ppid": None, "pgid": None, "sid": None, "proc_status": None,
         "cwd": None, "cpu_ticks": None, "rss_kb": None,
         "io_read_bytes": None, "io_write_bytes": None}

        pass
    try:
        raw   = pathlib.Path(f"/proc/{pid_str}/stat").read_text()
        after = raw[raw.rfind(")") + 2:]
        f     = after.split()
        m["proc_status"] = f[0]
        m["ppid"]        = int(f[1])
        m["pgid"]        = int(f[2])
        m["sid"]         = int(f[3])

    except Exception:
        pass
    try:
        for line in pathlib.Path(f"/proc/{pid_str}/io").read_text().splitlines():

            if k == "read_bytes":

_CLAUDE_SESSIONS_DIR  = home / ".claude/sessions"
_CODEX_SESSION_FD_RE  = re.compile(r"/\.codex/sessions/.+/([^/]+)\.jsonl$")
_OPENCODE_LOG_FD_RE   = re.compile(r"/\.local/share/opencode/log/[^/]+\.log$")

def _fd_entries(pid_str):
    """Yield readlink targets for /proc/<pid>/fd, silently skipping permission errors."""
    try:
        for entry in pathlib.Path(f"/proc/{pid_str}/fd").iterdir():
            try:
                yield os.readlink(str(entry))
            except Exception:
                pass
    except Exception:
        pass

    """Try non-cmdline session ID extraction. Returns (sid, source, fd_path, fd_kind)."""
    # claude-family: read ~/.claude/sessions/<pid>.json written by claude at startup

# ── PPID ancestry helpers for tmux linkage ────────────────────────────────────
def ppid_read(pid_str):
    """Return parent PID string from /proc/<pid>/stat, or None."""
    try:
        raw   = pathlib.Path(f"/proc/{pid_str}/stat").read_text()
        after = raw[raw.rfind(")") + 2:]
        return after.split()[1]
    except Exception:
        return None

def ppid_chain_walk(pid_str, pane_pid_set, max_depth=32):
    """Walk parent chain; return (matched_pane_pid_str, chain) or (None, chain)."""
    chain = []
    cur   = ppid_read(pid_str)
    for _ in range(max_depth):
        if cur is None or cur in ("0", "1"):
            break
        chain.append(cur)
        if cur in pane_pid_set:
            return cur, chain
        cur = ppid_read(cur)
    return None, chain

# ── codex model probe (read logs_2.sqlite) ───────────────────────────────────
# NOTE: logs_2.sqlite can be multi-GB / tens of millions of rows. A per-PID
# query (WHERE process_uuid LIKE 'pid:<pid>:%') forces SQLite to walk the
# ts index across the whole table looking for LIKE matches, which can take
# minutes per PID with several live codex processes. Instead we run ONE
# bounded query per collection cycle (ts >= now - horizon, LIMIT N), cache
# the most-recent match per pid in a module-level dict, and turn
# _codex_model_probe() into a plain dict lookup. The ts lower bound lets
# the descending index scan terminate at the horizon regardless of how
# many rows match feedback_log_body LIKE '%model=%'.
_CODEX_LOG_DB       = home / ".codex/logs_2.sqlite"
_CODEX_MODEL_RE     = re.compile(r'\bmodel=([A-Za-z0-9._:/-]+)')
_CODEX_PID_RE       = re.compile(r'^pid:(\d+):')
_CODEX_PROBE_HORIZON_S = 48 * 3600
_CODEX_PROBE_LIMIT     = 2000
_codex_model_by_pid    = None  # populated on first probe of this cycle

def _codex_model_probe_scan():
    """One bounded pass over logs_2.sqlite, cached per collection cycle.

    Returns {pid_str: extra_dict} built from the most-recent (ts DESC)
    matching row per pid within the horizon window."""
    result = {}
    if not _CODEX_LOG_DB.exists():
        return result
    try:
        conn = sqlite3.connect(f"file:{_CODEX_LOG_DB}?mode=ro", uri=True, timeout=5)
        try:
            conn.execute("PRAGMA busy_timeout=2000")
            horizon = time.time() - _CODEX_PROBE_HORIZON_S
            rows = conn.execute(
                "SELECT process_uuid, feedback_log_body, ts FROM logs "
                "WHERE ts >= ? AND feedback_log_body LIKE '%model=%' "
                "ORDER BY ts DESC, ts_nanos DESC LIMIT ?",
                (horizon, _CODEX_PROBE_LIMIT)
            ).fetchall()
        finally:
            conn.close()
        for (process_uuid, body, ts) in rows:
            if not body or not process_uuid:
                continue
            pm = _CODEX_PID_RE.match(process_uuid)
            if not pm:
                continue
            pid_str = pm.group(1)
            if pid_str in result:
                continue  # already have the most-recent (rows are ts DESC)
            mm = _CODEX_MODEL_RE.search(body)
            if mm:
                result[pid_str] = {
                    "provider":              "openai",
                    "provider_source":       "codex_log",
                    "model":                 mm.group(1),
                    "model_source":          "codex_log",
                    "model_confidence":      "high",
                    "context_size":          None,
                    "context_size_source":   "unknown",
                    "context_window":        None,
                    "context_window_source": "unknown",
                    "context_percent":       None,
                    "metadata_ts":           float(ts) if ts else None,
                }
    except Exception:
        return {}
    return result

def _codex_model_probe(pid_str):
    """Extract most-recent model for this PID from Codex logs_2.sqlite.

    Backed by a single bounded scan per collection cycle (see
    _codex_model_probe_scan) instead of a per-PID query."""
    global _codex_model_by_pid
    if _codex_model_by_pid is None:
        _codex_model_by_pid = _codex_model_probe_scan()
    return _codex_model_by_pid.get(str(pid_str), {})

TAIL_SCAN_MAX_BYTES = 65536      # bounded tail read
EXACT_COUNT_MAX_BYTES = 300_000  # below this, count lines exactly; above, estimate


def iso_from_epoch(ts):
    try:
        return (datetime.fromtimestamp(float(ts), tz=timezone.utc)
                 .isoformat(timespec="seconds").replace("+00:00", "Z"))
    except Exception:
        return None


def iso_from_mtime(path):
    try:
        return iso_from_epoch(path.stat().st_mtime)
    except Exception:
        return None


def read_head(path, max_bytes=HEAD_SCAN_MAX_BYTES):
    """Read a bounded prefix; never seeks past max_bytes."""
    try:
        with open(path, "rb") as fh:
            data = fh.read(max_bytes)
        return data.decode(errors="replace")
    except Exception as exc:
        warnings_global.append(f"{path.name}: head read failed: {exc}")
        return ""


def read_tail(path, max_bytes=TAIL_SCAN_MAX_BYTES):
    """Read a bounded suffix; drops a possibly-truncated first partial line."""
    try:
        size = path.stat().st_size
        with open(path, "rb") as fh:
            fh.seek(max(0, size - max_bytes))
            data = fh.read()
        text = data.decode(errors="replace")
        lines = text.split("\n")
        if size > max_bytes and lines:
            lines = lines[1:]
        return lines
    except Exception as exc:
        warnings_global.append(f"{path.name}: tail read failed: {exc}")
        return []


def iter_json_lines(lines):
    """Yield parsed dicts, skipping unparseable or empty lines silently."""
    for ln in lines:
        ln = ln.strip()
        if not ln:
            continue
        try:
            obj = json.loads(ln)
        except Exception:
            continue
        if isinstance(obj, dict):
            yield obj


def estimate_message_count(path, tail_lines):
    """Exact line count for small files; tail-density estimate for large ones."""
    try:
        size = path.stat().st_size
    except Exception:
        return None, "unknown"
    if size <= EXACT_COUNT_MAX_BYTES:
        try:
            with open(path, "rb") as fh:
                count = sum(1 for _ in fh)
            return count, "exact"
        except Exception:
            return None, "unknown"
    sample = [l for l in tail_lines if l.strip()]
    if not sample:
        return None, "unknown"
    tail_bytes = sum(len(l.encode(errors="replace")) + 1 for l in sample)
    if tail_bytes <= 0:
        return None, "unknown"
    avg_len = tail_bytes / len(sample)
    if avg_len <= 0:
        return None, "unknown"
    return round(size / avg_len), "estimated_from_tail_sample"

        return []

            continue
        try:
            files.append((f.stat().st_mtime, f))
        except OSError:
            continue
    files.sort(key=lambda t: t[0], reverse=True)
    return [f for _, f in files[:n]]

def fmt_bytes(n):
    if not n:
        return "—"
    for unit, thr in [("G", 1 << 30), ("M", 1 << 20), ("K", 1 << 10)]:
        if n >= thr:
            return f"{n/thr:.0f}{unit}"
    return f"{n}B"


def fmt_kb(kb):
    return fmt_bytes((kb or 0) * 1024)


def short_sid(s):
    if not s:
        return "—"
    return s[-12:] if len(s) > 12 else s


def short_machine(host):
    if not host:
        return "—"
    h = str(host)
    hl = h.lower()
    if "t480s" in hl: return "T480s"
    if "x230"  in hl: return "X230"
    if "t430"  in hl: return "T430"
    if "vultr" in hl: return "Vultr"
    if "vps"   in hl: return "VPS"
    if h.startswith("ThinkPad-"): return h[9:]
    return h[:8]


def short_model(m, width=20):
    if not m:
        return "—"
    if len(m) <= width:
        return m
    return ".." + m[-(width - 2):]


def cwd_name(path):
    if not path:
        return "—"
    home = os.path.expanduser("~")
    if path.startswith(home):
        path = "~" + path[len(home):]
    return path.rstrip("/").split("/")[-1] or path


def tmux_label(a):
    sn = a.get("tmux_session_name") or ""
    wn = a.get("tmux_window_name") or ""
    pi = a.get("tmux_pane_id") or ""
    if sn and wn and pi:
        return f"{sn}/{wn}/{pi}"
    if sn and wn:
        return f"{sn}/{wn}"
    if sn:
        return sn
    return "—"


def fmt_date(s):
    return (s or "")[:10] or "—"


def age_str(secs):
    if secs is None:
        return "?"
    if secs < 60:
        return f"{int(secs)}s"
    if secs < 3600:
        return f"{int(secs // 60)}m"
    if secs < 86400:
        return f"{int(secs // 3600)}h"
    return f"{int(secs // 86400)}d"


def freshness_badge(secs):
    if secs is None:
        return "OLD"
    if secs < 60:
        return "FRESH"
    if secs < 300:
        return "WARM"
    if secs < 3600:
        return "STALE"
    return "OLD"


def state_age_seconds(state, now_utc):
    try:
        g = (state.get("generated_at") or "")[:19]
        if not g:
            return None
        return max(0, (now_utc - datetime.datetime.fromisoformat(g)).total_seconds())
    except Exception:
        return None

def compute_tier(agent):
    has_sid = bool(agent.get("session_id"))
    has_pane = bool(agent.get("tmux_pane_id"))
    if has_sid and has_pane:
        return ("HIGH", GRN)
    if has_sid:
        return ("MED", CYN)
    if has_pane:
        return ("LOW", YLW)
    return ("NONE", DIM)


def discover_state_files(dirs):
    found = {}
    for d in dirs:
        if not os.path.isdir(d):
            continue
        try:
            for fn in os.listdir(d):
                if not fn.startswith("state.") or fn == "state.latest.json" or not fn.endswith(".json"):
                    continue
                host_key = fn[6:-5]
                path = os.path.join(d, fn)
                if not os.path.isfile(path):
                    continue
                mtime = os.path.getmtime(path)
                if host_key not in found or mtime > found[host_key][1]:
                    found[host_key] = (path, mtime)
        except OSError:
            continue
    return found


def load_all_states(single_file, all_hosts, state_dir, extra_dirs):
    if not all_hosts:
        f = single_file or os.path.join(state_dir, "state.latest.json")
        if not os.path.isfile(f):
            return []
        try:
            return [(json.load(open(f)), f)]
        except Exception as e:
            print(f"\033[31mERROR: {e}\033[0m", file=sys.stderr)
            sys.exit(1)
    dirs = [state_dir] + extra_dirs
    found = discover_state_files(dirs)
    if not found:
        f = os.path.join(state_dir, "state.latest.json")
        if os.path.isfile(f):
            try:
                return [(json.load(open(f)), f)]
            except Exception:
                pass
        return []
    results = []
    for _, (path, _) in sorted(found.items()):
        try:
            results.append((json.load(open(path)), path))
        except Exception:
            continue
    return results

run_watch() {
  local show_all="${1:-false}"
  local all_hosts="${2:-false}"
  local no_refresh="${3:-false}"
  local agentd local_pid="" remote_pid="" last_local=0 last_remote=0 last_reload=0
  agentd="$(_find_agentd)"
  trap 'exit 0' INT TERM
  export AGENT_TOP_ALL_HOSTS="$all_hosts"
  mkdir -p "$AGENT_TOP_STATE_DIR"

  _age_label() {
    local last="${1:-0}" now age
    now="$(date +%s)"
    if [ "$last" -le 0 ] 2>/dev/null; then
      printf 'never'
      return
    fi
    age=$((now - last))
    if [ "$age" -lt 60 ]; then printf '%ss ago' "$age"
    elif [ "$age" -lt 3600 ]; then printf '%sm ago' "$((age / 60))"
    else printf '%sh ago' "$((age / 3600))"
    fi
  }

  _poll_refresh_jobs() {
    if [ -n "$local_pid" ] && ! ps -p "$local_pid" >/dev/null 2>&1; then
      if wait "$local_pid" 2>/dev/null; then
        last_local="$(date +%s)"
      fi
      local_pid=""
    fi
    if [ -n "$remote_pid" ] && ! ps -p "$remote_pid" >/dev/null 2>&1; then
      if wait "$remote_pid" 2>/dev/null; then
        last_remote="$(date +%s)"
      fi
      remote_pid=""
    fi
  }

  _start_local_refresh() {
    if [ "$no_refresh" = "true" ] || { [ -n "$local_pid" ] && ps -p "$local_pid" >/dev/null 2>&1; }; then
      return
    fi
    ( timeout 30 "$agentd" once >/dev/null 2>&1 || true ) &
    local_pid="$!"
  }

  _start_remote_refresh() {
    if [ "$no_refresh" = "true" ] || [ "$all_hosts" != "true" ] || { [ -n "$remote_pid" ] && ps -p "$remote_pid" >/dev/null 2>&1; }; then
      return
    fi
    (
      _json="$(timeout 45 ssh -o ConnectTimeout=5 -o BatchMode=yes x230 "sh -c 'agent-sessiond once >/dev/null 2>&1; cat ~/HelmCortex/ROOTS/DotRoot/agent-session/state.\$(hostname).json'" 2>/dev/null || true)"
      [ -n "$_json" ] || exit 1
      if [ -n "$_json" ]; then
        _host="$(printf '%s' "$_json" | python3 -c 'import json,sys; print(json.load(sys.stdin).get("host",""))' 2>/dev/null || true)"
        [ -n "$_host" ] || exit 1
        if [ -n "$_host" ]; then
          printf '%s' "$_json" > "${AGENT_TOP_STATE_DIR}/state.${_host}.json"
        fi
      fi
    ) &
    remote_pid="$!"
  }

  while true; do
    now="$(date +%s)"
    _poll_refresh_jobs
    if [ "$no_refresh" != "true" ]; then
      if [ "$last_local" -eq 0 ] || [ $((now - last_local)) -ge "$AGENT_TOP_LOCAL_REFRESH_INTERVAL" ]; then
        _start_local_refresh
      fi
      if [ "$all_hosts" = "true" ] && { [ "$last_remote" -eq 0 ] || [ $((now - last_remote)) -ge "$AGENT_TOP_REMOTE_REFRESH_INTERVAL" ]; }; then
        _start_remote_refresh
      fi
    fi
    if [ "$last_reload" -eq 0 ] || [ $((now - last_reload)) -ge "$AGENT_TOP_RELOAD_INTERVAL" ]; then
      clear
      printf 'refreshed local: %s' "$(_age_label "$last_local")"
      if [ "$all_hosts" = "true" ]; then
        printf '   refreshed X230: %s' "$(_age_label "$last_remote")"
      fi
      printf '   reload: %ss\n\n' "$AGENT_TOP_RELOAD_INTERVAL"
      run_once "$show_all" || true
      last_reload="$now"
    fi
    sleep 1
  done
}

def fmt_kb(kb):
    n = (kb or 0) * 1024
    if not n:
        return "—"
    for unit, thr in [("G", 1 << 30), ("M", 1 << 20), ("K", 1 << 10)]:
        if n >= thr:
            return f"{n/thr:.0f}{unit}"
    return f"{n}B"

def fmt_ctx(n):
    if n is None: return "—"
    if n >= 1_000_000: return f"{n/1_000_000:.1f}M"
    if n >= 1_000:     return f"{n/1_000:.1f}k"
    return str(n)

def short_sid(s):
    return (s[-12:] if s and len(s) > 12 else s) or "—"

def short_machine(host):
    if not host:
        return "—"
    h = str(host)
    hl = h.lower()
    if "t480s" in hl: return "T480s"
    if "x230"  in hl: return "X230"
    if "t430"  in hl: return "T430"
    if "vultr" in hl: return "Vultr"
    if "vps"   in hl: return "VPS"
    if h.startswith("ThinkPad-"): return h[9:]
    return h[:8]

def local_hostname():
    return socket.gethostname()

def ssh_alias_for_host(host):
    h = short_machine(host).lower()
    if h == "x230":
        return "x230"
    return None

def is_local_host(host):
    h = short_machine(host).lower()
    lh = short_machine(local_hostname()).lower()
    return not h or h == lh

def row_owner_host(row):
    return row.get("_state_host") or row.get("machine") or local_hostname()

def shell_join_remote(cwd, argv):
    cmd = shlex.join(argv)
    return f"cd {shlex.quote(cwd)} && {cmd}" if cwd else cmd

def build_tmux_pane_script(pane_id, attach=True):
    pane_q = shlex.quote(str(pane_id))
    parts = [
        f"pane={pane_q}",
        "session=$(tmux display-message -p -t \"$pane\" '#S') || exit 1",
        "window=$(tmux display-message -p -t \"$pane\" '#I') || exit 1",
    ]
    if attach:
        parts.append(
            "tmux select-window -t \"$session:$window\" \\; "
            "select-pane -t \"$pane\" \\; "
            "attach-session -t \"$session\""
        )
    else:
        parts.append(
            "tmux switch-client -t \"$session\" \\; "
            "select-window -t \"$session:$window\" \\; "
            "select-pane -t \"$pane\""
        )
    return "; ".join(parts)

def build_remote_tmux_attach_cmd(host_alias, pane_id):
    return ["ssh", "-t", host_alias, build_tmux_pane_script(pane_id, attach=True)]

def ssh_command_for(host, remote_script):
    alias = ssh_alias_for_host(host)
    if not alias:
        return None
    return ["ssh", "-t", alias, remote_script]

def ssh_available(alias):
    try:
        return subprocess.run(
            ["ssh", "-o", "ConnectTimeout=5", "-o", "BatchMode=yes", alias, "true"],
            stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, timeout=8,
        ).returncode == 0
    except Exception:
        return False

def resume_argv_for(agent, sid):
    fn = RESUME_ARGV.get(agent)
    return fn(sid) if fn else []

def resume_command_for_row(r):
    idx = r.get("_idx") or {}
    agent = r.get("pattern") or ""
    sid = r.get("session_id") or ""
    cwd_v = idx.get("cwd") or r.get("cwd") or ""
    argv = resume_argv_for(agent, sid)
    if not argv:
        return "", [], cwd_v, agent, sid
    host = row_owner_host(r)
    if is_local_host(host):
        return shell_join_remote(cwd_v, argv), argv, cwd_v, agent, sid
    alias = ssh_alias_for_host(host)
    if not alias:
        return f"(no ssh alias for {host})", [], cwd_v, agent, sid
    remote = shell_join_remote(cwd_v, argv)
    cmd = ["ssh", "-t", alias, remote]
    return shlex.join(cmd), cmd, cwd_v, agent, sid

def jump_command_for_row(r):
    pane = r.get("tmux_pane_id")
    if not pane:
        return "", []
    host = row_owner_host(r)
    if is_local_host(host):
        script = build_tmux_pane_script(pane, attach=not bool(os.environ.get("TMUX")))
        return script, ["sh", "-c", script]
    alias = ssh_alias_for_host(host)
    if not alias:
        return f"(no ssh alias for {host})", []
    cmd = build_remote_tmux_attach_cmd(alias, pane)
    return shlex.join(cmd), cmd

def short_model(m, width=16):
    if not m:
        return "—"
    if len(m) <= width:
        return m
    return ".." + m[-(width - 2):]

def cwd_base(path):
    if not path:
        return "—"
    home = os.path.expanduser("~")
    p = ("~" + path[len(home):]) if path.startswith(home) else path
    return p.rstrip("/").split("/")[-1] or p

def age_str(secs):
    if secs is None:
        return "?"
    if secs < 60:
        return f"{int(secs)}s"
    if secs < 3600:
        return f"{int(secs // 60)}m"
    if secs < 86400:
        return f"{int(secs // 3600)}h"
    return f"{int(secs // 86400)}d"

def state_age_seconds(state):
    try:
        g = (state.get("generated_at") or "")[:19]
        if not g:
            return None
        return max(0, (datetime.datetime.utcnow() - datetime.datetime.fromisoformat(g)).total_seconds())
    except Exception:
        return None

def freshness_badge(secs):
    if secs is None:
        return "OLD"
    if secs < 60:
        return "FRESH"
    if secs < 300:
        return "WARM"
    if secs < 3600:
        return "STALE"
    return "OLD"

def tier_of(a):
    if a.get("session_id") and a.get("tmux_pane_id"):
        return "HIGH"
    if a.get("session_id"):
        return "MED"
    if a.get("tmux_pane_id"):
        return "LOW"
    return "NONE"

def tier_ord(t):
    return {"HIGH": 0, "MED": 1, "LOW": 2, "NONE": 3}.get(t, 4)

def discover_state_files_tui(dirs):
    found = {}
    for d in dirs:
        if not os.path.isdir(d):
            continue
        try:
            for fn in os.listdir(d):
                if not fn.startswith("state.") or fn == "state.latest.json" or not fn.endswith(".json"):
                    continue
                host_key = fn[6:-5]
                path = os.path.join(d, fn)
                if not os.path.isfile(path):
                    continue
                mtime = os.path.getmtime(path)
                if host_key not in found or mtime > found[host_key][1]:
                    found[host_key] = (path, mtime)
        except OSError:
            continue
    return found


def load_all_states_tui(single_file, all_hosts, state_dir, extra_dirs):
    if not all_hosts:
        f = single_file or os.path.join(state_dir, "state.latest.json")
        if not os.path.isfile(f):
            return []
        try:
            return [(json.load(open(f)), f)]
        except Exception:
            return []
    dirs = [state_dir] + extra_dirs
    found = discover_state_files_tui(dirs)
    if not found:
        f = os.path.join(state_dir, "state.latest.json")
        if os.path.isfile(f):
            try:
                return [(json.load(open(f)), f)]
            except Exception:
                pass
        return []
    results = []
    for _, (path, _) in sorted(found.items()):
        try:
            results.append((json.load(open(path)), path))
        except Exception:
            continue
    return results


def load_data(show_closed=False, all_hosts=False):
    states_loaded = load_all_states_tui(
        STATE_FILE if not all_hosts else "",
        all_hosts, STATE_DIR, EXTRA_DIRS)
    index = {}
    if INDEX_FILE and os.path.isfile(INDEX_FILE):
        try:
            raw = json.loads(open(INDEX_FILE).read())
            lst = raw.get("sessions", []) if isinstance(raw, dict) else raw
            index = {s.get("id"): s for s in lst if isinstance(s, dict) and s.get("id")}
        except Exception:
            pass
    rows = []
    seen_keys = set()
    for st, _path in states_loaded:
        state_host = st.get("host") or "?"
        for a in (st.get("active_agents") or []):
            key = (state_host, a.get("pid"), a.get("session_id"))
            if key in seen_keys:
                continue
            seen_keys.add(key)
            tier = tier_of(a)
            idx  = index.get(a.get("session_id")) or {}
            ist  = idx.get("status") or ""
            if not show_closed and ist in ("closed", "abandoned"):
                continue
            r = dict(a)
            r["_tier"]       = tier
            r["_prov"]       = a.get("provider")              or "—"
            r["_prov_src"]   = a.get("provider_source")       or "unknown"
            r["_model"]      = a.get("model")                 or "—"
            r["_model_src"]  = a.get("model_source")          or "unknown"
            r["_model_conf"] = a.get("model_confidence")      or "none"
            r["_ctx_size"]   = a.get("context_size")
            r["_ctx_src"]    = a.get("context_size_source")   or "unknown"
            r["_ctx_win"]    = a.get("context_window")
            r["_ctx_win_src"]= a.get("context_window_source") or "unknown"
            r["_ctx_pct"]    = a.get("context_percent")
            r["_meta_stale"] = a.get("metadata_stale") or False
            r["_meta_age"]   = a.get("metadata_age_s")
            r["_title"]      = idx.get("openquestion") or idx.get("title") or ""
            r["_ist"]        = ist or "—"
            r["_state_host"] = state_host
            r["_mach"]       = idx.get("machine") or state_host
            r["_mach_short"] = short_machine(r["_mach"])
            r["_idx"]        = idx
            rows.append(r)
    primary = states_loaded[0][0] if states_loaded else {}
    return states_loaded, primary, rows

def sort_rows(rows, sf):
    def key(r):
        if sf == "tier":     return (tier_ord(r["_tier"]), r.get("pattern", ""))
        if sf == "agent":    return (r.get("pattern", ""), tier_ord(r["_tier"]))
        if sf == "pid":      return (r.get("pid") or 0,)
        if sf == "rss":      return (-(r.get("rss_kb") or 0),)
        if sf == "io_write": return (-(r.get("io_write_bytes") or 0),)
        if sf == "status":   return (r.get("_ist", ""), r.get("pattern", ""))
        if sf == "title":    return (r.get("_title", "").lower(),)
        if sf == "ctx":      return (-(r.get("_ctx_pct") or 0),)
        if sf == "model":    return (r.get("_model", ""), r.get("pattern", ""))
        if sf == "prov":     return (r.get("_prov", ""), r.get("pattern", ""))
        return (0,)
    return sorted(rows, key=key)

def filter_rows(rows, q):
    if not q:
        return rows
    ql = q.lower()
    def m(r):
        return any(ql in (v or "").lower() for v in [
            r.get("pattern"), str(r.get("pid") or ""),
            r.get("session_id"), r.get("_title"), r.get("_mach"),
            r.get("cwd"), r.get("_ist"), r.get("_prov"),
        ])
    return [r for r in rows if m(r)]

    def __init__(self, scr):
        self.scr         = scr
        self.sel         = 0
        self.offset      = 0
        self.sort_idx    = 0
        self.query       = ""
        self.searching   = False
        self.show_closed = False
        self.all_hosts   = ALL_HOSTS_FLAG
        self.status_msg  = ""
        self.status_ts   = 0.0
        self.refresh_procs = {}
        self.refresh_started = {}
        self.last_reload = 0.0
        self.last_local_refresh = 0.0
        self.last_remote_refresh = 0.0
        self.refresh_enabled = not NO_REFRESH
        self.states      = []
        self.state       = {}
        self.all_rows    = []
        self.visible     = []
        self.reload()
        scr.keypad(True)
        curses.noecho()
        curses.cbreak()
        try: curses.curs_set(0)
        except curses.error: pass
        scr.timeout(500)
        self._init_colors()
        if self.refresh_enabled:
            self._start_refresh("local", force=True)
            if self.all_hosts:
                self._start_refresh("remote", force=True)

    def _init_colors(self):
        curses.start_color()
        try:
            curses.use_default_colors()
            bg = -1
        except Exception:
            bg = curses.COLOR_BLACK
        curses.init_pair(1, curses.COLOR_GREEN,  bg)
        curses.init_pair(2, curses.COLOR_CYAN,   bg)
        curses.init_pair(3, curses.COLOR_YELLOW, bg)
        curses.init_pair(7, curses.COLOR_BLACK,  curses.COLOR_WHITE)

    def tier_attr(self, t):
        return {
            "HIGH": curses.color_pair(1) | curses.A_BOLD,
            "MED":  curses.color_pair(2),
            "LOW":  curses.color_pair(3),
            "NONE": curses.A_DIM,
        }.get(t, curses.A_NORMAL)

    def reload(self):
        self.states, self.state, self.all_rows = load_data(self.show_closed, self.all_hosts)
        self.refresh_visible()

    def refresh_visible(self):
        sf = SORT_FIELDS[self.sort_idx]
        rows = sort_rows(self.all_rows, sf)
        self.visible = filter_rows(rows, self.query)
        if self.sel >= len(self.visible):
            self.sel = max(0, len(self.visible) - 1)

    def set_status(self, msg):
        self.status_msg = msg
        self.status_ts  = time.time()

    def _start_refresh(self, kind, force=False):
        if not self.refresh_enabled and not force:
            return
        if kind in self.refresh_procs and self.refresh_procs[kind].poll() is None:
            return
        os.makedirs(STATE_DIR, exist_ok=True)
        if kind == "local":
            self.set_status("refreshing local...")
            self.refresh_procs[kind] = subprocess.Popen(
                [AGENTD, "once"],
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
            )
            self.refresh_started[kind] = time.time()
        elif kind == "remote":
            self.set_status("refreshing X230...")
            self.refresh_procs[kind] = subprocess.Popen(
                ["ssh", "-o", "ConnectTimeout=5", "-o", "BatchMode=yes",
                 "x230", "sh", "-c",
                 "agent-sessiond once >/dev/null 2>&1; "
                 "cat ~/HelmCortex/ROOTS/DotRoot/agent-session/state.$(hostname).json"],
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
            )
            self.refresh_started[kind] = time.time()

    def _poll_refreshes(self):
        now = time.time()
        for kind, proc in list(self.refresh_procs.items()):
            if proc.poll() is None:
                limit = 30 if kind == "local" else 45
                if now - self.refresh_started.get(kind, now) > limit:
                    proc.terminate()
                    proc.communicate()
                    del self.refresh_procs[kind]
                    self.set_status("X230 offline / timeout" if kind == "remote" else "local refresh timed out")
                continue
            out, err = proc.communicate()
            del self.refresh_procs[kind]
            if kind == "local":
                self.last_local_refresh = now
                self.set_status("refreshed local")
            else:
                if proc.returncode == 0 and out:
                    try:
                        d = json.loads(out)
                        rh = d.get("host", "")
                        if rh:
                            with open(os.path.join(STATE_DIR, f"state.{rh}.json"), "wb") as fh:
                                fh.write(out)
                            self.last_remote_refresh = now
                            self.set_status("fetched X230 state")
                        else:
                            self.set_status("X230 refresh returned no host")
                    except Exception as e:
                        self.set_status(f"X230 refresh parse error: {e}")
                else:
                    self.set_status("X230 offline / timeout")

    def _background_tick(self):
        now = time.time()
        self._poll_refreshes()
        if now - self.last_reload >= RELOAD_INTERVAL:
            self.reload()
            self.last_reload = now
        if self.refresh_enabled:
            if now - self.last_local_refresh >= LOCAL_REFRESH_INTERVAL:
                self._start_refresh("local")
            if self.all_hosts and now - self.last_remote_refresh >= REMOTE_REFRESH_INTERVAL:
                self._start_refresh("remote")

    def _end_curses(self):
        try: curses.nocbreak()
        except curses.error: pass
        try: self.scr.keypad(False)
        except curses.error: pass
        try: curses.echo()
        except curses.error: pass
        try: curses.endwin()
        except curses.error: pass

    def draw(self):
        h, w = self.scr.getmaxyx()
        if h < 5 or w < 40:
            try:
                self.scr.addstr(0, 0, "Terminal too small (need 40x5)")
            except curses.error:
                pass
            self.scr.refresh()
            return
        self.scr.erase()
        self._draw_header(0, w)
        self._draw_col_hdr(1, w)
        table_h = h - 3
        if self.sel < self.offset:
            self.offset = self.sel
        elif self.sel >= self.offset + table_h:
            self.offset = self.sel - table_h + 1
        for i in range(table_h):
            ri = self.offset + i
            if ri >= len(self.visible):
                break
            self._draw_row(2 + i, w, self.visible[ri], ri == self.sel)
        self._draw_status(h - 1, w)
        self.scr.refresh()

    def _col_widths(self, w):
        fixed = [
            ("TIER", 5), ("AGENT", 10), ("PID", 7), ("SESSION", 14),
            ("STATUS", 6), ("MACH", 6), ("PROV", 9), ("MODEL", 16), ("CTX%", 5), ("RSS", 6), ("PS", 3),
            ("PANE", 10), ("CWD", 12),
        ]
        used = 2 + sum(cw for _, cw in fixed) + len(fixed)
        return fixed + [("TITLE", max(4, w - used - 1))]

    def _draw_header(self, y, w):
        sch = self.state.get("schema", "?")
        sf  = SORT_FIELDS[self.sort_idx]
        cf  = "  +closed" if self.show_closed else ""
        ah  = "  [all-hosts]" if self.all_hosts else ""
        stale_parts = []
        newest_gat  = ""
        host_shorts = []
        for st, _p in (self.states if self.states else [(self.state, "")]):
            _hst = st.get("host", "?")
            _hs  = short_machine(_hst)
            if _hs not in host_shorts:
                host_shorts.append(_hs)
            _gat = (st.get("generated_at") or "")[:19]
            if _gat > newest_gat:
                newest_gat = _gat
            _age = state_age_seconds(st)
            stale_parts.append(f"{_hs}:{freshness_badge(_age)}")
        hosts_str = ",".join(host_shorts) if host_shorts else "?"
        stale_flag = "  " + ",".join(stale_parts) if stale_parts else ""
        hdr = (f" agent-top  {hosts_str}  {newest_gat}  schema:{sch}"
               f"  active:{len(self.all_rows)}  vis:{len(self.visible)}"
               f"  sort:{sf}{cf}{ah}{stale_flag}")
        try:
            self.scr.addstr(y, 0, hdr[:w - 1].ljust(w - 1),
                            curses.A_REVERSE | curses.A_BOLD)
        except curses.error:
            pass

    def _draw_col_hdr(self, y, w):
        cols = self._col_widths(w)
        line = "  " + " ".join(f"{name:<{cw}}" for name, cw in cols)
        try:
            self.scr.addstr(y, 0, line[:w - 1], curses.A_DIM)
        except curses.error:
            pass

    def _draw_row(self, y, w, r, sel):
        tier = r["_tier"]
        cols = self._col_widths(w)
        tw   = cols[-1][1]
        ps   = (r.get("proc_status") or "?")[:3]
        mach  = (r.get("_mach_short") or "—")[:6]
        model = short_model(r.get("_model"))
        cpct  = r.get("_ctx_pct")
        if cpct is not None:
            ctx_s = ("~" if r.get("_meta_stale") else "") + f"{cpct:.0f}%"
        else:
            ctx_s = "—"
        fields = [
            f"{tier:<5}",
            f"{(r.get('pattern') or '?')[:10]:<10}",
            f"{str(r.get('pid') or '?'):<7}",
            f"{short_sid(r.get('session_id')):<14}",
            f"{r.get('_ist', '—')[:6]:<6}",
            f"{mach:<6}",
            f"{r.get('_prov', '—')[:9]:<9}",
            f"{model:<16}",
            f"{ctx_s[:5]:<5}",
            f"{fmt_kb(r.get('rss_kb'))[:6]:<6}",
            f"{ps:<3}",
            f"{(r.get('tmux_pane_id') or '—')[:10]:<10}",
            f"{cwd_base(r.get('cwd'))[:12]:<12}",
            f"{(r.get('_title') or '')[:tw]:<{tw}}",
        ]
        line = "  " + " ".join(fields)
        try:
            if sel:
                self.scr.addstr(y, 0, line[:w - 1], curses.color_pair(7))
            else:
                self.scr.addstr(y, 0, line[:w - 1])
                self.scr.addstr(y, 2, f"{tier:<5}", self.tier_attr(tier))
        except curses.error:
            pass

    def _draw_status(self, y, w):
        if self.searching:
            bar = f" /{self.query}█  Esc:clear  Enter:done"
        elif self.status_msg and (time.time() - self.status_ts) < 5:
            bar = f" {self.status_msg}"
        else:
            now = time.time()
            local_age = age_str(now - self.last_local_refresh) if self.last_local_refresh else "never"
            remote_age = age_str(now - self.last_remote_refresh) if self.last_remote_refresh else "never"
            refresh_s = (
                f" refreshed local:{local_age}"
                + (f" X230:{remote_age}" if self.all_hosts else "")
                + f" reload:{int(RELOAD_INTERVAL)}s"
            )
            bar = (" ?:help  s:sort  /:search  a:±closed"
                   "  r:refresh  Enter:jump  Spc:show  R:resume  c:close  t:stop  q:quit"
                   + refresh_s)
        try:
            self.scr.addstr(y, 0, bar[:w - 1].ljust(w - 1), curses.A_REVERSE)
        except curses.error:
            pass

    def run(self):
        while True:
            self._background_tick()
            self.draw()
            key = self.scr.getch()
            if key == curses.ERR:
                continue
            dbg(f"key={key} name={curses.keyname(key)}")
            if self.searching:
                self._search_key(key)
            elif not self._normal_key(key):
                break

    def _search_key(self, key):
        if key == 27:
            self.searching = False
            self.query = ""
            self.refresh_visible()
        elif key in (curses.KEY_BACKSPACE, 127, 8):
            self.query = self.query[:-1]
            self.refresh_visible()
        elif key == ord('\n'):
            self.searching = False
        elif 32 <= key < 127:
            self.query += chr(key)
            self.refresh_visible()

    def _normal_key(self, key):
        n  = len(self.visible)
        h, _ = self.scr.getmaxyx()
        pg = max(1, h - 3)
        # Movement — Colemak-NEIO primary; arrow keys always work
        if key in (ord('q'), curses.KEY_F10):
            return False
        elif key in (curses.KEY_UP, ord('i')):
            self.sel = max(0, self.sel - 1)
        elif key in (curses.KEY_DOWN, ord('e')):
            self.sel = min(max(0, n - 1), self.sel + 1)
        elif key == curses.KEY_PPAGE:
            self.sel = max(0, self.sel - pg)
        elif key == curses.KEY_NPAGE:
            self.sel = min(max(0, n - 1), self.sel + pg)
        elif key == curses.KEY_HOME:
            self.sel = 0
        elif key == curses.KEY_END:
            self.sel = max(0, n - 1)
        # Jump / inspect
        elif key in (curses.KEY_ENTER, ord('\n'), ord('\r'), ord('o')):
            self._action_jump()
        elif key == ord(' '):
            self._action_show()
        # Help
        elif key in (curses.KEY_F1, ord('?')):
            self._overlay(self.HELP, " Help ")
        # Sort
        elif key in (curses.KEY_F2, ord('s')):
            self.sort_idx = (self.sort_idx + 1) % len(SORT_FIELDS)
            self.refresh_visible()
            self.set_status(f"Sort: {SORT_FIELDS[self.sort_idx]}")
        # Search
        elif key in (curses.KEY_F3, ord('/')):
            self.searching = True
        # Toggle closed
        elif key in (curses.KEY_F4, ord('a')):
            self.show_closed = not self.show_closed
            self.reload()
            self.set_status(f"Closed sessions: {'shown' if self.show_closed else 'hidden'}")
        # Refresh
        elif key in (curses.KEY_F5, ord('r')):
            self._action_refresh()
        # Resume
        elif key in (curses.KEY_F6, ord('R')):
            self._action_resume()
        # Close
        elif key in (curses.KEY_F7, ord('c')):
            self._action_close()
        # Polite stop (SIGTERM)
        elif key in (curses.KEY_F8, ord('t')):
            self._action_stop()
        elif key == curses.KEY_F9:
            self.set_status("F9 kill (SIGKILL) not wired yet")
        elif key == curses.KEY_RESIZE:
            self.scr.clear()
            dbg("resize")
        return True

    def _sel_row(self):
        if self.visible and 0 <= self.sel < len(self.visible):
            return self.visible[self.sel]
        return None

    def _action_refresh(self):
        self._start_refresh("local", force=True)
        if self.all_hosts:
            self._start_refresh("remote", force=True)
        self.set_status("refreshing local..." + (" refreshing X230..." if self.all_hosts else ""))

    def _action_close(self):
        r = self._sel_row()
        if not r:
            self.set_status("No row selected")
            return
        sid = r.get("session_id") or ""
        if not sid:
            self.set_status(
                f"No session_id  pid={r.get('pid')} [{r.get('pattern')}]"
            )
            return
        ist = r.get("_ist") or ""
        if ist in ("closed", "abandoned"):
            self.set_status(f"Already {ist}: {short_sid(sid)}")
            return
        agent  = r.get("pattern") or "?"
        title_s = (r.get("_title") or "")[:25]
        tail   = f" {title_s}" if title_s else ""
        prompt = f" Close {short_sid(sid)} [{agent}]{tail}? [y/N] "
        if not self._confirm(prompt):
            self.set_status("Close cancelled")
            return
        dbg(f"close: {agent} sid={sid[:12]}")
        try:
            result = subprocess.run(
                [AGENT_SESSION, "close", sid],
                capture_output=True, text=True, timeout=15,
            )
            if result.returncode == 0:
                self.reload()
                self.set_status(f"Closed: {short_sid(sid)} [{agent}]")
            else:
                err = (result.stderr or result.stdout or "unknown error").strip()[:60]
                self.set_status(f"Close failed: {err}")
        except subprocess.TimeoutExpired:
            self.set_status("Close timed out")
        except Exception as e:
            self.set_status(f"Close error: {e}")

    def _action_stop(self):
        r = self._sel_row()
        if not r:
            self.set_status("No row selected")
            return
        pid = r.get("pid")
        if not pid:
            self.set_status(f"No PID  [{r.get('pattern')}]")
            return
        try:
            pid_int = int(pid)
        except (TypeError, ValueError):
            self.set_status(f"Invalid PID: {pid!r}")
            return
        try:
            os.kill(pid_int, 0)
        except ProcessLookupError:
            self.reload()
            self.set_status(f"PID {pid_int} no longer exists — reloaded")
            return
        except PermissionError:
            pass  # process exists, owned by another uid
        agent  = r.get("pattern") or "?"
        sid    = r.get("session_id") or ""
        label  = short_sid(sid) if sid else (r.get("_title") or "")[:20]
        tail   = f" {label}" if label else ""
        prompt = f" Stop PID {pid_int} [{agent}]{tail}? [y/N] "
        if not self._confirm(prompt):
            self.set_status("Stop cancelled")
            return
        dbg(f"stop: SIGTERM pid={pid_int} agent={agent}")
        try:
            os.kill(pid_int, signal.SIGTERM)
        except ProcessLookupError:
            self.reload()
            self.set_status(f"PID {pid_int} already gone — reloaded")
            return
        except PermissionError as e:
            self.set_status(f"SIGTERM denied: {e}")
            return
        except Exception as e:
            self.set_status(f"Stop error: {e}")
            return
        self.reload()
        self.set_status(f"SIGTERM → PID {pid_int} [{agent}]")

    def _action_resume(self):
        r = self._sel_row()
        if not r:
            self.set_status("No row selected")
            return
        idx   = r.get("_idx") or {}
        agent = r.get("pattern") or ""
        sid   = r.get("session_id") or ""
        cwd_v = idx.get("cwd") or r.get("cwd") or ""
        if not sid:
            self.set_status("No session_id for selected row")
            return
        argv = resume_argv_for(agent, sid)
        if not argv:
            self.set_status(f"No resume command for agent '{agent}'")
            return
        ist = r.get("_ist") or ""
        if ist in ("closed", "abandoned"):
            if not self._confirm(f" Session is {ist}. Resume anyway? [y/N] "):
                self.set_status("Resume cancelled")
                return
        host = row_owner_host(r)
        if not is_local_host(host):
            alias = ssh_alias_for_host(host)
            if not alias:
                self.set_status(f"No ssh alias for host '{host}'")
                return
            remote = shell_join_remote(cwd_v, argv)
            cmd = ["ssh", "-t", alias, remote]
            display = shlex.join(cmd)
            self.set_status(display)
            if not ssh_available(alias):
                self.set_status(f"{short_machine(host)} offline; resume command not run.")
                return
            dbg(f"resume remote: {display}")
            self._end_curses()
            print(f"  {display}")
            try:
                os.execvp(cmd[0], cmd)
            except OSError as e:
                print(f"agent-top: exec failed: {e}", file=sys.stderr)
                sys.exit(1)
        dbg(f"resume: exec {agent} sid={sid[:12]} cwd={cwd_v[:30]!r}")
        self._end_curses()
        if cwd_v:
            print(f"  cd {shlex.quote(cwd_v)}")
            try:
                os.chdir(cwd_v)
            except Exception:
                pass
        print(f"  {shlex.join(argv)}")
        try:
            os.execvp(argv[0], argv)
        except OSError as e:
            print(f"agent-top: exec failed: {e}", file=sys.stderr)
            sys.exit(1)

    def _confirm(self, prompt):
        h, w = self.scr.getmaxyx()
        cw  = min(len(prompt) + 6, w - 4)
        win = curses.newwin(3, cw, (h - 3) // 2, (w - cw) // 2)
        win.box()
        try:
            win.addstr(1, 1, prompt[:cw - 2])
        except curses.error:
            pass
        win.refresh()
        k = win.getch()
        return 32 <= k < 127 and chr(k).lower() == 'y'

    def _overlay(self, lines, title="", w_hint=None):
        h, w = self.scr.getmaxyx()
        mw = w_hint if w_hint else min(60, w - 4)
        mh = min(len(lines) + 2, h - 2)
        win = curses.newwin(mh, mw, (h - mh) // 2, (w - mw) // 2)
        win.box()
        if title:
            tx = max(0, (mw - len(title)) // 2)
            try:
                win.addstr(0, tx, title[:mw - 2], curses.A_BOLD)
            except curses.error:
                pass
        for i, line in enumerate(lines):
            if i + 1 >= mh - 1:
                break
            try:
                win.addstr(i + 1, 1, line[:mw - 2])
            except curses.error:
                pass
        win.refresh()
        win.getch()

* Lid Switch Fix — T480s (elogind)
T480s runs SLiM + elogind + xfce4-power-manager. By default
xfce4-power-manager sets =logind-handle-lid-switch=false=, meaning it tries
to handle the lid switch itself over a D-Bus call to
=org.freedesktop.login1=. That connection has been observed to die
(=xfce4-power-manager: Error: The connection is closed= in
=~/.xsession-errors=) and silently fail every lid-close action afterward —
screen stays lit all night, sometimes with a "couldn't do it" popup. SLiM
also leaves the graphical session mis-classified (=Class=greeter=) since it
predates logind session-class transitions, and its lack of a clean
greeter→user handoff correlates with zombie sessions accumulating in
=State=closing= in =loginctl list-sessions=.

Fix: let elogind own the lid switch natively via a config drop-in — this
sidesteps xfce4-power-manager's D-Bus proxy entirely — and flip
=logind-handle-lid-switch= to =true= in xfconf so it steps out of the way.

** logind drop-in
#+BEGIN_SRC conf :tangle t480s/.local/share/dotcortex/power/logind-lid.conf :padline no :comments none
[Login]
HandleLidSwitch=suspend
HandleLidSwitchExternalPower=suspend
LidSwitchIgnoreInhibited=yes
#+END_SRC

** Setup script
Idempotent — safe to re-run. Installs the drop-in, restarts elogind, sets
the xfconf key so xfce4-power-manager stops fighting elogind for the lid
switch, and clears any zombie =closing= sessions left over from prior
SLiM/elogind session-handoff hiccups.

sudo install -d -m 755 /etc/elogind/logind.conf.d
sudo install -m 644 "$SRC" "$DEST"

echo "[2/4] xfconf: step xfce4-power-manager out of the lid switch..."
XFCE_PM_PID="$(pgrep -u "$USER" -f '(^|/)xfce4-power-manager$' | head -n1 || true)"
if [ -n "$XFCE_PM_PID" ] && [ -r "/proc/${XFCE_PM_PID}/environ" ]; then
  DBUS_SESSION_BUS_ADDRESS="$(tr '\0' '\n' < "/proc/${XFCE_PM_PID}/environ" | sed -n 's/^DBUS_SESSION_BUS_ADDRESS=//p')"
  export DBUS_SESSION_BUS_ADDRESS
  xfconf-query -c xfce4-power-manager -p /xfce4-power-manager/logind-handle-lid-switch -n -t bool -s true 2>/dev/null \
    || xfconf-query -c xfce4-power-manager -p /xfce4-power-manager/logind-handle-lid-switch -s true
else
  echo "WARNING: no running xfce4-power-manager found for this user — set logind-handle-lid-switch=true manually once a session is up"
fi

echo "[3/4] clearing zombie 'closing' sessions..."
loginctl list-sessions --no-legend | awk '$6=="closing"{print $1}' | while read -r sid; do
  sudo loginctl terminate-session "$sid" 2>/dev/null || true
done

echo "[4/4] restarting elogind..."
sudo service elogind restart

echo
echo "=== T480s lid-switch fix applied ==="
echo "Test by physically closing the lid — this is the only real verification."
#+END_SRC
