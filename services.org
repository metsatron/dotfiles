* Services Loom
This loom tracks small service wrappers and helper scripts that live in
the =.local/bin= overlays and are deployed via stow.

* Early OOM
Scripts to apply a conservative, balanced OOM policy.
The X230 runs systemd; the T480s runs Devuan/sysvinit.
Both expose the same =MODE= knob (=balanced= | =proactive=).

** earlyoom-balanced — X230 (systemd drop-in)
Writes a systemd drop-in at =/etc/systemd/system/earlyoom.service.d/override.conf=
and restarts the service.  Avoids =/etc/default/= quoting issues by setting
=ExecStart= directly.

#+BEGIN_SRC bash :tangle x230/.local/bin/earlyoom-balanced :shebang "#!/bin/env bash"
set -euo pipefail

MODE="${MODE:-balanced}"   # balanced or proactive

# Kill this stuff first if we MUST kill something:
#  - non-critical XFCE daemons
#  - media / background apps
#  - tools that autosave (Obsidian, Telegram)
PREFER='^(xfce4-screensaver|xfce4-clipman|xfdesktop|xfce4-notifyd|kdeconnectd|fluidsynth|jellyfin|[Oo]bsidian|Telegram|WebExtensions|qmmp)$'

# Protect these at all costs:
#  - browsers (no autosave guarantees)
#  - Codium / Code (no autosave guarantees in your current setup)
#  - file managers + GVFS plumbing
#  - file transfer / backup tools (rsync, etc.)
#  - xfce4-session so we don't drop the whole session
AVOID='^(firedragon|floorp|zen|ungoogled-chromium|firefox|codium|code|thunar|gvfs.*|rsync|scp|sftp|rclone|syncthing|borg|restic|xfce4-session|barrier|barriers|deskflow|signal-desktop|Betterbird|betterbird|thunderbird)$'

if [[ "$MODE" = "balanced" ]]; then
  M_TERM=5;  M_KILL=3;  S_TERM=15; S_KILL=10
elif [[ "$MODE" = "proactive" ]]; then
  M_TERM=7;  M_KILL=5;  S_TERM=20; S_KILL=10
else
  echo "Unknown MODE='$MODE' (use balanced or proactive)" >&2
  exit 1
fi

DROPIN_DIR="/etc/systemd/system/earlyoom.service.d"
DROPIN="${DROPIN_DIR}/override.conf"

sudo install -d -m 0755 "$DROPIN_DIR"

# Write drop-in atomically
TMP="$(mktemp)"
cat >"$TMP" <<EOF
[Service]
EnvironmentFile=
ExecStart=
ExecStart=/usr/bin/earlyoom -r 3600 -m ${M_TERM},${M_KILL} -s ${S_TERM},${S_KILL} \
  --prefer ${PREFER} \
  --avoid  ${AVOID}
OOMScoreAdjust=-100
Nice=-20
EOF
sudo install -m 0644 "$TMP" "$DROPIN"
rm -f "$TMP"

sudo systemctl daemon-reload
sudo systemctl restart earlyoom

echo "Applied Early OOM policy: mem ${M_TERM}%/${M_KILL}%, swap ${S_TERM}%/${S_KILL}%"
echo "Prefer: ${PREFER}"
echo "Avoid:  ${AVOID}"

# Show the running command and thresholds
if pidof earlyoom >/dev/null; then
  tr '\0' ' ' </proc/$(pidof earlyoom)/cmdline; echo
  journalctl -u earlyoom -b --no-pager | tail -n 20
else
  echo "earlyoom is not running" >&2
  exit 2
fi
#+END_SRC

** Jellyfin toggle (ThinkPad X230 server)
This script lives in the =x230= overlay, it is specific to the current
X230 that is running Jellyfin as a systemd service.
#+BEGIN_SRC bash :tangle x230/.local/bin/jellyfin-toggle :shebang "#!/bin/sh"
# jellyfin-toggle
# Toggle Jellyfin Media Server on and off via systemd.
# Requires sudoers entry:
#   metsatron ALL=(root) NOPASSWD: /usr/bin/systemctl

if systemctl is-active --quiet jellyfin; then
  exec sudo -n /usr/bin/systemctl stop jellyfin
else
  exec sudo -n /usr/bin/systemctl start jellyfin
fi
#+END_SRC

After first tangle:

#+BEGIN_SRC bash
chmod +x ~/DotCortex/x230/.local/bin/jellyfin-toggle
#+END_SRC

Panel launcher command:
#+BEGIN_SRC bash
sh -c 'if systemctl is-active --quiet jellyfin; then sudo -n /usr/bin/systemctl stop jellyfin; else sudo -n /usr/bin/systemctl start jellyfin; fi'
#+END_SRC

*** Jellyfin sudoers drop in (manual step)
This is not tangled, since =/etc/sudoers.d= is root territory. The doc
lives here so the setup is reproducible.

Install or re install with:
#+BEGIN_SRC bash
sudo visudo -f /etc/sudoers.d/jellyfin-toggle
#+END_SRC

Then type this line by hand inside =visudo=:
#+BEGIN_SRC sudoers
metsatron ALL=(root) NOPASSWD: /usr/bin/systemctl
#+END_SRC

Test:
#+BEGIN_SRC bash
sudo -n /usr/bin/systemctl status jellyfin
sudo -n /usr/bin/systemctl stop jellyfin
sudo -n /usr/bin/systemctl start jellyfin
#+END_SRC

If there is no password prompt, the panel toggle is good.

* Swapfetch
#+NAME: swapfetch
#+BEGIN_SRC bash :tangle linux/.local/bin/swapfetch :shebang "#!/usr/bin/env bash"
set -euo pipefail

# Ensure swapon/free are reachable (Devuan keeps them in /sbin)
[[ ":$PATH:" != *":/sbin:"* ]] && export PATH="$PATH:/sbin"

# Nerd Font icons
ICON_ZRAM="󰓅"   # ZRAM / compression
ICON_SWAP="󰓾"   # swap/disk
ICON_MEM=""    # RAM

# Colors (always on; tune shades in style.org if you like)
C_HDR=$'\e[1;36m'     # cyan bold (headers)
C_LABEL=$'\e[1;33m'   # yellow bold (labels)
C_VAL=$'\e[1;37m'     # white bold  (values)
C_DIM=$'\e[2;37m'     # dim white   (background / secondary)
C_OK=$'\e[1;32m'      # green
C_WARM=$'\e[1;33m'    # yellow
C_HOT=$'\e[1;31m'     # red
C_RESET=$'\e[0m'

C_BOLD=$'\033[1m'

# Bar background: “12th” color – you already settled on #454545
C_BAR_BG=$'\e[38;5;11m'

# Logo color: *this* is the one that makes the banner colorful.
# Use header cyan by default; change to C_LABEL if you want gold.
: "${C_LOGO:=$C_HDR}"

# --- optional watch mode: swapfetch --watch [seconds] ---
if [[ "${SWAPFETCH_SKIP_WATCH:-0}" != "1" ]]; then
  case "${1-}" in
    -w|--watch)
      shift
      interval="${1-5}"
      # default to 5 if not a simple integer
      [[ "$interval" =~ ^[0-9]+$ ]] || interval=5

      while :; do
        clear
        printf "swapfetch (every %ss) - %s\n\n" "$interval" "$(date '+%H:%M:%S')"
        SWAPFETCH_SKIP_WATCH=1 "$0"
        sleep "$interval"
      done
      exit 0
      ;;
  esac
fi

print_header() {
  printf '%b%s%b\n' "$C_HDR" "$1" "$C_RESET"
}

gauge_color_for() {
  local pct=$1
  if   (( pct >= 90 )); then printf '%s' "$C_HOT"
  elif (( pct >= 70 )); then printf '%s' "$C_WARM"
  else                       printf '%s' "$C_OK"
  fi
}

gauge_bar() {
  local pct=$1
  local width=${2:-24}
  local col="${3:-$C_OK}"      # filled color
  local bg="${4:-$C_BAR_BG}"   # empty + right cap color

  ((pct < 0))   && pct=0
  ((pct > 100)) && pct=100

  local filled=$(( (pct * width + 50) / 100 ))
  ((filled > width)) && filled=$width

  local filled_str="" empty_str=""
  local i
  for ((i=1; i<=filled; i++)); do
    filled_str+="█"
  done
  for ((; i<=width; i++)); do
    empty_str+="█"
  done

  # left cap + filled in severity color; empty + right cap in bar-bg color
  printf '%b' "${col}${filled_str}${bg}${empty_str}${C_RESET}"
}

# ---------- ZRAM (zram0) ----------
if [ -r /sys/block/zram0/mm_stat ]; then
  print_header "$ICON_ZRAM  ZRAM (zram0)"

  eval "$(
    awk '{
      og = $1/1024/1024/1024;   # orig_data
      cg = $2/1024/1024/1024;   # compr_data
      mg = $3/1024/1024/1024;   # mem_used_total
      lg = $4/1024/1024/1024;   # mem_limit
      xg = $5/1024/1024/1024;   # max_used
      ratio = ($2 > 0 ? $1 / $2 : 0);
      printf "Z_ORIG=%.2f\n",  og;
      printf "Z_COMPR=%.2f\n", cg;
      printf "Z_MEM=%.2f\n",   mg;
      printf "Z_LIMIT=%.2f\n", lg;
      printf "Z_MAX=%.2f\n",   xg;
      printf "Z_RATIO=%.2f\n", ratio;
    }' /sys/block/zram0/mm_stat
  )"

  # Big ASCII logo (no awk, no escaping hell)
  mapfile -t ZLOGO <<'EOF'
 ____ _ __ __ _ _ __ _ _     ___ __      __ _   _ __
|_  /| '__/ _` | '_ ` _ \ _ / __|\ \ /\ / / _` | '_ \
 / / | | | (_| | | | | | |_|\__ \ \ V  V / (_| | |_) |
/___||_|  \__,_|_| |_| |_|  |___/  \_/\_/ \__,_| .__/
                                                  |_|
EOF

  printf "  %sorig_data%s   : %s%.2f GiB%s  %s%s%s\n" \
    "$C_LABEL" "$C_RESET" "$C_VAL" "$Z_ORIG"  "$C_RESET" "$C_LOGO" "${ZLOGO[0]}" "$C_RESET"

  printf "  %scompr_data%s  : %s%.2f GiB%s  %s%s%s\n" \
    "$C_LABEL" "$C_RESET" "$C_VAL" "$Z_COMPR" "$C_RESET" "$C_LOGO" "${ZLOGO[1]}" "$C_RESET"

  printf "  %smem_used%s    : %s%.2f GiB%s  %s%s%s\n" \
    "$C_LABEL" "$C_RESET" "$C_VAL" "$Z_MEM"   "$C_RESET" "$C_LOGO" "${ZLOGO[2]}" "$C_RESET"

  printf "  %smax_used%s    : %s%.2f GiB%s  %s%s%s\n" \
    "$C_LABEL" "$C_RESET" "$C_VAL" "$Z_MAX"   "$C_RESET" "$C_LOGO" "${ZLOGO[3]}" "$C_RESET"

  if [[ "${Z_LIMIT%.*}" -gt 0 ]] 2>/dev/null; then
    printf "  %smem_limit%s   : %s%.2f GiB%s\n" \
      "$C_LABEL" "$C_RESET" "$C_VAL" "$Z_LIMIT" "$C_RESET"
  fi

  printf "  %scompression%s : %s%.2fx%s  %s%s%s\n\n" \
    "$C_LABEL" "$C_RESET" "$C_VAL" "$Z_RATIO" "$C_RESET" \
    "$C_LOGO" "${ZLOGO[4]}" "$C_RESET"
else
  # No ZRAM — show swap-only banner
  mapfile -t SLOGO <<'EOF'
                            __     _       _
 _____      ____ _ _ __   / _| ___| |_ ___| |__
/ __\ \ /\ / / _` | '_ \ | |_ / _ \ __/ __| '_ \
\__ \\ V  V / (_| | |_) ||  _|  __/ || (__| | | |
|___/ \_/\_/ \__,_| .__/ |_|  \___|\__\___|_| |_|
                  |_|
EOF
  for line in "${SLOGO[@]}"; do
    printf "  %s%s%s\n" "$C_LOGO" "$line" "$C_RESET"
  done
  echo
fi

# --- Swap devices ---------------------------------------------------------
print_header "$ICON_SWAP  Swap devices"

swapon --show=NAME,SIZE,USED,PRIO --bytes 2>/dev/null | \
  awk 'NR==1 {next} {print $1, $2, $3, $4}' | \
  while read -r name size used prio; do
    [ -z "${name:-}" ] && continue
    [ "${size:-0}" -eq 0 ] && continue

    # percentage (rounded)
    pct=$(( (used * 100 + size/2) / size ))
    ((pct < 0))   && pct=0
    ((pct > 100)) && pct=100

    col="$(gauge_color_for "$pct")"
    bar="$(gauge_bar "$pct" 24 "$col")"   # bg defaults to C_BAR_BG now

    # human-readable GiB
    read -r used_gib size_gib <<EOF
$(awk -v u="$used" -v s="$size" 'BEGIN {printf "%.1f %.1f\n", u/1024/1024/1024, s/1024/1024/1024}')
EOF

    printf "  %s%-16s%s %s %s%5.1f/%-5.1f GiB%s (%3d%%) prio %d\n" \
      "$C_LABEL" "$name" "$C_RESET" \
      "$bar" \
      "$C_VAL" "$used_gib" "$size_gib" "$C_RESET" \
      "$pct" "$prio"
  done

echo

# --- Memory summary -------------------------------------------------------
print_header "$ICON_MEM  Memory"

# Raw bytes for math
eval "$(free -b | awk '
/^Mem:/  { printf "MEM_TOTAL=%s MEM_USED=%s MEM_FREE=%s MEM_AVAIL=%s\n", $2,$3,$4,$7 }
/^Swap:/ { printf "SWAP_TOTAL=%s SWAP_USED=%s SWAP_FREE=%s\n", $2,$3,$4 }
')"

mem_pct=0
swap_pct=0
if [[ ${MEM_TOTAL:-0} -gt 0 ]]; then
  mem_pct=$(( (MEM_USED * 100 + MEM_TOTAL/2) / MEM_TOTAL ))
fi
if [[ ${SWAP_TOTAL:-0} -gt 0 ]]; then
  swap_pct=$(( (SWAP_USED * 100 + SWAP_TOTAL/2) / SWAP_TOTAL ))
fi

mem_col=$(gauge_color_for "$mem_pct")
swap_col=$(gauge_color_for "$swap_pct")
mem_bar=$(gauge_bar "$mem_pct" 30 "$mem_col")
swap_bar=$(gauge_bar "$swap_pct" 30 "$swap_col")

# Pretty numbers from free -h
free -h | awk -v label="$C_LABEL" -v val="$C_VAL" -v reset="$C_RESET" '
NR==2 {
  printf "  %sMem %s total=%s%s%s used=%s%s%s free=%s%s%s avail=%s%s%s\n",
         label, reset,
         val, $2, reset,
         val, $3, reset,
         val, $4, reset,
         val, $7, reset;
}
NR==3 {
  printf "  %sSwap%s total=%s%s%s used=%s%s%s free=%s%s%s\n",
         label, reset,
         val, $2, reset,
         val, $3, reset,
         val, $4, reset;
}
'

printf "  %sMem use%s   %s %s%3d%%%s\n" \
  "$C_LABEL" "$C_RESET" "$mem_bar" "$C_VAL" "$mem_pct" "$C_RESET"
printf "  %sSwap use%s  %s %s%3d%%%s\n" \
  "$C_LABEL" "$C_RESET" "$swap_bar" "$C_VAL" "$swap_pct" "$C_RESET"

#+END_SRC

* Swap tools
** swaptop – top by VmSwap (btop-ish)
#+BEGIN_SRC bash :tangle linux/.local/bin/swaptop :shebang "#!/usr/bin/env bash"
set -euo pipefail

interval="${1:-5}"

# === Colors – keep in sync with swapfetch ===

# Nerd Font style: same palette as swapfetch
C_HDR=$'\e[1;36m'     # cyan bold (headers)
C_LABEL=$'\e[1;33m'   # yellow bold (labels)
C_VAL=$'\e[1;37m'     # white bold  (values)
C_DIM=$'\e[2;37m'     # dim white   (background / secondary)
C_OK=$'\e[1;32m'      # green
C_WARM=$'\e[1;33m'    # yellow
C_HOT=$'\e[1;31m'     # red
C_RESET=$'\e[0m'
C_BOLD=$'\033[1m'

# Bar background: “12th” color – aligned with swapfetch’s style
C_BAR_BG=$'\e[38;5;11m'

# Logo/header color
: "${C_LOGO:=$C_HDR}"

gauge_color_for() {
  local pct=$1
  if   (( pct >= 90 )); then printf '%s' "$C_HOT"
  elif (( pct >= 70 )); then printf '%s' "$C_WARM"
  else                       printf '%s' "$C_OK"
  fi
}

gauge_bar() {
  local pct=$1
  local width=${2:-24}
  local col="${3:-$C_OK}"      # filled color
  local bg="${4:-$C_BAR_BG}"   # unfilled “background”

  ((pct < 0))   && pct=0
  ((pct > 100)) && pct=100

  local filled=$(( (pct * width + 50) / 100 ))
  ((filled > width)) && filled=$width

  local filled_str="" empty_str=""
  local i
  for ((i=1; i<=filled; i++)); do
    filled_str+="█"
  done
  for ((; i<=width; i++)); do
    empty_str+="█"
  done

  printf '%b' "${col}${filled_str}${bg}${empty_str}${C_RESET}"
}

# Use tput if available so we can move the cursor without nuking scrollback
HAS_TPUT=0
if command -v tput >/dev/null 2>&1; then
  HAS_TPUT=1
  tput civis 2>/dev/null || true
  trap 'tput cnorm 2>/dev/null || true; exit' INT TERM
fi

# How many process rows we printed last time
LAST_ROWS=0

print_row() {
  local idx="$1"
  local line pid kb pct col bar mib cmdline

  line=${_rows[idx]}
  set -- $line
  pid=$1
  kb=$2

  [[ -z "$pid" || -z "$kb" ]] && return 0

  # percent of total swap
  pct=$(( (kb * 100 + local_total_swap_kb/2) / local_total_swap_kb ))
  ((pct < 0))   && pct=0
  ((pct > 100)) && pct=100

  col="$(gauge_color_for "$pct")"
  bar="$(gauge_bar "$pct" 24 "$col" "$C_BAR_BG")"

  # human readable MiB
  mib=$(awk -v k="$kb" 'BEGIN {printf "%.1f", k/1024}')

  # command line (shortened single-line)
  cmdline=$(ps -p "$pid" -o comm=,args= 2>/dev/null || true)
  cmdline=${cmdline# }
  cmdline=${cmdline//$'\n'/ }
  cmdline=${cmdline:0:60}

  printf "%7s %7s MiB %3d%% %s %s\n" \
    "$pid" "$mib" "$pct" "$bar" "$cmdline"
}

while :; do
  # Terminal height (only if tput exists)
  if (( HAS_TPUT )); then
    term_lines=$(tput lines 2>/dev/null || echo 24)
  else
    term_lines=24
  fi

  # Total swap in KiB
  local_total_swap_kb=$(awk '/^SwapTotal:/ {print $2}' /proc/meminfo)
  (( local_total_swap_kb == 0 )) && local_total_swap_kb=1

  # Collect processes sorted by VmSwap DESC
  mapfile -t _rows < <(
    awk '/^VmSwap:/ {print FILENAME, $2}' /proc/[0-9]*/status 2>/dev/null \
      | sed 's#/proc/\([0-9]*\)/status#\1#' \
      | sort -k2,2nr
  )

  rows_len=${#_rows[@]}
  if (( rows_len == 0 )); then
    current_rows=0
  else
    # 1 title + 1 header + 1 separator + 1 breathing line = 4
    max_rows=$((term_lines - 4))
    ((max_rows < 1)) && max_rows=1
    current_rows=$rows_len
    ((current_rows > max_rows)) && current_rows=$max_rows
  fi

  if (( HAS_TPUT )); then
    # --- Header lines (each one: move, clear line, print) ---

    tput cup 0 0 2>/dev/null
    printf '\033[K'
    printf '%b%s%b  (interval %ss)\n' \
      "$C_HDR" "swaptop – top VmSwap users" "$C_RESET" "$interval"

    tput cup 1 0 2>/dev/null
    printf '\033[K'
    printf '%b%-7s %-8s %-8s %-5s %-s%b\n' \
      "$C_LABEL" "PID" "SWAP" "PCT" "BAR" "COMMAND" "$C_RESET"

    tput cup 2 0 2>/dev/null
    printf '\033[K'
    printf '%s\n' "--------------------------------------------------------------------------"

    if (( current_rows == 0 )); then
      # No rows: print a single status line
      tput cup 3 0 2>/dev/null
      printf '\033[K'
      printf "   (no processes with VmSwap)\n"
    else
      # Print each row at a fixed line, clearing only that line
      for ((idx=0; idx<current_rows; idx++)); do
        line_no=$((3 + idx))
        tput cup "$line_no" 0 2>/dev/null
        printf '\033[K'
        print_row "$idx"
      done
    fi

    # If we had more rows last time, blank the leftovers
    if (( LAST_ROWS > current_rows )); then
      for ((idx=current_rows; idx<LAST_ROWS; idx++)); do
        line_no=$((3 + idx))
        tput cup "$line_no" 0 2>/dev/null
        printf '\033[K'
      done
    fi

  else
    # Fallback: dumb terminal – just print a fresh block each tick
    printf '%b%s%b  (interval %ss)\n' \
      "$C_HDR" "swaptop – top VmSwap users" "$C_RESET" "$interval"
    printf '%b%-7s %-8s %-8s %-5s %-s%b\n' \
      "$C_LABEL" "PID" "SWAP" "PCT" "BAR" "COMMAND" "$C_RESET"
    printf '%s\n' "--------------------------------------------------------------------------"

    if (( current_rows == 0 )); then
      printf "   (no processes with VmSwap)\n"
    else
      for ((idx=0; idx<current_rows; idx++)); do
        print_row "$idx"
      done
    fi
  fi

  LAST_ROWS=$current_rows
  sleep "$interval"
done

  printf '[xfce4-refresh] %s\n' "$*"

icon_theme_ready() {
  local icon_theme icon_root
  command -v xfconf-query >/dev/null 2>&1 || return 0
  icon_theme="$(xfconf-query -c xsettings -p /Net/IconThemeName 2>/dev/null || true)"
  [ -n "$icon_theme" ] || return 0

    "${XDG_DATA_HOME:-$HOME/.local/share}/icons/$icon_theme" \

       [ -e "$icon_root/status/scalable/image-missing.svg" ]; then

  return 1

kill_if_running() {
  local name="$1"
  if pgrep -x "$name" >/dev/null 2>&1; then
    log "Killing $name…"
    pkill -x "$name" || true
  fi
}

start_with_host_wrap_if_missing() {
  local name="$1"
  if ! pgrep -x "$name" >/dev/null 2>&1; then
    if command -v host-wrap >/dev/null 2>&1; then
      log "Starting $name via host-wrap…"
      nohup host-wrap "$name" >/dev/null 2>&1 &
    else
      log "host-wrap not found; starting $name directly…"
      nohup "$name" >/dev/null 2>&1 &
    fi
  fi
}

log "Refreshing XFCE desktop daemons"

# Refuse to enter the kill phase when GTK cannot render the panel's mandatory
# fallback icon. A failed panel restart otherwise leaves the whole desktop
# looking dead while the terminal that launched this helper survives.
if ! icon_theme_ready; then
  log "Aborting before stopping desktop daemons; run loom stow:cortex first"
  exit 1
fi

if pgrep -x xfce4-screensav >/dev/null 2>&1 || pgrep -f /usr/bin/xfce4-screensaver >/dev/null 2>&1; then
  log "Killing xfce4-screensaver..."
  pkill -x xfce4-screensav || pkill -f /usr/bin/xfce4-screensaver || true
fi

# 2) Clipman: kill + restart (via host-wrap)
kill_if_running xfce4-clipman
start_with_host_wrap_if_missing xfce4-clipman

# 3) Desktop: kill + restart (via host-wrap)
kill_if_running xfdesktop
start_with_host_wrap_if_missing xfdesktop

# 4) Notifications: kill + restart (via host-wrap)
if command -v xfce4-notifyd >/dev/null 2>&1; then
  kill_if_running xfce4-notifyd
  start_with_host_wrap_if_missing xfce4-notifyd
fi

# 5) xfrun4 - kill if it is hanging around - do not restart
kill_if_running xfrun4

# 6) Panel + settings: delegate to canonical healer
if command -v xfce4-heal >/dev/null 2>&1; then
  log "Delegating panel + settings restart to xfce4-heal…"
  xfce4-heal
elif command -v xfce4-panel >/dev/null 2>&1; then
  # Fallback path if xfce4-heal is missing for some reason
  if pgrep -x xfce4-panel >/dev/null 2>&1; then
    log "xfce4-heal missing; trying xfce4-panel -r fallback…"
    if ! xfce4-panel -r >/dev/null 2>&1; then

      kill_if_running xfce4-panel
      start_with_host_wrap_if_missing xfce4-panel
    fi
  else
    log "xfce4-heal missing; xfce4-panel not running, starting fresh via host-wrap…"
    start_with_host_wrap_if_missing xfce4-panel
  fi
fi

log "Done."

** xfce4-refresh-guardian – early panel refresh on high swap + panel revive
#+BEGIN_SRC bash :tangle x230/.local/bin/xfce4-refresh-guardian :shebang "#!/bin/env bash"
#!/bin/env bash
# xfce4-refresh-guardian – early refresh trigger based on swap
# and “is the panel actually visible?”

set -euo pipefail

SWAP_LIMIT="${SWAP_LIMIT:-80}"

DO_RESTART=0
if [[ "${1:-}" == "--restart" || "${1:-}" == "-r" ]]; then
  DO_RESTART=1
fi

# --- helper: is the xfce4-panel actually alive (proc + window)? ---
panel_is_alive=0

if pgrep -x xfce4-panel >/dev/null 2>&1; then
  # We have a process – now check for a real panel window if wmctrl exists.
  if command -v wmctrl >/dev/null 2>&1; then
    if wmctrl -lx 2>/dev/null | grep -qi "xfce4-panel.Xfce4-panel"; then
      panel_is_alive=1
    else
      printf '[xfce4-refresh-guardian] xfce4-panel process found, but no panel window.\n'
    fi
  else
    # No wmctrl? fall back to trusting the process
    panel_is_alive=1
  fi
else
  printf '[xfce4-refresh-guardian] No xfce4-panel process.\n'
fi

# If the panel is missing (no process OR no window), healing is top priority.
if (( panel_is_alive == 0 )); then
  printf '[xfce4-refresh-guardian] xfce4-panel is NOT alive, attempting heal via xfce4-heal…\n'
  if command -v xfce4-heal >/dev/null 2>&1; then
    xfce4-heal
    # Give it a moment, then re-check both process + window.
    sleep 1

    panel_is_alive_after=0
    if pgrep -x xfce4-panel >/dev/null 2>&1; then
      if command -v wmctrl >/dev/null 2>&1; then
        if wmctrl -lx 2>/dev/null | grep -qi "xfce4-panel.Xfce4-panel"; then
          panel_is_alive_after=1
        fi
      else
        panel_is_alive_after=1
      fi
    fi

    if (( panel_is_alive_after == 1 )); then
      printf '[xfce4-refresh-guardian] Panel is now running (process + window detected).\n'
    else
      printf '[xfce4-refresh-guardian] Heal attempted, but panel still not detected.\n'
    fi
  else
    printf '[xfce4-refresh-guardian] xfce4-heal not found; cannot restore panel.\n'
  fi

  # Panel health is a higher priority than swap tuning – stop here.
  exit 0
fi

# --- at this point: panel_is_alive == 1, proceed with swap threshold logic ---

# Read swap from /proc/meminfo (KiB)
swap_total_kb=$(awk '/^SwapTotal:/ {print $2}' /proc/meminfo)
swap_free_kb=$(awk '/^SwapFree:/  {print $2}' /proc/meminfo)

if (( swap_total_kb == 0 )); then
  SWAP_PCT=0
else
  swap_used_kb=$((swap_total_kb - swap_free_kb))
  SWAP_PCT=$(( (swap_used_kb * 100 + swap_total_kb/2) / swap_total_kb ))
fi

printf '[xfce4-refresh-guardian] THRESHOLD: SWAP=%d%% (limit %d%%)\n' \
  "$SWAP_PCT" "$SWAP_LIMIT"

if (( SWAP_PCT >= SWAP_LIMIT )); then
  if (( DO_RESTART )); then
    if command -v xfce4-refresh >/dev/null 2>&1; then
      printf '[xfce4-refresh-guardian] Triggering xfce4-refresh (early recycle)…\n'
      xfce4-refresh
    else
      printf '[xfce4-refresh-guardian] xfce4-refresh not found, nothing to do.\n'
    fi
  else
    printf '[xfce4-refresh-guardian] Would run xfce4-refresh (dry run).\n'
  fi
else
  printf '[xfce4-refresh-guardian] Below threshold, nothing to do.\n'
fi
#+END_SRC

  fi

** desktop-heal - DE-aware desktop repair entry point

detect_desktop() {
  local current="${XDG_CURRENT_DESKTOP:-}"

     pgrep -x ksmserver >/dev/null 2>&1 ||

     [[ "$session" == xfce* ]]; then

  printf '%s\n' unknown

  fi

  fi

  if command -v kbuildsycoca6 >/dev/null 2>&1; then
    log "Rebuilding KDE service cache..."
    kbuildsycoca6 --noincremental >/dev/null 2>&1 || true
  elif command -v kbuildsycoca5 >/dev/null 2>&1; then
    log "Rebuilding KDE service cache..."
    kbuildsycoca5 --noincremental >/dev/null 2>&1 || true
  fi

  local x11_session=0
  if [[ "${XDG_SESSION_TYPE:-}" == x11 ]]; then
    x11_session=1

    x11_session=1
  fi

      log "Restoring kwin_x11 compositor..."
      nohup env DISPLAY="$DISPLAY" DBUS_SESSION_BUS_ADDRESS="$DBUS_SESSION_BUS_ADDRESS" XDG_RUNTIME_DIR="$XDG_RUNTIME_DIR" XDG_DATA_DIRS="$XDG_DATA_DIRS" PATH="$PATH" GUIX_PROFILE="${GUIX_PROFILE:-}" kwin_x11 --replace >/tmp/dotcortex-kwin_x11.log 2>&1 &

      log "kwin_x11 not found in PATH."
    fi

  fi

  if pgrep -x krunner >/dev/null 2>&1; then
    log "Restarting krunner..."
    kquitapp6 krunner >/dev/null 2>&1 || pkill -x krunner >/dev/null 2>&1 || true
    sleep 0.5
  fi
  if command -v krunner >/dev/null 2>&1; then
    nohup env DISPLAY="$DISPLAY" DBUS_SESSION_BUS_ADDRESS="$DBUS_SESSION_BUS_ADDRESS" XDG_RUNTIME_DIR="$XDG_RUNTIME_DIR" XDG_DATA_DIRS="$XDG_DATA_DIRS" PATH="$PATH" GUIX_PROFILE="${GUIX_PROFILE:-}" krunner --daemon >/tmp/dotcortex-krunner.log 2>&1 &
  fi

  log "Starting plasmashell..."
  if command -v plasmashell >/dev/null 2>&1; then
    if pgrep -x plasmashell >/dev/null 2>&1; then
      nohup env DISPLAY="$DISPLAY" DBUS_SESSION_BUS_ADDRESS="$DBUS_SESSION_BUS_ADDRESS" XDG_RUNTIME_DIR="$XDG_RUNTIME_DIR" XDG_DATA_DIRS="$XDG_DATA_DIRS" PATH="$PATH" GUIX_PROFILE="${GUIX_PROFILE:-}" plasmashell --replace >/tmp/dotcortex-plasmashell.log 2>&1 &

    fi
  else
    log "plasmashell not found in PATH."
  fi

  sleep 2
  pgrep -af 'plasmashell|krunner|kwin_x11|ksmserver' || true
}

** swap-heal – safe swap purge + desktop/Flatpak heal
#+BEGIN_SRC bash :tangle x230/.local/bin/swap-heal :shebang "#!/bin/env bash"
#!/bin/env bash
set -euo pipefail

log() { printf '[swap-heal] %s\n' "$*"; }

# 1) Inspect RAM + swap
mem_available_kb=$(awk '/^MemAvailable:/ {print $2}' /proc/meminfo)
swap_total_kb=$(awk '/^SwapTotal:/     {print $2}' /proc/meminfo)
swap_free_kb=$(awk '/^SwapFree:/      {print $2}' /proc/meminfo)
swap_used_kb=$((swap_total_kb - swap_free_kb))

log "MemAvailable: ${mem_available_kb} KiB"
log "Swap used:    ${swap_used_kb} KiB"

if (( swap_total_kb == 0 )); then
  log "No swap configured, nothing to do."
  exit 0
fi

# Simple safety rule: need at least as much free RAM as swap in use
needed_kb=${swap_used_kb}

if (( mem_available_kb < needed_kb )) && [[ "${SWAP_HEAL_FORCE:-0}" != 1 ]]; then
  log "Not enough free RAM to safely clear swap (need ${needed_kb} KiB)."
  log "Set SWAP_HEAL_FORCE=1 if you really want to override this (risk of OOM)."
  exit 1
fi

if [[ "${SWAP_HEAL_FORCE:-0}" == 1 ]]; then
  log "SWAP_HEAL_FORCE=1 set, ignoring RAM safety check. This may OOM the system."
else
  log "Enough RAM available, proceeding with swap purge."
fi

# 2) Drop caches
log "Dropping caches (vm.drop_caches=3)…"
if ! sudo sh -c 'echo 3 > /proc/sys/vm/drop_caches'; then
  log "WARNING: could not write vm.drop_caches; continuing anyway."
fi

# 3) swapoff + zramswap restart + swapon
log "swapoff -a…"
sudo swapoff -a

log "Restarting zramswap.service if present…"
if ! sudo systemctl restart zramswap.service 2>/dev/null; then
  log "zramswap.service restart failed or unit not present; continuing."
fi

log "swapon -a…"
sudo swapon -a

# Make sure zram actually ends up as swap if it exists
if [ -e /dev/zram0 ]; then
  disksize=$(cat /sys/block/zram0/disksize 2>/dev/null || echo 0)
  if [ "$disksize" -gt 0 ] && ! grep -q '^/dev/zram0' /proc/swaps; then
    log "Enabling zram swap on /dev/zram0…"
    if ! sudo swapon /dev/zram0 2>/dev/null; then
      log "WARNING: swapon /dev/zram0 failed; zram still not active."
    fi
  else
    log "zram0 already active or no disksize set, skipping swapon."
  fi
else
  log "/dev/zram0 not present; nothing to enable."
fi

# 4) Heal XFCE desktop
if command -v xfce4-heal >/dev/null 2>&1; then
  log "Running xfce4-heal…"
  xfce4-heal
else
  log "xfce4-heal not found; skipping."
fi

# 5) Heal Flatpak bridge + fontcache
if command -v flatpak-desktop-bridge >/dev/null 2>&1; then
  log "Refreshing Flatpak desktop bridge…"
  flatpak-desktop-bridge
fi

if command -v flatpak-fontcache-refresh >/dev/null 2>&1; then
  log "Refreshing Flatpak font cache…"
  flatpak-fontcache-refresh
fi

log "swap-heal ritual complete."
#+END_SRC

It is started by =FORGE/bin/recall embedder start=, which is idempotent (it exits 0 if the PID file names a live process) and detaches via =start_new_session=, so it survives the autostart shell exiting.

*Bind ruling (2026-07-28):* the spawn binds =0.0.0.0= (set via =bind_host= in =ROOTS/RAG/recall.toml=), not loopback — until the T1700 takes over, ALL fleet inference routes through the T480s, so the X230's ingest and daemon must be able to reach this embedder over the tailnet (=t480s= node in =ROOTS/inference/nodes.toml=).  The X230 itself runs no inference or compilation duty, ever — it is the fileserver.

Both =FORGE/bin/recall= and the model file live under =$HOME/mnt/x230=, which is an NFS mount from the X230, so the entry cannot assume the tree is present at login. * Lid Switch Fix — T480s (elogind + acpid)
T480s runs SLiM + elogind + xfce4-power-manager. By default
xfce4-power-manager sets =logind-handle-lid-switch=false=, meaning it tries
to handle the lid switch itself over a D-Bus call to
=org.freedesktop.login1=. That connection has been observed to die
(=xfce4-power-manager: Error: The connection is closed= in
=~/.xsession-errors=) and silently fail every lid-close action afterward —
screen stays lit all night, sometimes with a "couldn't do it" popup. SLiM
also leaves the graphical session mis-classified (=Class=greeter=) since it
predates logind session-class transitions, and its lack of a clean
greeter→user handoff correlates with zombie sessions accumulating in
=State=closing= in =loginctl list-sessions=.

First pass: let elogind own the lid switch natively via a config drop-in
(=HandleLidSwitch=suspend=) and flip =logind-handle-lid-switch= to =true= in
xfconf so xfce4-power-manager steps out of the way. This fixed the
stuck-on-screen symptom but overshot the actual ask — =suspend= is a real
ACPI suspend-to-RAM, not a display blank. It froze every running process on
lid close (background agents included) and stopped audio playback, both
resuming only on lid open. The only thing actually wanted was the screen
turning off.

Second pass: flip elogind's action to =ignore= (logind takes zero action on
the lid switch — no suspend, no lock, nothing) and give the lid switch
exclusively to acpid, which was already running on this box with no lid
rule installed. acpid's action is DPMS-only (=xset dpms force off/on=
against the real Xorg server), which blanks the monitor without touching
any process. This does not double up with elogind: acpid listens on the
ACPI netlink event stream independently of whatever evdev handling logind
does, so both can hold a lid rule at once — the fix is which *action* each
one takes on it (elogind: none; acpid: DPMS blank), not deciding which
daemon gets the event.

** logind drop-in
#+BEGIN_SRC conf :tangle t480s/.local/share/dotcortex/power/logind-lid.conf :padline no :comments none
[Login]
HandleLidSwitch=ignore
HandleLidSwitchExternalPower=ignore
#+END_SRC

** acpid lid event
This box's acpid keeps action scripts flat in =/etc/acpi/= (alongside the
stock =powerbtn-acpi-support.sh=), not under an =/etc/acpi/actions/=

#+BEGIN_SRC conf :tangle t480s/.local/share/dotcortex/power/acpi-lid.event :padline no :comments none

** acpid lid action — DPMS blank/unblank only
Reads the authoritative lid state from =/proc/acpi/button/lid= rather than
trusting the acpi event string (its close/open payload format is not
consistent across kernel versions). Runs as root (acpid's own user), so it
talks to the real Xorg server via SLiM's own auth cookie rather than any
per-user =~/.Xauthority= — this keeps it independent of which desktop
session (host or a nested sanctuary Xephyr/Xnest one) happens to be
focused.
#+BEGIN_SRC bash :tangle t480s/.local/share/dotcortex/power/dotcortex-lid-dpms.sh :shebang "#!/bin/sh" :tangle-mode (identity #o755)
export DISPLAY=:0
export XAUTHORITY=/var/run/slim.auth

state="$(cat /proc/acpi/button/lid/*/state 2>/dev/null | awk '{print $NF}')"
logger -t dotcortex-lid "invoked, lid state=${state:-unknown}"

case "$state" in
  closed) xset dpms force off; logger -t dotcortex-lid "dpms force off rc=$?" ;;
  open)   xset dpms force on;  logger -t dotcortex-lid "dpms force on rc=$?" ;;
  *)      logger -t dotcortex-lid "unrecognized lid state '$state', no DPMS action taken" ;;
esac
#+END_SRC

** Setup script
Idempotent — safe to re-run. Installs the elogind drop-in and the acpid lid
event+action pair, restarts elogind and acpid, sets the xfconf key so
xfce4-power-manager stops fighting for the lid switch, and clears any
zombie =closing= sessions left over from prior SLiM/elogind session-handoff
hiccups.

#+BEGIN_SRC bash :tangle t480s/.local/bin/t480s-lid-fix-setup :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

POWER_DIR="${HOME}/.local/share/dotcortex/power"
LOGIND_SRC="${POWER_DIR}/logind-lid.conf"
LOGIND_DEST=/etc/elogind/logind.conf.d/dotcortex-lid.conf
ACPI_EVENT_SRC="${POWER_DIR}/acpi-lid.event"
ACPI_EVENT_DEST=/etc/acpi/events/dotcortex-lid
ACPI_ACTION_SRC="${POWER_DIR}/dotcortex-lid-dpms.sh"
ACPI_ACTION_DEST=/etc/acpi/dotcortex-lid-dpms.sh

die() { printf 't480s-lid-fix-setup: %s\n' "$*" >&2; exit 1; }

for f in "$LOGIND_SRC" "$ACPI_EVENT_SRC" "$ACPI_ACTION_SRC"; do
  [ -f "$f" ] || die "missing $f — run: tangle-one services.org && loom stow:t480s"
done

echo "[1/5] elogind drop-in (lid switch: ignore)..."
sudo install -d -m 755 /etc/elogind/logind.conf.d
sudo install -m 644 "$LOGIND_SRC" "$LOGIND_DEST"

echo "[2/5] acpid lid event + DPMS-only action..."
sudo install -m 644 "$ACPI_EVENT_SRC" "$ACPI_EVENT_DEST"
sudo install -m 755 "$ACPI_ACTION_SRC" "$ACPI_ACTION_DEST"

echo "[3/5] xfconf: step xfce4-power-manager out of the lid switch..."
XFCE_PM_PID="$(pgrep -u "$USER" -f '(^|/)xfce4-power-manager$' | head -n1 || true)"
if [ -n "$XFCE_PM_PID" ] && [ -r "/proc/${XFCE_PM_PID}/environ" ]; then
  DBUS_SESSION_BUS_ADDRESS="$(tr '\0' '\n' < "/proc/${XFCE_PM_PID}/environ" | sed -n 's/^DBUS_SESSION_BUS_ADDRESS=//p')"
  export DBUS_SESSION_BUS_ADDRESS
  xfconf-query -c xfce4-power-manager -p /xfce4-power-manager/logind-handle-lid-switch -n -t bool -s true 2>/dev/null \
    || xfconf-query -c xfce4-power-manager -p /xfce4-power-manager/logind-handle-lid-switch -s true
else
  echo "no running xfce4-power-manager for this user — nothing to step out of the way (fine on an IceWM-only session)"
fi

echo "[4/5] clearing zombie 'closing' sessions..."
loginctl list-sessions --no-legend | awk '$6=="closing"{print $1}' | while read -r sid; do
  sudo loginctl terminate-session "$sid" 2>/dev/null || true
done

echo "[5/5] restarting elogind and acpid..."
sudo service elogind restart
sudo service acpid restart

echo
echo "=== T480s lid-switch fix applied ==="
echo "Test by physically closing the lid — this is the only real verification."
echo "Expect: screen off within ~1s, everything (audio, background agents) keeps running, screen back on when reopened."
#+END_SRC
