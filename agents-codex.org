* Codex live config apply
Codex keeps credentials, state, marketplace cache paths, and machine-local
project trust entries under =~/.codex=. DotCortex should not own that whole
tree. This helper patches only the declarative TUI status-line list and the
single DotCortex stow-target hook in the live private config. The hook source
itself is public and tangled from =agents-hooks.org=; only the private
=~/.codex/hooks.json= registration is applied here.

Codex's =PreToolUse= hook sees Bash and unified =exec_command= calls. It runs
the same target guard as Claude Code and =safe-stow=, so a mismatched
=loom stow:<machine>= is denied before the shell starts. Codex requires changed
command hooks to be reviewed/trusted in =/hooks= before they execute.

#+BEGIN_SRC bash :tangle all/.local/bin/codex-config-apply :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

config="${CODEX_CONFIG:-$HOME/.codex/config.toml}"
desired='status_line = ["model-with-reasoning", "current-dir", "context-used", "five-hour-limit", "weekly-limit", "context-window-size"]'

apply_codex_stow_hook() {
  local hooks_config guard_command hooks_input hooks_new hooks_backup
  hooks_config="${CODEX_HOOKS:-$HOME/.codex/hooks.json}"
  guard_command="${DOTCORTEX_STOW_GUARD:-$HOME/.local/bin/dotcortex-stow-target-guard}"
  [ -x "$guard_command" ] || guard_command="$HOME/DotCortex/all/.local/bin/dotcortex-stow-target-guard"
  [ -x "$guard_command" ] || {
    echo "codex-config-apply: missing dotcortex-stow-target-guard" >&2
    return 1
  }
  command -v jq >/dev/null 2>&1 || {
    echo "codex-config-apply: jq is required to update $hooks_config" >&2
    return 1
  }

  mkdir -p "$(dirname "$hooks_config")"
  hooks_input="$(mktemp)"
  hooks_new="$(mktemp)"
  if [ -f "$hooks_config" ]; then
    cp -f "$hooks_config" "$hooks_input"
  else
    printf '{"hooks":{}}\n' >"$hooks_input"
  fi

  jq --arg command "$guard_command" '
    .hooks = (.hooks // {}) |
    .hooks.PreToolUse = (.hooks.PreToolUse // []) |
    if any(.hooks.PreToolUse[]?;
      ((.matcher // "") == "^Bash$" or (.matcher // "") == "Bash") and
      any(.hooks[]?; .type == "command" and .command == ($command + " --codex-json")))
    then .
    else .hooks.PreToolUse += [{
      "matcher": "^Bash$",
      "hooks": [{
        "type": "command",
        "command": $command + " --codex-json",
        "timeout": 30,
        "statusMessage": "Checking DotCortex stow target"
      }]
    }]
    end
  ' "$hooks_input" >"$hooks_new"

  if cmp -s "$hooks_new" "$hooks_config" 2>/dev/null; then
    rm -f "$hooks_input" "$hooks_new"
    echo "codex-config-apply: stow hook already current"
    return 0
  fi
  if [ -f "$hooks_config" ]; then
    hooks_backup="${hooks_config}.backup-$(date +%Y%m%d-%H%M%S)"
    cp -a "$hooks_config" "$hooks_backup"
    echo "codex-config-apply: hook backup $hooks_backup"
  fi
  cat "$hooks_new" >"$hooks_config"
  chmod 600 "$hooks_config"
  rm -f "$hooks_input" "$hooks_new"
  echo "codex-config-apply: installed stow-target PreToolUse hook in $hooks_config"
}

apply_codex_stow_hook

mkdir -p "$(dirname "$config")"

# Codex's workspace-write sandbox only makes the invocation cwd writable. The
# companion pins cwd to the Claude session workspace (e.g. DotCortex), so the
# HelmCortex NFS mount is read-only under Landlock and every HelmCortex build
# fails. Declare HelmCortex as an additional writable root. Landlock resolves
# realpaths, so the actual mount path must be listed, not just the ~ symlink.
[ -f "$config" ] || : >"$config"
if ! grep -q '^\[sandbox_workspace_write\]' "$config"; then
  {
    printf '\n[sandbox_workspace_write]\n'
    printf 'writable_roots = ["%s", "%s"]\n' "$HOME/HelmCortex" "$HOME/mnt/x230/HelmCortex"
  } >>"$config"
  echo "codex-config-apply: added sandbox_workspace_write writable_roots (HelmCortex)"
fi

if [ ! -f "$config" ]; then
  {
    printf '[tui]\n'
    printf '%s\n' "$desired"
  } >"$config"
  chmod 600 "$config"
  echo "codex-config-apply: created $config"
  exit 0
fi

tmp="$(mktemp)"
awk -v desired="$desired" '
  BEGIN {
    in_tui = 0
    seen_tui = 0
    wrote_status = 0
  }

  /^\[[^]]+\]$/ {
    if (in_tui && !wrote_status) {
      print desired
      wrote_status = 1
    }
    in_tui = ($0 == "[tui]")
    if (in_tui) {
      seen_tui = 1
    }
    print
    next
  }

  in_tui && /^[[:space:]]*status_line[[:space:]]*=/ {
    if (!wrote_status) {
      print desired
      wrote_status = 1
    }
    next
  }

  { print }

  END {
    if (!seen_tui) {
      print ""
      print "[tui]"
      print desired
    } else if (in_tui && !wrote_status) {
      print desired
    }
  }
' "$config" >"$tmp"

if cmp -s "$tmp" "$config"; then
  rm -f "$tmp"
  echo "codex-config-apply: already current"
  exit 0
fi

backup="${config}.backup-$(date +%Y%m%d-%H%M%S)"
cp -a "$config" "$backup"
cat "$tmp" >"$config"
rm -f "$tmp"
chmod 600 "$config"
echo "codex-config-apply: updated $config"
echo "codex-config-apply: backup $backup"
#+END_SRC
