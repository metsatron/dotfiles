backup_once() {

  "$KWRITE" --file "$KGLOBAL" --group "$1" --key "$2" "$3"

# Reload kglobalaccel so the shortcuts above are actually GRABBED, not merely written.
# A file write is not a live grab: the running kglobalacceld must re-read the config,
# and it only does so from inside the Plasma session (it needs the session bus). Run
# this from SSH, cron, or an agent subprocess and the file lands while the daemon never
# reloads — so detect an unreachable bus and fail LOUD instead of reporting a false
# success. This exact trap cost a long session: "applied" was reported while a 2-day-old
# kglobalacceld still held the old grabs, and the keys silently never worked.
KGA_DAEMON=/usr/lib/x86_64-linux-gnu/libexec/kglobalacceld
if command -v "$QDBUS" >/dev/null 2>&1 && "$QDBUS" org.kde.kglobalaccel >/dev/null 2>&1; then
  # In-session: bus reachable. Rebuild the service cache (so new .desktop launchers
  # resolve) then bounce kglobalacceld so it re-reads and re-grabs.
  "$QDBUS" org.kde.KWin /KWin org.kde.KWin.reconfigure >/dev/null 2>&1 || true
  command -v kbuildsycoca6 >/dev/null 2>&1 && kbuildsycoca6 >/dev/null 2>&1 || true
  if command -v kquitapp6 >/dev/null 2>&1; then
    kquitapp6 kglobalacceld >/dev/null 2>&1 || true
  else
    "$QDBUS" org.kde.kglobalaccel /MainApplication org.qtproject.Qt.QCoreApplication.quit >/dev/null 2>&1 || true
  fi
  [ -x "$KGA_DAEMON" ] && setsid -f env -u LD_PRELOAD "$KGA_DAEMON" >/dev/null 2>&1 || true
  printf 'Plasma/SonicDE preferences applied and kglobalaccel reloaded.\n'
else
  printf 'Plasma/SonicDE preferences WRITTEN to %s (daemon NOT reloaded).\n' "$KGLOBAL" >&2
  cat >&2 <<'WARN'
WARNING: could not reach the Plasma session bus, so the live kglobalaccel daemon was
NOT reloaded — the new shortcuts are on disk but NOT active yet. This happens when the
script runs outside your session (SSH, cron, an agent subprocess). A config write is
not a live grab. Load them by running INSIDE your Plasma session:

    kbuildsycoca6
    kquitapp6 kglobalacceld
    setsid -f env -u LD_PRELOAD /usr/lib/x86_64-linux-gnu/libexec/kglobalacceld

…or simply log out and back in (the guaranteed clean reload).
WARN
fi

gone in Plasma 6), so each rofi launcher needs a stable =.desktop= target. The
critical part, paid for across a full debugging session: the shortcut must be
declared *in the .desktop itself* via =X-KDE-Shortcuts== — exactly how the
built-in Emojier and Spectacle declare theirs. A hand-written =kglobalshortcutsrc=
=[services][X.desktop] _launch== entry ALONE is ignored for a launcher KDE does
not already know as a shortcut owner: kglobalaccel only grabs shortcuts for
services that declare =X-KDE-Shortcuts= (or were registered live via System
Settings). Symptom: the =_launch= entries were present and the daemon reloaded,
the Emojier displacement even took — yet =Meta+Space= fired nothing, because the
launcher =.desktop=s never declared their own keys. All are =NoDisplay= so they

* Plasmoid Manifest — pinned panel applets
Pinned upstream Plasma 6 applets installed per-user with =kpackagetool6=.
Each row is locked to a tag *and* its resolved commit; =plasmoid-apply=
refuses to install if the tag has moved. Plugin IDs are recorded from each
package's =metadata.json=, and re-verified against the source at apply time.

Update flow: bump TAG, resolve =git ls-remote --tags <upstream>= for the new
commit, update COMMIT, then =loom plasmoid:apply= and =loom plasmoid:reload=.
COMMIT must be the *peeled* hash: for annotated tags use the =^{}= line from
ls-remote, not the tag object hash.

plasma6-window-title-applet org.kde.windowtitle https://github.com/dhruv8sh/plasma6-window-title-applet.git v0.9.0 a6eaf5086a473919ed2fffc5d3b8d98237c2dd41 . plasma-applet
application-title-bar com.github.antroids.application-title-bar https://github.com/antroids/application-title-bar.git v0.9.1 7a1f0ef0d1c7ae789f0a30026ca2115e7a0167fe package plasma-applet

* Plasmoid Apply
Clones/updates each upstream into the shared DotCortex source cache,
checks out the pinned commit, verifies the lock, then installs or upgrades
via =kpackagetool6=. Never restarts plasmashell — use =plasmoid-reload=.

#+BEGIN_SRC bash :tangle linux/.local/bin/plasmoid-apply :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

MANIFEST="${PLASMOID_SSV:-$HOME/.plasmoid/manifest/applets.ssv}"
SRC_ROOT="${PLASMOID_SRC_ROOT:-$HOME/.local/share/dotcortex/src}"
KPKG="${KPKG:-kpackagetool6}"
USER_PLASMOIDS="$HOME/.local/share/plasma/plasmoids"

die() { printf 'plasmoid-apply: %s\n' "$*" >&2; exit 1; }

command -v "$KPKG" >/dev/null 2>&1 || die "missing $KPKG (not a Plasma 6 machine?)"
command -v git >/dev/null 2>&1 || die "missing git"
[ -f "$MANIFEST" ] || die "missing manifest: $MANIFEST"

meta_field() {
  python3 -c 'import json,sys; print(json.load(open(sys.argv[1]))["KPlugin"][sys.argv[2]])' "$1" "$2"
}

mkdir -p "$SRC_ROOT"

while read -r name id upstream tag commit pkgroot type; do
  case "$type" in
    plasma-applet)
      src="$SRC_ROOT/$name"
      echo "== $name $tag ($commit)"
      if [ -d "$src/.git" ]; then
        git -C "$src" fetch --quiet --tags origin
      else
        git clone --quiet "$upstream" "$src"
      fi
      git -C "$src" checkout --quiet --detach "$commit" \
        || die "$name: pinned commit not present in $upstream"
      head="$(git -C "$src" rev-parse HEAD)"
      [ "$head" = "$commit" ] || die "$name: HEAD $head != pinned $commit"
      tag_commit="$(git -C "$src" rev-list -n1 "refs/tags/$tag" 2>/dev/null || true)"
      if [ -n "$tag_commit" ] && [ "$tag_commit" != "$commit" ]; then
        die "$name: tag $tag moved upstream to $tag_commit (pinned $commit) — re-resolve and update the manifest"
      fi
      dir="$src"
      [ "$pkgroot" = "." ] || dir="$src/$pkgroot"
      ;;
    plasma-applet-local)
      dir="${pkgroot/#\~/$HOME}"
      echo "== $name (local) $dir"
      [ -d "$dir" ] || die "$name: missing local package dir $dir — tangle + stow first"
      ;;
    *)
      die "$name: unsupported install_type '$type'"
      ;;
  esac
  meta="$dir/metadata.json"
  [ -f "$meta" ] || die "$name: no metadata.json under $dir"
  real_id="$(meta_field "$meta" Id)"
  [ "$real_id" = "$id" ] || die "$name: metadata Id '$real_id' != manifest ID '$id'"
  want_ver="$(meta_field "$meta" Version)"

  installed_meta="$USER_PLASMOIDS/$id/metadata.json"
  if [ -f "$installed_meta" ]; then
    have_ver="$(meta_field "$installed_meta" Version)"
    if [ "$have_ver" = "$want_ver" ] && [ -z "${PLASMOID_FORCE:-}" ]; then
      echo "   up-to-date: $id $have_ver"
    else
      "$KPKG" --type Plasma/Applet --upgrade "$dir"
      echo "   upgraded: $id $have_ver -> $want_ver"
    fi
  else
    "$KPKG" --type Plasma/Applet --install "$dir"
    echo "   installed: $id $want_ver"
  fi
done < <(grep -vE '^[[:space:]]*(#|$)' "$MANIFEST")

echo
echo "plasmoid-apply: done — reload the shell with 'plasmoid-reload' to pick up changes"
#+END_SRC

* Plasmoid Diff
Reports manifest vs installed state without touching the network.

#+BEGIN_SRC bash :tangle linux/.local/bin/plasmoid-diff :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

MANIFEST="${PLASMOID_SSV:-$HOME/.plasmoid/manifest/applets.ssv}"
USER_PLASMOIDS="$HOME/.local/share/plasma/plasmoids"

[ -f "$MANIFEST" ] || { printf 'plasmoid-diff: missing manifest: %s\n' "$MANIFEST" >&2; exit 1; }

ver_of() {
  python3 -c 'import json,sys; print(json.load(open(sys.argv[1]))["KPlugin"]["Version"])' "$1" 2>/dev/null || printf 'unknown'
}

while read -r name id upstream tag commit pkgroot type; do
  if [ "$type" = "plasma-applet-local" ]; then
    want="$(ver_of "${pkgroot/#\~/$HOME}/metadata.json")"
    pin="local-$want"
  else
    want="${tag#v}"
    pin="$tag"
  fi
  installed_meta="$USER_PLASMOIDS/$id/metadata.json"
  if [ -f "$installed_meta" ]; then
    have="$(ver_of "$installed_meta")"
    if [ "$have" = "$want" ]; then
      printf 'ok        %-50s %s\n' "$id" "$have"
    else
      printf 'outdated  %-50s installed=%s pinned=%s\n' "$id" "$have" "$want"
    fi
  else
    printf 'missing   %-50s pinned=%s\n' "$id" "$pin"
  fi
done < <(grep -vE '^[[:space:]]*(#|$)' "$MANIFEST")
#+END_SRC

* KWin Borderless Maximized Toggle
Opt-in: hide titlebars/borders only when windows are maximized, so the top
panel (window buttons + title + global menu) carries the chrome instead.
Backs up =kwinrc= before every write; reloads KWin via D-Bus reconfigure
(non-destructive — no session restart).

#+BEGIN_SRC bash :tangle linux/.local/bin/kwin-borderless-maximized :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

KREAD="${KREAD:-kreadconfig6}"
KWRITE="${KWRITE:-kwriteconfig6}"
QDBUS="${QDBUS:-qdbus6}"

usage() {
  cat <<'EOF'
usage: kwin-borderless-maximized {status|enable|disable}

Controls [Windows] BorderlessMaximizedWindows in kwinrc.
enable  = maximized windows lose titlebar/borders
disable = normal titlebars everywhere
EOF
  exit "${1:-0}"
}

current() {
  "$KREAD" --file kwinrc --group Windows --key BorderlessMaximizedWindows 2>/dev/null || true
}

reconfigure() {
  if command -v "$QDBUS" >/dev/null 2>&1 && "$QDBUS" org.kde.KWin /KWin org.kde.KWin.reconfigure >/dev/null 2>&1; then
    echo "kwin reconfigured live"
  else
    echo "could not reach KWin over D-Bus — log out/in to apply"
  fi
}

setval() {
  command -v "$KWRITE" >/dev/null 2>&1 || { echo "missing $KWRITE" >&2; exit 1; }
  [ ! -f "$KWINRC" ] || cp "$KWINRC" "$KWINRC.backup-$(date +%Y%m%d-%H%M%S)"
  "$KWRITE" --file kwinrc --group Windows --key BorderlessMaximizedWindows "$1"
  echo "BorderlessMaximizedWindows=$1"
  reconfigure
}

case "${1:-status}" in

    echo "BorderlessMaximizedWindows: ${val:-unset (default: false)}"
    ;;
  enable)  setval true ;;
  disable) setval false ;;

  *) usage 1 ;;
esac

PlasmoidItem {

    property string machineName: ""

    function decodeEntities(s) {
        return s.replace(/&lt;/g, "<").replace(/&gt;/g, ">").replace(/&quot;/g, "\"").replace(/&apos;/g, "'").replace(/&amp;/g, "&");
    }

    function parsePlaces(xml) {
        placesModel.clear();
        const re = /<bookmark href="([^"]*)">([\s\S]*?)<\/bookmark>/g;
        let m;
        while ((m = re.exec(xml)) !== null) {
            const href = decodeEntities(m[1]);
            const body = m[2];
            if (/<isHidden>true<\/isHidden>/.test(body))
                continue;
            if (/<OnlyInApp>[^<]+<\/OnlyInApp>/.test(body))
                continue;
            const tm = body.match(/<title>([^<]*)<\/title>/);
            if (!tm)
                continue;
            const im = body.match(/<bookmark:icon name="([^"]*)"/);
            placesModel.append({
                title: decodeEntities(tm[1]),
                href: href,
                iconName: im ? im[1] : "folder"
            });
        }
    }

    function reloadPlaces() {
        const path = String(StandardPaths.writableLocation(StandardPaths.GenericDataLocation)).replace(/^file:\/\//, "") + "/user-places.xbel";
        placesReader.connectSource("cat " + quoteArg(path));
    }

    function templateIcon(name) {
        const lower = String(name).toLowerCase();
        if (lower.endsWith(".desktop"))
            return "application-x-desktop";
        if (lower.endsWith(".txt") || lower.endsWith(".md") || lower.endsWith(".org"))
            return "text-x-generic";
        if (lower.endsWith(".odt") || lower.endsWith(".doc") || lower.endsWith(".docx"))
            return "x-office-document";
        if (lower.endsWith(".ods") || lower.endsWith(".xls") || lower.endsWith(".xlsx"))
            return "x-office-spreadsheet";
        if (lower.endsWith(".odp") || lower.endsWith(".ppt") || lower.endsWith(".pptx"))
            return "x-office-presentation";
        return "document-new";
    }

    function parseTemplates(output) {
        templatesModel.clear();
        const lines = String(output).split("\n");
        for (let i = 0; i < lines.length; i++) {
            const line = lines[i];
            if (!line)
                continue;
            const tab = line.indexOf("\t");
            if (tab < 0)
                continue;
            const path = line.slice(0, tab);
            const name = line.slice(tab + 1);
            templatesModel.append({
                path: path,
                name: name,
                iconName: templateIcon(name)
            });
        }
    }

    function reloadTemplates() {
        const dir = String(StandardPaths.writableLocation(StandardPaths.HomeLocation)).replace(/^file:\/\//, "") + "/Templates";
        templatesReader.connectSource("bash -c " + quoteArg("mkdir -p \"$1\" && find \"$1\" -maxdepth 1 -mindepth 1 \\( -type f -o -type l \\) -printf '%p\\t%f\\n' | sort") + " _ " + quoteArg(dir));
    }

    function createFromTemplate(path, name) {
        root.run("bash -c " + root.quoteArg("name=$(kdialog --title 'Create From Template' --inputbox 'File name:' \"$2\") && [ -n \"$name\" ] && cp -R -- \"$1\" \"$HOME/Desktop/$name\"") + " _ " + root.quoteArg(path) + " " + root.quoteArg(name));
    }

    function homeBin(name) {
        return String(StandardPaths.writableLocation(StandardPaths.HomeLocation)).replace(/^file:\/\//, "") + "/.local/bin/" + name;
    }

    function effectiveInt(value, fallback) {
        if (value === undefined || value === null || value === "")
            return fallback;
        const parsed = parseInt(value);
        return isNaN(parsed) ? fallback : parsed;
    }

    function effectiveBool(value, fallback) {
        if (value === undefined || value === null || value === "")
            return fallback;
        if (value === true || value === "true" || value === "1")
            return true;
        if (value === false || value === "false" || value === "0")
            return false;
        return fallback;
    }

        desktopStateReader.connectSource(homeBin("plasma-desktop-containment-report"));
    }

    function writeDesktopKey(key, value) {
        root.run("bash -lc " + root.quoteArg(homeBin("plasma-desktop-containment-set") + " " + root.quoteArg(key) + " " + root.quoteArg(value) + " >>/tmp/dotcortex-desktopmenu-actions.log 2>&1"));
        Qt.callLater(root.reloadDesktopState);
    }

    function parseDesktopState(output) {
        let report;
        try {
            report = JSON.parse(output);
        } catch (e) {
            return;
        }
        if (!report || report.length < 1 || !report[0].keys)
            return;

        const keys = report[0].keys;
        const sortMode = effectiveInt(keys.sortMode, 0);
        const iconSize = effectiveInt(keys.iconSize, 3);
        const arrangement = effectiveInt(keys.arrangement, 0);
        const alignment = effectiveInt(keys.alignment, 0);

        sortUnsortedItem.checked = sortMode === -1;
        sortNameItem.checked = sortMode === 0;
        sortSizeItem.checked = sortMode === 1;
        sortTypeItem.checked = sortMode === 6;
        sortDateItem.checked = sortMode === 2;
        sortDescendingItem.checked = effectiveBool(keys.sortDesc, false);
        sortFoldersFirstItem.checked = effectiveBool(keys.sortDirsFirst, true);

        iconTinyItem.checked = iconSize === 0;
        iconVerySmallItem.checked = iconSize === 1;
        iconSmallItem.checked = iconSize === 2;
        iconSmallMediumItem.checked = iconSize === 3;
        iconMediumItem.checked = iconSize === 4;
        iconLargeItem.checked = iconSize === 5;
        iconHugeItem.checked = iconSize === 6;

        arrangeLeftToRightItem.checked = arrangement === 0;
        arrangeTopToBottomItem.checked = arrangement === 1;
        alignLeftItem.checked = alignment === 0;
        alignRightItem.checked = alignment === 1;
        previewsItem.checked = effectiveBool(keys.previews, true);
        lockedItem.checked = effectiveBool(keys.locked, false);
    }

    function setDesktopSortMode(mode) {
        sortUnsortedItem.checked = mode === -1;
        sortNameItem.checked = mode === 0;
        sortSizeItem.checked = mode === 1;
        sortTypeItem.checked = mode === 6;
        sortDateItem.checked = mode === 2;
