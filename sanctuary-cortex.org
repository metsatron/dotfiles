* Purpose and ownership

The Cortex sanctuary is the X230's XFCE desktop shape moved onto the headless
T1700 Kikin-Kushi. Xpra owns two persistent displays on the T1700 host:

- display 101 is the complete XFCE desktop;
- display 100 is the seamless-app lane used by FireDragon and Obsidian.

The public, reproducible layer is kept here: the Guix package profile, Xpra
host/guest launchers, default application shims, and migration tools. The live
XFCE panel tree, mutable XFCE state, and FireDragon profile are private runtime
data under =~/HelmCortex/NEXUS/sanctuaries/sanctuary-cortex=. They are captured
from the X230 and projected into the T1700 guest by the helpers below; no
cookies, logins, browser databases, or panel captures are committed here.

=firedragon-cortex= is an additional seamless-app route into this sanctuary.
It does not replace the migrated X230 launcher contract: =firedragon-x230= now
keeps its historical behavior, using the client machine's local FireDragon
Flatpak against the T1700 profile reached through the NFS mount at
=~/mnt/t1700=. The sanctuary route and the profile-mounted route are separate
front doors.

The migration is staged. Until the private Cortex profile exists on the target
host, =vm-homes-init= continues to use the existing X230 profile as a fallback.

* Guix profile

XFCE is a guest profile rather than a host desktop dependency. The host Xpra
binary remains owned by the Devuan package lane; the guest profile contains the
session, panel, settings, desktop, Thunar, and the Flatpak bridge dependencies.

;; Guix's stock appmenu-gtk-module build sees GTK and libwnck, but not the
;; XFCE development interfaces, so its auto-detection omits the XFCE panel
;; applet. Build the same pinned source with both integrations enabled and
;; keep the panel descriptor, module, and registrar in the sanctuary profile.
(define appmenu-gtk-module-xfce
  (package
    (inherit appmenu-gtk-module)
    (name "appmenu-gtk-module-xfce")
    (arguments
     (substitute-keyword-arguments (package-arguments appmenu-gtk-module)
       ((#:configure-flags flags #~'())
        #~(cons* "-Dxfce=enabled"
                 "-Dappmenu-gtk-module=enabled"
                 "-Dappmenu-gtk-module:gtk=3"
                 #$flags))
       ((#:phases phases)
        #~(modify-phases #$phases
            ;; The pinned 0.7.6 checkout lists nb twice in po/LINGUAS;
            ;; Meson rejects the resulting duplicate .mo target.
            (add-after 'unpack 'deduplicate-linguas
              (lambda _
                (substitute* "po/LINGUAS"
                  ((" nn nb nr ") " nn nr "))))
            ;; Upstream uses the consuming panel's prefix/libdir, which is an
            ;; immutable input store path in Guix. Keep the plugin and its
            ;; descriptor inside this profile instead.
            (add-after 'deduplicate-linguas 'use-profile-install-paths
              (lambda _
                (substitute*
                    '("applets/meson.build" "data/meson.build")
                  (("xp.get_pkgconfig_variable\\('libdir'\\)")
                   "join_paths(prefix, get_option('libdir'))")
                  (("xp.get_pkgconfig_variable\\('prefix'\\)")
                   "prefix"))))
            ;; The private Cortex boot helper resolves the registrar through
            ;; PATH. The upstream install keeps it in libexec, so expose a
            ;; profile-local command without moving the real service binary.
            (add-after 'install 'expose-registrar-command
              (lambda* (#:key outputs #:allow-other-keys)
                (let ((out (assoc-ref outputs "out")))
                  (mkdir-p (string-append out "/bin"))
                  (symlink
                   (string-append out "/libexec/vala-panel/appmenu-registrar")
                   (string-append out "/bin/appmenu-registrar")))))
            (replace 'fix-install-gtk-module
              (lambda* (#:key outputs #:allow-other-keys)
                (let ((out (assoc-ref outputs "out")))
                  (substitute*
                      "subprojects/appmenu-gtk-module/src/gtk-3.0/meson.build"
                    (("gtk3.get_pkgconfig_variable\\('libdir'\\)")
                     (string-append "'" out "/lib'"))))))))))
    (inputs
     (modify-inputs (package-inputs appmenu-gtk-module)
       (append libxfce4ui xfce4-panel xfconf)))
    (native-inputs
     (modify-inputs (package-native-inputs appmenu-gtk-module)
       (append gettext-minimal)))))

* Private migration bridges

The following helpers deliberately write only below the ignored HelmCortex
namespace and the host-visible Cortex guest home. They do not tangle private
payloads into an overlay and do not replace a real guest directory without a
timestamped private backup.

#+BEGIN_SRC bash :tangle linux/.local/bin/sanctuary-cortex-profile-capture :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

PRIVATE_ROOT="${SANCTUARY_CORTEX_PRIVATE_ROOT:-$HOME/HelmCortex/NEXUS/sanctuaries/sanctuary-cortex}"
SOURCE_HOME="${SANCTUARY_CORTEX_SOURCE_HOME:-$HOME/mnt/x230}"
SOURCE="${SOURCE_HOME}/.firedragon/flatpak-migrate"
DEST="${PRIVATE_ROOT}/firefox-profile"

[[ -d "$SOURCE" ]] || { echo "profile-capture: source missing: $SOURCE" >&2; exit 1; }
[[ -f "$SOURCE/prefs.js" && -f "$SOURCE/places.sqlite" ]] || {
  echo "profile-capture: source is not a FireDragon profile: $SOURCE" >&2
  exit 1
}

for lock in lock .parentlock parent.lock SingletonLock SingletonSocket SingletonCookie; do
  if [[ -e "$SOURCE/$lock" || -L "$SOURCE/$lock" ]]; then
    echo "profile-capture: refusing a live or stale profile lock: $SOURCE/$lock" >&2
    echo "Close FireDragon on the X230, then remove only a verified stale lock." >&2
    exit 1
  fi
done

command -v rsync >/dev/null 2>&1 || { echo "profile-capture: rsync is required" >&2; exit 127; }
mkdir -p "$PRIVATE_ROOT"
if [[ -d "$DEST" ]]; then
  backup="${DEST}.before-capture.$(date +%Y%m%d-%H%M%S)"
  cp -a "$DEST" "$backup"
  echo "profile-capture: previous private copy preserved at $backup"
fi
mkdir -p "$DEST"
rsync -a \
  --exclude='lock' \
  --exclude='.parentlock' \
  --exclude='parent.lock' \
  --exclude='Singleton*' \
  "$SOURCE/" "$DEST/"
printf 'source=%s\ncaptured=%s\n' "$SOURCE" "$(date --iso-8601=seconds)" > "$PRIVATE_ROOT/firefox-profile.capture"
echo "profile-capture: private profile ready at $DEST"
#+END_SRC

  echo "config-capture: captured wallpaper $WALLPAPER_SOURCE"

* T1700 Xpra runtime

The Xpra server is host-side and the XFCE session is guest-side. The session
helper enters through =distrobox enter= so the container's UID and guest-home
mounts remain intact; it does not use a root =podman exec= path.

The migrated X230 panel contains the real =appmenu= plugin. The preflight below
requires its XFCE descriptor, panel module, GTK module, and D-Bus registrar
before any desktop daemon is restarted. A missing AppMenu therefore stops the
session or refresh before =xfce4-panel= can load a half-provisioned plugin.

#+BEGIN_SRC bash :tangle linux/.local/bin/sanctuary-cortex-panel-preflight :shebang "#!/bin/bash" :tangle-mode (identity #o755)
set -euo pipefail

PROFILE="${SANCTUARY_CORTEX_PROFILE:-$HOME/.guix-extra-profiles/sanctuary-cortex/sanctuary-cortex}"
PLUGIN_DESKTOP="$PROFILE/share/xfce4/panel/plugins/appmenu.desktop"
PLUGIN_SO="$(find -L "$PROFILE/lib" -type f -name 'libappmenu-xfce.so' -print -quit 2>/dev/null || true)"
GTK_SO="$(find -L "$PROFILE/lib" -type f -path '*/gtk-3.0/modules/libappmenu-gtk-module.so' -print -quit 2>/dev/null || true)"
REGISTRAR="$PROFILE/libexec/vala-panel/appmenu-registrar"
REGISTRAR_SERVICE="$PROFILE/share/dbus-1/services/com.canonical.AppMenu.Registrar.service"

fail() {
  printf 'sanctuary-cortex-panel-preflight: %s\n' "$*" >&2
  printf 'sanctuary-cortex-panel-preflight: apply the sanctuary profile, then stow Cortex again\n' >&2
  return 1
}

[[ -r "$PLUGIN_DESKTOP" ]] || fail "missing XFCE AppMenu descriptor: $PLUGIN_DESKTOP"
grep -q '^X-XFCE-Module=appmenu-xfce$' "$PLUGIN_DESKTOP" \
  || fail "AppMenu descriptor does not name appmenu-xfce: $PLUGIN_DESKTOP"
[[ -n "$PLUGIN_SO" ]] || fail "missing XFCE AppMenu panel module in $PROFILE/lib"

#+BEGIN_SRC bash :tangle linux/.local/bin/sanctuary-cortex-xpra :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

FULL_DISPLAY="${SANCTUARY_CORTEX_FULL_DISPLAY:-:101}"
APPS_DISPLAY="${SANCTUARY_CORTEX_APPS_DISPLAY:-:100}"
GEOMETRY="${SANCTUARY_CORTEX_GEOMETRY:-1920x1080}"
SESSION="${SANCTUARY_CORTEX_SESSION:-$HOME/.local/bin/sanctuary-cortex-session}"

usage() { echo "Usage: sanctuary-cortex-xpra start-full|start-apps|stop|status" >&2; }
command -v xpra >/dev/null 2>&1 || { echo "sanctuary-cortex-xpra: xpra is required on the T1700 host" >&2; exit 127; }

start_full() {
  [[ -x "$SESSION" ]] || { echo "sanctuary-cortex-xpra: missing session helper: $SESSION" >&2; exit 1; }
  xpra start-desktop "$FULL_DISPLAY" --daemon=yes --exit-with-children=no \
    --resize-display="$GEOMETRY" --mdns=no --start-child="$SESSION"
}

start_apps() {
  xpra start "$APPS_DISPLAY" --daemon=yes --exit-with-children=no \
    --resize-display="$GEOMETRY" --mdns=no --start-new-commands=yes
}

case "${1:-}" in
  start-full) start_full ;;
  start-apps) start_apps ;;
  stop) xpra stop "$FULL_DISPLAY" 2>/dev/null || true; xpra stop "$APPS_DISPLAY" 2>/dev/null || true ;;
  status) xpra list ;;
  *) usage; exit 2 ;;
esac
#+END_SRC
