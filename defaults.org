* Purpose
Small default-application state belongs in platform-native control planes, not in whole mutable GUI config files. This file declares the exact bits DotCortex should assert.

* Capture lanes

Web capture is a two-stage route: KDE Connect opens a received HTTP(S) URL in
the sanctuary-local browser, then Obsidian Web Clipper emits an =obsidian://=
URI.  Both stages must stay in the same machine and sanctuary.  The registry
below is the fleet control plane for that route; launchers select a sanctuary
row, while the machine name is discovered at runtime and becomes part of the
KDE Connect identity.  A machine-specific row may replace the =any= selector
without changing any launcher code.

KDE Connect instances still share the host network namespace.  Separate guest
homes, private D-Bus sessions, and distinct generated device names isolate
identity and pairing state, but they are a compatibility boundary rather than
a hard transport boundary; strict network sovereignty would require separate
network namespaces and discovery relays.

Columns: MACHINE, SANCTUARY, LANE, OBSIDIAN_APP_ID, VAULT_URI_SELECTOR,
VAULT_PATH, BROWSER_DESKTOP_ID, KDE_NAME_PREFIX, KDE_ENABLED, THEME.

The URI dispatcher changes only the =vault= query token on Obsidian =new= and
=daily= actions.  It rejects =path= because Obsidian gives that parameter
precedence over =vault=, preserves every other token byte-for-byte (including
Web Clipper's bare =clipboard= token), and invokes the concrete sanctuary
Flatpak runner directly so it can never recurse through =xdg-open=.

set -euo pipefail

die() {
  printf 'capture-lane: %s\n' "$*" >&2
  exit 1
}

usage() {
  cat >&2 <<'EOF'
Usage: capture-lane apply [sanctuary]
       capture-lane health [sanctuary]
       capture-lane obsidian-uri URI
       capture-lane obsidian [arguments...]
EOF
}

config_home="${XDG_CONFIG_HOME:-${HOME}/.config}"
data_home="${XDG_DATA_HOME:-${HOME}/.local/share}"
lane_dir="${config_home}/dotcortex"
lane_env="${lane_dir}/capture-lane.env"

detect_sanctuary() {
  if [[ -n "${CAPTURE_LANE_SANCTUARY:-}" ]]; then
    printf '%s\n' "${CAPTURE_LANE_SANCTUARY}"
    return
  fi
  case "${HOME}" in
    */.local/share/dotcortex/guests/*/home)
      basename "$(dirname "${HOME}")"
      ;;
    *)
      die 'sanctuary is ambiguous; pass it to apply/health or set CAPTURE_LANE_SANCTUARY'
      ;;
  esac
}

detect_machine() {
  local value=''
  if [[ -n "${CAPTURE_LANE_MACHINE:-}" ]]; then
    value="${CAPTURE_LANE_MACHINE}"
  elif [[ -r /run/host/etc/hostname ]]; then
    IFS= read -r value < /run/host/etc/hostname || true
  else
    value="$(hostname -s 2>/dev/null || true)"
  fi
  value="${value%%.*}"
  [[ "${value}" =~ ^[A-Za-z0-9][A-Za-z0-9._-]*$ ]] \
    || die "unsafe or empty machine identity: ${value:-unset}"
  printf '%s\n' "${value}"
}

find_manifest() {
  local candidate
  for candidate in \
    "${CAPTURE_LANE_MANIFEST:-}" \
    "${lane_dir}/capture-lanes.ssv" \
    "${HOME}/DotCortex/linux/.defaults/capture-lanes.ssv" \
    "/home/metsatron/DotCortex/linux/.defaults/capture-lanes.ssv"
  do
    [[ -n "${candidate}" && -r "${candidate}" ]] || continue
    printf '%s\n' "${candidate}"
    return
  done
  die 'capture-lanes.ssv not found'
}

load_row() {
  local sanctuary="$1" manifest row
  CAPTURE_LANE_MACHINE="$(detect_machine)"
  manifest="$(find_manifest)"
  CAPTURE_LANE_MANIFEST_PATH="${manifest}"
  row="$(awk -v machine="${CAPTURE_LANE_MACHINE}" -v sanctuary="${sanctuary}" '
    $0 !~ /^[[:space:]]*#/ && NF >= 10 && $1 == machine && $2 == sanctuary { print; exit }
  ' "${manifest}")"
  if [[ -z "${row}" ]]; then
    row="$(awk -v sanctuary="${sanctuary}" '
      $0 !~ /^[[:space:]]*#/ && NF >= 10 && $1 == "any" && $2 == sanctuary { print; exit }
    ' "${manifest}")"
  fi
  [[ -n "${row}" ]] || die "no capture lane for machine=${CAPTURE_LANE_MACHINE} sanctuary=${sanctuary}"

  read -r _machine CAPTURE_LANE_SANCTUARY CAPTURE_LANE_ID \
    CAPTURE_LANE_OBSIDIAN_APP_ID CAPTURE_LANE_VAULT_SELECTOR \
    _vault_path CAPTURE_LANE_BROWSER_DESKTOP CAPTURE_LANE_KDE_PREFIX \
    CAPTURE_LANE_KDE_ENABLED CAPTURE_LANE_THEME <<< "${row}"

  case "${_vault_path}" in
    HOME/*) CAPTURE_LANE_VAULT_PATH="${HOME}/${_vault_path#HOME/}" ;;
    /*) CAPTURE_LANE_VAULT_PATH="${_vault_path}" ;;
    *) die "vault path must be HOME/... or absolute: ${_vault_path}" ;;
  esac
  [[ "${CAPTURE_LANE_VAULT_SELECTOR}" =~ ^([A-Za-z0-9._~-]|%[0-9A-Fa-f]{2})+$ ]] \
    || die "vault URI selector is not percent-safe: ${CAPTURE_LANE_VAULT_SELECTOR}"
  CAPTURE_LANE_KDE_NAME="${CAPTURE_LANE_KDE_PREFIX}-${CAPTURE_LANE_MACHINE}"
}

write_env() {
  local temporary
  mkdir -p "${lane_dir}"
  temporary="$(mktemp "${lane_dir}/capture-lane.env.XXXXXX")"
  {
    printf '# generated by capture-lane; do not edit\n'
    printf 'CAPTURE_LANE_MACHINE=%q\n' "${CAPTURE_LANE_MACHINE}"
    printf 'CAPTURE_LANE_SANCTUARY=%q\n' "${CAPTURE_LANE_SANCTUARY}"
    printf 'CAPTURE_LANE_ID=%q\n' "${CAPTURE_LANE_ID}"
    printf 'CAPTURE_LANE_OBSIDIAN_APP_ID=%q\n' "${CAPTURE_LANE_OBSIDIAN_APP_ID}"
    printf 'CAPTURE_LANE_VAULT_SELECTOR=%q\n' "${CAPTURE_LANE_VAULT_SELECTOR}"
    printf 'CAPTURE_LANE_VAULT_PATH=%q\n' "${CAPTURE_LANE_VAULT_PATH}"
    printf 'CAPTURE_LANE_BROWSER_DESKTOP=%q\n' "${CAPTURE_LANE_BROWSER_DESKTOP}"
    printf 'CAPTURE_LANE_KDE_NAME=%q\n' "${CAPTURE_LANE_KDE_NAME}"
    printf 'CAPTURE_LANE_KDE_ENABLED=%q\n' "${CAPTURE_LANE_KDE_ENABLED}"
    printf 'CAPTURE_LANE_THEME=%q\n' "${CAPTURE_LANE_THEME}"
  } > "${temporary}"
  chmod 600 "${temporary}"
  mv -f "${temporary}" "${lane_env}"
}

apply_kde_name() {
  [[ "${CAPTURE_LANE_KDE_ENABLED}" == yes ]] || return 0
  local config temporary backup current=''
  config="${config_home}/kdeconnect/config"
  if [[ -r "${config}" ]]; then
    current="$(awk -F= '
      /^\[device\]$/ { in_device=1; next }
      /^\[/ { in_device=0 }
      in_device && $1 == "name" { sub(/^[^=]*=/, ""); print; exit }
    ' "${config}")"
  fi
  [[ "${current}" == "${CAPTURE_LANE_KDE_NAME}" ]] && return 0

  mkdir -p "$(dirname "${config}")"
  if [[ -e "${config}" ]]; then
    backup="${config}.backup-$(date +%Y%m%d-%H%M%S)"
    cp -a -- "${config}" "${backup}"
    printf 'capture-lane: preserved KDE Connect config at %s\n' "${backup}" >&2
  fi
  temporary="$(mktemp "${config}.XXXXXX")"
  awk -v name="${CAPTURE_LANE_KDE_NAME}" '
    BEGIN { in_device=0; found_device=0; wrote_name=0 }
    /^\[device\]$/ {
      print
      in_device=1
      found_device=1
      next
    }
    /^\[/ {
      if (in_device && !wrote_name) { print "name=" name; wrote_name=1 }
      in_device=0
      print
      next
    }
    in_device && /^name=/ {
      if (!wrote_name) { print "name=" name; wrote_name=1 }
      next
    }
    { print }
    END {
      if (in_device && !wrote_name) print "name=" name
      if (!found_device) {
        print ""
        print "[device]"
        print "name=" name
      }
    }
  ' "${config}" 2>/dev/null > "${temporary}" || {
    printf '[device]\nname=%s\n' "${CAPTURE_LANE_KDE_NAME}" > "${temporary}"
  }
  chmod 600 "${temporary}"
  mv -f "${temporary}" "${config}"
}

mime_backup_done=0
backup_mimeapps() {
  local mimeapps backup
  mimeapps="${config_home}/mimeapps.list"
  [[ "${mime_backup_done}" -eq 0 && -e "${mimeapps}" ]] || return 0
  backup="${mimeapps}.backup-$(date +%Y%m%d-%H%M%S)"
  cp -a -- "${mimeapps}" "${backup}"
  printf 'capture-lane: preserved MIME defaults at %s\n' "${backup}" >&2
  mime_backup_done=1
}

apply_mime_default() {
  local mime="$1" desktop="$2" current
  current="$(xdg-mime query default "${mime}" 2>/dev/null || true)"
  [[ "${current}" == "${desktop}" ]] && return 0
  backup_mimeapps
  xdg-mime default "${desktop}" "${mime}"
  current="$(xdg-mime query default "${mime}" 2>/dev/null || true)"
  [[ "${current}" == "${desktop}" ]] \
    || die "failed to assert ${mime} -> ${desktop}; got ${current:-unset}"
}

raw_obsidian_desktops() {
  awk '
    $0 !~ /^[[:space:]]*#/ && NF >= 4 {
      app=$4
      ids[app ".desktop"]=1
      ids["sanctuary-flatpak-" app "-" app ".desktop"]=1
    }
    END { for (id in ids) print id }
  ' "${CAPTURE_LANE_MANIFEST_PATH}" | sort
}

removed_obsidian_associations() {
  local mimeapps="${config_home}/mimeapps.list"
  awk -F= '
    /^\[Removed Associations\]$/ { in_removed=1; next }
    /^\[/ { in_removed=0 }
    in_removed && $1 == "x-scheme-handler/obsidian" {
      sub(/^[^=]*=/, "")
      print
      exit
    }
  ' "${mimeapps}" 2>/dev/null || true
}

apply_obsidian_mime_exclusions() {
  local mimeapps current desired combined temporary
  mimeapps="${config_home}/mimeapps.list"
  current="$(removed_obsidian_associations)"
  desired="$(raw_obsidian_desktops | awk '{ printf "%s;", $0 }')"
  combined="$({ printf '%s' "${current}"; printf '%s' "${desired}"; } \
    | tr ';' '\n' \
    | awk 'NF && !seen[$0]++ { printf "%s;", $0 }')"
  [[ "${current}" == "${combined}" ]] && return 0

  backup_mimeapps
  temporary="$(mktemp "${mimeapps}.XXXXXX")"
  awk -v key='x-scheme-handler/obsidian' -v desired="${combined}" '
    BEGIN { in_removed=0; found_section=0; wrote=0 }
    /^\[Removed Associations\]$/ {
      in_removed=1
      found_section=1
      print
      next
    }
    /^\[/ {
      if (in_removed && !wrote) { print key "=" desired; wrote=1 }
      in_removed=0
      print
      next
    }
    in_removed && index($0, key "=") == 1 {
      if (!wrote) { print key "=" desired; wrote=1 }
      next
    }
    { print }
    END {
      if (in_removed && !wrote) print key "=" desired
      if (!found_section) {
        print ""
        print "[Removed Associations]"
        print key "=" desired
      }
    }
  ' "${mimeapps}" > "${temporary}"
  chmod --reference="${mimeapps}" "${temporary}"
  mv -f "${temporary}" "${mimeapps}"
}

apply_lane() {
  local sanctuary="${1:-}"
  [[ -n "${sanctuary}" ]] || sanctuary="$(detect_sanctuary)"
  load_row "${sanctuary}"
  if [[ ! -d "${CAPTURE_LANE_VAULT_PATH}" ]]; then
    if [[ "${CAPTURE_LANE_THEME}" == win98 ]]; then
      mkdir -p "${CAPTURE_LANE_VAULT_PATH}"
    else
      die "selected vault is unavailable: ${CAPTURE_LANE_VAULT_PATH}"
    fi
  fi
  write_env
  [[ -r "${data_home}/applications/capture-lane-obsidian.desktop" ]] \
    || die "capture handler desktop is missing from ${data_home}/applications"
  if command -v update-desktop-database >/dev/null 2>&1; then
    update-desktop-database "${data_home}/applications" >/dev/null 2>&1 || true
  fi
  command -v xdg-mime >/dev/null 2>&1 || die 'xdg-mime is required to assert capture routing'
  apply_mime_default x-scheme-handler/obsidian capture-lane-obsidian.desktop
  apply_obsidian_mime_exclusions
  apply_mime_default x-scheme-handler/http "${CAPTURE_LANE_BROWSER_DESKTOP}"
  apply_mime_default x-scheme-handler/https "${CAPTURE_LANE_BROWSER_DESKTOP}"
  apply_kde_name
  printf 'capture-lane: %s/%s -> %s (%s), browser=%s, KDE=%s\n' \
    "${CAPTURE_LANE_MACHINE}" "${CAPTURE_LANE_ID}" \
    "${CAPTURE_LANE_VAULT_SELECTOR}" "${CAPTURE_LANE_OBSIDIAN_APP_ID}" \
    "${CAPTURE_LANE_BROWSER_DESKTOP}" "${CAPTURE_LANE_KDE_NAME}"
}

load_env() {
  [[ -r "${lane_env}" ]] || die "lane is not applied: ${lane_env}"
  # shellcheck disable=SC1090
  . "${lane_env}"
  : "${CAPTURE_LANE_OBSIDIAN_APP_ID:?}" \
    "${CAPTURE_LANE_VAULT_SELECTOR:?}" "${CAPTURE_LANE_VAULT_PATH:?}"
}

runner() {
  local candidate
  for candidate in \
    "${CAPTURE_LANE_RUNNER:-}" \
    "${HOME}/.local/bin/sanctuary-flatpak-run" \
    "/home/metsatron/.local/bin/sanctuary-flatpak-run"
  do
    [[ -n "${candidate}" && -x "${candidate}" ]] || continue
    printf '%s\n' "${candidate}"
    return
  done
  die 'sanctuary-flatpak-run not found'
}

rewrite_uri() {
  local uri="$1" before_fragment fragment='' base query='' token remaining
  local rewritten='' separator='' saw_vault=0 last=0 key
  [[ "${uri}" == obsidian://* ]] || die 'refusing a non-Obsidian URI'
  before_fragment="${uri%%#*}"
  [[ "${uri}" == *'#'* ]] && fragment="#${uri#*#}"
  if [[ "${before_fragment}" == *'?'* ]]; then
    base="${before_fragment%%\?*}"
    query="${before_fragment#*\?}"
  else
    base="${before_fragment}"
  fi
  case "${base}" in
    obsidian://new|obsidian://daily) ;;
    *) die "unsupported Obsidian URI action: ${base}" ;;
  esac

  remaining="${query}"
  if [[ "${before_fragment}" == *'?'* ]]; then
    while [[ "${last}" -eq 0 ]]; do
      if [[ "${remaining}" == *'&'* ]]; then
        token="${remaining%%&*}"
        remaining="${remaining#*&}"
      else
        token="${remaining}"
        last=1
      fi
      key="${token%%=*}"
      case "${key,,}" in
        path) die 'refusing Obsidian URI containing path=; path overrides the selected vault' ;;
        vault)
          if [[ "${saw_vault}" -eq 0 ]]; then
            rewritten+="${separator}vault=${CAPTURE_LANE_VAULT_SELECTOR}"
            separator='&'
            saw_vault=1
          fi
          ;;
        *)
          rewritten+="${separator}${token}"
          separator='&'
          ;;
      esac
    done
  fi
  if [[ "${saw_vault}" -eq 0 ]]; then
    rewritten+="${separator}vault=${CAPTURE_LANE_VAULT_SELECTOR}"
  fi
  printf '%s?%s%s\n' "${base}" "${rewritten}" "${fragment}"
}

launch_obsidian() {
  local executable
  load_env
  [[ -d "${CAPTURE_LANE_VAULT_PATH}" ]] \
    || die "selected vault is unavailable: ${CAPTURE_LANE_VAULT_PATH}"
  executable="$(runner)"
  exec "${executable}" "${CAPTURE_LANE_OBSIDIAN_APP_ID}" \
    "--filesystem=${CAPTURE_LANE_VAULT_PATH}" -- "$@"
}

health_lane() {
  local sanctuary="${1:-}" expected actual removed missing=0 desktop
  [[ -n "${sanctuary}" ]] || sanctuary="$(detect_sanctuary)"
  load_row "${sanctuary}"
  printf 'lane:       %s/%s (%s)\n' "${CAPTURE_LANE_MACHINE}" "${CAPTURE_LANE_ID}" "${CAPTURE_LANE_SANCTUARY}"
  printf 'vault:      %s -> %s [%s]\n' "${CAPTURE_LANE_VAULT_SELECTOR}" \
    "${CAPTURE_LANE_VAULT_PATH}" "$([[ -d "${CAPTURE_LANE_VAULT_PATH}" ]] && echo ready || echo missing)"
  printf 'Obsidian:   %s\n' "${CAPTURE_LANE_OBSIDIAN_APP_ID}"
  printf 'browser:    %s\n' "${CAPTURE_LANE_BROWSER_DESKTOP}"
  printf 'KDE name:   %s (enabled=%s)\n' "${CAPTURE_LANE_KDE_NAME}" "${CAPTURE_LANE_KDE_ENABLED}"
  if command -v xdg-mime >/dev/null 2>&1; then
    for expected in \
      'x-scheme-handler/obsidian capture-lane-obsidian.desktop' \
      "x-scheme-handler/http ${CAPTURE_LANE_BROWSER_DESKTOP}" \
      "x-scheme-handler/https ${CAPTURE_LANE_BROWSER_DESKTOP}"
    do
      read -r _mime _desktop <<< "${expected}"
      actual="$(xdg-mime query default "${_mime}" 2>/dev/null || true)"
      printf 'MIME:       %s -> %s [%s]\n' "${_mime}" "${actual:-unset}" \
        "$([[ "${actual}" == "${_desktop}" ]] && echo ok || echo expected-${_desktop})"
    done
  fi
  removed=";$(removed_obsidian_associations);"
  while IFS= read -r desktop; do
    [[ "${removed}" == *";${desktop};"* ]] || missing=$((missing + 1))
  done < <(raw_obsidian_desktops)
  printf 'chooser:    Obsidian — Current Sanctuary Vault [%s]\n' \
    "$([[ "${missing}" -eq 0 ]] && echo exclusive || echo "${missing}-raw-handlers-visible")"
}

command="${1:-}"
[[ -n "${command}" ]] || { usage; exit 2; }
shift
case "${command}" in
  apply) apply_lane "${1:-}" ;;
  health) health_lane "${1:-}" ;;
  obsidian-uri)
    [[ "$#" -eq 1 ]] || { usage; exit 2; }
    load_env
    rewritten_uri="$(rewrite_uri "$1")"
    launch_obsidian "${rewritten_uri}"
    ;;
  obsidian) launch_obsidian "$@" ;;
  *) usage; exit 2 ;;
esac

#!/usr/bin/env bash
set -euo pipefail

ROOT="${DOTCORTEX_ROOT:-$HOME/DotCortex}"
MIME_MAN="$ROOT/linux/.defaults/mime.ssv"
XFCE_MAN="$ROOT/linux/.defaults/xfce-helpers.ssv"
XFCE_RC="$HOME/.config/xfce4/helpers.rc"

apply_mime_defaults() {
    [ -f "$MIME_MAN" ] || return 0
    awk '$0 !~ /^[[:space:]]*#/ && NF >= 2 {print $1, $2}' "$MIME_MAN" |
        while read -r mime desktop_id; do
            xdg-mime default "$desktop_id" "$mime"
        done
}

apply_xfce_helpers() {
    [ -f "$XFCE_MAN" ] || return 0
    mkdir -p "$(dirname "$XFCE_RC")"
    touch "$XFCE_RC"

    tmp="$(mktemp)"
    cp "$XFCE_RC" "$tmp"

    awk '$0 !~ /^[[:space:]]*#/ && NF >= 2 {print $1, $2}' "$XFCE_MAN" |
        while read -r category helper_id; do
            if grep -q "^${category}=" "$tmp"; then
                sed -i "s|^${category}=.*|${category}=${helper_id}|" "$tmp"
            else
                printf '%s=%s\n' "$category" "$helper_id" >> "$tmp"
            fi
        done

    if ! cmp -s "$tmp" "$XFCE_RC"; then
        cp "$tmp" "$XFCE_RC"
    fi
    rm -f "$tmp"
}

apply_mime_defaults
apply_xfce_helpers

echo "Defaults applied."
#+END_SRC

#+BEGIN_SRC bash :tangle linux/.local/bin/defaults-diff :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
#!/usr/bin/env bash
set -euo pipefail

ROOT="${DOTCORTEX_ROOT:-$HOME/DotCortex}"
MIME_MAN="$ROOT/linux/.defaults/mime.ssv"
XFCE_MAN="$ROOT/linux/.defaults/xfce-helpers.ssv"
XFCE_RC="$HOME/.config/xfce4/helpers.rc"

echo "== MIME defaults =="
if [ -f "$MIME_MAN" ]; then
    awk '$0 !~ /^[[:space:]]*#/ && NF >= 2 {print $1, $2}' "$MIME_MAN" |
        while read -r mime desktop_id; do
            have="$(xdg-mime query default "$mime" 2>/dev/null || true)"
            if [ "$have" = "$desktop_id" ]; then
                printf '  = %s %s\n' "$mime" "$desktop_id"
            else
                printf '  ~ %s %s -> %s\n' "$mime" "${have:-unset}" "$desktop_id"
            fi
        done
fi

echo "== XFCE helpers =="
if [ -f "$XFCE_MAN" ]; then
    awk '$0 !~ /^[[:space:]]*#/ && NF >= 2 {print $1, $2}' "$XFCE_MAN" |
        while read -r category helper_id; do
            have="$(awk -F= -v key="$category" '$1 == key {print $2}' "$XFCE_RC" 2>/dev/null | tail -n1)"
            if [ "$have" = "$helper_id" ]; then
                printf '  = %s %s\n' "$category" "$helper_id"
            else
                printf '  ~ %s %s -> %s\n' "$category" "${have:-unset}" "$helper_id"
            fi
        done
fi

#!/usr/bin/env bash
set -euo pipefail

echo "MIME:"
printf '  http   = %s\n' "$(xdg-mime query default x-scheme-handler/http 2>/dev/null || echo unset)"
printf '  https  = %s\n' "$(xdg-mime query default x-scheme-handler/https 2>/dev/null || echo unset)"
printf '  mailto = %s\n' "$(xdg-mime query default x-scheme-handler/mailto 2>/dev/null || echo unset)"
printf '  rfc822 = %s\n' "$(xdg-mime query default message/rfc822 2>/dev/null || echo unset)"

echo "XFCE helpers:"
if [ -f "$HOME/.config/xfce4/helpers.rc" ]; then
    sed -n '/^[A-Za-z].*=.*/p' "$HOME/.config/xfce4/helpers.rc" | sed 's/^/  /'
else
    echo "  helpers.rc missing"
fi
