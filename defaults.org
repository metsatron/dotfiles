* Apply defaults
#+BEGIN_SRC bash :tangle linux/.local/bin/defaults-apply :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
#!/usr/bin/env bash
set -euo pipefail

ROOT="${DOTCORTEX_ROOT:-$HOME/DotCortex}"
MIME_MAN="$ROOT/linux/.defaults/mime.ssv"
XFCE_MAN="$ROOT/linux/.defaults/xfce-helpers.ssv"
XFCE_RC="$HOME/.config/xfce4/helpers.rc"

apply_mime_defaults() {
    [ -f "$MIME_MAN" ] || return 0
    awk '$0 !~ /^[[:space:]]*#/ && NF >= 2 {print $1, $2}' "$MIME_MAN" |
        while read -r mime desktop_id; do
            xdg-mime default "$desktop_id" "$mime"
        done
}

apply_xfce_helpers() {
    [ -f "$XFCE_MAN" ] || return 0
    mkdir -p "$(dirname "$XFCE_RC")"
    touch "$XFCE_RC"

    tmp="$(mktemp)"
    cp "$XFCE_RC" "$tmp"

    awk '$0 !~ /^[[:space:]]*#/ && NF >= 2 {print $1, $2}' "$XFCE_MAN" |
        while read -r category helper_id; do
            if grep -q "^${category}=" "$tmp"; then
                sed -i "s|^${category}=.*|${category}=${helper_id}|" "$tmp"
            else
                printf '%s=%s\n' "$category" "$helper_id" >> "$tmp"
            fi
        done

    if ! cmp -s "$tmp" "$XFCE_RC"; then
        cp "$tmp" "$XFCE_RC"
    fi
    rm -f "$tmp"
}

apply_mime_defaults
apply_xfce_helpers

echo "Defaults applied."
#+END_SRC

* Diff defaults
#+BEGIN_SRC bash :tangle linux/.local/bin/defaults-diff :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
#!/usr/bin/env bash
set -euo pipefail

ROOT="${DOTCORTEX_ROOT:-$HOME/DotCortex}"
MIME_MAN="$ROOT/linux/.defaults/mime.ssv"
XFCE_MAN="$ROOT/linux/.defaults/xfce-helpers.ssv"
XFCE_RC="$HOME/.config/xfce4/helpers.rc"

echo "== MIME defaults =="
if [ -f "$MIME_MAN" ]; then
    awk '$0 !~ /^[[:space:]]*#/ && NF >= 2 {print $1, $2}' "$MIME_MAN" |
        while read -r mime desktop_id; do
            have="$(xdg-mime query default "$mime" 2>/dev/null || true)"
            if [ "$have" = "$desktop_id" ]; then
                printf '  = %s %s\n' "$mime" "$desktop_id"
            else
                printf '  ~ %s %s -> %s\n' "$mime" "${have:-unset}" "$desktop_id"
            fi
        done
fi

echo "== XFCE helpers =="
if [ -f "$XFCE_MAN" ]; then
    awk '$0 !~ /^[[:space:]]*#/ && NF >= 2 {print $1, $2}' "$XFCE_MAN" |
        while read -r category helper_id; do
            have="$(awk -F= -v key="$category" '$1 == key {print $2}' "$XFCE_RC" 2>/dev/null | tail -n1)"
            if [ "$have" = "$helper_id" ]; then
                printf '  = %s %s\n' "$category" "$helper_id"
            else
                printf '  ~ %s %s -> %s\n' "$category" "${have:-unset}" "$helper_id"
            fi
        done
fi
#+END_SRC

#+BEGIN_SRC bash :tangle linux/.local/bin/defaults-health :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
#!/usr/bin/env bash
set -euo pipefail

echo "MIME:"
printf '  http   = %s\n' "$(xdg-mime query default x-scheme-handler/http 2>/dev/null || echo unset)"
printf '  https  = %s\n' "$(xdg-mime query default x-scheme-handler/https 2>/dev/null || echo unset)"
printf '  mailto = %s\n' "$(xdg-mime query default x-scheme-handler/mailto 2>/dev/null || echo unset)"
printf '  rfc822 = %s\n' "$(xdg-mime query default message/rfc822 2>/dev/null || echo unset)"

echo "XFCE helpers:"
if [ -f "$HOME/.config/xfce4/helpers.rc" ]; then
    sed -n '/^[A-Za-z].*=.*/p' "$HOME/.config/xfce4/helpers.rc" | sed 's/^/  /'
else
    echo "  helpers.rc missing"
fi
