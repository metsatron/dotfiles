set -euo pipefail

# Canonical, host-wide Rust locations, not under ~/DotCortex and not under HelmCortex.
export RUSTUP_HOME="${RUSTUP_HOME:-$HOME/.rustup}"
export CARGO_HOME="${CARGO_HOME:-$HOME/.cargo}"
export PATH="$CARGO_HOME/bin:/usr/bin:/bin:$PATH"

# Only care whether *our* cargo exists, not Guix or apt.
if [ ! -x "$CARGO_HOME/bin/cargo" ]; then
  echo ">> Installing rustup into $RUSTUP_HOME and cargo into $CARGO_HOME"
  curl -fsSL https://sh.rustup.rs | \
    RUSTUP_HOME="$RUSTUP_HOME" CARGO_HOME="$CARGO_HOME" \
    sh -s -- -y --no-modify-path --default-toolchain stable
fi

# Ensure a default toolchain is ready in this env.
if ! "$CARGO_HOME/bin/rustup" show >/dev/null 2>&1; then
  RUSTUP_HOME="$RUSTUP_HOME" CARGO_HOME="$CARGO_HOME" \
    "$CARGO_HOME/bin/rustup" default stable
fi

exec "$@"

# Reads cargo install --list, appends new crates to DotCortex manifest
#+BEGIN_SRC bash :tangle all/.local/bin/cargo-capture :shebang "#!/usr/bin/env bash"
# ~/DotCortex/all/.local/bin/cargo-capture
set -euo pipefail

MAN="$HOME/DotCortex/all/.local/share/dotcortex/manifests/cargo/crates.ssv"
RUN="$HOME/.local/bin/dotcortex-rust-env"

mkdir -p "$(dirname "$MAN")"
touch "$MAN"

tmp_existing="$(mktemp)"
tmp_have="$(mktemp)"
tmp_new="$(mktemp)"

# Existing crates already tracked in manifest
awk '!/^[[:space:]]*#/ && NF { print $1 }' "$MAN" | sort -u >"$tmp_existing"

# Crates currently installed by cargo
"$RUN" cargo install --list 2>/dev/null \
  | awk '/^[^[:space:]]+ v[0-9]+\.[0-9]+\.[0-9]+:/ { print $1 }' \
  | sort -u >"$tmp_have"

# new = have - existing
comm -13 "$tmp_existing" "$tmp_have" >"$tmp_new"

rm -f "$tmp_existing" "$tmp_have"

if [ ! -s "$tmp_new" ]; then
  rm -f "$tmp_new"
  echo "No new cargo crates to capture into $MAN"
  exit 0
fi

while read -r crate; do
  [ -n "$crate" ] || continue
  # Default: floating latest on crates.io. You can edit VERSION later to pin.
  printf '%s "" "" "" crates.io "" "" ""\n' "$crate" >>"$MAN"
done <"$tmp_new"

rm -f "$tmp_new"

echo "Updated $MAN"
#+END_SRC

* Apply manifest inside DotCortex env
#+BEGIN_SRC bash :tangle all/.local/bin/cargo-apply :shebang "#!/usr/bin/env bash"
#!/usr/bin/env bash
set -euo pipefail

MAN="$HOME/DotCortex/all/.local/share/dotcortex/manifests/cargo/crates.ssv"

# Default runner (your existing choice)
RUN="$HOME/.local/bin/dotcortex-rust-env"

# Crates that failed to install. Collected rather than fatal — see the install loop.
failed=()

# Host-link runner (Guix toolchain, host libs)
HOST_RUN="$HOME/.local/bin/cargo-wrap"

DEFAULT_LANE="${DEFAULT_LANE:-host}"

ENFORCE="${ENFORCE:-1}"
UNINSTALL="${UNINSTALL:-1}"

# Thermal guard. A host that cannot survive a rustc storm does not run one.
#
# The X230 died this way on 2026-07-13: cargo:apply pinned all four cores, the
# ACPI zone hit its 100C critical trip, and the kernel hardware-protection-killed
# the machine -- taking the fleet's NFS/Samba/ZFS down with it, unattended, at
# 23:11. The lesson is not "be careful when running sweeps": an automated sweep
# has no one to be careful. The refusal has to live in the tool.
#
# The marker is stowed from the x230/ overlay, so it exists only on hosts that
# have actually been declared thermally constrained. Clear it after the teardown.
if [[ -e "$HOME/.config/dotcortex/no-local-builds" && "${ALLOW_LOCAL_BUILD:-0}" != 1 ]]; then
  echo "REFUSING: $(hostname) is marked thermally constrained (no-local-builds)."
  echo "  Reason: $(head -1 "$HOME/.config/dotcortex/no-local-builds" 2>/dev/null)"
  echo
  echo "  Build on the fleet build host instead:  cargo-remote-apply"
  echo "  To override once (you had better mean it):  ALLOW_LOCAL_BUILD=1 cargo-apply"
  exit 1
fi

echo "== cargo apply =="

# Check for host libraries required by crates that link against C libs
_check_host_deps() {
  local missing=()
  local -A libs=([freetype2]=libfreetype-dev [fontconfig]=libfontconfig-dev)
  for pc in "${!libs[@]}"; do
    if ! pkg-config --exists "$pc" 2>/dev/null; then
      missing+=("${libs[$pc]}")
    fi
  done
  if (( ${#missing[@]} )); then
    echo "ERROR: missing host libraries: ${missing[*]}"
    echo "Install them:  sudo apt install ${missing[*]}"
    exit 1
  fi
}
_check_host_deps

_emit_norm() {
  awk -v FS='[ \t]+' '
    function dequote(s){ return (s=="\"\"" ? "" : s) }
    $0 ~ /^[ \t]*#/ || NF==0 { next }
    {
      printf "%s|%s|%s|%s|%s|%s|%s|%s|%s\n",
        dequote($1),dequote($2),dequote($3),dequote($4),
        dequote($5),dequote($6),dequote($7),dequote($8),dequote($9)
    }
  ' "$MAN"
}

while IFS='|' read -r crate version features bins source git_url git_ref target lane
do
  [[ -n "$crate" ]] || continue

  lane="${lane:-$DEFAULT_LANE}"

  args=(install "$crate")

  if [[ -n "$version" ]]; then
    args+=(--version "$version" --locked)
  fi

  [[ -n "$features" ]] && args+=(--features "$features")

  if [[ -n "$bins" ]]; then
    IFS=',' read -ra BB <<<"$bins"
    for b in "${BB[@]}"; do
      [[ -n "$b" ]] && args+=(--bin "$b")
    done
  fi

  [[ -n "$target" ]] && args+=(--target "$target")

  if [[ "${source:-crates.io}" = "git" && -n "${git_url:-}" ]]; then
    args+=(--git "$git_url")
    [[ -n "${git_ref:-}" ]] && args+=(--rev "$git_ref")
  fi

  # One crate that fails to build must NOT abort the rest of the manifest.
  # Under `set -e` it did: on X230 (cargo 1.90) the git build of neovide dies
  # parsing its own Cargo.toml, and because neovide sits near the top of the
  # manifest, every crate below it — fuelcheck-cli and agent-code included —
  # was silently never installed. A declared package that never lands, with no
  # summary saying so, is the same failure that hid fuelcheck-cli for eleven
  # weeks. Collect failures, keep going, and exit non-zero with a list.
  if [[ "$lane" = "host" ]]; then
    "$RUN" "$HOST_RUN" "${args[@]}" || failed+=("$crate")
  else
    "$RUN" cargo "${args[@]}" || failed+=("$crate")
  fi
done < <(_emit_norm)

if [[ "$ENFORCE" = 1 && "$UNINSTALL" = 1 ]]; then
  echo "== uninstall extras =="

  want_tmp="$(mktemp)"
  have_tmp="$(mktemp)"

  awk '!/^[[:space:]]*#/ && NF { print $1 }' "$MAN" | sort -u >"$want_tmp"

  "$RUN" cargo install --list 2>/dev/null \
    | awk '/^[^[:space:]]+ v[0-9]+\.[0-9]+\.[0-9]+:/ { print $1 }' \
    | sort -u >"$have_tmp"

  comm -13 "$want_tmp" "$have_tmp" | while read -r extra; do
    [[ -n "$extra" ]] || continue
    "$RUN" cargo uninstall "$extra" || true
  done

  rm -f "$want_tmp" "$have_tmp"
fi

# Loud, and non-zero: a declared crate that did not install is a failure the
# sweep must surface, not bury under a cheerful "Done."
if [[ "${#failed[@]}" -gt 0 ]]; then
  echo "FAILED (${#failed[@]}): ${failed[*]}"
  exit 1
fi

echo "Done."
#+END_SRC

* Remote build (thermally-constrained hosts)
Build the manifest on a build host, inside a container whose userland ABI matches
*this* machine, then verify and ship the binaries back. See =cargo-remote-apply=
for why each guard exists.

#+BEGIN_SRC bash :tangle all/.local/bin/cargo-remote-apply :shebang "#!/usr/bin/env bash"
#!/usr/bin/env bash
set -euo pipefail

# Build this machine's cargo manifest somewhere else, because building it here
# would destroy the machine.
#
# On 2026-07-13 a cargo:apply sweep on the X230 pinned all four cores in rustc.
# The ACPI thermal zone hit its 100C critical trip and the kernel executed a
# HARDWARE PROTECTION shutdown -- which took NFS, Samba, ZFS and Jellyfin down
# with it, because that laptop is the fleet's fileserver. The heat path on a
# 13-year-old chassis is degraded (dust, worn thermal interface); the compile
# storm is what exposed it. Until the teardown, this host does not compile Rust.
#
# Three constraints shape the design, and all three are load-bearing:
#
# 1. glibc is forward-compatible ONLY. Every build host in the fleet is newer
#    than the X230 (T480s 2.41 vs X230 2.39). A binary built natively on the
#    build host links symbols this machine does not have and will not start.
#    So the build runs in a container pinned to THIS machine's os-release.
#
# 2. The X230 is Ivy Bridge and has NO AVX2; every other machine in the fleet
#    has it. A -C target-cpu=native build would emit AVX2 and SIGILL on arrival.
#    RUSTFLAGS is deliberately scrubbed, never forwarded.
#
# 3. "It will run over there" is a claim, so it gets machine-checked, not
#    assumed: every built binary is verified against this host's glibc floor
#    and then actually executed here before anything is installed.

MAN="$HOME/DotCortex/all/.local/share/dotcortex/manifests/cargo/crates.ssv"

BUILD_HOST="${CARGO_BUILD_HOST:-t480s}"
DEST="$HOME/.cargo/bin"

# The container must match THIS machine's userland, not the build host's.
. /etc/os-release
BUILD_IMAGE="${CARGO_BUILD_IMAGE:-docker.io/library/${ID}:${VERSION_ID}}"
TAG="dotcortex-cargo-build:${ID}-${VERSION_ID}"

# This host's glibc -- the ABI ceiling every returned binary must respect.
# NB: awk without exit, deliberately. `ldd --version | head -1` looks equivalent
# but head closes the pipe early, ldd dies of SIGPIPE, and under `set -o pipefail`
# that kills this script with 141 before it prints a single line.
HOST_GLIBC="$(ldd --version | awk 'NR==1 {print $NF}')"

REMOTE_WORK=".cache/dotcortex/cargo-remote/${ID}-${VERSION_ID}"
STAGE="$(mktemp -d)"
trap 'rm -rf "$STAGE"' EXIT

echo "== cargo remote apply =="
echo "   build host : $BUILD_HOST"
echo "   image      : $BUILD_IMAGE  (matches this host: ${ID} ${VERSION_ID}, glibc ${HOST_GLIBC})"
echo "   dest       : $DEST"

[[ -r "$MAN" ]] || { echo "ERROR: manifest not found: $MAN"; exit 1; }
ssh -o BatchMode=yes -o ConnectTimeout=10 "$BUILD_HOST" true 2>/dev/null \
  || { echo "ERROR: build host '$BUILD_HOST' unreachable over ssh"; exit 1; }
ssh -o BatchMode=yes "$BUILD_HOST" 'command -v podman >/dev/null' \
  || { echo "ERROR: podman not present on $BUILD_HOST"; exit 1; }

# --- ship the manifest ------------------------------------------------------
ssh -o BatchMode=yes "$BUILD_HOST" "mkdir -p '$REMOTE_WORK'/{out,cargo-home,rustup-home}"
scp -q "$MAN" "$BUILD_HOST:$REMOTE_WORK/crates.ssv"

# --- build the toolchain image once, reuse thereafter -----------------------
ssh -o BatchMode=yes "$BUILD_HOST" bash -s -- "$REMOTE_WORK" "$BUILD_IMAGE" "$TAG" <<'REMOTE_IMAGE'
set -euo pipefail
work="$1"; image="$2"; tag="$3"
if podman image exists "$tag"; then
  echo ">> reusing build image $tag"
  exit 0
fi
echo ">> building build image $tag from $image"
cat >"$work/Containerfile" <<EOF
FROM $image
ENV DEBIAN_FRONTEND=noninteractive
RUN apt-get update && apt-get install -y --no-install-recommends \
      ca-certificates curl git pkg-config build-essential \
      libssl-dev libfreetype-dev libfontconfig-dev \
 && rm -rf /var/lib/apt/lists/*
EOF
podman build -t "$tag" -f "$work/Containerfile" "$work"
REMOTE_IMAGE

# --- build the manifest inside the ABI-matched container --------------------
# CARGO_HOME/RUSTUP_HOME are bind-mounted and persist between runs, so the
# registry cache survives and repeat sweeps are cheap.
ssh -o BatchMode=yes "$BUILD_HOST" bash -s -- "$REMOTE_WORK" "$TAG" <<'REMOTE_BUILD'
set -euo pipefail
work="$1"; tag="$2"
abs="$HOME/$work"

podman run --rm \
  -v "$abs:/work:Z" \
  -w /work \
  "$tag" \
  bash -euo pipefail -c '
    export CARGO_HOME=/work/cargo-home
    export RUSTUP_HOME=/work/rustup-home
    export PATH="$CARGO_HOME/bin:$PATH"

    # Never tune for the build host CPU: the target has no AVX2.
    unset RUSTFLAGS CARGO_BUILD_RUSTFLAGS

    if [ ! -x "$CARGO_HOME/bin/cargo" ]; then
      curl -fsSL https://sh.rustup.rs \
        | sh -s -- -y --no-modify-path --default-toolchain stable
    fi

    failed=()
    while read -r crate version features bins source git_url git_ref target lane; do
      case "$crate" in ""|\#*) continue ;; esac
      dq() { [ "$1" = "\"\"" ] && echo "" || echo "$1"; }
      version="$(dq "$version")"; features="$(dq "$features")"; bins="$(dq "$bins")"
      source="$(dq "$source")";   git_url="$(dq "$git_url")"; git_ref="$(dq "$git_ref")"

      args=(install "$crate" --root /work/out)
      [ -n "$version" ] && args+=(--version "$version" --locked)
      [ -n "$features" ] && args+=(--features "$features")
      if [ -n "$bins" ]; then
        IFS="," read -ra BB <<<"$bins"
        for b in "${BB[@]}"; do [ -n "$b" ] && args+=(--bin "$b"); done
      fi
      if [ "${source:-crates.io}" = "git" ] && [ -n "$git_url" ]; then
        args+=(--git "$git_url")
        [ -n "$git_ref" ] && args+=(--rev "$git_ref")
      fi

      echo ">> building $crate"
      # One bad crate must not silently skip the rest of the manifest.
      cargo "${args[@]}" || failed+=("$crate")
    done < /work/crates.ssv

    if [ "${#failed[@]}" -gt 0 ]; then
      echo "BUILD FAILED (${#failed[@]}): ${failed[*]}"
      exit 1
    fi
  '
REMOTE_BUILD

# --- bring the binaries back into a staging area, NOT straight into PATH -----
echo "== fetching build output =="
rsync -a --info=stats1 "$BUILD_HOST:$REMOTE_WORK/out/bin/" "$STAGE/"

shopt -s nullglob
built=( "$STAGE"/* )
if (( ${#built[@]} == 0 )); then
  echo "ERROR: build host returned no binaries"
  exit 1
fi

# --- VERIFY before installing -----------------------------------------------
# The whole point of the container is an ABI claim. Check the claim.
echo "== verifying ${#built[@]} binaries against this host (glibc ${HOST_GLIBC}, $(uname -m)) =="

_max_glibc_req() {
  # Highest GLIBC_x.y.z this binary demands. Empty if statically linked.
  # The `|| true` is load-bearing: on a static binary objdump exits non-zero and
  # grep matches nothing, and under `set -o pipefail` that would abort the whole
  # verification on a binary that is in fact perfectly fine.
  { objdump -T "$1" 2>/dev/null || true; } \
    | { grep -o 'GLIBC_[0-9.]*' || true; } \
    | sed 's/GLIBC_//' \
    | sort -V \
    | tail -1
}

bad=()
good=()
for bin in "${built[@]}"; do
  name="$(basename "$bin")"

  # (a) glibc floor.
  req="$(_max_glibc_req "$bin")"
  if [[ -n "$req" ]] && [[ "$(printf '%s\n%s\n' "$req" "$HOST_GLIBC" | sort -V | tail -1)" != "$HOST_GLIBC" ]]; then
    echo "   FAIL $name: needs glibc $req, this host has $HOST_GLIBC"
    bad+=("$name"); continue
  fi

  # (b) Every library and symbol resolves against THIS host's loader. Catches
  # missing sonames and undefined symbols without executing anything.
  if ldd_out="$(ldd -r "$bin" 2>&1)"; then
    if grep -qE 'not found|undefined symbol' <<<"$ldd_out"; then
      echo "   FAIL $name: unresolved at load time:"
      grep -E 'not found|undefined symbol' <<<"$ldd_out" | sed 's/^/          /'
      bad+=("$name"); continue
    fi
  fi

  # (c) It starts. NOT "it exits 0" -- that was a bug: tpfancontrol is a daemon
  # with no --help handler, exits 1 complaining about its config file, and got
  # branded an ABI failure for it. A non-zero exit means the binary LOADED and
  # RAN and then made its own decision, which is exactly what we are testing for.
  # Only a loader failure (126/127) or death by signal (>=128 -- SIGILL=132 is
  # the AVX2-on-Ivy-Bridge case) means it cannot execute on this host.
  set +e
  "$bin" --version >/dev/null 2>&1
  rc=$?
  set -e
  if (( rc == 126 || rc == 127 || rc >= 128 )); then
    echo "   FAIL $name: will not execute here (exit $rc$( ((rc>=128)) && echo ", signal $((rc-128))"))"
    bad+=("$name"); continue
  fi

  printf '   ok   %-16s glibc<=%-6s loads, runs\n' "$name" "${req:-static}"
  good+=("$bin")
done

# --- install the verified ones ----------------------------------------------
# One bad binary must not block the other twelve. That is the same law as
# b4b2c1aa9 ("one bad package must not silently skip the rest of the manifest"),
# just from the other end -- and the failure is loud and the exit is non-zero,
# so nothing is buried.
if (( ${#good[@]} )); then
  mkdir -p "$DEST"
  install -m 0755 "${good[@]}" "$DEST/"
  echo
  echo "Installed ${#good[@]} verified binaries into $DEST"
fi

if (( ${#bad[@]} )); then
  echo
  echo "NOT INSTALLED (${#bad[@]} failed verification): ${bad[*]}"
  echo "These built on $BUILD_HOST but will not run here. Nothing was overwritten for them."
  exit 1
fi

echo "Done."
#+END_SRC

* Detach live Cargo home from DotCortex
#+BEGIN_SRC python :tangle all/.local/bin/cargo-home-detach :shebang "#!/usr/bin/env python3" :tangle-mode (identity #o755)
from __future__ import annotations

import argparse
import shutil
import sys
from pathlib import Path


SKIP = {"manifest"}


def copy_entry(src: Path, dst: Path) -> None:
    if src.is_dir() and not src.is_symlink():
        shutil.copytree(src, dst, symlinks=True, dirs_exist_ok=True)
    else:
        dst.parent.mkdir(parents=True, exist_ok=True)
        if dst.exists() or dst.is_symlink():
            if dst.is_dir() and not dst.is_symlink():
                shutil.rmtree(dst)
            else:
                dst.unlink()
        shutil.copy2(src, dst, follow_symlinks=False)


def remove_entry(path: Path) -> None:
    if path.is_dir() and not path.is_symlink():
        shutil.rmtree(path)
    elif path.exists() or path.is_symlink():
        path.unlink()


def main() -> int:
    parser = argparse.ArgumentParser(
        description="Detach ~/.cargo from DotCortex and keep live Cargo state in $HOME."
    )
    parser.add_argument(
        "--repo-home",
        default=str(Path.home() / "DotCortex/all/.cargo"),
        help="repo-owned cargo path to migrate away from",
    )
    parser.add_argument(
        "--home",
        default=str(Path.home() / ".cargo"),
        help="real cargo home destination",
    )
    parser.add_argument(
        "--keep-repo-copy",
        action="store_true",
        help="do not remove migrated live state from the repo path",
    )
    args = parser.parse_args()

    repo_home = Path(args.repo_home).expanduser().resolve()
    home = Path(args.home).expanduser()

    if not repo_home.exists():
        print(f"repo cargo path missing: {repo_home}")
        return 0

    migrated = 0
    if home.is_symlink():
        home.unlink()
    home.mkdir(parents=True, exist_ok=True)

    for src in sorted(repo_home.iterdir()):
        if src.name in SKIP:
            continue
        dst = home / src.name
        copy_entry(src, dst)
        migrated += 1
        if not args.keep_repo_copy:
            remove_entry(src)

    if not args.keep_repo_copy:
        leftovers = [p.name for p in repo_home.iterdir() if p.name not in SKIP]
        if leftovers:
            print(f"left repo leftovers untouched: {', '.join(leftovers)}")

    print(f"detached cargo home: migrated={migrated} home={home} repo={repo_home}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
#+END_SRC

** Dedicated wrapper: cargo-wrap
This is basically the host-wrap philosophy, but tuned for builds, not GUI apps. It runs in a near empty env, keeps Guix cargo and rustc reachable, and forces pkg-config to look only at host paths so OpenSSL comes from /usr, not Guix.
#+BEGIN_SRC bash :tangle linux/.local/bin/cargo-wrap :shebang "#!/usr/bin/env bash" :mkdirp yes
#!/usr/bin/env bash
# Guix Cargo, Host Link Wrapper
# Goal: keep Guix cargo/rustc, but force host pkg-config and prevent Guix libs from being selected.

set -euo pipefail

# Prefer DotCortex/rustup cargo when present. Fresh hosts may not have
# ~/.cargo/bin on PATH yet, but cargo-apply invokes this through
# dotcortex-rust-env so CARGO_HOME is already authoritative.
CARGO_HOME="${CARGO_HOME:-$HOME/.cargo}"
RUSTUP_HOME="${RUSTUP_HOME:-$HOME/.rustup}"

CARGO_BIN="${CARGO_BIN:-}"
if [[ -z "${CARGO_BIN:-}" && -x "$CARGO_HOME/bin/cargo" ]]; then
  CARGO_BIN="$CARGO_HOME/bin/cargo"
fi
if [[ -z "${CARGO_BIN:-}" ]]; then
  CARGO_BIN="$(command -v cargo || true)"
fi
[[ -n "${CARGO_BIN:-}" && -x "$CARGO_BIN" ]] || { echo "cargo-wrap: cargo not found" >&2; exit 127; }

RUSTC_BIN="${RUSTC_BIN:-}"
if [[ -z "${RUSTC_BIN:-}" && -x "$CARGO_HOME/bin/rustc" ]]; then
  RUSTC_BIN="$CARGO_HOME/bin/rustc"
fi
if [[ -z "${RUSTC_BIN:-}" ]]; then
  RUSTC_BIN="$(command -v rustc || true)"
fi

cargo_dir="$(cd "$(dirname "$CARGO_BIN")" && pwd -P)"
rustc_dir=""
if [[ -n "${RUSTC_BIN:-}" && -x "$RUSTC_BIN" ]]; then
  rustc_dir="$(cd "$(dirname "$RUSTC_BIN")" && pwd -P)"
fi

# Build a minimal PATH:
# - Guix toolchain dir(s) so cargo can find rustc
# - host /usr/bin,/bin for cc, ld, pkg-config, git, etc
clean_path="$cargo_dir"
if [[ -n "${rustc_dir:-}" && "$rustc_dir" != "$cargo_dir" ]]; then
  clean_path="$clean_path:$rustc_dir"
fi
clean_path="$clean_path:/usr/bin:/bin"

# Host pkg-config search roots (Debian family)
triplet=""
if command -v dpkg-architecture >/dev/null 2>&1; then
  triplet="$(dpkg-architecture -qDEB_HOST_MULTIARCH 2>/dev/null || true)"
fi

pc=""
for d in \
  "/usr/lib/${triplet}/pkgconfig" \
  "/usr/lib/pkgconfig" \
  "/usr/share/pkgconfig"
do
  [[ -d "$d" ]] && pc="${pc:+$pc:}$d"
done
pc="${pc:-/usr/lib/pkgconfig:/usr/share/pkgconfig}"

LANG="${LANG:-C.UTF-8}"
LC_ALL="${LC_ALL:-}"
TERM="${TERM:-xterm-256color}"

# Host openssl libdir (multiarch aware)
openssl_lib=""
if [[ -n "${triplet:-}" && -d "/usr/lib/${triplet}" ]]; then
  openssl_lib="/usr/lib/${triplet}"
elif [[ -d "/usr/lib64" ]]; then
  openssl_lib="/usr/lib64"
else
  openssl_lib="/usr/lib"
fi

exec env -i \
  HOME="$HOME" \
  PATH="$clean_path" \
  LANG="$LANG" \
  LC_ALL="$LC_ALL" \
  TERM="$TERM" \
  CARGO_HOME="$CARGO_HOME" \
  RUSTUP_HOME="$RUSTUP_HOME" \
  PKG_CONFIG_LIBDIR="$pc" \
  PKG_CONFIG_PATH="" \
  PKG_CONFIG_SYSROOT_DIR="" \
  OPENSSL_DIR="/usr" \
  OPENSSL_INCLUDE_DIR="/usr/include" \
  OPENSSL_LIB_DIR="$openssl_lib" \
  OPENSSL_NO_VENDOR="1" \
  "$CARGO_BIN" "$@"
#+END_SRC
