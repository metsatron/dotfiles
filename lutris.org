* Purpose

This file declares the shared Lutris storage topology.  It does not reconstruct
games entry-by-entry and it does not move user-owned game payloads.

Canonical payload storage:

#+BEGIN_SRC text :tangle linux/.config/lutris-shared/topology.txt
X230 canonical payload roots:
  ~/Games
  ~/RetroPie

Shared Lutris state:
  ~/.config/lutris       -> ~/RetroPie/stow/.config/lutris
  ~/.local/share/lutris  -> ~/RetroPie/stow/.local/share/lutris

Fleet links:
  T480/T480s ~/Games                 -> ~/mnt/x230/Games
  T480/T480s ~/RetroPie              -> ~/mnt/x230/RetroPie
  T480/T480s ~/.config/lutris        -> ~/mnt/x230/RetroPie/stow/.config/lutris
  T480/T480s ~/.local/share/lutris   -> ~/mnt/x230/RetroPie/stow/.local/share/lutris

Sanctuary guest-home links:
  ~/Games                -> /run/host/home/metsatron/Games
  ~/RetroPie             -> /run/host/home/metsatron/RetroPie
  ~/.config/lutris       -> /run/host/home/metsatron/RetroPie/stow/.config/lutris
  ~/.local/share/lutris  -> /run/host/home/metsatron/RetroPie/stow/.local/share/lutris

Policy:
  Lutris-created Wine prefixes under ~/Games are intentional portable payloads.
  Lutris cache remains machine-local.
  Do not install or download game data from DotCortex.
  Do not delete or replace existing real directories automatically.

* Host Link Apply

Creates or refreshes host-side links for machines that mount X230 at
=~/mnt/x230=.  Existing real directories are preserved and reported.

#+BEGIN_SRC bash :tangle linux/.local/bin/lutris-shared-link-host :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

host_name="$(hostname -s 2>/dev/null || hostname 2>/dev/null || printf unknown)"
x230_root="${X230_MOUNT:-$HOME/mnt/x230}"

safe_link() {
  local target="$1"
  local link="$2"

  if [[ ! -e "$target" ]]; then
    echo "lutris-shared-link-host: missing target $target" >&2
    return 1
  fi

  mkdir -p "$(dirname "$link")"
  if [[ -e "$link" && ! -L "$link" ]]; then
    echo "lutris-shared-link-host: preserving existing real path $link" >&2
    return 0
  fi

  ln -sfnT "$target" "$link"
  printf '%s -> %s\n' "$link" "$target"
}

case "$host_name" in
  x230|X230|ThinkPad-X230)
    safe_link "$HOME/RetroPie/stow/.config/lutris" "$HOME/.config/lutris"
    safe_link "$HOME/RetroPie/stow/.local/share/lutris" "$HOME/.local/share/lutris"
    ;;
  *)
    safe_link "$x230_root/Games" "$HOME/Games"
    safe_link "$x230_root/RetroPie" "$HOME/RetroPie"
    safe_link "$x230_root/RetroPie/stow/.config/lutris" "$HOME/.config/lutris"
    safe_link "$x230_root/RetroPie/stow/.local/share/lutris" "$HOME/.local/share/lutris"
    ;;
esac

echo "lutris-shared-link-host: cache stays local at $HOME/.cache/lutris"
#+END_SRC

* Health Check

Reports the effective shared paths without mutating them.

#+BEGIN_SRC bash :tangle linux/.local/bin/lutris-shared-health :shebang "#!/usr/bin/env bash" :tangle-mode (identity #o755)
set -euo pipefail

show_path() {
  local path="$1"
  if [[ -e "$path" || -L "$path" ]]; then
    printf '%s\n' "$path"
    ls -ld "$path"
    printf '  link: %s\n' "$(readlink "$path" 2>/dev/null || true)"
    printf '  real: %s\n' "$(readlink -f "$path" 2>/dev/null || true)"
  else
    printf '%s\n  MISSING\n' "$path"
  fi
}

echo "Host paths"
show_path "$HOME/Games"
show_path "$HOME/RetroPie"
show_path "$HOME/.config/lutris"
show_path "$HOME/.local/share/lutris"
show_path "$HOME/.cache/lutris"

echo
echo "Shared Lutris roots"
show_path "$HOME/RetroPie/stow/.config/lutris"
show_path "$HOME/RetroPie/stow/.local/share/lutris"

echo
echo "Guest projections"
guests_root="${XDG_DATA_HOME:-$HOME/.local/share}/dotcortex/guests"
if [[ -d "$guests_root" ]]; then
  for home in "$guests_root"/*/home; do
    [[ -d "$home" ]] || continue
    printf '%s\n' "$home"
    for rel in Games RetroPie .config/lutris .local/share/lutris; do
      path="$home/$rel"
      if [[ -e "$path" || -L "$path" ]]; then
        printf '  %s -> %s\n' "$rel" "$(readlink "$path" 2>/dev/null || true)"
      else
        printf '  %s MISSING\n' "$rel"
      fi
    done
  done
else
  echo "  no guest root at $guests_root"
fi
#+END_SRC
