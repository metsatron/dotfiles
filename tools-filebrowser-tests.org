    def run_script(self, name, *args, ok=True, env=None):
        path = BIN / name if name != 'filebrowser-serve' else REPO / 'all/.local/bin'/name
        r = subprocess.run(['/usr/bin/env', '-i', *[k+'='+v for k,v in (env or self.env).items()], str(path), *args], text=True, capture_output=True)
        if ok:
            self.assertEqual(r.returncode, 0, r.stderr + r.stdout)
        else:
            self.assertNotEqual(r.returncode, 0)
        return r

    def fake_binary(self, path, version, parses=True):
        body = ('#!/bin/bash\nif [ "${1:-}" = version ]; then echo "FileBrowser Quantum"; printf "\\tVersion \\t : %s\\n" "' + version + '"; exit 0; fi\n'
                + ('echo "[INFO ] Running at               : http://127.0.0.1:1/"; exec sleep 30\n' if parses else 'echo "[FATAL] error unmarshaling YAML data: unknown field"; exit 1\n'))
        pathlib.Path(path).write_text(body); pathlib.Path(path).chmod(0o755)
        return hashlib.sha256(pathlib.Path(path).read_bytes()).hexdigest()
    def install_env(self):
        (self.home / '.config/filebrowser/config.yaml').write_text('server:\n  port: 9091\n  database: ' + str(self.home / 'db.db') + '\n  cacheDir: ' + str(self.home / 'cache') + '\n')
        (self.home / 'db.db').write_bytes(b'users')
        env = dict(self.env)
        env.update({'FILEBROWSER_BIN': str(self.home / '.local/bin/filebrowser-milkdown'), 'FILEBROWSER_CONFIG': str(self.home / '.config/filebrowser'),
                    'FILEBROWSER_INSTALL_RESTART': 'echo restarted >> "$HOME/restart.log"', 'FILEBROWSER_INSTALL_WAIT': '2', 'FILEBROWSER_INSTALL_PRECHECK_WAIT': '5',
                    'FILEBROWSER_INSTALL_HEALTH': 'test "$(sha256sum "$HOME/.local/bin/filebrowser-milkdown" | cut -d" " -f1)" = "$1" && test "$1" != "$(cat "$HOME/sickfor" 2>/dev/null)"'})
        return env
    def installed(self, version='v1.5.6-stable'):
        return self.fake_binary(self.home / '.local/bin/filebrowser-milkdown', version)
    def sha_of(self, path):
        return hashlib.sha256(pathlib.Path(path).read_bytes()).hexdigest()
    def test_installer_dry_run_changes_nothing(self):
        env = self.install_env(); old = self.installed(); cand = self.home / 'cand'; self.fake_binary(cand, 'v1.5.8-stable')
        r = self.run_script('filebrowser-milkdown-install', '--candidate', str(cand), env=env)
        self.assertIn('dry run', r.stdout)
        self.assertEqual(self.sha_of(self.home / '.local/bin/filebrowser-milkdown'), old)
        self.assertFalse((self.home / '.local/state/filebrowser/install-backups').exists()); self.assertFalse((self.home / 'restart.log').exists())
    def test_installer_apply_backs_up_swaps_restarts_and_proves_the_running_binary(self):
        env = self.install_env(); old = self.installed(); cand = self.home / 'cand'; new = self.fake_binary(cand, 'v1.5.8-stable')
        r = self.run_script('filebrowser-milkdown-install', '--candidate', str(cand), '--expect-version', 'v1.5.8-stable', '--apply', env=env)
        self.assertEqual(self.sha_of(self.home / '.local/bin/filebrowser-milkdown'), new)
        backup = sorted((self.home / '.local/state/filebrowser/install-backups').iterdir())[-1]
        self.assertEqual(self.sha_of(backup / 'filebrowser-milkdown'), old)
        self.assertTrue((backup / 'config.yaml').exists()); self.assertEqual((backup / 'database.db').read_bytes(), b'users')
        self.assertIn('installed_version=v1.5.6-stable', (backup / 'manifest').read_text())
        self.assertEqual((self.home / 'restart.log').read_text().count('restarted'), 1)
